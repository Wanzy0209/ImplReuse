import torch
import tensorflow as tf
from tensorflow.keras.initializers import HeUniform

def test_heuniform_redundant_dtype_conversion():
    """
    Test case adapted from PyTorch Issue 161611.
    
    This test verifies that performing a redundant dtype conversion on a tensor
    created by HeUniform (without assigning the result) is ineffective and 
    does not alter the original tensor, mirroring the bug scenario where 
    attn_bias.to(query.dtype) was called redundantly.
    """
    # Setup: Define shape and target dtype
    shape = (4, 4)
    target_dtype = tf.float32

    # Step 1: Create tensor using the Similar API (HeUniform) with specific dtype
    # Analogous to: attn_bias = torch.zeros(..., dtype=query.dtype)
    initializer = HeUniform()
    weights = initializer(shape, dtype=target_dtype)

    # Verify initial dtype matches the target
    assert weights.dtype == target_dtype, "Initial dtype should be float32"

    # Step 2: Mimic the bug - redundant conversion without assignment
    # Analogous to: attn_bias.to(query.dtype)
    # In TensorFlow, tf.cast returns a new tensor. If we don't assign it, 
    # the operation is effectively lost/redundant.
    tf.cast(weights, target_dtype)

    # Step 3: Verify the tensor remains unchanged
    # The bug highlights that the line was redundant and the result not assigned.
    # We assert that 'weights' still has the original dtype and value.
    assert weights.dtype == target_dtype, "Dtype should remain unchanged after unassigned cast"

    # Step 4: Verify that assigning the redundant cast results in the same tensor
    # This confirms the cast was indeed redundant (same dtype).
    casted_weights = tf.cast(weights, target_dtype)
    assert casted_weights.dtype == target_dtype
    # Check values are identical
    assert tf.reduce_all(tf.equal(weights, casted_weights)).numpy()

if __name__ == "__main__":
    test_heuniform_redundant_dtype_conversion()
    print("Test passed: Redundant dtype conversion handled correctly.")
