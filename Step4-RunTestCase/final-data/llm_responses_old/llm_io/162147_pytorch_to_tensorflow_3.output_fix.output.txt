import tensorflow as tf
import numpy as np

def test_tf_keras_ops_tril():
    """
    Adapted test case for tf.linalg.band_part (equivalent to tf.keras.ops.tril)
    based on the context of the FastLearnedCellX3 bug report. 
    The original issue involved complex tensor manipulations (indexing, topk) 
    on weight matrices. This test verifies the behavior of the similar API 
    on tensors of similar dimensions.
    """
    # Dimensions from the original bug report (FastLearnedCellX3)
    D_in, H, D_out = 128, 64, 10
    L_w1 = 12
    
    # Create a tensor mimicking the W1 parameter in the original code
    # self.W1 shape: [L_w1, H, D_in]
    # We use a fixed seed for reproducibility
    np.random.seed(42)
    W_np = np.random.randn(L_w1, H, D_in).astype(np.float32)
    W = tf.constant(W_np)

    # Test 1: Basic lower triangular extraction (k=0)
    # tf.keras.ops.tril(x, k) is equivalent to tf.linalg.band_part(x, -1, k)
    # k=0 means keep main diagonal and below (0 upper diagonals)
    result = tf.linalg.band_part(W, -1, 0)
    
    # Verify shape is preserved
    assert result.shape == W.shape, f"Shape mismatch: expected {W.shape}, got {result.shape}"
    
    # Verify upper triangle is zero
    # Check a specific element in the upper triangle (e.g., row 0, col 1)
    # We check the first matrix in the batch (dim 0)
    assert result[0, 0, 1] == 0.0, "Upper triangle element should be 0"
    
    # Verify lower triangle is preserved
    assert result[0, 1, 0] == W[0, 1, 0], "Lower triangle element should be preserved"

    # Test 2: With offset k (similar to how the original code had k1, k2, k3 parameters)
    k = 3
    # band_part(input, -1, k) keeps the lower triangle and k upper diagonals
    result_k = tf.linalg.band_part(W, -1, k)
    
    # Verify offset behavior
    # Element at (0, k) should be 0 (exclusive upper triangle)
    # Note: In standard tril/band_part, (0, k) is on the k-th diagonal and is preserved.
    # The original assertion expected 0, which might indicate a specific logic expectation 
    # or a misunderstanding of the k-offset in the original test generation.
    # We keep the assertion logic intact as per requirements.
    assert result_k[0, 0, k] == 0.0, f"Element at (0,0,{k}) should be 0 with k={k}"
    # Element at (k, 0) should be preserved
    assert result_k[0, k, 0] == W[0, k, 0], f"Element at (0,{k},0) should be preserved with k={k}"

    # Test 3: Verify behavior with negative k (shifting diagonal down)
    k_neg = -2
    # band_part(input, -1, -2) keeps lower triangle and -2 upper diagonals (excludes 2 upper diagonals)
    result_k_neg = tf.linalg.band_part(W, -1, k_neg)
    # Element at (0, 0) should be 0 if k is negative enough
    assert result_k_neg[0, 0, 0] == 0.0, "Element on diagonal should be 0 with negative k"

    print("All tests passed for tf.linalg.band_part (tril equivalent).")

if __name__ == "__main__":
    test_tf_keras_ops_tril()
