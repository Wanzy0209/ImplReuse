import tensorflow as tf

class M(tf.Module):
    def __init__(self, n_fft=512, hop=160, win=320):
        super().__init__()
        self.n_fft = n_fft
        self.hop = hop
        self.win = win
        # Register window as a variable/buffer equivalent
        self.window = tf.signal.hann_window(win, dtype=tf.float32)
        # Introduce parameter for broadcasting / stride ops
        self.p = tf.Variable(2.0, dtype=tf.float32)

    @tf.function
    def __call__(self, x: tf.Tensor) -> tf.Tensor:
        # tf.signal.stft returns complex tensor by default
        S = tf.signal.stft(
            x, frame_length=self.win, frame_step=self.hop, fft_length=self.n_fft,
            window=self.window, pad_end=True
        )
        
        R = tf.abs(tf.math.real(S))   # unary op on real
        
        # Use the similar API: tf.compat.v1.math.add
        # Replaces the scalar divide with an add operation involving the Parameter
        # to leverage the identified similar API pattern.
        I = tf.compat.v1.math.add(tf.math.imag(S), self.p) 
        
        Z = tf.complex(R, I) # recombine
        return Z

def main():

    tf.random.set_seed(0)
    # Input tensor

    # Check if complex
    assert z_eager.dtype == tf.complex64

    # Compiled execution (tf.function)

    # We call the module again to ensure the graph path works.
    z_compiled = m(x)

    assert tf.reduce_all(tf.equal(tf.math.real(z_eager), tf.math.real(z_compiled)))
