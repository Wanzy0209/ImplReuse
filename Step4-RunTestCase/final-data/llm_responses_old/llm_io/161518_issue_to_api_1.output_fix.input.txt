import torch
import tensorflow as tf
import time

# Mapping dtypes to their size in bytes
dtype_size_map = {
    tf.float16: 2,
    tf.bfloat16: 2,
    tf.float32: 4
}

def test_linear_operator_identity_throughput():
    """
    Test case adapted from Issue 161518 (PyTorch copy_ throughput).
    
    Original Issue Logic:
    The original issue benchmarks memory throughput for torch.cat (which involves 
    copying data from source tensors to a destination tensor) on B200 hardware.
    
    Similar API Logic (tf.linalg.LinearOperatorIdentity):
    The LinearOperatorIdentity is an implicit representation. Materializing it 
    via to_dense() involves allocating memory and writing the identity matrix 
    (copying data from implicit definition to explicit memory). The API docs 
    specifically mention that broadcasting "will require making copies".
    
    Adaptation:
    This test benchmarks the materialization (to_dense) of a large 
    LinearOperatorIdentity to measure memory write throughput, analogous to 
    the copy operations in the original issue.
    """
    target_size_mib = 128  # Target 128 MiB data size, matching the original issue

    for dtype in [tf.float16, tf.bfloat16, tf.float32]:
        element_size = dtype_size_map[dtype]

        # Calculate dimensions to match target data size
        # N x N matrix where N*N*element_size = 128 MiB
        num_rows = int(((target_size_mib * 1024 * 1024) / element_size) ** 0.5)

        # Initialize LinearOperatorIdentity
        # This operator is efficient and does not materialize the matrix initially.
        operator = tf.linalg.LinearOperatorIdentity(num_rows=num_rows, dtype=dtype)

        # Warmup run to avoid initialization overhead in benchmark
        _ = operator.to_dense()

        # Benchmark run
        start = time.time()
        rep = 100
        for _ in range(rep):
            dense_matrix = operator.to_dense()
        end = time.time()

        avg_time_ms = (end - start) * 1000 / rep

        # Calculate throughput (TiB/s)
        # Data processed: 128 MiB written to memory
        data_size_tib = (target_size_mib / 1024 / 1024)
        bdwidth = data_size_tib / (avg_time_ms / 1000)

        print("\t".join([str(dtype), f"{avg_time_ms:.2f} ms", f"{bdwidth:.2f} TiB/s"]))

        # Assertions to verify correctness of the operation
        assert dense_matrix.shape == (num_rows, num_rows), f"Shape mismatch for {dtype}"
        assert dense_matrix.dtype == dtype, f"Dtype mismatch for {dtype}"
        
        # Verify it is actually an identity matrix (spot check diagonal)
        # This ensures the "copy" (materialization) was accurate
        assert tf.reduce_all(tf.linalg.diag_part(dense_matrix) == 1.0), "Diagonal elements should be 1"

if __name__ == "__main__":
    test_linear_operator_identity_throughput()
