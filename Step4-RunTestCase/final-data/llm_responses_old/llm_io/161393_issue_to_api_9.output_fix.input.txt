import torch
import tensorflow as tf
import numpy as np

def test_bias_add_with_dynamic_slicing():
    """
    Test case adapted from PyTorch Issue 161393.
    Original Issue: Hard error when slicing a tensor with unbacked sizes inside torch.compile.
    Adaptation: Tests tf.keras.backend.bias_add inside a tf.function (compiled context)
    with dynamic input shapes, followed by a slicing operation.
    """
    
    # Define a function with dynamic input shape (None in the first dimension)
    # This mimics the "unbacked" or dynamic nature of the PyTorch issue
    @tf.function(input_signature=[
        tf.TensorSpec(shape=[None, 4], dtype=tf.float32),
        tf.TensorSpec(shape=[4], dtype=tf.float32)
    ])
    def f(x, bias):
        # Use the similar API: bias_add
        # This replaces 'nz = x.nonzero()' in the original bug logic
        y = tf.keras.backend.bias_add(x, bias, data_format='channels_last')
        
        # Perform the slicing operation that caused the error in PyTorch
        # Original: return nz[:-1]
        return y[:-1]

    # Create test data
    x_data = np.random.randn(3, 4).astype(np.float32)
    bias_data = np.random.randn(4).astype(np.float32)

    # Execute
    try:
        out = f(x_data, bias_data)
        print("Test passed. Output shape:", out.shape)
        
        # Assertions to verify correctness
        # The input has 3 rows, slicing [:-1] should result in 2 rows
        assert out.shape[0] == 2, f"Expected 2 rows after slicing, got {out.shape[0]}"
        assert out.shape[1] == 4, f"Expected 4 columns, got {out.shape[1]}"
        
        # Verify values (bias_add adds bias to each channel/row for channels_last)
        expected = x_data[:-1] + bias_data
        np.testing.assert_allclose(out.numpy(), expected, rtol=1e-5)
        
    except Exception as e:
        print(f"Test failed with error: {e}")
        raise

if __name__ == "__main__":
    test_bias_add_with_dynamic_slicing()
