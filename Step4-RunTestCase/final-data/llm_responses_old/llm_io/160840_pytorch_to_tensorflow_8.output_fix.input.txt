    # Input format is NHWC for TensorFlow

        input,
        ksize=2,
        strides=2,
        padding='VALID',
        data_format='NHWC'
    )

    squeeze = tf.squeeze(pool, axis=0)
    # Argmin along dimension 1 (Height dimension in NHWC)
    argmin = tf.argmin(squeeze, axis=1)
    return argmin

# Assert results are close
