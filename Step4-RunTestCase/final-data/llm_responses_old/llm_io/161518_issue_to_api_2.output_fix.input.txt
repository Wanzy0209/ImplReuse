import tensorflow as tf
import time

def benchmark_moving_average_update():
    """
    Benchmarks memory throughput for tf.keras.backend.moving_average_update.
    This mirrors the logic of the PyTorch issue #161518 which benchmarks
    memory throughput for elementwise ops (copy_/cat).
    """
    print("Benchmarking tf.keras.backend.moving_average_update memory throughput")
    
    # Check for GPU
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("No GPU detected. Running on CPU (results will be slow).")
    
    # Dtypes to test, matching the spirit of the original issue (various precisions)
    dtypes = [tf.float16, tf.bfloat16, tf.float32]

    for dtype in dtypes:
        element_size = dtype.size # bytes per element
        
        # Create tensors of approx 128 MiB
        # Shape calculation: (128 MiB / element_size) elements
        # We use a 2D shape similar to the original script
        num_elements = (128 * 1024 * 1024) // element_size
        dim1 = num_elements // 1024
        dim2 = 1024
        
        # x is the variable to be updated, value is the new data
        x = tf.Variable(tf.zeros((dim1, dim2), dtype=dtype))
        value = tf.zeros((dim1, dim2), dtype=dtype)
        momentum = 0.9

        # Warmup
        tf.keras.backend.moving_average_update(x, value, momentum)
        _ = x.numpy() # Sync

        # Benchmark
        reps = 100
        start_time = time.perf_counter()
        
        for _ in range(reps):
            tf.keras.backend.moving_average_update(x, value, momentum)
            
        # Synchronize to ensure all ops finished
        _ = x.numpy()
        
        end_time = time.perf_counter()
        
        avg_time_ms = (end_time - start_time) * 1000 / reps
        
        # Calculate IO
        # Operation: x = x * momentum + value * (1 - momentum)
        # Read x (128MB) + Read value (128MB) + Write x (128MB) = 384 MB total IO
        total_io_mib = 384 
        total_io_tib = total_io_mib / 1024 / 1024
        
        # Bandwidth in TiB/s
        # TiB/s = TiB / (ms / 1000)
        bdwidth = total_io_tib / (avg_time_ms / 1000)
        
        print(f"\t".join([str(dtype), f"{avg_time_ms:.2f} ms", f"{bdwidth:.2f} TiB/s"]))

if __name__ == "__main__":
    benchmark_moving_average_update()
