import torch
import tensorflow as tf
import numpy as np

def test_distributed_values_addmm():
    """
    Test case adapted from PyTorch Issue 161618.
    
    Original Issue: Inductor fails on triton main branch when executing 
    torch.addmm with specific configurations and tensor shapes.
    
    Adaptation: This test translates the logic to TensorFlow using the 
    tf.distribute.DistributedValues API. It verifies that a distributed 
    matrix multiplication (equivalent to addmm) executes correctly 
    when inputs are handled as DistributedValues across replicas.
    """
    
    # Dimensions from the original bug report
    m = 20120
    k = 1536
    n = 512

    # Initialize distribution strategy (MirroredStrategy for multi-GPU/CPU simulation)
    strategy = tf.distribute.MirroredStrategy()

    # Define the operation equivalent to torch.addmm(a, mat1, mat2)
    # which computes mat1 @ mat2 + a
    def distributed_addmm(a, mat1, mat2):
        return tf.add(a, tf.matmul(mat1, mat2))

    with strategy.scope():
        # Function to create DistributedValues for inputs
        # This mimics the setup of tensors in the original script
        def input_fn(ctx):
            # Calculate the slice of data for this specific replica
            num_replicas = ctx.num_replicas_in_sync
            replica_batch_size = m // num_replicas
            
            # Create local slices for 'a' and 'mat1' (dimension m is split)
            # Note: In the original bug, these were random tensors.
            local_a = tf.random.normal((replica_batch_size, n))
            local_mat1 = tf.random.normal((replica_batch_size, k))
            
            # 'mat2' is shared across all replicas (dimension k, n)
            # In a real distributed scenario, this might be broadcast or sharded by k.
            # Here we replicate it fully to match the semantic usage of mat2 in addmm.
            local_mat2 = tf.random.normal((k, n))
            
            return local_a, local_mat1, local_mat2

        # Create DistributedValues using the experimental API
        # This explicitly leverages the 'Similar API' identified in the prompt
        dist_a, dist_mat1, dist_mat2 = strategy.experimental_distribute_values_from_function(input_fn)

        # Execute the operation
        # The result will be a DistributedValues object containing the per-replica outputs
        result = strategy.run(distributed_addmm, args=(dist_a, dist_mat1, dist_mat2))

        # Assertions to verify correctness and API interaction
        # 1. Verify the result is a DistributedValues (or PerReplica) object
        assert isinstance(result, tf.distribute.DistributedValues), \
            "Result should be a DistributedValues object"

        # 2. Verify we can retrieve values (mimicking the _get logic in the Similar API info)
        per_replica_results = strategy.experimental_local_results(result)
        
        # 3. Check shape consistency
        # Each replica should produce a tensor of shape (batch_size, n)
        expected_batch_size = m // strategy.num_replicas_in_sync
        for tensor in per_replica_results:
            assert tensor.shape == (expected_batch_size, n), \
                f"Expected shape ({expected_batch_size}, {n}), got {tensor.shape}"

        print("Test passed: DistributedValues handled addmm operation successfully.")

if __name__ == "__main__":
    test_distributed_values_addmm()
