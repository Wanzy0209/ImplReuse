import sys
import torch

# Handle environment dependency issues for TensorFlow
try:
    import tensorflow as tf
except ImportError as e:
    # Check if the error is related to the GLIBC version mismatch
    if "GLIBCXX" in str(e):
        print("Test skipped: TensorFlow import failed due to environment issue (GLIBC version).")
        print(f"Details: {e}")
        sys.exit(0)
    else:
        # If it's a different import error, raise it
        raise

def test_sparse_concat_complex_empty():
    """
    Adapted from PyTorch Issue 160495.
    Original issue: Adding a non-empty complex tensor with an empty (0-dim) complex tensor 
    caused a crash in PyTorch Inductor.
    
    This test verifies the behavior of tf.sparse.concat when concatenating a non-empty 
    complex sparse tensor with an empty complex sparse tensor.
    """
    
    # 1. Create a non-empty complex sparse tensor
    # Analogous to x = torch.randn(1, dtype=torch.complex64)
    indices_a = [[0, 0], [1, 1]]
    values_a = [1.0 + 2.0j, 3.0 + 4.0j]
    shape_a = [2, 2]
    sparse_a = tf.SparseTensor(indices=indices_a, values=values_a, dense_shape=shape_a)
    
    # 2. Create an empty complex sparse tensor
    # Analogous to y = torch.empty((), dtype=torch.complex64) in terms of being empty/edge-case
    # Note: Sparse tensors must have rank >= 1, so we use a 0-element tensor instead of 0-dim.
    indices_b = tf.zeros((0, 2), dtype=tf.int64)
    values_b = tf.zeros((0,), dtype=tf.complex64)
    shape_b = [2, 2] # Shape must match sparse_a for concatenation on axis 0
    sparse_b = tf.SparseTensor(indices=indices_b, values=values_b, dense_shape=shape_b)
    
    # 3. Perform the operation
    # Analogous to fn(x, y) -> x + y
    try:
        result = tf.sparse.concat([sparse_a, sparse_b], axis=0)
        
        # 4. Verify behavior
        # The operation should succeed and return the non-empty tensor's content
        # effectively, as concatenating nothing adds nothing.
        assert result.dtype == tf.complex64, "Output dtype should be complex64"
        # Result shape should be sum of shapes along axis 0: 2 + 0 = 2
        assert result.dense_shape[0] == 2, "Output shape dimension 0 should be 2"
        
        print("tf.sparse.concat success: Handled empty complex tensor correctly.")
        print("Result:", result)
        
    except Exception as e:
        print(f"tf.sparse.concat failed with error: {e}")

if __name__ == "__main__":
    test_sparse_concat_complex_empty()
