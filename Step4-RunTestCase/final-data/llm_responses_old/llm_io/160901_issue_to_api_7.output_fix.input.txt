class MyTraceFn(torch.autograd.Function):

    def forward(ctx, x):

        return torch.trace(x)

    def backward(ctx, grad_out):
        # Propagate gradient to ensure requires_grad is handled correctly.
        # We return a matrix filled with the output gradient to verify

        return torch.ones_like(x) * grad_out

    assert grad_eager is not None, "Eager gradient is None"

    # Check that gradients are correct
    assert torch.allclose(grad_eager, grad_compiled), \
        f"Gradients differ between eager and compiled modes.\nEager: {grad_eager}\nCompiled: {grad_compiled}"
