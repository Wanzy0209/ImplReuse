import torch
import sys

# torch.export is available in PyTorch 2.1+. 
# We handle the ImportError for older versions by skipping the test.
try:
    from torch import export
except ImportError:
    print("Skipping test: torch.export is not available (requires PyTorch >= 2.1).")
    sys.exit(0)

from typing import List

def view_decomposition(x: torch.Tensor, size: List[torch.SymInt]) -> torch.Tensor:
    """
    Custom decomposition function that uses torch.ops.aten._reshape_copy.
    This mirrors the logic in the bug report where view is replaced by _reshape_copy.
    """
    return torch.ops.aten._reshape_copy.default(x, size)

class SimpleModel(torch.nn.Module):
    """
    A minimal model to replace the heavy Qwen3MoeModel dependency.
    It performs a view operation which is the target of the decomposition.
    """
    def forward(self, x):
        return x.view(x.size(0), -1)

def test_reshape_copy_unbacked_semantics():
    """
    Test case for Issue ID: 162110.
    Verifies that torch.ops.aten._reshape_copy works correctly within
    torch.export when handling dynamic shapes (SymInts).
    """
    # Define a dynamic dimension similar to the bug report
    seq_len = export.Dim("seq_len", min=1, max=128)
    
    model = SimpleModel()
    inputs = torch.randn(1, 12)
    
    # Export the model with dynamic shapes
    # strict=False is used as in the original bug report
    ep = export.export(
        model,
        args=(inputs,),
        dynamic_shapes=({1: seq_len},),
        strict=False,
    )
    
    # Set up the decomposition table
    decomp_table = export.default_decompositions()
    # Replace the default view op with our custom decomposition using _reshape_copy
    decomp_table[torch.ops.aten.view.default] = view_decomposition

    # Run decompositions
    # In the bug report, this step fails due to unbacked semantics not being defined.
    # A successful run indicates the fix or expected behavior.
    after_decomp = ep.run_decompositions(decomp_table=decomp_table)
    
    # Verify that the decomposition was applied
    graph_str = str(after_decomp.graph)
    assert "aten::_reshape_copy" in graph_str, "Decomposition did not apply _reshape_copy"
    
    print("Test passed: _reshape_copy handles dynamic shapes correctly.")

if __name__ == "__main__":
    test_reshape_copy_unbacked_semantics()
