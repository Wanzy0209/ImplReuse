import torch
import unittest
import gc
import psutil

# Handle environment/dependency issues for TensorFlow
TF_AVAILABLE = False
try:
    import tensorflow as tf
    import tf.experimental.numpy as tnp
    TF_AVAILABLE = True
except ImportError as e:
    # If the specific GLIBCXX error or any other import error occurs, we skip.
    print(f"Warning: Could not import TensorFlow. Skipping related tests. Error: {e}")

def mem_mb_print():
    gc.collect()
    process = psutil.Process().memory_info().rss
    print(f"Process Memory: {process // 1024**2} MB")

@unittest.skipIf(not TF_AVAILABLE, "TensorFlow not available or environment issue (e.g., GLIBCXX)")
class TestFullLikeMemory(unittest.TestCase):
    # We must check TF_AVAILABLE here to prevent accessing tf.config if tf is None
    @unittest.skipIf(TF_AVAILABLE and len(tf.config.list_physical_devices('GPU')) == 0, "GPU not available")
    def test_full_like_memory_leak(self):
        # Setup device (TensorFlow handles device placement implicitly or via strategy)
        
        # leaf GPU tensor
        # In PyTorch: leaf = torch.randn(..., requires_grad=True)
        # In TF: We use a tf.Variable with trainable=True to mimic requires_grad=True
        leaf = tf.Variable(tnp.random.randn(8, 1, 24), trainable=True)
        
        # non-leaf tensor
        # In PyTorch: no_leaf = leaf * 1.0
        # In TF: Operations on Variables yield Tensors. To perform in-place assignment 
        # (which is the core of the bug reproduction), we wrap the result in a Variable.
        no_leaf = tf.Variable(leaf * 1.0, trainable=True)

        for _ in range(100):
            window_left = no_leaf[:, :, :-1]          # [8, 1, 23]
            
            # inplace operators
            # PyTorch: no_leaf[:, :, 1:] = torch.max(no_leaf[:, :, 1:], window_left)
            # TF Adaptation: Using .assign() for in-place update and tnp.full_like as the target API.
            # tnp.full_like creates a tensor with the same shape as window_left filled with 1.0.
            no_leaf[:, :, 1:].assign(tnp.full_like(window_left, 1.0))
            
            # Python arithmetic operators
            mask = (no_leaf >= 0)
            mem_mb_print()

        loss = tnp.sum(mask)

if __name__ == "__main__":
    unittest.main()
