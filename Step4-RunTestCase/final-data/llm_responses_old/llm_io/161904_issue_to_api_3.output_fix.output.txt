"""
Test case for Issue 161904: ZeroBubble and DualPipeV pipeline parallel schedules fail with torch.compiled model.

This test case preserves the original bug reproduction logic (PyTorch Pipeline Parallelism + torch.compile)
while leveraging the code pattern and semantics of the similar API (tf.io.decode_proto) within the model definition.
The tf.io.decode_proto logic involves iterating through a list of codecs to find a matching decoder,
which is translated here into a PyTorch module to test compatibility with torch.compile.
"""

import torch
import torch.nn as nn
import torch.distributed as dist
from torch.distributed.device_mesh import init_device_mesh
from torch.distributed.pipelining import PipelineStage
from torch.distributed.pipelining.schedules import get_schedule_class
import functools

# Translating the semantics of tf.io.decode_proto to a PyTorch Module
# The original API iterates through a list of codecs to decode a structure.
class DecodeProtoStyleLayer(nn.Module):
    """
    Mimics the logic of tensorflow.python.saved_model.nested_structure_coder._map_structure
    which is utilized by tf.io.decode_proto.
    """
    def __init__(self):
        super().__init__()
        # Simulating a list of codecs (can_decode, do_decode)
        # We use a ModuleList for the 'do' part (the actual layers)
        self.codecs = nn.ModuleList([
            nn.Linear(32, 32, bias=False),
            nn.Linear(32, 32, bias=False)
        ])

    def forward(self, x):
        # Mimic the iteration logic: iterate through codecs to process input
        for layer in self.codecs:
            x = layer(x)
        return x
