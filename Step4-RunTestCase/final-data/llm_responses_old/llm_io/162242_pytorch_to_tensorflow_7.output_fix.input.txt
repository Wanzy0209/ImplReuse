def test_determinism_normalize():
    """
    Adapted test for tf.linalg.normalize based on the torch.scatter determinism issue.
    The original issue questioned the non-deterministic behavior of scatter with duplicate indices.
    This test verifies if tf.linalg.normalize produces deterministic results and gradients
    across multiple iterations.
    """
    # Setup input data similar to the original test's shape
    # Original inputs: (2, 3, 4)
    x_np = np.arange(24, dtype=np.float32).reshape([2, 3, 4])

    # Determine device (GPU if available, mimicking the original .cuda() call)
    device = '/GPU:0' if tf.config.list_physical_devices('GPU') else '/CPU:0'
    print(f"Running test on device: {device}")

    with tf.device(device):

        # Note: tf.linalg.normalize returns (normalized_tensor, norm)
        x = tf.Variable(x_np)
        with tf.GradientTape() as tape:
            res, _ = tf.linalg.normalize(x, axis=1, order=2)
        
        grad = tape.gradient(res, x)
        
        gt_res = res.numpy()
        gt_grad = grad.numpy()

        # Loop to check for non-determinism (Original test ran 1000 times)
        for i in range(1000):
            # Re-initialize variable to ensure clean state
            x = tf.Variable(x_np)
            
            with tf.GradientTape() as tape:
                res, _ = tf.linalg.normalize(x, axis=1, order=2)
            
            grad = tape.gradient(res, x)

            # Verify consistency against the ground truth
            # The original issue was about results looking deterministic despite docs saying otherwise.
            # We assert that the results match the first run exactly.
            np.testing.assert_allclose(res.numpy(), gt_res, rtol=1e-5, atol=1e-5)
            np.testing.assert_allclose(grad.numpy(), gt_grad, rtol=1e-5, atol=1e-5)

            if (i + 1) % 100 == 0:
                print(f"Test {i + 1}/1000 passed")

    print("All tests passed. Behavior is deterministic.")


if __name__ == "__main__":
