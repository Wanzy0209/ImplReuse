import torch
import tensorflow as tf
import numpy as np

def test_adam_with_addmm_operation():
    """
    Test case for tf.keras.optimizers.Adam based on the logic of PyTorch issue #161618.
    
    The original issue involved a failure in torch.compile when performing an addmm 
    operation (matrix multiplication + addition) with specific tensor dimensions 
    (m=20120, k=1536, n=512) under specific backend configurations.
    
    This test adapts that scenario to TensorFlow by using the Adam optimizer to 
    perform a training step involving the equivalent operation (tf.add + tf.matmul)
    with the same tensor dimensions, wrapped in a tf.function to mimic compilation.
    """
    # Dimensions from the original bug report
    m = 20120
    k = 1536
    n = 512

    # Initialize tensors matching the original shapes
    # In the original bug, these were inputs. Here we make mat1 and mat2 variables
    # to be optimized by Adam, simulating a training step involving this operation.
    a = tf.constant(np.random.randn(m, n).astype(np.float32))
    mat1 = tf.Variable(np.random.randn(m, k).astype(np.float32))
    mat2 = tf.Variable(np.random.randn(k, n).astype(np.float32))

    # Instantiate the Similar API: Adam optimizer
    # Using explicit configuration to mirror the config patching in the original issue
    optimizer = tf.keras.optimizers.Adam(
        learning_rate=0.001,
        beta_1=0.9,
        beta_2=0.999,
        epsilon=1e-7
    )

    # Define the computation graph equivalent to torch.addmm(a, mat1, mat2)
    # which computes a + mat1 @ mat2
    # We use @tf.function to mimic the torch.compile aspect of the original issue
    @tf.function(jit_compile=True)
    def compute_step():
        with tf.GradientTape() as tape:
            # Operation: addmm equivalent
            # torch.addmm(a, mat1, mat2) -> a + (mat1 * mat2)
            result = tf.add(a, tf.matmul(mat1, mat2))
            
            # Define a simple loss function (sum of squares) to generate gradients
            loss = tf.reduce_sum(result ** 2)
        
        # Calculate gradients and apply them using Adam
        gradients = tape.gradient(loss, [mat1, mat2])
        optimizer.apply_gradients(zip(gradients, [mat1, mat2]))
        
        return loss

    # Execute the step
    # This verifies that Adam can handle the specific workload that caused 
    # the failure in the PyTorch Inductor/Triton backend.
    loss = compute_step()

    # Assertion to ensure execution completed and returned a valid loss
    assert loss is not None
    print("Test passed: Adam optimizer successfully handled addmm operation with dimensions {}x{}x{}.".format(m, k, n))

if __name__ == "__main__":
    test_adam_with_addmm_operation()
