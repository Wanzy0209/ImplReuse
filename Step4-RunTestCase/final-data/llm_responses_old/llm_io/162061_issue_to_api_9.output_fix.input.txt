def test_dynamo_pixel_shuffle_export():

    unnecessary operations to the graph around pixel shuffle.

            self.pixel_shuffle = nn.PixelShuffle(upscale_factor)

            input_names=["input"],
            output_names=["output"],
            dynamic_shapes={"input": {0: "batch", 2: "width", 3: "height"}},
