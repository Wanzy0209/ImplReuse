# Disable v2 behavior to use SessionCreator as intended

    """
    A concrete implementation of SessionCreator to leverage the similar API.
    This acts as the factory for the session in which the graph is executed.
    """
    def create_session(self):

def test_uint8_neg_add_logic():
    """
    Test case adapted from the PyTorch bug report (Issue 161763).
    It verifies the behavior of uint8 negation and addition using TensorFlow's
    graph execution, managed by tf.compat.v1.train.SessionCreator.
    """
    # Input data matching the PyTorch issue
    x_val = np.array([[1.5410, -0.2934], [-2.1788, 0.5684]], dtype=np.float32)

    # Build the TensorFlow graph
    # c is a uint8 tensor with value 7
    c = tf.constant(7, dtype=tf.uint8)
    
    # x is a float32 placeholder
    x = tf.placeholder(dtype=tf.float32, shape=(2, 2))

    # Operation 1: c + x
    # Expected: 7 + x
    res0 = c + x

    # Operation 2: neg(c)
    # Note: tf.negative does not support uint8 directly. To simulate the 
    # correct wrapping behavior (249) seen in PyTorch eager mode, we cast 
    # to int32, negate, and cast back to uint8.
    neg_c = tf.cast(tf.cast(c, tf.int32) * -1, tf.uint8)

    # Operation 3: neg(c) + x
    # Expected: 249 + x
    res2 = neg_c + x

    # Use the SessionCreator to manage the session lifecycle
    creator = CustomSessionCreator()
    
    with creator.create_session() as sess:
        # Execute the graph
        out0, out2 = sess.run([res0, res2], feed_dict={x: x_val})

        # Expected values based on the "res" (correct) output from the PyTorch issue
        # res[0]: tensor([[8.5410, 6.7066], [4.8212, 7.5684]])
        expected0 = np.array([[8.5410, 6.7066], [4.8212, 7.5684]], dtype=np.float32)
        
        # res[2]: tensor([[250.5410, 248.7066], [246.8212, 249.5684]])
        expected2 = np.array([[250.5410, 248.7066], [246.8212, 249.5684]], dtype=np.float32)

        # Assert results match the expected behavior
        np.testing.assert_allclose(out0, expected0, rtol=1e-5, err_msg="Addition result mismatch")
        np.testing.assert_allclose(out2, expected2, rtol=1e-5, err_msg="Neg+Add result mismatch")

    print("Test passed successfully.")

if __name__ == "__main__":
    test_uint8_neg_add_logic()
