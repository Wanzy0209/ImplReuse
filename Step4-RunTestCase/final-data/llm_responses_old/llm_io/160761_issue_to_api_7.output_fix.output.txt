import torch
import numpy as np
import tempfile
import os
import sys

# Handle environment dependency issues (e.g., GLIBCXX) by skipping the test if TF fails to load
try:
    import tensorflow as tf
except ImportError as e:
    print(f"Skipping test: TensorFlow import failed due to environment incompatibility.")

def test_gemma3_like_export_with_context_check():
    """
    Test case adapted from Issue 160761 (Gemma3 ONNX Export Fails due to vmap).
    
    This test replaces the PyTorch `torch.vmap` context issue with the TensorFlow
    `tf.compat.v1.executing_eagerly_outside_functions` API to demonstrate handling
    execution modes during model export/tracing.
    
    Original Issue: torch.onnx.export fails because of vmap usage in masking_utils.
    Adaptation: Verify that `executing_eagerly_outside_functions` allows a model to
    correctly identify its context during a trace (export), ensuring logic branches
    are taken correctly.
    """
    
    # 1. Setup Model (Mimicking AutoModelForCausalLM)
    # We define a custom model that uses the similar API
    class ContextSensitiveModel(tf.Module):
        def __init__(self):
            super().__init__()
            self.dense_layer = tf.keras.layers.Dense(10)

        @tf.function
        def __call__(self, input_ids, attention_mask):
            # This is the usage of the similar API: tf.compat.v1.executing_eagerly_outside_functions
            # It checks if the outermost context is eager, even if we are inside a tf.function (graph).
            # This relates to the original bug where vmap (a functional transformation) caused issues
            # during export because the context/execution mode was not compatible.
            is_outer_eager = tf.compat.v1.executing_eagerly_outside_functions()

            # Logic mimicking the issue: 
            # In the original bug, vmap is used internally and causes ONNX export to fail.
            # Here, we check the context. If we are in eager mode (outer), we might perform
            # operations that are safe in eager but tricky in graph (or vice versa).
            # For this test, we assert that we can detect the context correctly during tracing.
            
            if is_outer_eager:
                # Path taken when running in eager mode (even if tracing)
                # Simulating a vmap-like operation or eager-specific logic
                x = self.dense_layer(input_ids)
                return x * attention_mask
            else:
                # Path taken if running in a pure graph/session context (e.g. TF1 session)
                x = self.dense_layer(input_ids)
                return x + attention_mask

    # 2. Prepare Inputs (Mimicking tokenizer.apply_chat_template)
    batch_size = 1
    seq_length = 10
    input_ids = tf.random.uniform((batch_size, seq_length), dtype=tf.float32)
    attention_mask = tf.ones((batch_size, seq_length), dtype=tf.float32)

    model = ContextSensitiveModel()

    # 3. "Export" / Trace the model
    # In TensorFlow, the equivalent of torch.onnx.export is often tf.saved_model.save 
    # or obtaining a concrete function to trace the graph.
    
    print("Attempting to trace/export model...")
    
    # We simulate the export process by getting the concrete function.
    # This triggers the tracing similar to torch.onnx.export.
    try:
        concrete_fn = model.__call__.get_concrete_function(input_ids, attention_mask)
        print("Model traced successfully.")
    except Exception as e:
        print(f"Tracing failed: {e}")
        raise

    # 4. Verify Execution
    # Since this script runs in eager mode (TF2 default), 
    # executing_eagerly_outside_functions should return True inside the function.
    # Therefore, the result should follow the 'if is_outer_eager' branch (multiplication).
    
    output = concrete_fn(input_ids, attention_mask)
    
    # Calculate expected output based on the eager branch logic
    expected = model.dense_layer(input_ids) * attention_mask
    
    # Assert that the traced function produced the result corresponding to the eager context
    np.testing.assert_allclose(output.numpy(), expected.numpy(), rtol=1e-5)
    
    print("Test passed: Context check correctly identified outer eager mode during export.")

if __name__ == "__main__":
    test_gemma3_like_export_with_context_check()
