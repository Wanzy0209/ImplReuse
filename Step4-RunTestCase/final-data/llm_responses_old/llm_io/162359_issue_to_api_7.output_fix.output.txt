import tensorflow as tf

# Ensure eager execution is enabled so that .numpy() and immediate evaluation work correctly.
# This is necessary because the test logic relies on eager tensor behaviors.
if not tf.executing_eagerly():
    tf.compat.v1.enable_eager_execution()

def test_required_space_to_batch_paddings_aliasing():
    """
    Test case adapted from PyTorch Issue #162359.
    
    The original issue describes a bug where SequentialLR aliases the optimizer's 
    lr Tensor with initial_lr. When the first scheduler updates the lr in-place,
    it corrupts the initial_lr and base_lrs of subsequent schedulers.
    
    This test checks if tf.required_space_to_batch_paddings exhibits similar 
    aliasing behavior where the returned 'paddings' tensor might share memory 
    with the input 'base_paddings', potentially leading to unintended corruption 
    if the output is modified in-place (though TF tensors are mostly immutable, 
    we check for object identity and reference sharing).
    """
    
    # 1. Setup inputs
    # Using a Variable to mimic the mutable state of an optimizer's param group
    input_shape = tf.constant([10, 20])
    block_shape = tf.constant([2, 5])
    
    # base_paddings corresponds to 'initial_lr' in the PyTorch issue
    base_paddings = tf.Variable([[1, 1], [2, 2]], dtype=tf.int32)
    base_paddings_id = id(base_paddings)
    base_paddings_val = base_paddings.read_value()

    # 2. Call the API
    # This corresponds to initializing the SequentialLR
    paddings, crops = tf.required_space_to_batch_paddings(
        input_shape, 
        block_shape, 
        base_paddings=base_paddings
    )

    # 3. Check for aliasing
    # In the PyTorch bug, the scheduler's base_lrs were aliased with the optimizer's initial_lr.
    # We check if the returned 'paddings' is the same object as 'base_paddings'.
    # Note: In TF, paddings is usually a new Tensor, but we verify it is not the Variable itself.
    assert paddings is not base_paddings, \
        "Returned 'paddings' should not be the same object as input 'base_paddings'."

    # 4. Check for state corruption
    # In the PyTorch bug, the 'initial_lr' value was mutated by the scheduler's step.
    # We verify that calling the function did not mutate the input 'base_paddings'.
    assert tf.reduce_all(base_paddings.read_value() == base_paddings_val).numpy(), \
        "Input 'base_paddings' should not be mutated by the function call."

    # 5. Check chained usage (SequentialLR aspect)
    # If we were to use the output of one call as the input to another (chaining),
    # we want to ensure that modifying the intermediate result (if possible via Variable)
    # doesn't corrupt the original source.
    
    # Simulate a chain: Pass the result as base_paddings to another call
    # (Conceptually similar to LinearLR -> CosineAnnealingLR)
    # Note: paddings is a Tensor, so we use it directly.
    paddings_2, _ = tf.required_space_to_batch_paddings(
        input_shape,
        block_shape,
        base_paddings=paddings
    )
    
    # Verify that paddings_2 is distinct from paddings
    assert paddings_2 is not paddings, \
        "Chained calls should produce distinct tensor objects."

    print("Test passed: No aliasing or unintended mutation detected.")

if __name__ == "__main__":
    test_required_space_to_batch_paddings_aliasing()
