def test_swapaxes_deterministic():
    """
    Adapted test for tf.keras.ops.swapaxes based on the structure of the 
    torch.scatter determinism test.
    
    Verifies that tf.keras.ops.swapaxes produces deterministic results 
    and gradients over multiple iterations.
    """
    # Setup input tensor similar to the original test's 'inputs'
    # Original shape: [2, 3, 4]
    inputs = tf.range(24, dtype=tf.float32)
    inputs = tf.reshape(inputs, [2, 3, 4])

    # Define axes to swap (e.g., axis 0 and 1)
    axis1, axis2 = 0, 1

    # Calculate ground truth result using numpy
    np_inputs = np.arange(24, dtype=np.float32).reshape(2, 3, 4)
    gt_res = np.swapaxes(np_inputs, axis1, axis2)

    # Calculate ground truth gradient
    # For swapaxes, if upstream gradient is ones, the input gradient is ones.
    gt_input_grad = np.ones((2, 3, 4), dtype=np.float32)

    for i in range(1000):
        # Use GradientTape to record operations for automatic differentiation
        with tf.GradientTape() as tape:
            tape.watch(inputs)
            # Perform the swapaxes operation
            res = tf.keras.ops.swapaxes(inputs, axis1, axis2)

        # Calculate gradients
        # Simulates res.backward(torch.ones_like(res))
        grads = tape.gradient(res, inputs, tf.ones_like(res))

        print(f"Test {i + 1}/{1000}")
        
        # Assert results match ground truth (Determinism check)
        np.testing.assert_allclose(res.numpy(), gt_res)
        
        # Assert gradients match ground truth (Determinism check)
        np.testing.assert_allclose(grads.numpy(), gt_input_grad)


if __name__ == "__main__":
