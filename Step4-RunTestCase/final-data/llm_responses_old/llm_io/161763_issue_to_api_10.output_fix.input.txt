import torch

def test_uint_neg_add_inductor():
    """
    Test case for Issue #161763: neg+add computation including uint tensor is incorrect under inductor.
    
    The bug manifests when torch.compile (using the inductor backend) processes a negation of a uint8 
    tensor followed by an addition with a float tensor. The compiler incorrectly promotes the uint8 
    to float before negation, resulting in -7.0 instead of the wrapped value 249.
    """
    def func(x):
        c = torch.tensor(7, dtype=torch.uint8)
        return c + x, torch.neg(c), torch.neg(c) + x

    torch.manual_seed(0)
    x = torch.randn(2, 2, dtype=torch.float32)

    # Eager execution
    res_eager = func(x)

    # Compiled execution
    compiled_func = torch.compile(func)
    res_compiled = compiled_func(x)

    # Assert all outputs match between eager and compiled modes
    # res[0]: c + x (uint + float)
    torch.testing.assert_close(res_eager[0], res_compiled[0])
    
    # res[1]: torch.neg(c) (uint negation)
    torch.testing.assert_close(res_eager[1], res_compiled[1])
    
    # res[2]: torch.neg(c) + x (uint negation + float)
    # This is the specific case that failed in the bug report.
    # Expected behavior: neg(uint8(7)) wraps to 249, then adds x.
    # Buggy behavior: neg(uint8(7)) becomes -7.0 (float), then adds x.
    torch.testing.assert_close(res_eager[2], res_compiled[2])

if __name__ == "__main__":
    test_uint_neg_add_inductor()
    print("Test passed successfully.")
