import tensorflow as tf
import tf.experimental.numpy as tnp
import numpy as np

# Note: The prompt metadata mentions tf.experimental.numpy.ix_, but the bug report 
# is specifically about einsum behavior. ix_ is semantically different (constructs open meshes).
# Therefore, this test case uses tf.experimental.numpy.einsum to verify the 
# cross-library behavior for the reported issue.

def test_einsum_contiguity():
    # Setup dimensions matching the bug report
    batch, in_dim, out_dim = 128, 1024, 4096
    
    # Create inputs using tf.experimental.numpy to mimic the numpy interface
    # We use float32 to match the original torch setup (default float is usually float32 in torch)
    x = tnp.random.randn(batch, in_dim).astype(np.float32)
    w = tnp.random.randn(out_dim, in_dim).astype(np.float32)

    # Perform the einsum operation
    # Original bug: torch.einsum("fd,bd->bf", w, x)
    out_tf = tnp.einsum("fd,bd->bf", w, x)

    # Perform the same operation with standard numpy for comparison
    out_np = np.einsum("fd,bd->bf", w.numpy(), x.numpy())

    print("TensorFlow (experimental.numpy) Output:")
    print(f"Shape: {out_tf.shape}")
    print(f"Strides: {out_tf.strides}")

    print("\nNumPy Output:")
    print(f"Shape: {out_np.shape}")
    print(f"Strides: {out_np.strides}")

    # Assertions
    # 1. Shape must match
    assert out_tf.shape == out_np.shape, f"Shape mismatch: TF {out_tf.shape} vs NumPy {out_np.shape}"
    
    # 2. Strides must match (The bug report highlights PyTorch returning transposed strides (1, 128) 
    # instead of contiguous strides (16384, 4) for float32).
    # We expect TensorFlow to match NumPy's contiguous behavior.
    assert out_tf.strides == out_np.strides, f"Strides mismatch: TF {out_tf.strides} vs NumPy {out_np.strides}"
    
    print("\nTest passed: TensorFlow einsum matches NumPy behavior (contiguous output).")

if __name__ == "__main__":
    test_einsum_contiguity()
