```python
import torch
from collections import namedtuple
import sys

# Handle environment dependency issues (GLIBCXX version mismatch)
try:
    import tensorflow as tf
except ImportError as e:
    print(f"Skipping test due to missing dependencies or environment issues: {e}")
    sys.exit(0)

def test_namedtuple_clone_model():
    """
    Adapts the PyTorch torch.export.export NamedTuple bug reproduction 
    to tf.keras.models.clone_model.
    
    The original bug involves passing a NamedTuple containing tensors 
    directly to the export function. Here, we test passing a NamedTuple 
    to the input_tensors argument of clone_model.
    """
    # Define a NamedTuple similar to the PyTorch example
    Point = namedtuple('Point', 'x y')

    # Create a simple Keras model that takes two inputs and adds them
    # This mimics the PyTorch Module M
    input_x = tf.keras.Input(shape=(3,), name='x')
    input_y = tf.keras.Input(shape=(3,), name='y')
    output = tf.keras.layers.Add()([input_x, input_y])
    model = tf.keras.Model(inputs=[input_x, input_y], outputs=output)

    # Create input data using the NamedTuple
    inp = Point(tf.ones((3,)), tf.ones((3,)))

    # Verify the original model works with unpacked inputs
    # PyTorch equivalent: M()(*inp)
    print("Original model output:", model([inp.x, inp.y]))

    # Attempt to clone the model passing the NamedTuple directly as input_tensors
    # PyTorch equivalent: torch.export.export(M(), inp, strict=False) -> errors
    print("\nAttempting clone_model with NamedTuple input...")
    try:
        cloned_model = tf.keras.models.clone_model(model, input_tensors=inp)
        print("Clone with NamedTuple succeeded (unexpected if similar to PyTorch bug)")
        print(cloned_model([inp.x, inp.y]))
    except Exception as e:
        print(f"Clone with NamedTuple failed: {type(e).__name__}: {e}")

    # Attempt the workaround: converting NamedTuple to a list (standard Keras format)
    # PyTorch equivalent: converting to kwargs
    print("\nAttempting clone_model with list input (workaround)...")
    try:
        inp_list = [getattr(inp, field) for field in inp._fields]
        cloned_model_workaround = tf.keras.models.clone_model(model, input_tensors=inp_list)
        print("Clone with list succeeded")
        print(cloned_model_workaround(inp_list))
    except Exception as e:
        print(f"Clone with list failed: {e}")

if __name__ == "__main__":
    test_namedtuple_clone_model()
```
