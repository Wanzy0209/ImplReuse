import sys
import torch

# Handle environment dependency issues (e.g., missing GLIBCXX) gracefully
try:
    import tensorflow as tf
except ImportError as e:
    print(f"[SKIP] Cannot import TensorFlow due to environment/dependency issues: {e}")
    sys.exit(0)

def test_nccl_all_reduce_huge_packs():
    """
    Test case for tf.raw_ops.NcclAllReduce (via tf.distribute.NcclAllReduce)
    inspired by PyTorch Issue 160868.
    
    The original bug involves passing a huge integer (2**63 - 1) as a 'step'
    to torch.slice_copy, causing a segfault in the Inductor backend due to
    integer overflow or invalid loop bounds.
    
    This test checks if tf.distribute.NcclAllReduce handles a similarly huge
    integer for its 'num_packs' parameter, which controls tensor partitioning.
    """
    
    # Check for GPU availability, as NCCL requires NVIDIA GPUs
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("[SKIP] Test requires GPUs to run NcclAllReduce.")
        return

    # Mimic the "huge step" parameter from the PyTorch bug report
    huge_step = 2**63 - 1

    try:
        # Initialize the strategy with the extreme parameter.
        # In the PyTorch bug, the crash occurred during the execution/compilation
        # triggered by the parameter. Here we check initialization and usage.
        print(f"[TEST] Initializing NcclAllReduce with num_packs={huge_step}...")
        
        strategy = tf.distribute.MirroredStrategy(
            cross_device_ops=tf.distribute.NcclAllReduce(num_packs=huge_step)
        )

        with strategy.scope():
            # Create a minimal input tensor
            x = tf.constant([1.0, 2.0, 3.0])

            @tf.function
            def reduce_fn(x):
                # Perform a simple all-reduce operation
                return tf.reduce_sum(x)

            print("[TEST] Running reduction...")
            # If the bug exists (similar to PyTorch Inductor), this might segfault
            # or raise an error due to integer overflow in the packing logic.
            result = strategy.run(reduce_fn, args=(x,))
            
            print(f"[PASS] Operation completed without segfault. Result: {result}")

    except ValueError as e:
        # TensorFlow might handle this gracefully by raising a ValueError
        # (e.g., "num_packs must be non-negative"). This is acceptable behavior
        # (better than a segfault).
        print(f"[INFO] Caught ValueError (graceful handling): {e}")
    except Exception as e:
        # Catch other unexpected errors
        print(f"[FAIL] Unexpected exception: {type(e).__name__}: {e}")

if __name__ == "__main__":
    test_nccl_all_reduce_huge_packs()
