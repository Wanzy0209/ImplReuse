import tensorflow as tf
import numpy as np

def test_linear_operator_block_diag_complex_integrity():
    """
    Test case adapted from PyTorch Issue 160845.
    Verifies that tf.linalg.LinearOperatorBlockDiag correctly preserves
    the imaginary components of complex tensors, specifically when the
    real part is zero.
    """
    # Setup: Create complex tensors with zero real part and random imaginary part
    # This mirrors the 'src' tensor creation in the original bug report.
    dtype = tf.complex64
    
    # Block 1
    shape_a = (2, 2)
    imag_a = tf.random.normal(shape_a, dtype=tf.float32)
    real_a = tf.zeros_like(imag_a)
    tensor_a = tf.complex(real_a, imag_a)

    # Block 2
    shape_b = (3, 3)
    imag_b = tf.random.normal(shape_b, dtype=tf.float32)
    real_b = tf.zeros_like(imag_b)
    tensor_b = tf.complex(real_b, imag_b)

    # Operation: Construct the Block Diagonal Operator
    # Fix: LinearOperatorBlockDiag expects a list of LinearOperator objects, not Tensors.
    # We wrap the tensors in LinearOperatorFullMatrix first.
    operator_a = tf.linalg.LinearOperatorFullMatrix(tensor_a, is_non_singular=True)
    operator_b = tf.linalg.LinearOperatorFullMatrix(tensor_b, is_non_singular=True)
    op = tf.linalg.LinearOperatorBlockDiag([operator_a, operator_b], is_non_singular=True)
    result = op.to_dense()

    # Ground Truth: Manually construct the block diagonal matrix
    # This serves as the reference (like 't_cpu' in the original bug).
    expected = tf.linalg.block_diag(tensor_a, tensor_b)

    # Verification: Check imaginary parts and max difference
    # Mirrors the print statements in the original bug.
    result_imag_sum = tf.reduce_sum(tf.abs(tf.math.imag(result))).numpy()
    expected_imag_sum = tf.reduce_sum(tf.abs(tf.math.imag(expected))).numpy()
    max_abs_diff = tf.reduce_max(tf.abs(result - expected)).numpy()

    print(f"Result imag sum: {result_imag_sum}")
    print(f"Expected imag sum: {expected_imag_sum}")
    print(f"Max abs diff: {max_abs_diff}")

    # Assertions
    # 1. Ensure imaginary part is not lost (was 0.0 in the MPS bug)
    assert result_imag_sum > 1e-5, "Imaginary part was lost (sum is near zero)."
    # 2. Ensure exact match with ground truth
    assert max_abs_diff < 1e-5, f"Result differs from expected by {max_abs_diff}"

if __name__ == "__main__":
    test_linear_operator_block_diag_complex_integrity()
