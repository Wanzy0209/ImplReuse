import torch
import tensorflow as tf
import sys

def test_large_tensor_initialization():
    """
    Test case to verify large tensor (>4GB) initialization.
    This mirrors the PyTorch MPS bug (Issue 161265) where torch.ones 
    failed to correctly fill tensors larger than 4GB.
    
    The similar API context points to TensorFlow user_ops, but semantically
    we map torch.full/torch.ones to tf.ones/tf.fill to test the specific
    buffer filling logic described in the bug.
    """
    
    # Shape: 2 rows, (2^31 + 5) columns.
    # Total elements: 2 * (2^31 + 5) > 2^32.
    # With int8 (1 byte), size > 4GB.
    shape = [2, (1 << 31) + 5]
    dtype = tf.int8
    
    try:
        # Attempt to place on GPU if available to mimic the 'mps' device context.
        # Note: TensorFlow on macOS uses the Metal plugin for GPU acceleration.
        device_name = '/GPU:0' if tf.config.list_physical_devices('GPU') else '/CPU:0'
        
        with tf.device(device_name):
            # Create a tensor filled with ones (equivalent to torch.ones)
            a = tf.ones(shape, dtype=dtype)
        
        # Verify specific elements as in the original bug report
        # Original: a[1, -2] should be 1
        val = a[1, -2]
        assert val == 1, f"Expected a[1, -2] to be 1, got {val}"
        
        # Original: a[:, -2] should be [1, 1]
        slice_val = a[:, -2]
        # tf.reduce_all checks if all elements in the tensor are True
        assert tf.reduce_all(slice_val == 1).numpy(), \
            f"Expected a[:, -2] to be [1, 1], got {slice_val}"
        
        print("Test Passed: Large tensor filled correctly.")
        
    except tf.errors.ResourceExhaustedError:
        print("Test Skipped: Not enough memory to allocate 4GB+ tensor.")
    except Exception as e:
        print(f"Test Failed: {e}")
        raise

if __name__ == "__main__":
    test_large_tensor_initialization()
