```python
import torch
import tensorflow as tf
import numpy as np

# Disable eager execution to use the graph-based execution model,
# which is required for tf.compat.v1.train.add_queue_runner.
tf.compat.v1.disable_eager_execution()

def addcmul_func(x, y, z):
    """Equivalent function to the PyTorch addcmul_func."""
    return x + (y * z)

# Define a FIFOQueue to mimic the input tensors.
# In TensorFlow graph mode, queues are often used to pipeline data.
# We use shape [128] to match the PyTorch example.
queue = tf.compat.v1.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[[128]])

# Generate random data to mimic torch.randn(128).to("xpu")
# Note: TensorFlow does not natively support "xpu" device placement in standard builds,
# so we use the default device (CPU) to ensure the test case is runnable.
random_data = tf.random.normal([128])

# Enqueue the data
enqueue_op = queue.enqueue_many([random_data])

# Dequeue three tensors to serve as x, y, z
x = queue.dequeue()
y = queue.dequeue()
z = queue.dequeue()

# Define the operation
out = addcmul_func(x, y, z)

# Create a QueueRunner to manage the queue operations in background threads
qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op] * 2)

# --- TARGET API USAGE ---
# This corresponds to the setup phase in PyTorch (torch.compile).
# We add the queue runner to the graph collection.
tf.compat.v1.train.add_queue_runner(qr)
# ------------------------

with tf.compat.v1.Session() as sess:
    # Initialize variables
    sess.run(tf.compat.v1.global_variables_initializer())
    sess.run(tf.compat.v1.local_variables_initializer())

    # Start the queue runners
    coord = tf.compat.v1.train.Coordinator()
    threads = tf.compat.v1.train.start_queue_runners(sess=sess, coord=coord)

    try:
        # Run the operation
        result = sess.run(out)
        
        # Verify the result shape and type
        assert result.shape == (128,), f"Expected shape (128,), got {result.shape}"
        assert result.dtype == np.float32, f"Expected dtype float32, got {result.dtype}"
        
        print("Graph mode passed")
        print("tf.compat.v1.train.add_queue_runner passed")
        
    except Exception as e:
        print(f"Test failed with error: {e}")
        raise
    finally:
        # Stop the threads
        coord.request_stop()
        coord.join(threads)
```
