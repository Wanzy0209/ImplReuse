import torch
import tensorflow as tf

def test_large_tensor_fill_with_epsilon():
    """
    Test case adapted from Issue 161265 (torch.full fails for 4+Gb tensors).
    This test verifies that large tensors (>4GB) can be created and filled correctly
    using TensorFlow, leveraging tf.keras.backend.epsilon as the fill value.
    """
    # Retrieve the value from the similar API
    fill_value = tf.keras.backend.epsilon()

    # Reproduce the logic: Create a tensor larger than 4GB
    # Shape [2, (1 << 31) + 5] results in > 4GB of elements
    # We use float32 to match the type of epsilon
    shape = [2, (1 << 31) + 5]

    try:
        # Create the tensor using tf.fill (equivalent to torch.full)
        # This tests the backend's ability to handle large buffer allocation and filling
        large_tensor = tf.fill(shape, fill_value)

        # Check the tail element (beyond the 4GB boundary)
        # Corresponds to a[1, -2] in the original bug report
        tail_val = large_tensor[1, -2]

        # Assert that the tail value matches the expected fill value
        # This verifies that the fill operation worked correctly for the entire buffer
        assert tail_val.numpy() == fill_value.numpy(), \
            f"Large tensor fill failed: expected {fill_value.numpy()}, got {tail_val.numpy()}"

        print("Test Passed: Large tensor filled correctly with epsilon.")

    except tf.errors.ResourceExhaustedError:
        print("Test Skipped: Insufficient memory to allocate 4GB+ tensor.")
    except Exception as e:
        print(f"Test Failed: {e}")

if __name__ == "__main__":
