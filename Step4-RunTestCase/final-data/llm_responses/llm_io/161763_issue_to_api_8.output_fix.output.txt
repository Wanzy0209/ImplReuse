import torch
import torch.nn as nn

# Fix: Handle environments where torch.compile is not available (PyTorch < 2.0)
if not hasattr(torch, 'compile'):
    # Define a mock torch.compile that acts as an identity function.
    # This allows the test to run without crashing, though it effectively
    # tests eager mode against eager mode in older environments.
    torch.compile = lambda model, *args, **kwargs: model

# Adapting the Wrapper pattern from the similar API (tf.keras.layers.Wrapper)
# to structure the test case for torch.compile.
class FunctionWrapper(nn.Module):
    """
    A wrapper module that delegates the forward pass to an inner function.
    This mimics the structure of tf.keras.layers.Wrapper which wraps an inner layer.
    """
    def __init__(self, inner_func):
        super().__init__()
        self.inner_func = inner_func

    def forward(self, x):
        return self.inner_func(x)

def foo(x):
    # The original bug reproduction logic
    c = torch.tensor(7, dtype=torch.uint8)
    return c+x, torch.neg(c), torch.neg(c)+x

def test_uint_neg_add_compile():
    torch.manual_seed(0)
    x = torch.randn(2, 2, dtype=torch.float32)
    
    # Instantiate the wrapper with the logic function
    model = FunctionWrapper(foo)
    
    # Compile the model using the API under test
    compiled_model = torch.compile(model)
    
    # Get results from eager execution
    res_eager = model(x)
    
    # Get results from compiled execution
    res_compiled = compiled_model(x)
    
    # Assertions to check for correctness
    # res[0]: c + x
    assert torch.allclose(res_eager[0], res_compiled[0]), \
        f"Mismatch in c+x:\nEager: {res_eager[0]}\nCompiled: {res_compiled[0]}"
        
    # res[1]: torch.neg(c)
    assert torch.equal(res_eager[1], res_compiled[1]), \
        f"Mismatch in neg(c):\nEager: {res_eager[1]}\nCompiled: {res_compiled[1]}"
        
    # res[2]: torch.neg(c) + x
    # This is the specific case failing in the bug report (Issue 161763)
    assert torch.allclose(res_eager[2], res_compiled[2]), \
        f"Mismatch in neg(c)+x:\nEager: {res_eager[2]}\nCompiled: {res_compiled[2]}"

if __name__ == "__main__":
    test_uint_neg_add_compile()
