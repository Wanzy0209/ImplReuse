import sys

# Handle environment dependency issues (e.g., missing libstdc++ version) gracefully

    import torch

# We leverage the similar API (tf.custom_gradient) to define a custom operation
# that mimics the behavior of the original API (torch.index_select).
@tf.custom_gradient
def custom_index_select_like(x, index):
    """
    A custom gradient function that performs a selection operation similar to index_select.
    This tests if the custom gradient mechanism handles scalar index tensors correctly.
    """
    # Forward pass: Use tf.gather to select indices along axis 0.
    # Note: In PyTorch, index_select with a scalar index keeps the dimension (e.g., [1, 3]).
    # In TensorFlow, tf.gather with a scalar index reduces the rank (e.g., [3]).
    # We use tf.expand_dims on the index to mimic PyTorch's behavior of keeping the dimension.
    y = tf.gather(x, tf.expand_dims(index, 0), axis=0)

    # Define the gradient function
    def grad(dy):
        # For index_select, gradients are typically scattered back.
        # Here we return a zero gradient for x and None for index to keep it minimal.
        return tf.zeros_like(x), None

    return y, grad

def test_custom_index_select_with_scalar():
    # Reproduce the original bug logic: using a scalar (0-dimensional) tensor as index
    x = tf.ones([2, 3])
    index = tf.constant(1)  # zero-dimensional index tensor

    try:
        # Call the operation wrapped with the similar API
        output = custom_index_select_like(x, index)
        # Verify the output shape matches the expected behavior (keeping the dimension)
        assert output.shape == (1, 3), f"Expected shape (1, 3), got {output.shape}"
        print(f"Test succeeds. Output shape: {output.shape}")
    except Exception as e:
        print(f"Test fails: {e}")

if __name__ == "__main__":
    test_custom_index_select_with_scalar()
