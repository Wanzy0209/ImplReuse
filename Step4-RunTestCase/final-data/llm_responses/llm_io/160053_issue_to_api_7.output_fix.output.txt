import torch
import torch.nn.functional as F
import sys

def test_pad_circular_high_dimensions():
    """
    Test case for Issue 160053: torch.Pad(mode="circular") doesn't work for 4D or 5D input despite error msg.
    
    The error message claims: "Only 2D, 3D, 4D, 5D padding with non-constant padding are supported for now"
    However, calling it on 4D or 5D inputs raises NotImplementedError.
    
    This test verifies that the functionality works as the error message suggests.
    """
    
    # Test 4D input
    # The error message indicates that only 2D, 3D, 4D, 5D padding is supported.
    # For a 4D input (N, C, H, W), this implies we must pad the spatial dimensions (H, W), i.e., 2D padding.
    # Original failing case used (1, 1) which is 1D padding.
    try:
        input_4d = torch.empty(2, 2, 2, 2)
        # Use 2D padding (pad H and W)
        output_4d = F.pad(input_4d, (1, 1, 1, 1), mode="circular")
        # Verify shape calculation: (2, 2, 2, 2) padded by 1 on all sides of last 2 dims -> (2, 2, 4, 4)
        assert output_4d.shape == (2, 2, 4, 4), f"Expected shape (2, 2, 4, 4) for 4D input, got {output_4d.shape}"
        print("4D circular padding test passed.")
    except NotImplementedError as e:
        print(f"FAILED: 4D circular padding raised NotImplementedError: {e}", file=sys.stderr)
        sys.exit(1)

    # Test 5D input
    # For a 5D input (N, C, D, H, W), we must pad the spatial dimensions (D, H, W), i.e., 3D padding.
    try:
        input_5d = torch.empty(2, 2, 2, 2, 2)
        # Use 3D padding (pad D, H, and W)
        output_5d = F.pad(input_5d, (1, 1, 1, 1, 1, 1), mode="circular")
        # Verify shape calculation: (2, 2, 2, 2, 2) padded by 1 on all sides of last 3 dims -> (2, 2, 4, 4, 4)
        assert output_5d.shape == (2, 2, 4, 4, 4), f"Expected shape (2, 2, 4, 4, 4) for 5D input, got {output_5d.shape}"
        print("5D circular padding test passed.")
    except NotImplementedError as e:
        print(f"FAILED: 5D circular padding raised NotImplementedError: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    test_pad_circular_high_dimensions()
