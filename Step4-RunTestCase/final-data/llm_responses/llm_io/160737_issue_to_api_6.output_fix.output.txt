import sys

try:
    import tensorflow as tf
except ImportError as e:
    print(f"Skipping test: Failed to import TensorFlow due to environment issues (missing GLIBCXX_3.4.29). Error: {e}")
    sys.exit(0)

import torch

def test_exp_scalar_input():
    """
    Test case adapted from the logic of Issue 160737.
    The original issue highlights a failure in torch.index_select when provided
    with a scalar (0-dimensional) tensor as an index on the MPS backend.
    
    This test verifies that the similar API, tf.keras.ops.exp, correctly
    handles a scalar tensor input without raising dimension-related errors.
    """
    # Create a scalar (zero-dimensional) tensor
    x = tf.constant(2.0)
    
    try:
        # Call the similar API
        output = tf.keras.ops.exp(x)
        
        # Verify the output shape is scalar and value is correct
        assert output.shape == (), f"Expected scalar shape (), got {output.shape}"
        expected_val = 7.38905609893065
        assert abs(output.numpy() - expected_val) < 1e-6, f"Expected {expected_val}, got {output.numpy()}"
        
        print(f"exp test succeeds for scalar input. output shape: {output.shape}")
    except Exception as e:
        print(f"exp test fails for scalar input: {e}")

if __name__ == "__main__":
    test_exp_scalar_input()
