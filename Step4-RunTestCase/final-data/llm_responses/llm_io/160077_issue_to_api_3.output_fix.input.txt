import torch
import unittest

class TestTorchCompileSplitContext(unittest.TestCase):
    """
    Regression test for Issue 160077.
    Verifies that tensor.split works correctly with torch.compile
    inside a torch.device context.
    """
    
    def test_split_with_compile_in_device_context(self):
        # Define the function using the tensor method split
        def f(xs):
            return xs.split(1, dim=0)

        # Use 'cpu' to ensure the test runs on all environments.
        # The original bug was reported on 'cuda', but the issue stems from
        # the interaction between the device context manager and torch.compile.
        device_type = "cpu"

        with torch.device(device_type):
            xs = torch.randn(2, 2, device=device_type)

            # 1. Verify Eager Mode works
            eager_result = f(xs)
            self.assertEqual(len(eager_result), 2)
            self.assertEqual(eager_result[0].shape, (1, 2))

            # 2. Verify Compiled Mode works
            # The bug manifested here as: `module 'torch._tensor' has no attribute 'split'`
            # This test ensures the attribute is found and the operation succeeds.
            compiled_f = torch.compile(f)
            compiled_result = compiled_f(xs)

            # 3. Verify Correctness
            self.assertEqual(len(compiled_result), 2)
            for e, c in zip(eager_result, compiled_result):
                self.assertTrue(torch.allclose(e, c))

if __name__ == "__main__":
