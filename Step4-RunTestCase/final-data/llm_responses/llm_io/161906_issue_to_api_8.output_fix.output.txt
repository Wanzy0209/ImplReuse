import torch
import numpy as np
import sys

# Handle environment dependency issues gracefully
try:
    import tensorflow as tf
except ImportError as e:
    print("Error: Failed to import TensorFlow.")
    print(f"Details: {e}")
    print("This test requires a compatible TensorFlow environment (GLIBCXX_3.4.29 or higher).")
    print("Skipping test execution.")
    sys.exit(0)

class ComplexRecombinationModel(tf.keras.Model):
    """
    TensorFlow equivalent of the PyTorch model in the bug report.
    Replaces torch.complex with tf.complex and leverages tf.keras.layers.add
    for the parameter interaction step.
    """
    def __init__(self, n_fft=512, hop=160, win=320):
        super().__init__()
        self.n_fft = n_fft
        self.hop = hop
        self.win = win
        
        # Register window as a variable/buffer equivalent
        # In TF, we can just store the tensor or use a Variable
        self.window = tf.signal.hann_window(win)
        
        # Introduce parameter for broadcasting / stride ops
        # Using a Variable to mimic nn.Parameter
        self.p = tf.Variable(2.0, trainable=True, dtype=tf.float32)
        
        # Leverage the similar API: tf.keras.layers.add
        # We will use this to combine the imaginary part with the parameter
        self.add_layer = tf.keras.layers.Add()

    def call(self, x: tf.Tensor) -> tf.Tensor:
        # Perform STFT
        # Note: tf.signal.stft returns a complex64 tensor
        S = tf.signal.stft(
            x, 
            frame_length=self.n_fft, 
            frame_step=self.hop, 
            fft_length=self.n_fft, 
            window_fn=lambda _: self.window,
            pad_end=True
        )
        
        # Extract real and imaginary parts
        # R = torch.abs(S.real)
        R = tf.abs(tf.math.real(S))
        
        # I = S.imag / self.p
        # Adaptation: Use tf.keras.layers.add to add the parameter (broadcasting)
        # instead of division, to leverage the similar API.
        # We ensure p is cast to float32 to match S.imag
        I = tf.math.imag(S)
        I_modified = self.add_layer([I, self.p])
        
        # Z = torch.complex(R, I)
        # Recombine into complex tensor
        Z = tf.complex(R, I_modified)
        
        return Z

def test_tf_complex_recombination():
    # Setup
    # Use CPU to ensure consistency if CUDA is not available/configured in TF
    with tf.device('/CPU:0'):
        tf.random.set_seed(0)
        x = tf.random.normal((1, 16000))
        
        model = ComplexRecombinationModel()

        # 1. Eager execution
        print("Testing Eager Mode...")
        z_eager = model(x)
        assert z_eager.dtype == tf.complex64, f"Expected complex64, got {z_eager.dtype}"
        print(f"Eager mode OK: Shape={z_eager.shape}, Dtype={z_eager.dtype}")

        # 2. Compiled execution (tf.function mimics torch.compile)
        # This tests if the graph mode (XLA/AutoGraph) handles the strides/shapes correctly
        print("\nTesting Compiled Mode (tf.function)...")
        compiled_model = tf.function(model)
        z_compiled = compiled_model(x)
        
        assert z_compiled.dtype == tf.complex64, f"Expected complex64, got {z_compiled.dtype}"
        print(f"Compiled mode OK: Shape={z_compiled.shape}, Dtype={z_compiled.dtype}")

        # 3. Consistency Check
        # Ensure compiled and eager produce the same results
        assert np.allclose(z_eager.numpy(), z_compiled.numpy()), "Mismatch between eager and compiled output"
        print("Consistency Check Passed: Eager and Compiled outputs match.")

if __name__ == "__main__":
    test_tf_complex_recombination()
