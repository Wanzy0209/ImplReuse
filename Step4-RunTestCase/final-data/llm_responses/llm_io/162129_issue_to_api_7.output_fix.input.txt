class SimpleTensorMode(TorchDispatchMode):
    """
    A dispatch mode that records the function calls made during its context.
    Used to verify if c10d ops are being dispatched correctly.
    """
    def __init__(self):
        self.dispatched_ops = []

    def __torch_dispatch__(self, func, types, args=(), kwargs=None):
        if kwargs is None:
            kwargs = {}
        self.dispatched_ops.append(func)
        return func(*args, **kwargs)


def test_fake_process_group_dispatch():

        dist.all_reduce(tensor, group=fake_pg)

    # Check dispatched operations
    dispatched_funcs = [str(op) for op in mode.dispatched_ops]
