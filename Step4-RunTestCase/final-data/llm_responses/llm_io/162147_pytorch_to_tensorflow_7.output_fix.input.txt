class TFFastLearnedCellX3(tf.keras.layers.Layer):
    """
    TensorFlow adaptation of the FastLearnedCellX3 module.
    Replaces torch.index_select logic with tf.gather and integrates
    tf.keras.backend.dropout as requested.
    """
    def __init__(self, D_in, H, D_out,
                 L_w1=12, L_w2=12, L_b2=12,
                 k1=3, k2=3, k3=3, tau1=1.0, tau2=1.0, tau3=1.0,
                 d_addr=64,
                 learn_addr=False,
                 learn_tape_w1=True, learn_tape_w2=True, learn_tape_b2=True,
                 rate=0.1, # Dropout rate
                 **kwargs):
        super().__init__(**kwargs)
        self.D_in, self.H, self.D_out = D_in, H, D_out
        self.L_w1, self.L_w2, self.L_b2 = int(L_w1), int(L_w2), int(L_b2)
        self.k1, self.k2, self.k3 = int(k1), int(k2), int(k3)
        self.t1, self.t2, self.t3 = float(tau1), float(tau2), float(tau3)
        self.rate = rate

        # Address bottleneck P
        self.P = self.add_weight(name='P', shape=(D_in, d_addr), 
                                 initializer='random_normal', trainable=learn_addr)

        # Unembeddings U1, U2, U3
        self.U1 = self.add_weight(name='U1', shape=(self.L_w1, d_addr), initializer='random_normal', trainable=learn_addr)
        self.U2 = self.add_weight(name='U2', shape=(self.L_w2, d_addr), initializer='random_normal', trainable=learn_addr)
        self.U3 = self.add_weight(name='U3', shape=(self.L_b2, d_addr), initializer='random_normal', trainable=learn_addr)

        # Value tapes W1, W2, b2
        self.W1 = self.add_weight(name='W1', shape=(self.L_w1, H, D_in), initializer='random_normal', trainable=learn_tape_w1)
        self.W2 = self.add_weight(name='W2', shape=(self.L_w2, D_out, H), initializer='random_normal', trainable=learn_tape_w2)
        self.b2 = self.add_weight(name='b2', shape=(self.L_b2, D_out), initializer='random_normal', trainable=learn_tape_b2)

    def _tk(self, z, k, tau):
        # TensorFlow equivalent of torch.topk
        top_values, top_indices = tf.math.top_k(z, k=k, sorted=False)
        w = tf.nn.softmax(top_values / (tau + 1e-8), axis=1)
        return top_indices, w

    def _address(self, x_addr):
        # Pack U
        U_pack = tf.concat([self.U1, self.U2, self.U3], axis=0)
        
        # Normalize U (mimicking the PyTorch normalization logic)
        U_mean = tf.reduce_mean(U_pack, axis=1, keepdims=True)
        U_norm = tf.norm(U_pack - U_mean, axis=1, keepdims=True) + 1e-8
        U_pack = (U_pack - U_mean) / U_norm

        Z = tf.matmul(x_addr, U_pack, transpose_b=True)
        s1, s2, s3 = self.L_w1, self.L_w2, self.L_b2
        z1, z2, z3 = tf.split(Z, [s1, s2, s3], axis=1)
        
        i1, w1 = self._tk(z1, self.k1, self.t1)
        i2, w2 = self._tk(z2, self.k2, self.t2)
        i3, w3 = self._tk(z3, self.k3, self.t3)
        return (i1, w1), (i2, w2), (i3, w3)

    def _apply_mixture(self, x_flat, topi, weights, W):
        """
        Mix-then-apply logic using tf.gather (equivalent to torch.index_select).
        x_flat : [N, in]
        topi   : [N, k]
        weights: [N, k]
        W      : [L, out, in]
        return : [N, out]
        """
        # Normalize W
        W_norm = tf.norm(W, axis=(1,2), keepdims=True) + 1e-8
        W = W / W_norm

        # Select specific rows from W based on indices topi
        # PyTorch: W[indices] -> TF: tf.gather
        gathered_W = tf.gather(W, topi, axis=0) # Shape: [N, k, out, in]
        
        # Apply mixture: Einsum 'nkoi,ni->nko'
        # Multiply selected weights by input x_flat
        mixed = tf.einsum('nkoi,ni->nko', gathered_W, x_flat)
        
        # Apply attention weights
        w_exp = tf.expand_dims(weights, -1)
        res = tf.reduce_sum(mixed * w_exp, axis=1)
        return res

    def call(self, inputs, training=None):
        # inputs: [N, D_in]
        
        # Address calculation
        x_addr = tf.matmul(inputs, self.P)
        (i1, w1), (i2, w2), (i3, w3) = self._address(x_addr)
        
        # Layer 1
        h = self._apply_mixture(inputs, i1, w1, self.W1)
        h = tf.nn.gelu(h)
        
        # --- TARGET API INTEGRATION ---
        # Apply tf.keras.backend.dropout
        # Note: In graph mode (training=True), dropout applies noise. 
        # In inference (training=False), it acts as identity.
        h = tf.keras.backend.dropout(h, level=self.rate, noise_shape=None, seed=123)
        # ------------------------------

        # Layer 2
        out = self._apply_mixture(h, i2, w2, self.W2)
        
        # Bias handling
        b2_norm = tf.norm(self.b2, axis=1, keepdims=True) + 1e-8
        b2 = self.b2 / b2_norm
        b2_gathered = tf.gather(b2, i3, axis=0)
        b2_mixed = tf.reduce_sum(b2_gathered * tf.expand_dims(w3, -1), axis=1)
        
        out = out + b2_mixed
        return out
