import torch
import tensorflow as tf
import tensorflow.experimental.numpy as tnp
import time

def test_memory_throughput_with_arange():
    """
    Test case to benchmark memory throughput for concatenation operations 
    using tf.experimental.numpy.arange for data generation, mirroring the 
    logic of the PyTorch issue #161518.
    """
    
    # Check for GPU availability to ensure meaningful throughput results
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("This test requires a GPU to measure memory throughput. Skipping.")
        return

    # Enable numpy behavior for tensorflow.experimental.numpy
    tnp.experimental_enable_numpy_behavior()

    # Define dtypes to test. 
    # Note: float8_e4m3fn support depends on the TensorFlow version and hardware.
    dtypes = []
    if hasattr(tf, 'float8_e4m3fn'):
        dtypes.append(tf.float8_e4m3fn)
    dtypes.extend([tf.bfloat16, tf.float32])

    print(f"{'Dtype':<20} {'Time (ms)':<15} {'Bandwidth (TiB/s)':<15}")

    for dtype in dtypes:
        try:
            # Calculate element size to match 128 MiB data size
            # We create a dummy tensor to check the byte size
            dummy = tf.constant(0, dtype=dtype)
            element_size = dummy.numpy().nbytes
            
            # Calculate dimensions: (128 // element_size) * 1024 * 1024 elements
            # PyTorch script uses: (128 // element_size) * 1024 rows, 1024 cols
            # Total elements = 128 * 1024 * 1024 / element_size
            num_elements = (128 * 1024 * 1024) // element_size
            
            dim0 = num_elements // 1024
            dim1 = 1024

            # Generate input data using the similar API: tf.experimental.numpy.arange
            # This replaces torch.zeros from the original script.
            # We generate a sequence and reshape it to the target dimensions.
            
            # Input 1: 128 MiB
            input1_flat = tnp.arange(0, num_elements, 1, dtype=dtype)
            input1 = tf.reshape(input1_flat, (dim0, dim1))
            
            # Input 2: 128 MiB
            input2_flat = tnp.arange(num_elements, 2 * num_elements, 1, dtype=dtype)
            input2 = tf.reshape(input2_flat, (dim0, dim1))

            # Pre-allocate output tensor (256 MiB) to mimic the 'out' parameter in torch.cat
            # PyTorch: output = torch.zeros(..., 2048)
            # TF concat usually allocates new memory, but we can prepare a variable 
            # or just let concat allocate. To strictly follow the "IO" calculation logic 
            # (512 MiB total movement), we assume standard concat behavior (Read 128+128, Write 256).
            
            # Define the operation to benchmark
            @tf.function
            def concat_op():
                return tf.concat([input1, input2], axis=1)

            # Warmup
            concat_op()

            # Benchmark
            # Repetition count similar to the original script (rep=100)
            num_reps = 100
            start_time = time.perf_counter()
            
            for _ in range(num_reps):
                output = concat_op()
                
            # Synchronize to ensure accurate timing (TF operations are async)
            # Accessing .numpy() forces synchronization
            _ = output.numpy()
            
            end_time = time.perf_counter()
            
            avg_time_ms = (end_time - start_time) * 1000 / num_reps
            
            # Calculate Bandwidth
            # Total IO = 128 (in1) + 128 (in2) + 256 (out) = 512 MiB
            # Formula from issue: bdwidth = 512 / output * 1000 / 1000 / 1000 # TiB/s
            # Note: The issue uses 1000 for TiB conversion (decimal), not 1024 (binary).
            io_mib = 512.0
            bdwidth = io_mib / avg_time_ms * 1000 / 1000 / 1000

            print(f"{str(dtype):<20} {avg_time_ms:.2f}           {bdwidth:.2f}")

        except Exception as e:
            # Handle cases where dtype might not be fully supported or device errors
            print(f"{str(dtype):<20} Error: {str(e)}")

if __name__ == "__main__":
    test_memory_throughput_with_arange()
