import torch
import unittest

# Check if torch.compile is available (introduced in PyTorch 2.0)
HAS_TORCH_COMPILE = hasattr(torch, 'compile')

@unittest.skipIf(not HAS_TORCH_COMPILE, "torch.compile requires PyTorch 2.0+")
class TestFillDiagonalDynamicShapes(unittest.TestCase):
    """
    Test case for Tensor.fill_diagonal_ with dynamic shapes.
    Related to Issue ID: 162271.
    
    This test verifies that fill_diagonal_ works correctly when used 
    within a torch.compile context with dynamic=True, which forces 
    the use of symbolic sizes (similar to how tf.broadcast_dynamic_shape 
    handles symbolic shapes).
    """

    def test_fill_diagonal_with_dynamic_compile(self):
        """
        Reproduces the logic from the bug report:
        Using torch.compile(dynamic=True) triggers the usage of FakeTensors
        with symbolic sizes. The original bug caused a RuntimeError:
        'Cannot call storage_offset() on tensor with symbolic sizes/strides'
        """
        
        @torch.compile(dynamic=True)
        def fill_diagonal_compiled(x):
            # This operation previously failed with symbolic sizes
            x.fill_diagonal_(True)
            return x

        # Test with a fixed size tensor (4, 4) as in the original report
        # The 'dynamic=True' flag implies the graph should handle symbolic shapes
        # even if the input size is static during this specific call.
        x = torch.zeros(4, 4)
        
        try:
            result = fill_diagonal_compiled(x)
            
            # Verify the operation actually worked
            self.assertTrue(result[0, 0].item(), "Diagonal element (0,0) should be True")
            self.assertTrue(result[1, 1].item(), "Diagonal element (1,1) should be True")
            self.assertTrue(result[3, 3].item(), "Diagonal element (3,3) should be True")
            
            # Verify non-diagonal elements are unchanged
            self.assertFalse(result[0, 1].item(), "Non-diagonal element (0,1) should be False")
            self.assertFalse(result[1, 0].item(), "Non-diagonal element (1,0) should be False")
            
        except RuntimeError as e:
            if "storage_offset()" in str(e) and "symbolic" in str(e):
                self.fail(f"fill_diagonal_ failed with symbolic shapes: {e}")
            else:
                raise

    def test_fill_diagonal_dynamic_varying_sizes(self):
        """
        Further test to ensure dynamic shape handling works across different sizes,
        mimicking the flexibility of tf.broadcast_dynamic_shape.
        """
        
        @torch.compile(dynamic=True)
        def fill_diag(x):
            x.fill_diagonal_(1.0)
            return x

        # Call with different sizes to exercise the dynamic path
        x1 = torch.zeros(3, 3)
        res1 = fill_diag(x1)
        self.assertTrue(torch.all(res1.diag() == 1.0))

        x2 = torch.zeros(5, 5)
        res2 = fill_diag(x2)
        self.assertTrue(torch.all(res2.diag() == 1.0))

if __name__ == "__main__":
    unittest.main()
