import torch
import torch.nn as nn

# Adapting the structure of tf.keras.initializers.LecunUniform
# to encapsulate the bug reproduction logic for torch.compile.
class Uint8NegAddModule(nn.Module):
    def __init__(self, scale=1., mode='fan_in', distribution='uniform', seed=None):
        super(Uint8NegAddModule, self).__init__()
        self.scale = scale
        self.mode = mode
        self.distribution = distribution
        self.seed = seed
        
        if seed is not None:
            torch.manual_seed(seed)
            
        # Constant uint8 tensor involved in the bug
        self.c = torch.tensor(7, dtype=torch.uint8)

    def forward(self, x):
        # Reproducing the logic: neg(uint8) + x
        # Eager: wraps (249 + x)
        # Inductor (buggy): casts to signed (-7 + x)
        return torch.neg(self.c) + x

def test_uint8_neg_add_compile():
    # Check for torch.compile availability (PyTorch 2.0+)
    if not hasattr(torch, 'compile'):
        print("Skipping test: torch.compile is not available (requires PyTorch 2.0+).")
        return

    # Setup from the original bug report
    torch.manual_seed(0)
    x = torch.randn(2, 2, dtype=torch.float32)
    
    # Initialize module mimicking the API pattern
    model = Uint8NegAddModule(seed=0)
    
    # Compile the model
    compiled_model = torch.compile(model)
    
    # Execute eager and compiled versions
    res_eager = model(x)
    res_compiled = compiled_model(x)
    
    # The bug manifests in the result of neg(uint8) + x
    # Expected (Eager): tensor([[250.5410, 248.7066], [246.8212, 249.5684]])
    # Buggy (Compiled): tensor([[-5.4590, -7.2934], [-9.1788, -6.4316]])
    
    print(f"Eager result:\n{res_eager}")
    print(f"Compiled result:\n{res_compiled}")
    
    # Assert equality to catch the regression
    torch.testing.assert_close(res_eager, res_compiled)

if __name__ == "__main__":
    test_uint8_neg_add_compile()
