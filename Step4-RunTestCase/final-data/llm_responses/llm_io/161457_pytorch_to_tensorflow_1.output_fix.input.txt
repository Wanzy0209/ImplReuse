    def computation_fn(inputs):

        return model(inputs[0]).logits

    # Execute using batch_parallel
    # num_shards=1 is used to match the batch size of 1 in the original test
    res2 = tf.compat.v1.tpu.batch_parallel(
        computation_fn,
        inputs=[input_ids],
        num_shards=1
    )

    # Verify correctness
