import tensorflow as tf
import numpy as np

# Define a model that uses the similar API: tf.linalg.lu_solve
# This mimics the structure of the original issue where a model is defined and then exported.
class LUSolverModel(tf.Module):
    def __init__(self):
        super(LUSolverModel, self).__init__()

    @tf.function(input_signature=[
        tf.TensorSpec(shape=[None, 3, 3], dtype=tf.float32, name="lower_upper"),
        tf.TensorSpec(shape=[None, 3], dtype=tf.int32, name="perm"),
        tf.TensorSpec(shape=[None, 3], dtype=tf.float32, name="rhs_vector")
    ])
    def solve(self, lower_upper, perm, rhs_vector):
        # The API info suggests handling vector cases by expanding dims
        # lu_solve(..., rhs[..., tf.newaxis])[..., 0]
        rhs_matrix = rhs_vector[..., tf.newaxis]
        solution_matrix = tf.linalg.lu_solve(lower_upper, perm, rhs_matrix)
        return solution_matrix[..., 0]

# Prepare inputs
# Mimicking the "messages" -> "inputs" preparation in the original issue
batch_size = 2
matrix_size = 3

# Create a batch of random invertible matrices
# We ensure invertibility by making them diagonally dominant
np.random.seed(42)
A_np = np.random.rand(batch_size, matrix_size, matrix_size).astype(np.float32)
A_np = A_np + np.eye(matrix_size)[np.newaxis, :, :] * matrix_size

# Compute LU factors (inputs to the model)
lower_upper_np, perm_np = tf.linalg.lu(tf.constant(A_np))

# Create RHS vectors (inputs to the model)
rhs_np = np.random.rand(batch_size, matrix_size).astype(np.float32)

# Instantiate model
model = LUSolverModel()

# Run inference (mimicking the model run before export)
print("Running inference...")
outputs = model.solve(lower_upper_np, perm_np, rhs_np)
print(f"Inference successful. Output shape: {outputs.shape}")

# Export (mimicking torch.onnx.export)
# In TensorFlow, the equivalent of exporting a model for deployment is saving a SavedModel.
export_dir = tempfile.mkdtemp()

tf.saved_model.save(model, export_dir)

# Verify the export (mimicking checking the ONNX file)
print("Loading exported model to verify...")
loaded = tf.saved_model.load(export_dir)
infer = loaded.signatures["serving_default"]

# Call the loaded model
loaded_outputs = infer(lower_upper=lower_upper_np, perm=perm_np, rhs_vector=rhs_np)

# Check consistency
# The output key might be 'output_0' or similar, let's inspect or just grab the first tensor
