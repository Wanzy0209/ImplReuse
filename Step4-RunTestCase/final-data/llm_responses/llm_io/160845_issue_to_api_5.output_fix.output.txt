import tensorflow as tf
import numpy as np

def test_linear_operator_low_rank_update_complex_imag():
    """
    Test case for tf.linalg.LinearOperatorLowRankUpdate handling complex tensors.
    
    This test is inspired by PyTorch Issue 160845, where index_add_ on MPS 
    incorrectly dropped the imaginary component of complex tensors during an update.
    
    Here, we verify that LinearOperatorLowRankUpdate correctly preserves the 
    imaginary components when performing a low-rank update (A = L + U D V^H).
    We specifically construct a scenario where the update term is purely 
    imaginary to stress-test the complex arithmetic, mirroring the original bug's
    conditions (src_real=0, src_imag!=0).
    """
    # Set random seed for reproducibility
    # Handle compatibility between TensorFlow 1.x and 2.x
    try:
        tf.random.set_seed(0)
    except AttributeError:
        # Fallback for TensorFlow 1.x where set_seed is a top-level function
        tf.set_random_seed(0)
    np.random.seed(0)

    # Dimensions
    n = 4  # Matrix size
    k = 2  # Rank of update

    # 1. Define the Base Operator L (Identity)
    # Using complex64 to match the original bug's dtype
    L = tf.linalg.LinearOperatorIdentity(
        num_rows=n, 
        dtype=tf.complex64
    )

    # 2. Define the Update Components U, D, V
    # To mimic the original bug where the source had zero real part and non-zero imaginary part:
    # We construct U to be purely imaginary, D to be real, and V to be real.
    # The resulting update term (U * D * V^H) should be purely imaginary.
    
    # U: [n, k] - Purely Imaginary
    u_imag = tf.random.normal((n, k), dtype=tf.float32)
    U = tf.complex(tf.zeros_like(u_imag), u_imag)

    # D: [k] - Real (Diagonal)
    d_vals = tf.random.normal((k,), dtype=tf.float32)
    D = tf.linalg.LinearOperatorDiag(d_vals, dtype=tf.float32)

    # V: [n, k] - Real
    V = tf.complex(tf.random.normal((n, k), dtype=tf.float32), 
                   tf.zeros((n, k), dtype=tf.float32))

    # 3. Construct the LowRankUpdate Operator
    # A = L + U * D * V^H
    op = tf.linalg.LinearOperatorLowRankUpdate(L, U, V, diag_update=D)

    # 4. Compute the result using the operator
    result_matrix = op.to_dense()

    # 5. Compute the expected result manually
    # V^H (Hermitian transpose of V). Since V is real, this is just transpose.
    V_h = tf.linalg.adjoint(V)
    
    # U * D
    # Note: D is real, U is complex. Result is complex (purely imaginary).
    UD = tf.matmul(U, D.to_dense())
    
    # (U * D) * V^H
    # Result should be purely imaginary.
    update_term = tf.matmul(UD, V_h)
    
    expected_matrix = L.to_dense() + update_term

    # 6. Verify the results
    # We check if the operator result matches the expected dense calculation.
    # Specifically, we ensure the imaginary part is not zeroed out.
    np.testing.assert_allclose(
        result_matrix.numpy(), 
        expected_matrix.numpy(), 
        rtol=1e-5, 
        atol=1e-5,
        err_msg="LinearOperatorLowRankUpdate failed to correctly handle complex update (imaginary part mismatch)."
    )

    # Explicit check to ensure imaginary part exists (mimicking the bug report's check)
    # The update term should have non-zero imaginary sum.
    imag_sum = tf.reduce_sum(tf.abs(tf.imag(update_term))).numpy()
    assert imag_sum > 1e-3, "Update term should have non-zero imaginary component."

    print("Test passed: LinearOperatorLowRankUpdate correctly preserves complex components.")

if __name__ == "__main__":
    test_linear_operator_low_rank_update_complex_imag()
