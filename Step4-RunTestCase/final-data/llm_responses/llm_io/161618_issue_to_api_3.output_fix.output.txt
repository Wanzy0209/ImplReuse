import torch
from torch import jit, Tensor, Parameter

# Handle missing torch._inductor gracefully
try:
    import torch._inductor.config as inductor_config
except (ImportError, ModuleNotFoundError):
    inductor_config = None

# Define the LSTMCell similar to the provided similar API context
class LSTMCell(jit.ScriptModule):
    def __init__(self, input_size, hidden_size):
        super().__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.weight_ih = Parameter(torch.randn(4 * hidden_size, input_size))
        self.weight_hh = Parameter(torch.randn(4 * hidden_size, hidden_size))
        self.bias_ih = Parameter(torch.randn(4 * hidden_size))
        self.bias_hh = Parameter(torch.randn(4 * hidden_size))

    @jit.script_method
    def forward(
        self, input: Tensor, state: tuple[Tensor, Tensor]
    ) -> tuple[Tensor, tuple[Tensor, Tensor]]:
        hx, cx = state
        gates = (
            torch.mm(input, self.weight_ih.t())
            + self.bias_ih
            + torch.mm(hx, self.weight_hh.t())
            + self.bias_hh
        )
        ingate, forgetgate, cellgate, outgate = gates.chunk(4, 1)

        ingate = torch.sigmoid(ingate)
        forgetgate = torch.sigmoid(forgetgate)
        cellgate = torch.tanh(cellgate)
        outgate = torch.sigmoid(outgate)

        cy = (forgetgate * cx) + (ingate * cellgate)
        hy = outgate * torch.tanh(cy)
        return hy, (hy, cy)

def test_inductor_triton_lstm():
    # Skip test if torch._inductor is not available
    if inductor_config is None:
        print("Skipping test: torch._inductor is not available in this environment.")
        return

    # Reuse dimensions from the original bug report to trigger similar kernel sizes
    m = 20120
    k = 1536
    n = 512

    batch_size = m
    input_size = k
    hidden_size = n

    # Instantiate the cell and move to CUDA
    lstm_cell = LSTMCell(input_size, hidden_size).cuda()
    
    # Prepare inputs
    input = torch.randn((batch_size, input_size)).cuda()
    hx = torch.randn((batch_size, hidden_size)).cuda()
    cx = torch.randn((batch_size, hidden_size)).cuda()

    # Apply the same Inductor configuration that caused the failure in the original issue
    with inductor_config.patch(
        max_autotune=True,
        max_autotune_gemm_backends="TRITON",
        autotune_fallback_to_aten=False,
    ):
        # Compile the LSTMCell forward pass
        compiled_lstm = torch.compile(lstm_cell, dynamic=False)
        
        # Run the compiled function
        # This should trigger the Triton GEMM kernels similar to torch.addmm
        hy, (hy_out, cy_out) = compiled_lstm(input, (hx, cx))

    # Basic sanity check to ensure execution completed and produced finite values
    assert torch.isfinite(hy).all(), "Output contains NaN or Inf"
    assert torch.isfinite(cy_out).all(), "Cell state contains NaN or Inf"

if __name__ == "__main__":
    test_inductor_triton_lstm()
