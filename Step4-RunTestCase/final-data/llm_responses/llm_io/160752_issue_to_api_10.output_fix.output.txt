import torch
import unittest

# Check if torch.func is available (requires PyTorch 2.0+)
HAS_TORCH_FUNC = hasattr(torch, 'func')

class TestJacfwdOneHotDynamicCompile(unittest.TestCase):
    """
    Test case for Issue ID: 160752.
    Verifies that torch.func.jacfwd works correctly with torch.nn.functional.one_hot
    when compiled with torch.compile(..., dynamic=True).
    """

    @unittest.skipIf(not HAS_TORCH_FUNC, "torch.func is not available (requires PyTorch 2.0+)")
    def test_jacfwd_one_hot_dynamic_compile(self):
        MAX = 3
        BATCH = 37

        def func(x, idxs):
            return x.square() * torch.nn.functional.one_hot(idxs, MAX)

        def jacfunc(x, idxs):
            return torch.func.jacfwd(func, argnums=(0,))(x, idxs)

        # Setup inputs
        idxs = torch.randint(MAX, (BATCH,), dtype=torch.int64)
        x = torch.rand((BATCH, MAX), dtype=torch.float64)

        # 1. Run in eager mode to get expected output
        expected_out = jacfunc(x, idxs)

        # 2. Run with torch.compile(dynamic=True)
        # This is the specific scenario that caused the failure in the bug report.
        compiled_jacfunc = torch.compile(jacfunc, dynamic=True)
        actual_out = compiled_jacfunc(x, idxs)

        # 3. Verify outputs match
        self.assertTrue(torch.allclose(expected_out, actual_out))

if __name__ == "__main__":
    unittest.main()
