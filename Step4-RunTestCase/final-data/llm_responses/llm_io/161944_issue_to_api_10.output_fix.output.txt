import sys
import torch

# Handle environment incompatibility issues (e.g., GLIBCXX version) gracefully
try:
    import tensorflow as tf
except ImportError as e:
    if "GLIBCXX" in str(e):
        print(f"Skipping test: TensorFlow import failed due to missing system libraries ({e}).")
        print("This is likely due to an outdated libstdc++.so.6 in the environment.")
        sys.exit(0)
    else:
        raise

def test_exp_precision_compilation():
    """
    Test case adapted from PyTorch issue #161944.
    Checks precision of tf.exp under XLA compilation (tf.function with jit_compile=True).
    This mirrors the logic of checking for precision loss due to compiler optimizations (fast math).
    """
    # Generate input
    inp = tf.random.normal((8192,))
    
    # Define the exponential function
    # Using jit_compile=True to trigger XLA compilation, similar to torch.compile
    @tf.function(jit_compile=True)
    def exp_func(x):
        return tf.exp(x)
    
    # 1. Eager execution (float32)
    out1 = tf.exp(inp)
    
    # 2. Compiled execution (float32)
    out2 = exp_func(inp)
    
    # 3. High precision execution (float64)
    out3_high = tf.exp(tf.cast(inp, tf.float64))
    
    # Calculate max absolute difference from high precision reference
    diff_eager = tf.reduce_max(tf.abs(out3_high - tf.cast(out1, tf.float64)))
    diff_compiled = tf.reduce_max(tf.abs(out3_high - tf.cast(out2, tf.float64)))
    
    print(f"Max diff (eager vs high): {diff_eager.numpy()}")
    print(f"Max diff (compiled vs high): {diff_compiled.numpy()}")

if __name__ == "__main__":
    test_exp_precision_compilation()
