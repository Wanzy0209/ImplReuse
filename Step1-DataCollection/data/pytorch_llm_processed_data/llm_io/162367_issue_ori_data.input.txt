```python
"""Reproduce NVRTC error from TestCompileKernel.test_compile_kernel"""

import torch
from torch.cuda import _compile_kernel

# Exact kernel from the failing test
kernel_source = """
__global__ void add_tensors(const float* a, const float* b, float* c, int n) {
    int i = threadIdx.x + blockIdx.x * blockDim.x;
    if (i < n)
        c[i] = a[i] + b[i];
}
"""

# This line triggers: OSError: libnvrtc.so: cannot open shared object file
add_kernel = _compile_kernel(kernel_source, "add_tensors")

# Test the kernel
N = 1024
a = torch.rand(N, device="cuda")
b = torch.rand(N, device="cuda")
c = torch.empty_like(a)

threads_per_block = 256
blocks_per_grid = (N + threads_per_block - 1) // threads_per_block

add_kernel(
    grid=(blocks_per_grid, 1, 1),
    block=(threads_per_block, 1, 1),
    args=[a, b, c, N],
)

# Verify results
expected = a + b
assert torch.allclose(c, expected)
print("Kernel test passed!")
```
