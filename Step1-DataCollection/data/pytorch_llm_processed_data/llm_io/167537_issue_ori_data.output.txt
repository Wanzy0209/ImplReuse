import os
import torch
import torch.distributed as dist

try:
    import torch.distributed._symmetric_memory as torch_symm_mem
    symm_mem_available = True
except ImportError:
    symm_mem_available = False

def run_rendezvous():
    rank = int(os.environ["RANK"])
    world_size = int(os.environ["WORLD_SIZE"])

    dist.init_process_group(backend="nccl", rank=rank, world_size=world_size)

    if not symm_mem_available:
        print(f"Rank {rank}: Symmetric memory is not available.")
        return

    device = torch.device(f"cuda:{rank}")
    torch.cuda.set_device(device)

    max_size = 1024
    dtype = torch.bfloat16
    buffer = torch_symm_mem.empty(
        max_size // dtype.itemsize,
        device=device,
        dtype=dtype,
    )

    group_name = dist.group.WORLD.group_name
    handle = torch_symm_mem.rendezvous(buffer, group_name)

    tensor_to_reduce = torch.ones(4, device=device, dtype=dtype) * (rank + 1)
    dist.all_reduce(tensor_to_reduce, op=dist.ReduceOp.SUM)
```
