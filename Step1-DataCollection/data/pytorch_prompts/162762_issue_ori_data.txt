import logging
import os

import torch
from transformers import AutoConfig, AutoProcessor, AutoTokenizer

from optimum.executorch import ExecuTorchModelForMultiModalToText
from optimum.exporters.executorch.tasks.multimodal_text_to_text import load_multimodal_text_to_text_model


logging.basicConfig(level=logging.DEBUG)


def compile_with_aoti(exported_programs, output_dir: str = "./voxtral_aoti_compiled"):
    """
    Compile exported programs with AOTI
    """
    os.makedirs(output_dir, exist_ok=True)
    compiled_programs = {}

    print(f"\nCompiling with AOTI to {output_dir}...")

    for name, exported_program in exported_programs.items():
        # text_decoder takes a while, better to skip it while debugging audio encoder
        # if name == "text_decoder":
        #     continue
        print(f"Compiling {name}...")

        # Use AOTI to compile the exported program
        package_path = os.path.join(output_dir, f"{name}.pt2")
        output_path = torch._inductor.aoti_compile_and_package(
            exported_program,
            package_path=package_path,
        )
        print(f"✓ {name} compiled to {output_path}")

        # Load the compiled program
        aoti_compiled_program = torch._inductor.aoti_load_package(output_path)
        print(f"✓ Loaded {name}")

        compiled_programs[name] = aoti_compiled_program

    return compiled_programs


def main(device, aoti_compile):
    model_id = "mistralai/Voxtral-Mini-3B-2507"
    config = AutoConfig.from_pretrained(model_id)
    module = load_multimodal_text_to_text_model(

        device=device,
        use_custom_sdpa=True,
        use_custom_kv_cache=True,
    )

    ep = module.export()

    if aoti_compile:
        compiled_programs = compile_with_aoti(ep)

                {"type": "text", "text": "What can you tell me about this audio?"},
            ],
        }
    ]
    processor = AutoProcessor.from_pretrained(model_id)
    inputs = processor.apply_chat_template(conversation)

    for name in inputs:
        inputs[name] = inputs[name].to(device)

    input_ids = inputs["input_ids"]

        token_embeddings = compiled_programs["token_embedding"](input_ids)
        if torch.allclose(token_embeddings, token_embeddings_ep, atol=1e-4):
            print("✓ Token embeddings are the same")
