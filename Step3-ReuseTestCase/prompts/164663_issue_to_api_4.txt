Issue ID: 164663
Title: FSDPMemTracker fails with KeyError for RMSNorm during backward
Bug Description:
### 🐛 Describe the bug

When sharding a model that uses nn.RMSNorm, FSDPMemTracker raises KeyError during pre-backward. The failure happens when RMSNorm is wrapped with another Linear module in a single fully_sharded().

from torch.distributed._tools.fsdp2_mem_tracker import FSDPMemTracker


class TestModule(nn.Module):

    def __init__(self, d_model: int):
        super().__init__()
        self.norm = nn.RMSNorm(d_model)
        self.output = nn.Linear(d_model, d_model)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.norm(x)
        x = self.output(x)
        return x


def main():
    torch.cuda.set_device(int(os.environ["LOCAL_RANK"]))

    dist.init_process_group(backend="nccl")

    d_model = 128

    model = TestModule(d_model)
    model = model.to('cuda:0')
    mesh = init_device_mesh("cuda", (dist.get_world_size(),))

    fully_shard([model.norm, model.output], mesh=mesh)   # shards the RMSNorm's parameters/state across ranks
    fully_shard(model, mesh=mesh)  # shards the Linear's parameters/state across ranks
    print(model)

    tracker = FSDPMemTracker(model)

    with tracker, implicit_replication():
        x = torch.randn(16, d_model, device='cuda:0')
        y = model(x)
        loss = y.sum()
        loss.backward()


if __name__ == "__main__":
    main()
