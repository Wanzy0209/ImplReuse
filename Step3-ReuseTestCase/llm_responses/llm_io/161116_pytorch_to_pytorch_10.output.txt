```python
import torch
import os

def main():
    # Adapted from the original bug report's device setup logic
    # Default to GPU 0 if LOCAL_RANK is not set (for standalone testing)
    gpu_id = int(os.environ.get("LOCAL_RANK", "0"))
    
    if torch.cuda.is_available():
        device = f"cuda:{gpu_id}"
        torch.cuda.set_device(device)

        # Create a random tensor on the specific GPU device
        tensor = torch.randn(10, 10, device=device)

        # Call the similar API: torch.argsort
        # This replaces the original dist.init_process_group call site
        sorted_indices = torch.argsort(tensor, dim=-1)

        # Basic verification of the similar API
        assert sorted_indices.shape == tensor.shape
        assert sorted_indices.device == tensor.device
        
        print(f"torch.argsort test passed on {device}")
    else:
        print("CUDA is not available. Test skipped.")

if __name__ == "__main__":
    main()
```
