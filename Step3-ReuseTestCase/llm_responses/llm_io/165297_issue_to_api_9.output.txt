import torch
import torch.nn as nn

def test_maxpool2d_channels_last_bfloat16():
    """
    Test case for Issue 165297: MaxPool2d with channels_last + bfloat16 on CUDA produces NaNs.
    
    This test verifies the behavior of MaxPool2d when the tensor memory format is explicitly
    set to channels_last (NHWC). This is analogous to enabling a specific global behavior
    mode (like tf.compat.v1.enable_v2_behavior) where the memory layout and execution 
    strategy differ from the default.
    """
    # Check for CUDA availability
    if not torch.cuda.is_available():
        print("CUDA not available, skipping test.")
        return

    device = torch.device("cuda")
    
    # Large input tensor dimensions from the bug report
    N, C, H, W = 84, 64, 512, 960

    # Case 1: bfloat16 + channels_last
    # Initialize tensor with bfloat16
    x = torch.randn(N, C, H, W, dtype=torch.bfloat16, device=device)

    # Convert to NHWC channels_last layout
    # This step changes the memory access pattern, similar to how enabling v2 behavior
    # changes execution strategies in other frameworks.
    x = x.to(memory_format=torch.channels_last)

    # Verify the memory format state
    # It should not be contiguous in the standard NCHW sense, but contiguous in NHWC
    assert not x.is_contiguous(), "Tensor should not be NCHW contiguous"
    assert x.is_contiguous(memory_format=torch.channels_last), "Tensor should be channels_last contiguous"

    # Initialize MaxPool2d
    pool = nn.MaxPool2d(kernel_size=3, stride=2, padding=1).to(device)
    
    # Execute pooling
    y = pool(x)

    # Check for the reported bugs (NaNs or Infs)
    has_nan = torch.isnan(y).any().item()
    has_inf = torch.isinf(y).any().item()

    # Assertions
    assert not has_nan, f"Detected NaNs in MaxPool output! Stats: min={y.min()}, max={y.max()}"
    assert not has_inf, f"Detected Infs in MaxPool output! Stats: min={y.min()}, max={y.max()}"

    print("Test passed: No NaNs or Infs detected with channels_last and bfloat16.")

if __name__ == "__main__":
    test_maxpool2d_channels_last_bfloat16()
