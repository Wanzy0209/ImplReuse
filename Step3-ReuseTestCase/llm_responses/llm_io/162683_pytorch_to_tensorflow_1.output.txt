```python
import tensorflow as tf
import time

# Set seed for reproducibility
tf.random.set_seed(0)

# Adapted configurations: 
# The original test used tensor shapes for matrix multiplication.
# Here, we adapt these to be the 'input_shape' for the padding calculation API.
# We define a 'block_shape' (e.g., [2, 2]) to make the operation valid.
configs = [
     ([1, 12, 10, 64], [2, 2]),
     ([1, 12, 10, 10], [2, 2]),
]

def benchmark_paddings(input_shape, block_shape, repeat=500):
    # Prepare inputs as tensors
    input_t = tf.constant(input_shape, dtype=tf.int32)
    block_t = tf.constant(block_shape, dtype=tf.int32)

    # Verify behavior (sanity check on output shapes)
    paddings, crops = tf.required_space_to_batch_paddings(input_t, block_t)
    assert paddings.shape == (len(input_shape), 2), "Paddings shape mismatch"
    assert crops.shape == (len(input_shape), 2), "Crops shape mismatch"

    # warm up
    for _ in range(5000):
        _ = tf.required_space_to_batch_paddings(input_t, block_t)

    # run
    times = []
    for i in range(repeat):
        start = time.time()
        _ = tf.required_space_to_batch_paddings(input_t, block_t)
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
    
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    for input_shape, block_shape in configs:
        t = benchmark_paddings(input_shape, block_shape)
        print(f"Input: {input_shape}, Block: {block_shape} -> {t:.3f} us")
```
