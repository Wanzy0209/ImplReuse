import tensorflow as tf
import numpy as np

class Foo(tf.Module):
    def __init__(self) -> None:
        super().__init__()

    def __call__(self, x: tf.Tensor) -> tf.Tensor:
        # Using the target API: tf.experimental.numpy.negative
        return tf.experimental.numpy.negative(x)

def test_device(device_name, x):
    # Move tensor to the specific device
    with tf.device(device_name):
        foo = Foo()
        # Create a compiled version (equivalent to torch.compile)
        foo_compiled = tf.function(foo)

        # warm up (tracing happens here)
        y_original = foo(x)
        y_compiled = foo_compiled(x)

        # proper inference
        y_original = foo(x)
        y_compiled = foo_compiled(x)

        diff = tf.reduce_max(tf.abs(y_original - y_compiled)).numpy()
        print(f'device: {device_name}, diff: {diff}')
        print('original', y_original[:5, :5].numpy())
        print('compiled', y_compiled[:5, :5].numpy())
        
        # Verify behavior: compiled and eager should match
        assert np.allclose(y_original.numpy(), y_compiled.numpy()), \
            f"Mismatch found on {device_name}"

def main():
    batch_size = 32
    feature_dim = 10
    
    # Setup data
    np.random.seed(42)
    x_np = np.random.randn(batch_size, feature_dim).astype(np.float32)
    x = tf.constant(x_np)

    # Test on CPU
    test_device('/CPU:0', x)

    # Test on GPU if available
    gpus = tf.config.list_physical_devices('GPU')
    if gpus:
        test_device('/GPU:0', x)
    else:
        print("GPU not available, skipping GPU test.")

if __name__ == '__main__':
    main()
