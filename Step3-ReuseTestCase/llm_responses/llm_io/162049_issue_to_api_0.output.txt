import torch
import pytest

def test_nested_tensor_int_min_max():
    """
    Test that min/max operations work correctly for integer NestedTensors.
    
    This test addresses the bug where calling .max(dim=...), .min(dim=...), 
    .argmax(dim=...), or .argmin(dim=...) on a nested jagged tensor with 
    integer datatype raised a TypeError. The root cause was the implementation 
    accessing torch.finfo (for floats) instead of torch.iinfo (for ints).
    
    This test ensures the fix handles integer types correctly, similar to 
    how type-specific logic might be handled in other linear algebra APIs 
    (like tf.linalg.lu_matrix_inverse handling specific input types).
    """
    # Create a nested jagged tensor with integer values
    # This matches the minimal reproducing example from the issue
    x = torch.nested.nested_tensor(
        [torch.arange(0, n) for n in (10, 20, 30)],
        layout=torch.jagged,
    )

    # Test max(dim=1)
    # Expected: tensor([9, 19, 29])
    max_result = x.max(dim=1)
    assert torch.equal(max_result.values, torch.tensor([9, 19, 29])), \
        f"max(dim=1) failed for int tensor. Got {max_result.values}"

    # Test min(dim=1)
    # Expected: tensor([0, 0, 0])
    min_result = x.min(dim=1)
    assert torch.equal(min_result.values, torch.tensor([0, 0, 0])), \
        f"min(dim=1) failed for int tensor. Got {min_result.values}"

    # Test argmax(dim=1)
    # Expected: tensor([9, 19, 29])
    argmax_result = x.argmax(dim=1)
    assert torch.equal(argmax_result.values, torch.tensor([9, 19, 29])), \
        f"argmax(dim=1) failed for int tensor. Got {argmax_result.values}"

    # Test argmin(dim=1)
    # Expected: tensor([0, 0, 0])
    argmin_result = x.argmin(dim=1)
    assert torch.equal(argmin_result.values, torch.tensor([0, 0, 0])), \
        f"argmin(dim=1) failed for int tensor. Got {argmin_result.values}"

    # Verify float types still work (regression check)
    x_float = torch.nested.nested_tensor(
        [torch.arange(0.0, float(n)) for n in (10, 20, 30)],
        layout=torch.jagged,
    )
    max_float_result = x_float.max(dim=1)
    assert torch.allclose(max_float_result.values, torch.tensor([9.0, 19.0, 29.0])), \
        "max(dim=1) failed for float tensor."

if __name__ == "__main__":
    test_nested_tensor_int_min_max()
    print("Test passed.")
