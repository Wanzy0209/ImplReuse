```python
import torch
import torch.nn as nn

def test_mseloss_zero_shape():
    """
    Test case to verify torch.nn.MSELoss handles tensors with 0-shape dimensions
    without crashing, similar to the issue reported for torch.nn.functional.pad.
    """
    # Create a tensor with a 0-shape dimension, similar to the bug report
    x = torch.zeros((6, 0))
    y = torch.zeros((6, 0))

    # Initialize MSELoss with default reduction (mean)
    criterion_mean = nn.MSELoss()
    
    # Execute the function
    # This should not crash like torch.nn.functional.pad did in the bug report
    loss_mean = criterion_mean(x, y)

    # Assertion: With reduction='mean', the result should be NaN (0/0 elements)
    assert torch.isnan(loss_mean), f"Expected NaN for mean reduction on empty tensor, got {loss_mean}"

    # Initialize MSELoss with sum reduction
    criterion_sum = nn.MSELoss(reduction='sum')
    loss_sum = criterion_sum(x, y)

    # Assertion: With reduction='sum', the result should be 0.0
    assert loss_sum.item() == 0.0, f"Expected 0.0 for sum reduction on empty tensor, got {loss_sum}"

    print("Test passed: MSELoss handles 0-shape tensors correctly.")

if __name__ == "__main__":
    test_mseloss_zero_shape()
```
