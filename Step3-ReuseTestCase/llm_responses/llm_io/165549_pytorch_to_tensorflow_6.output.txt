import tensorflow as tf
import numpy as np

def test_broadcast_to_shape_integrity():
    """
    Adapted from PyTorch Issue 165549.
    
    Original Bug Description:
    Operations like abs (dispatched via CompositeExplicitAutograd) return empty 
    tensors (shape [0]) when run through cpu_fallback on custom devices.
    
    Adaptation Logic:
    This test verifies that tf.keras.ops.broadcast_to correctly allocates and 
    returns a tensor with the expected broadcasted shape, rather than returning 
    an empty tensor or failing to resize the output buffer correctly.
    """
    
    # Setup: Create an input tensor
    # Analogous to torch.randn(4, 4, device='privateuse1')
    # We use a shape (1, 4) to broadcast to (4, 4)
    input_tensor = tf.random.normal((1, 4))
    target_shape = (4, 4)

    # Operation: Broadcast the tensor
    # In the PyTorch bug, this resulted in shape [0] instead of the expected shape
    result = tf.keras.ops.broadcast_to(input_tensor, target_shape)

    # Assertion: Check if the shape matches the target
    # The bug would manifest as result.shape being empty (e.g., (0,) or similar)
    expected_shape = tf.TensorShape(target_shape)
    
    assert result.shape == expected_shape, (
        f"Shape mismatch detected. Expected {expected_shape}, "
        f"but got {result.shape}. "
        "This indicates a potential allocation issue similar to the PyTorch bug."
    )
    
    # Verify values are actually broadcasted correctly to ensure the tensor is valid
    # (Checking that the first row matches the input)
    assert tf.reduce_all(result[0] == input_tensor[0]).numpy(), "Values were not broadcasted correctly."

    print("Test passed: Output shape is correct and values are broadcasted.")

if __name__ == "__main__":
    test_broadcast_to_shape_integrity()
