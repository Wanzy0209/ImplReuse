import tensorflow as tf
import numpy as np

def test_ctc_loss_memory_leak():
    """
    Adapted test case for tf.compat.v1.nn.ctc_loss based on the PyTorch 
    checkpoint memory leak issue.
    
    The original issue involves a memory leak when a custom autograd function 
    is used with activation checkpointing due to early-stopping exceptions. 
    This test adapts the loop and memory monitoring logic to verify if 
    tf.compat.v1.nn.ctc_loss exhibits similar memory accumulation behavior 
    during repeated forward/backward passes.
    """
    
    # Check for GPU availability to monitor memory
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("GPU not found. Skipping memory allocation test.")
        return

    # Configure memory growth to prevent TensorFlow from allocating all GPU memory at once
    try:
        for gpu in gpus:
            tf.config.experimental.set_memory_growth(gpu, True)
    except RuntimeError as e:
        print(e)

    # Parameters mimicking the scale of the original test
    batch_size = 4
    max_time = 50
    num_classes = 20
    iterations = 100

    # Create dummy input (Parameter equivalent)
    # Shape: [batch_size, max_time, num_classes]
    dummy_input = tf.Variable(tf.random.normal([batch_size, max_time, num_classes]), dtype=tf.float32)

    # Create dummy labels (SparseTensor required by ctc_loss)
    indices = [[0, 0], [0, 1], [1, 0], [1, 1], [2, 0], [3, 0]]
    values = [1, 2, 3, 4, 5, 6]
    dense_shape = [batch_size, 2]
    labels = tf.SparseTensor(indices=indices, values=values, dense_shape=dense_shape)

    # Sequence length
    sequence_length = tf.constant([max_time, max_time, max_time, max_time], dtype=tf.int32)

    print(f"Starting test for {iterations} iterations...")

    for i in range(iterations):
        with tf.GradientTape() as tape:
            # Call the API under test: tf.compat.v1.nn.ctc_loss
            # This replaces the torch.utils.checkpoint.checkpoint + MyOp logic
            loss = tf.compat.v1.nn.ctc_loss(
                labels=labels,
                inputs=dummy_input,
                sequence_length=sequence_length,
                preprocess_collapse_repeated=False,
                ctc_merge_repeated=True,
                ignore_longer_outputs_than_inputs=False,
                time_major=True
            )
            
            # Reduce loss to a scalar for backward pass
            total_loss = tf.reduce_mean(loss)

        # Compute gradients (equivalent to .backward())
        grads = tape.gradient(total_loss, dummy_input)
        
        # In the original PyTorch code, gradients are explicitly cleared. 
        # In TF eager mode, gradients are tied to the Tape context, but we can 
        # verify memory usage here.
        
        # Monitor Memory
        # Note: 'current' refers to current memory allocated by the process on the device
        mem_info = tf.config.experimental.get_memory_info('GPU:0')
        current_mem_mb = mem_info['current'] / (1024 ** 2)
        
        if i % 10 == 0 or i == 0:
            print(f"Iteration {i}: {current_mem_mb:.2f} MiB")

    print("Test completed.")

if __name__ == "__main__":
    test_ctc_loss_memory_leak()
