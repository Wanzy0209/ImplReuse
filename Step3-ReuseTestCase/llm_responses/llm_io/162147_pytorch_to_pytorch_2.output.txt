import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class FastLearnedCellCumsum(nn.Module):
    def __init__(self, D_in, H, D_out,
                 L_w1=12, L_w2=12, L_b2=12,
                 k1=3, k2=3, k3=3, tau1=1.0, tau2=1.0, tau3=1.0,
                 d_addr=64,           # address bottleneck
                 learn_addr=False,
                 learn_tape_w1=True, learn_tape_w2=True, learn_tape_b2=True):
        super().__init__()
        self.D_in, self.H, self.D_out = D_in, H, D_out

        # keep sizes as plain Python ints (compile-friendly)
        self.L_w1, self.L_w2, self.L_b2 = int(L_w1), int(L_w2), int(L_b2)
        self.k1, self.k2, self.k3 = int(k1), int(k2), int(k3)
        self.t1, self.t2, self.t3 = float(tau1), float(tau2), float(tau3)

        # address bottleneck
        self.P = nn.Linear(D_in, d_addr, bias=False)
        if not learn_addr:
            for p in self.P.parameters():
                p.requires_grad = False
            with torch.no_grad():
                nn.init.normal_(self.P.weight, std=1.0/math.sqrt(D_in))

        def init_U(L, d, learn):
            U = torch.randn(L, d)
            U = U - U.mean(dim=1, keepdim=True)
            U = U / (U.norm(dim=1, keepdim=True) + 1e-8)
            return nn.Parameter(U, requires_grad=learn)

        # three unembeddings in addr-space
        self.U1 = init_U(self.L_w1, d_addr, learn_addr)
        self.U2 = init_U(self.L_w2, d_addr, learn_addr)
        self.U3 = init_U(self.L_b2, d_addr, learn_addr)

        # value tapes
        self.W1 = nn.Parameter(F.normalize(torch.randn(self.L_w1, H, D_in), dim=(1,2)), requires_grad=learn_tape_w1)
        self.W2 = nn.Parameter(F.normalize(torch.randn(self.L_w2, D_out, H), dim=(1,2)), requires_grad=learn_tape_w2)
        self.b2 = nn.Parameter(F.normalize(torch.randn(self.L_b2, D_out), dim=1),      requires_grad=learn_tape_b2)

        self.act = nn.GELU()

    @staticmethod
    def _tk(z: torch.Tensor, k: int, tau: float):
        topv, topi = torch.topk(z, k, dim=1, largest=True, sorted=False)
        w = torch.softmax(topv / (tau + 1e-8), dim=1)
        return topi, w

    def _address(self, x_addr: torch.Tensor):
        # fused logits for all three heads
        U_pack = torch.cat([self.U1, self.U2, self.U3], dim=0)          # [Ltot, d]
        Z = x_addr @ U_pack.t()                                         # [N, Ltot]
        s1, s2, s3 = self.L_w1, self.L_w2, self.L_b2
        z1, z2, z3 = torch.split( Z, (s1, s2, s3), dim=1)
        i1, w1 = FastLearnedCellCumsum._tk(z1, self.k1, self.t1)
        i2, w2 = FastLearnedCellCumsum._tk(z2, self.k2, self.t2)
        i3, w3 = FastLearnedCellCumsum._tk(z3, self.k3, self.t3)
        return (i1, w1), (i2, w2), (i3, w3)

    @staticmethod
    def _apply_mixture(x_flat, topi, weights, W):
        """
        Adapted to use torch.cumsum.
        Original logic likely used index_select/gather followed by weighted sum.
        Here we use index_select to gather, then torch.cumsum to accumulate 
        the contributions across the k dimension before applying weights.
        """
        N, k = topi.shape
        L, out_dim, in_dim = W.shape

        # Select rows from W based on indices topi
        # This uses the original API (torch.index_select) to retrieve data
        flat_indices = topi.view(-1)
        selected_W = torch.index_select(W, 0, flat_indices) # [N*k, out, in]
        selected_W = selected_W.view(N, k, out_dim, in_dim)

        # Adaptation: Use torch.cumsum along the k dimension (dim=1)
        # This accumulates the weight matrices cumulatively.
        # This tests the similar API torch.cumsum in the graph.
        cum_W = torch.cumsum(selected_W, dim=1) # [N, k, out, in]

        # Apply weights
        # weights: [N, k] -> [N, k, 1, 1]
        w_exp = weights.unsqueeze(-1).unsqueeze(-1)

        # Calculate output: Weighted sum of the cumulatively selected matrices
        # [N, k, out, in] * [N, k, 1, 1] -> [N, k, out, in]
        weighted = cum_W * w_exp
        # Sum over k
        out = weighted.sum(dim=1) # [N, out, in]

        # Contract with x_flat
        # x_flat: [N, in] -> [N, 1, in]
        x_exp = x_flat.unsqueeze(1)
        # [N, out, in] * [N, 1, in] -> [N, out]
        result = (out * x_exp).sum(dim=-1)
        return result

    def forward(self, x):
        x_addr = self.P(x)
        (i1, w1), (i2, w2), (i3, w3) = self._address(x_addr)

        # Apply mixtures
        h = self._apply_mixture(x, i1, w1, self.W1)
        h = self.act(h)

        out = self._apply_mixture(h, i2, w2, self.W2)

        # Bias handling (simplified for test case)
        # Using index_select for bias as well, but could be adapted
        bias = torch.index_select(self.b2, 0, i3.view(-1)).view(x.size(0), self.k3, -1)
        bias = (bias * w3.unsqueeze(-1)).sum(dim=1)
        out = out + bias

        return out

# Test Case
def test_fast_learned_cell_cumsum():
    D_in, H, D_out = 128, 64, 10
    batch_size = 4
    
    model = FastLearnedCellCumsum(D_in, H, D_out)
    model.train()
    
    # Create dummy input
    x = torch.randn(batch_size, D_in)
    
    # Forward pass
    output = model(x)
    
    # Assertions
    assert output.shape == (batch_size, D_out), f"Expected shape {(batch_size, D_out)}, got {output.shape}"
    assert not torch.isnan(output).any(), "Output contains NaNs"
    assert not torch.isinf(output).any(), "Output contains Infs"
    
    # Check backward pass
    loss = output.sum()
    loss.backward()
    
    # Check if gradients exist for parameters that require grad
    for name, param in model.named_parameters():
        if param.requires_grad:
            assert param.grad is not None, f"No gradient for {name}"
            assert not torch.isnan(param.grad).any(), f"NaN gradient for {name}"

    print("Test passed.")

if __name__ == "__main__":
    test_fast_learned_cell_cumsum()
