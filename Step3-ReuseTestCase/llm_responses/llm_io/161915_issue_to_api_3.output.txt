import torch
import torch.nested

def assert_nested_tensor_sharable(nested_tensor):
    """
    Validates that the NestedTensor meets the requirements for share_memory_().
    This function mirrors the defensive checking pattern found in 
    tf.compat.v1.train.assert_global_step (checking type and properties).
    """
    # Check 1: Type validation
    # Corresponds to: isinstance(global_step_tensor, variables.Variable) or ...
    if not isinstance(nested_tensor, torch.nested.NestedTensor):
        raise TypeError(f"Input must be a NestedTensor, got {type(nested_tensor)}")

    # Check 2: Property validation (Contiguity)
    # Corresponds to: checking global_step_tensor.dtype or shape
    # share_memory_ generally requires the tensor to be contiguous.
    if not nested_tensor.is_contiguous():
        raise ValueError("NestedTensor must be contiguous to share memory.")

    # Check 3: Device validation
    # share_memory_ is typically a CPU-specific operation.
    if nested_tensor.device.type != 'cpu':
        raise RuntimeError("share_memory_ is only supported for CPU tensors")

def test_nested_tensor_share_memory():
    """
    Test case for Issue 161915: share_memory_support for NestedTensor.
    Reproduces the segmentation fault scenario while applying validation
    logic similar to tf.compat.v1.train.assert_global_step.
    """
    # Setup from the original bug report
    a = torch.randn(3)
    b = torch.randn(5)
    nt = torch.nested.nested_tensor([a, b], layout=torch.jagged)
    
    # Leverage the validation pattern from the similar API
    # to ensure the tensor is in a valid state before the crash.
    assert_nested_tensor_sharable(nt)
    
    # The operation that causes the segmentation fault in the bug report.
    # Note: If the bug is present, this will cause a Segmentation Fault.
    # If the bug is fixed, this should succeed or raise a proper Python exception.
    try:
        nt.share_memory_()
        print("Test passed: share_memory_ executed successfully.")
    except Exception as e:
        # Catching Python exceptions to differentiate from a hard segfault.
        print(f"Test raised an exception: {e}")

if __name__ == "__main__":
    test_nested_tensor_share_memory()
