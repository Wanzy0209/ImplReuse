```python
import tensorflow as tf
import numpy as np

# Constants
BATCH_SIZE = 4
NUM_CLASSES = 10
LEARNING_RATE = 0.01

# Define a simple ResNet-like model structure for the test
inputs = tf.keras.Input(shape=(224, 224, 3))
x = tf.keras.layers.Conv2D(64, 7, strides=2, padding='same')(inputs)
x = tf.keras.layers.BatchNormalization()(x)
x = tf.keras.layers.Activation('relu')(x)
x = tf.keras.layers.MaxPooling2D(3, strides=2, padding='same')(x)

# A residual block to mimic ResNet structure
shortcut = x
x = tf.keras.layers.Conv2D(64, 3, padding='same')(x)
x = tf.keras.layers.BatchNormalization()(x)
x = tf.keras.layers.Activation('relu')(x)
x = tf.keras.layers.Conv2D(64, 3, padding='same')(x)
x = tf.keras.layers.BatchNormalization()(x)
x = tf.keras.layers.Add()([x, shortcut])
x = tf.keras.layers.Activation('relu')(x)

x = tf.keras.layers.GlobalAveragePooling2D()(x)
outputs = tf.keras.layers.Dense(NUM_CLASSES)(x)

model = tf.keras.Model(inputs=inputs, outputs=outputs)

optimizer = tf.keras.optimizers.SGD(learning_rate=LEARNING_RATE)
criterion = tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True)

# Adaptation: Using tf.compat.v1.name_scope to wrap the training logic
# This mirrors the structure of the original PyTorch test where torch.compile
# wrapped the training function.
def train(images, labels):
    with tf.compat.v1.name_scope("train_step"):
        with tf.GradientTape() as tape:
            outputs = model(images, training=True)
            loss = criterion(labels, outputs)
        
        # Equivalent to loss.backward()
        gradients = tape.gradient(loss, model.trainable_variables)
        # Equivalent to optimizer.step()
        optimizer.apply_gradients(zip(gradients, model.trainable_variables))
        return loss

# Generate random data
images = np.random.randn(BATCH_SIZE, 224, 224, 3).astype(np.float32)
labels = np.random.randint(0, NUM_CLASSES, (BATCH_SIZE,))

# Run the test
loss = train(images, labels)

# Verify execution
assert loss is not None
print(f"Test completed successfully. Loss: {loss.numpy()}")
```
