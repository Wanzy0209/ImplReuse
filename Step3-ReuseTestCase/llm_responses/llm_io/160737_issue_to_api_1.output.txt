import tensorflow as tf

def test_global_variables_initializer(context_type):
    """
    Tests tf.compat.v1.global_variables_initializer in different execution contexts.
    This mirrors the logic of testing torch.index_select on different devices (CPU vs MPS).
    We use a scalar variable to reflect the 'scalar tensor' aspect of the original bug.
    """
    try:
        if context_type == "eager":
            # Analogous to CPU test in the original bug
            # In eager mode, variables are initialized on creation, 
            # but the initializer op should still be accessible.
            var = tf.Variable(1.0, dtype=tf.float32, name="scalar_var")
            init_op = tf.compat.v1.global_variables_initializer()
            print(f"global_variables_initializer test succeeds for context: {context_type}. Op type: {type(init_op)}")
            
        elif context_type == "graph":
            # Analogous to MPS test in the original bug (different execution path)
            with tf.compat.v1.Graph().as_default():
                with tf.compat.v1.Session() as sess:
                    # Create a scalar variable
                    var = tf.Variable(1.0, dtype=tf.float32, name="graph_scalar_var")
                    init_op = tf.compat.v1.global_variables_initializer()
                    
                    # Run the initializer
                    sess.run(init_op)
                    
                    # Verify the scalar variable is initialized
                    val = sess.run(var)
                    print(f"global_variables_initializer test succeeds for context: {context_type}. Value: {val}")
                    
    except Exception as e:
        print(f"global_variables_initializer test fails for context: {context_type}: {e}")

# Test in Eager mode (default in TF 2.x)
test_global_variables_initializer(context_type="eager")

# Test in Graph mode (using compat.v1)
test_global_variables_initializer(context_type="graph")
