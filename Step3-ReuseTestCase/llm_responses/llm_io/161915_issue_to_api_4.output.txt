import unittest
import tensorflow as tf

class TestCallbackListSimilarity(unittest.TestCase):
    """
    Test case generated based on Issue 161915 (torch.nested.nested_tensor share_memory_ segfault).
    This test adapts the logic to tf.keras.callbacks.CallbackList to ensure
    initialization with a list of objects and subsequent method calls do not cause crashes.
    """

    def test_callback_list_initialization_and_method_call(self):
        """
        Original Bug Logic:
        1. Create a list of tensors (a, b).
        2. Initialize NestedTensor with the list.
        3. Call share_memory_() -> Segmentation fault.

        Adapted Test Logic for tf.keras.callbacks.CallbackList:
        1. Create a list of callbacks.
        2. Initialize CallbackList with the list.
        3. Call a method (set_params) -> Should succeed without crashing.
        """
        # 1. Create a list of items (analogous to torch.randn(3), torch.randn(5))
        # We use simple Lambda callbacks to create distinct callback instances.
        cb_a = tf.keras.callbacks.LambdaCallback(on_epoch_end=lambda epoch, logs: None)
        cb_b = tf.keras.callbacks.LambdaCallback(on_epoch_end=lambda epoch, logs: None)

        # 2. Initialize the container with the list (analogous to torch.nested.nested_tensor([a, b]))
        # The similar API pattern involves initializing with a list of callbacks.
        # We also mirror the keyword argument usage from the original bug (layout=...) 
        # by using add_history/add_progbar here.
        callback_list = tf.keras.callbacks.CallbackList(
            callbacks=[cb_a, cb_b],
            add_history=False,
            add_progbar=False
        )

        # 3. Call a method on the container (analogous to nt.share_memory_())
        # In the original bug, share_memory_() caused a segmentation fault.
        # Here we ensure that calling a method on the initialized list works correctly.
        # set_params is a standard method called during model training setup.
        try:
            callback_list.set_params({'epochs': 10, 'batch_size': 32})
            # If we reach here, no segfault occurred and the API is stable.
            self.assertTrue(True)
        except Exception as e:
            self.fail(f"CallbackList method call failed unexpectedly: {e}")

if __name__ == '__main__':
    unittest.main()
