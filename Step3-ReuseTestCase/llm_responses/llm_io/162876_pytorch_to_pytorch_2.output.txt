```python
import torch

# Test case for torch.index_select
# Adapted from the torch.aminmax issue which highlighted a problem with the 
# return type representation not being executable code.
# torch.index_select returns a standard Tensor, so we verify its functionality.

def test_torch_index_select():
    # Setup input and index
    input_tensor = torch.tensor([[1, 2], [3, 4], [5, 6]])
    index_tensor = torch.tensor([0, 2])
    
    # Call the API
    result = torch.index_select(input_tensor, 0, index_tensor)
    
    # Define expected output
    expected = torch.tensor([[1, 2], [5, 6]])
    
    # Verify the result matches the expected tensor
    assert torch.equal(result, expected), f"Expected {expected}, but got {result}"
    
    # Verify the return type is a standard Tensor
    assert isinstance(result, torch.Tensor)

if __name__ == "__main__":
    test_torch_index_select()
    print("Test passed.")
```
