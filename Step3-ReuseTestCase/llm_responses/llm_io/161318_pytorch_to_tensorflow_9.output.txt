import tensorflow as tf

def test_data_dependent_slice():
    """
    Adapted test case for tf.name_scope based on the PyTorch issue.
    The original issue involved a crash in torch.compile when performing
    a data-dependent slice. This test verifies that TensorFlow handles
    the equivalent logic correctly within a tf.name_scope context.
    """
    # Setup inputs mirroring the PyTorch code
    # mask = (torch.arange(512) < 8).unsqueeze(0).cuda()
    mask = tf.expand_dims(tf.cast(tf.range(512) < 8, tf.int32), 0)
    
    # hidden = torch.randn((1, 512, 4096)).cuda()
    hidden = tf.random.normal((1, 512, 4096))

    # Define the function using tf.function (analogous to torch.compile)
    # and wrapping the logic in tf.name_scope as requested.
    @tf.function
    def fn(encoder_attention_mask, encoder_hidden_states):
        # encoder_hidden_states = encoder_hidden_states.new_zeros([1, 512, 3072])
        encoder_hidden_states = tf.zeros([1, 512, 3072], dtype=encoder_hidden_states.dtype)

        # Use tf.name_scope to group operations
        with tf.name_scope("dynamic_slice_operations"):
            # text_len = encoder_attention_mask.sum().item()
            # In TensorFlow, we keep the result as a Tensor (int32) for graph compatibility
            text_len = tf.cast(tf.reduce_sum(encoder_attention_mask), tf.int32)

            # encoder_hidden_states = encoder_hidden_states[:, :text_len]
            # Perform the data-dependent slice
            encoder_hidden_states = encoder_hidden_states[:, :text_len]

        return encoder_hidden_states

    # Execute the function
    result = fn(mask, hidden)

    # Verify the behavior
    # The mask sum is 8, so the slice should result in a shape of (1, 8, 3072)
    expected_shape = (1, 8, 3072)
    assert result.shape == expected_shape, f"Expected shape {expected_shape}, but got {result.shape}"
    
    print("Test passed: Data-dependent slice handled correctly within tf.name_scope.")

if __name__ == "__main__":
    test_data_dependent_slice()
