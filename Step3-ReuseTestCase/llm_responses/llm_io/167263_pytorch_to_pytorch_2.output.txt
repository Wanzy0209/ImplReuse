```python
import torch
import torch.nn.functional as F

# Setup dimensions for 3D convolution (Batch, Channel, Depth, Height, Width)
B, C, D, H, W = 2, 4, 5, 5, 5

x = torch.randn(B, C, D, H, W, requires_grad=True)

# Create a view to mimic the original scenario where stride might be an issue
# Permuting dimensions creates a non-contiguous tensor with specific strides
values_view = x.permute(0, 2, 1, 3, 4) # Shape becomes (B, D, C, H, W)
values_view.retain_grad()

# Define weights for conv_transpose3d
# Input channels must match values_view.shape[1] (D)
# Output channels can be arbitrary
in_channels = values_view.shape[1]
out_channels = 4
kernel_size = 3
weight = torch.randn(in_channels, out_channels, kernel_size, kernel_size, kernel_size)

def log_grad(name):
    def hook(grad):
        print(f"{name} hook - shape: {grad.shape}, stride: {grad.stride()}")
        return grad
    return hook

print("forward strides", values_view.shape, values_view.stride())
values_view.register_hook(log_grad("values_view"))

# Replace einsum with conv_transpose3d
result = F.conv_transpose3d(values_view, weight)

result.backward(torch.ones_like(result))
```
