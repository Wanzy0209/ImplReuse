import torch
import torch._inductor.config as inductor_config

def f(x, y):
    x.copy_(x.flip(1))
    y = y.sum(dim=1, keepdim=True) + y

x = torch.randn(20, 1024 * 1024, device="cuda")
x_copy = x.clone()

opt_f = torch.compile(f)
ref = f(x, y)
