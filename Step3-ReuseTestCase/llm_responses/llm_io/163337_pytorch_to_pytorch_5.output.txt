```python
import torch
import torch.nn as nn
from torch.utils.checkpoint import checkpoint_sequential

# Define a simple sequential model to be used with checkpointing
# This replaces the C++ extension sources from the original context
model = nn.Sequential(
    nn.Linear(10, 20),
    nn.ReLU(),
    nn.Linear(20, 10)
)

# Create a dummy input tensor
input_tensor = torch.randn(2, 10, requires_grad=True)

# Call the similar API: torch.utils.checkpoint.checkpoint_sequential
# We divide the model into 2 segments to test the checkpointing mechanism
# This replaces the torch.utils.cpp_extension.load call
output = checkpoint_sequential(model, segments=2, input=input_tensor)

# Verify the output shape is correct
assert output.shape == (2, 10), f"Expected output shape (2, 10), but got {output.shape}"

# Verify gradients are computed correctly (checkpointing should preserve the graph)
loss = output.sum()
loss.backward()
assert input_tensor.grad is not None, "Gradients were not computed"
assert input_tensor.grad.shape == (2, 10), "Gradient shape mismatch"

print("Test passed for torch.utils.checkpoint.checkpoint_sequential")
```
