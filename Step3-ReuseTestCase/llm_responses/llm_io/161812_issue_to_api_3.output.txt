import tensorflow as tf

def test_gather_jagged_dim_0():
    """
    Test case for tf.keras.backend.gather with a RaggedTensor (jagged) along dimension 0.
    This test is derived from a PyTorch issue (ID: 161812) where th.cat crashed on jagged 
    tensors along dimension 0 due to argument/schema handling issues.
    
    We verify that tf.keras.backend.gather handles the analogous operation (gathering along 
    the batch/first dimension) correctly for RaggedTensors.
    """
    # Create a jagged tensor (RaggedTensor) analogous to th.nested.nested_tensor
    # PyTorch: [th.ones(3, 2, 3), th.ones(4, 2, 3)]
    # TF: RaggedTensor with row lengths 3 and 4, inner shape (2, 3)
    values = tf.ones((7, 2, 3), dtype=tf.float32)
    row_splits = tf.constant([0, 3, 7], dtype=tf.int64)
    x = tf.RaggedTensor.from_row_splits(values, row_splits)

    # Test tf.keras.backend.gather along dimension 0 (default)
    # The original bug involved a crash when operating along dim 0 with jagged tensors.
    # We select indices [0, 1] which corresponds to the two jagged rows.
    indices = [0, 1]
    
    try:
        # Call the API
        result = tf.keras.backend.gather(x, indices)
        
        # Assertions to verify correctness
        assert isinstance(result, tf.RaggedTensor), "Result should be a RaggedTensor"
        assert result.shape[0] == 2, "Result should have 2 rows (gathered indices)"
        assert result.shape[1] is None, "Inner dimension should remain ragged (None)"
        assert result.shape[2] == 2, "Dimension 2 should be 2"
        assert result.shape[3] == 3, "Dimension 3 should be 3"
        
        print("Test passed: tf.keras.backend.gather handled RaggedTensor along dim 0 successfully.")
        
    except Exception as e:
        print(f"Test failed with error: {e}")
        raise

if __name__ == "__main__":
    test_gather_jagged_dim_0()
