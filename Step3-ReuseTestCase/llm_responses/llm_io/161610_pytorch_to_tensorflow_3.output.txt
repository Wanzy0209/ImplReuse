```python
import tensorflow as tf
from typing import NamedTuple

# Define the NamedTuple structure
class MyNamedTuple(NamedTuple):
    first: tf.Tensor
    second: tf.Tensor

# Subclass to allow dynamic attributes.
# Note: Standard Python NamedTuples are immutable. We define __slots__ = () 
# in the subclass to allow adding dynamic attributes at runtime, similar to 
# the behavior expected in the PyTorch bug report.
class MyNamedTupleSubclass(MyNamedTuple):
    __slots__ = ()

def fn(tup: MyNamedTuple) -> MyNamedTuple:
    extra_info = tf.constant(4.0)
    tup.extra_info = extra_info  # Add dynamic attribute
    return tup

# Enable eager execution using the API under test
tf.compat.v1.enable_eager_execution()

print("\nTesting NamedTuple with __setattr__ in TensorFlow Eager Mode:")
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))
setattr_result = fn(extended_tup)

# Verify the behavior
try:
    # In eager mode, Python object semantics are preserved, so we expect the attribute to exist.
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info.numpy()}")
    assert hasattr(setattr_result, 'extra_info'), "Attribute 'extra_info' not found!"
    assert setattr_result.extra_info.numpy() == 4.0
    print("Test Passed: Dynamic attribute persisted in eager execution.")
except AttributeError as e:
    print(f"Test Failed: {e}")
```
