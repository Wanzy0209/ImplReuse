import torch
import torch.nn as nn

device = torch.device("cuda")

def test_nllloss_nan():
    # Large input tensor dimensions matching the original bug report
    N, C, H, W = 84, 64, 512, 960

    # Case 1: bfloat16 + channels_last
    # NLLLoss expects log-probabilities. Using randn to stress the kernel.
    x = torch.randn(N, C, H, W, dtype=torch.bfloat16, device=device)

    # Target tensor (class indices) for NLLLoss
    target = torch.randint(0, C, (N, H, W), device=device)

    # Convert to NHWC channels_last layout
    x = x.to(memory_format=torch.channels_last)

    # Uncommenting the line below might avoid NaNs if the bug is layout-specific
    # x = x.contiguous()

    print(f"Input tensor: contiguous={x.is_contiguous()}, channels_last={x.is_contiguous(memory_format=torch.channels_last)}")
    print(f"Input stride: {x.stride()}")

    criterion = nn.NLLLoss().to(device)
    loss = criterion(x, target)

    print(f"Loss value: {loss.item()}")
    print(f"Loss contains NaN? {torch.isnan(loss).item()}")
    print(f"Loss contains Inf? {torch.isinf(loss).item()}")

    if torch.isnan(loss) or torch.isinf(loss):
        print("Detected NaNs or Infs in NLLLoss output!")

if __name__ == "__main__":
    test_nllloss_nan()
