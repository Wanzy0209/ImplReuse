```python
import tensorflow as tf

# Set seed for reproducibility
tf.random.set_seed(0)

# Define parameters adapted to the Multinomial API
# Mimicking the specific integer configuration style of the original bug
total_count = 10
# Create a probability tensor (batch of vectors)
# Original input shape was (4, 6, 7). We use (4, 6) for probs to represent a batch of 4 with 6 classes.
probs = tf.random.uniform((4, 6), minval=0.1, maxval=1.0)
probs = probs / tf.reduce_sum(probs, axis=-1, keepdims=True) # Normalize to sum to 1

# Instantiate the distribution
dist = tf.compat.v1.distributions.Multinomial(total_count=total_count, probs=probs)

# Run on CPU
with tf.device("/CPU:0"):
    out_cpu = dist.sample()

# Check for GPU availability and run on GPU
gpus = tf.config.list_physical_devices('GPU')
if gpus:
    with tf.device("/GPU:0"):
        out_gpu = dist.sample()

    # Compare CPU and GPU outputs
    # Note: With the global seed set, samples should be identical across devices
    if not tf.reduce_all(tf.equal(out_cpu, out_gpu)).numpy():
        print("Output does not match!")
        print("CPU Output:")
        print(out_cpu.numpy())
        print("GPU Output:")
        print(out_gpu.numpy())
    else:
        print("Outputs match.")
else:
    print("GPU not available. Only CPU output generated:")
    print(out_cpu.numpy())
```
