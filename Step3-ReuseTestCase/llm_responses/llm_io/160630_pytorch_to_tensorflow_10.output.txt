```python
import tensorflow as tf
import numpy as np

def test_quantized_broadcast_arrays():
    # Create a float tensor
    float_input = tf.constant([1.0, 2.0, 3.0])
    
    # Quantize the tensor (TensorFlow equivalent of torch.quantize_per_tensor)
    # tf.quantization.quantize returns a tuple: (quantized_tensor, min, max)
    quant_input, _, _ = tf.quantization.quantize(float_input, min_range=0.0, max_range=10.0, dtype=tf.qint8)
    
    print("Quantized input:", quant_input)
    
    # Attempt to broadcast the quantized tensor
    # This adapts the original logic of applying an operation to a quantized tensor
    # to the specified TensorFlow API.
    try:
        # broadcast_arrays can take a single argument, returning it in a list
        out_arrays = tf.experimental.numpy.broadcast_arrays(quant_input)
        print("Output arrays:", out_arrays)
    except Exception as e:
        print(f"Error occurred: {e}")

if __name__ == "__main__":
    test_quantized_broadcast_arrays()
```
