```python
import torch

def get_sample_inputs():
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    # Create a sample input tensor
    x = torch.randn(4, 10, device=device)
    return (x,)

def main():
    inputs = get_sample_inputs()
    x = inputs[0]
    device = x.device

    if not torch.cuda.is_available():
        print("CUDA not available, skipping test")
        return

    # Eager execution
    # We use torch.all to check if all elements are greater than 0
    original_output = torch.all(x > 0)
    print('Original output:', original_output.item())

    # CUDA Graph capture
    # We need a static output tensor to store the result of the graph capture
    static_output = torch.empty((), dtype=torch.bool, device=device)

    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        # Adaptation: Replace torch.compile(model) with torch.all operation
        # We perform the reduction and copy the result to the static output tensor
        res = torch.all(x > 0)
        static_output.copy_(res)

    # Replay the graph
    graph.replay()
    
    print('Graph output:', static_output.item())
    
    # Verify that the output matches the eager execution
    assert original_output.item() == static_output.item(), "Output mismatch between original and graph execution"

if __name__ == "__main__":
    main()
```
