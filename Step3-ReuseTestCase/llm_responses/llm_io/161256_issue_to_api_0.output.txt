import tensorflow as tf
import pytest

# The original issue describes a segmentation fault when performing float64 matmul on GPU.
# The similar API, tf.errors.OperatorNotAllowedInGraphError, is raised when unsupported 
# operations are used in Graph execution (tf.function).
# This test translates the PyTorch reproduction logic to TensorFlow, executing it 
# within a tf.function (Graph context) to verify that float64 matmul is supported 
# and does not raise the specific error associated with the similar API.

def test_float64_matmul_graph_execution():
    # Check for GPU availability to match the original issue's "device='cuda'"
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        pytest.skip("Test requires GPU to match original issue conditions")

    # Use tf.function to enter Graph execution context (relevant to the similar API)
    @tf.function
    def float64_matmul_op():
        # Reproduce the bug logic: float64 (double) tensors
        # PyTorch: torch.randn(1000, 1000, device="cuda", dtype=torch.float64)
        x = tf.random.normal((1000, 1000), dtype=tf.float64)
        y = tf.random.normal((1000, 1000), dtype=tf.float64)

        # PyTorch: z = x @ y
        z = tf.matmul(x, y)

        # PyTorch: print(z[0, 0])
        return z[0, 0]

    # Execute the operation and verify behavior
    try:
        result = float64_matmul_op()
        # Assert that the operation completed and returned the correct dtype
        assert result.dtype == tf.float64
        assert result.shape == ()
    except tf.errors.OperatorNotAllowedInGraphError as e:
        # Leverage the similar API to catch if the operation is disallowed in Graph mode
        pytest.fail(f"Float64 matmul raised OperatorNotAllowedInGraphError: {e}")
