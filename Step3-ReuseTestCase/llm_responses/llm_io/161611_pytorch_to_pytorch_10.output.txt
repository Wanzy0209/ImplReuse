import torch

def test_qr_redundant_dtype_conversion():
    """
    Test case adapted from Issue 161611 to verify similar behavior in torch.qr.
    
    This test verifies that calling .to(dtype) on the result of torch.linalg.qr
    without assigning the result is ineffective, mimicking the redundant 
    dtype conversion bug found in the scaled_dot_product_attention docstring.
    """
    # Setup: Create a matrix with a specific dtype (float32)
    input_dtype = torch.float32
    A = torch.randn(4, 4, dtype=input_dtype)

    # Call the similar API: torch.linalg.qr
    Q, R = torch.linalg.qr(A)

    # Scenario 1: Redundant conversion (same dtype)
    # Mimics: attn_bias.to(query.dtype) where attn_bias is already query.dtype
    # Expected: No change, and operation is effectively a no-op.
    Q.to(input_dtype)
    
    assert Q.dtype == input_dtype, \
        "Q.dtype should remain float32. The unassigned .to() call should have no effect."

    # Scenario 2: Ineffective conversion (different dtype, unassigned)
    # Mimics: The general issue of calling .to() without assignment.
    target_dtype = torch.float64
    Q.to(target_dtype)

    assert Q.dtype == input_dtype, \
        "Q.dtype should still be float32. The result of .to(float64) was not assigned."

    # Scenario 3: Correct usage (assignment)
    # Verify that assignment works as expected.
    Q_converted = Q.to(target_dtype)
    
    assert Q_converted.dtype == target_dtype, \
        "Assigned Q_converted should be float64."
    assert Q.dtype == input_dtype, \
        "Original Q should remain unchanged (float32)."

if __name__ == "__main__":
    test_qr_redundant_dtype_conversion()
    print("Test passed.")
