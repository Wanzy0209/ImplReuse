import torch
import gc

def test_torch_prod_compile_memory_leak():
    """
    Test case to verify if torch.compile causes a memory leak 
    when compiling a function using torch.prod.
    Adapted from Issue 165407.
    """
    
    # Define a function using the similar API: torch.prod
    # This replaces the original call site (flash_attn_varlen_func)
    def prod_func(x):
        return torch.prod(x)

    # Compile the function using torch.compile
    compiled_func = torch.compile(prod_func)

    def get_tensor_count():
        gc.collect()
        return sum(1 for obj in gc.get_objects() if isinstance(obj, torch.Tensor))

    print("Step | Tensors")
    print("-------------")

    # Run a loop to check for memory leaks, similar to the original bug report
    for step in range(300):
        # Create input data
        input_tensor = torch.randn(32, 128)
        
        # Execute the compiled function
        output = compiled_func(input_tensor)
        
        # Clean up references to allow garbage collection
        del input_tensor, output
        
        # Log status every 50 steps
        if step % 50 == 0:
            num_tensors = get_tensor_count()
            print(f"{step:4d} | {num_tensors}")

if __name__ == "__main__":
    test_torch_prod_compile_memory_leak()
