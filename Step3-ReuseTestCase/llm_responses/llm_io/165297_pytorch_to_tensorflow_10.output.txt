```python
import tensorflow as tf
import numpy as np

def test_dirichlet_large_tensor():
    """
    Adapted test case for tf.compat.v1.distributions.Dirichlet.
    Original issue: MaxPool2d with channels_last + bfloat16 on CUDA produces NaNs for large tensors.
    This test checks for numerical stability (NaNs/Infs) when using Dirichlet distribution
    with large tensors and bfloat16 dtype on GPU.
    """
    
    # Check for GPU availability
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("Test skipped: No GPU available.")
        return

    print(f"Running on GPU: {gpus[0].name}")

    # Adaptation of "Large input tensor"
    # PyTorch dimensions: N, C, H, W = 84, 64, 512, 960 (~2.6B elements)
    # We create a large concentration tensor for Dirichlet to stress memory.
    # Using a large batch and event size.
    batch_size = 100
    event_size = 1000000  # 100M elements total

    # Case 1: bfloat16 (matching the bug report's trigger)
    # Note: Dirichlet requires strictly positive concentrations.
    concentration = tf.random.uniform((batch_size, event_size), minval=0.1, maxval=1.0, dtype=tf.bfloat16)

    print(f"Input tensor: shape={concentration.shape}, dtype={concentration.dtype}")

    # Initialize Dirichlet distribution
    # This is the equivalent operation to the MaxPool layer instantiation
    dirichlet = tf.compat.v1.distributions.Dirichlet(concentration, validate_args=False)

    # Perform the operation (sampling)
    # This is the equivalent to y = pool(x)
    samples = dirichlet.sample()

    # Check for NaNs
    has_nan = tf.reduce_any(tf.math.is_nan(samples))
    has_inf = tf.reduce_any(tf.math.is_inf(samples))

    print(f"Output contains NaN? {has_nan.numpy()}")
    print(f"Output contains Inf? {has_inf.numpy()}")

    # Basic stats
    print(f"Stats: min={tf.reduce_min(samples).numpy()}, max={tf.reduce_max(samples).numpy()}")

    if has_nan or has_inf:
        print("Detected NaNs or Infs in Dirichlet output!")
    else:
        print("Test passed: No NaNs or Infs detected.")

if __name__ == "__main__":
    test_dirichlet_large_tensor()
```
