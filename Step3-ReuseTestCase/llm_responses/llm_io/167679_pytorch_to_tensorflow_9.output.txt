# Adapted Test Case for tf.compat.v1.math.log1p
# This test verifies the functionality of the TensorFlow math API on the specific OS/Arch context (macOS 26, arm64).

# 1. Setup Environment
conda create -n tf_test python=3.11 -y
conda activate tf_test
python -m pip install tensorflow

# 2. Run Test
python - << 'EOF'
import sys
import platform
import tensorflow as tf

print("Python:", sys.version)
print("Executable:", sys.executable)
print("TensorFlow:", tf.__version__)
print("macOS:", platform.mac_ver()[0])
print("platform:", platform.platform())
print("arch:", platform.machine())

# Original bug logic: Check if a backend feature is available on macOS 26 (Tahoe).
# Adapted logic: Check if the similar API (tf.compat.v1.math.log1p) is available and functional.

try:
    # Check availability by attempting to import and access the function
    api_func = tf.compat.v1.math.log1p
    print("API Accessible: True")
    
    # Verify behavior with a standard input
    input_val = 1.0
    # log1p(1) = log(2) approx 0.693147...
    result = api_func(input_val)
    
    print(f"Input: {input_val}")
    print(f"Result: {result.numpy()}")
    
    # Assertion to ensure correct calculation
    expected = tf.math.log(2.0).numpy()
    assert abs(result.numpy() - expected) < 1e-6, f"Calculation error: {result} != {expected}"
    
    print("Test Status: PASSED (API is available and functional)")

except AttributeError as e:
    print("API Accessible: False")
    print(f"Test Status: FAILED - {e}")
except Exception as e:
    print(f"Test Status: FAILED - {e}")
EOF
