import tensorflow as tf
import numpy as np

# This test case checks if tf.distribute.NcclAllReduce handles dynamic shapes,
# analogous to the PyTorch issue where Tensor.fill_diagonal_ failed with dynamic shapes.

def test_nccl_all_reduce_dynamic_shapes():
    # NcclAllReduce is specific to NVIDIA GPUs
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("Skipping test: No GPUs available for NcclAllReduce.")
        return

    # Setup strategy with NcclAllReduce
    # This mirrors the usage pattern found in the similar API documentation
    strategy = tf.distribute.MirroredStrategy(
        cross_device_ops=tf.distribute.NcclAllReduce()
    )

    # Define a function with a dynamic shape (None) in the input signature
    # This mimics the behavior of torch.compile(dynamic=True) in the original bug report
    @tf.function(input_signature=[tf.TensorSpec(shape=[None, 4], dtype=tf.float32)])
    def distributed_op(x):
        # Perform a reduction which utilizes NcclAllReduce
        # This corresponds to the operation call in the original bug (fill_diagonal_)
        return strategy.reduce(tf.distribute.ReduceOp.SUM, x, axis=None)

    # Create input tensor
    x = tf.constant(np.ones((4, 4), dtype=np.float32))

    # Execute
    try:
        result = distributed_op(x)
        # Verify result
        assert result.shape == (4, 4)
        print("Test Passed: NcclAllReduce supports dynamic shapes.")
    except Exception as e:
        print(f"Test Failed: {e}")

if __name__ == "__main__":
    test_nccl_all_reduce_dynamic_shapes()
