import pytest
import tensorflow as tf

def test_operator_not_allowed_in_graph():
    """
    Test case leveraging tf.errors.OperatorNotAllowedInGraphError.
    
    This test mirrors the structure of the original PyTorch issue:
    1. Define a computation (function/model).
    2. Define inputs.
    3. Execute within a specific context (tf.function vs torch.no_grad).
    4. Assert the expected behavior (Error raised vs NaN values).
    """
    # Define a function that performs an operation disallowed in Graph execution
    @tf.function
    def iterate_over(t):
        # Iterating over a tensor is not supported in Graph execution
        a, b, c = t
        return a

    # Define input tensor
    input_tensor = tf.constant([1, 2, 3])

    # Assert that the specific error is raised when the function is called
    # This corresponds to the "assert not torch.isnan(...)" check in the original,
    # verifying the behavior of the execution context.
    with pytest.raises(tf.errors.OperatorNotAllowedInGraphError):
        iterate_over(input_tensor)
