import torch
import torch.distributed as dist
import torch.multiprocessing as mp
import os

def run_gather_object_test(rank, world_size):
    """
    Test case for torch.distributed.gather_object using data generation
    logic adapted from the original torch.compile bug report.
    """
    # Initialize the distributed environment
    os.environ['MASTER_ADDR'] = 'localhost'
    os.environ['MASTER_PORT'] = '12355'
    
    # Use 'gloo' backend for CPU-based object gathering
    dist.init_process_group("gloo", rank=rank, world_size=world_size)

    # Replicate the logic from the original fuzzed_program to generate the object
    torch.manual_seed(19989)

    # arg_0 is a python int derived from a random int32 tensor
    arg_0 = torch.tensor(torch.randn(()), dtype=torch.int32).item()

    # Scalar operations from the original bug
    var_node_2 = -6
    var_node_3 = arg_0
    var_node_1 = var_node_2 * var_node_3
    
    # Note: Original code used CUDA, but gather_object typically handles CPU objects.
    # We use CPU here to ensure the test runs in all environments.
    var_node_5 = torch.full((), 1, dtype=torch.int64, device='cpu')
    var_node_4 = var_node_5.item()
    
    # Division operation
    var_node_0 = var_node_1 / var_node_4
    
    # Sentinel tensor to ensure gradient computation context (if needed)
    sentinel = torch.tensor(1.0, requires_grad=True)
    
    # Final result object to be gathered
    result = var_node_0 * sentinel
    if result.is_complex():
        result = result.real

    # Prepare the gather list
    if rank == 0:
        gather_list = [None for _ in range(world_size)]
    else:
        gather_list = None

    # Call the similar API: torch.distributed.gather_object
    # We verify that the specific scalar/tensor types generated by the fuzzed logic
    # can be successfully pickled and gathered.
    try:
        dist.gather_object(result, gather_list, dst=0)
        
        if rank == 0:
            print(f"✅ Rank {rank} gathered objects successfully.")
            # Assertions to verify correctness
            assert len(gather_list) == world_size, f"Expected {world_size} objects, got {len(gather_list)}"
            for i, obj in enumerate(gather_list):
                assert isinstance(obj, torch.Tensor), f"Object at index {i} is not a Tensor"
                # Check if the gathered object matches the expected computation
                # (Since all ranks run the same seed, they should produce the same value)
                if i > 0:
                    assert torch.allclose(obj, gather_list[0]), "Gathered objects are not identical across ranks"
            print(f"✅ All assertions passed for gathered objects.")
            
    except Exception as e:
        print(f"❌ Error in Rank {rank}: {e}")
        raise

    dist.destroy_process_group()

if __name__ == "__main__":
    world_size = 2
    # Spawn processes to simulate a distributed environment
    mp.spawn(run_gather_object_test, args=(world_size,), nprocs=world_size, join=True)
