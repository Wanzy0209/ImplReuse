```python
import tensorflow as tf
import time

# Set seed for reproducibility
tf.random.set_seed(0)

# Adapted shapes for tf.keras.metrics.sparse_categorical_accuracy
# The original test used shapes for matrix multiplication: ((1, 12, 10, 64), (1, 12, 64, 10)).
# The result of that matmul is (1, 12, 10, 10).
# We adapt these to fit the API: y_pred (logits) and y_true (labels).
# We use the output shape of the original matmul as y_pred, and a corresponding y_true shape.
shapes = [
     ((1, 12, 10, 10), (1, 12, 10)), # Corresponds to result of (1, 12, 10, 64) x (1, 12, 64, 10)
     ((1, 12, 10, 64), (1, 12, 10)), # Corresponds to input (1, 12, 10, 64) treated as logits
]

def benchmark_accuracy(y_pred_shape, y_true_shape, dtype=tf.float16, repeat=500):
    # Generate data
    # y_true must be integer labels for sparse_categorical_accuracy
    y_true = tf.random.uniform(y_true_shape, minval=0, maxval=y_pred_shape[-1], dtype=tf.int32)
    # y_pred are the logits/probabilities
    y_pred = tf.random.uniform(y_pred_shape, dtype=dtype)
    
    # warm up
    for _ in range(5000):
        _ = tf.keras.metrics.sparse_categorical_accuracy(y_true, y_pred)
        
    # run
    times = []
    for i in range(repeat):
        start = time.time()
        _ = tf.keras.metrics.sparse_categorical_accuracy(y_true, y_pred)
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
            
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    for y_pred_shape, y_true_shape in shapes:
        t = benchmark_accuracy(y_pred_shape, y_true_shape)
        print(f"{y_pred_shape} x {y_true_shape}  ->  {t:.3f} us")
```
