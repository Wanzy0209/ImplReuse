```python
import tensorflow as tf
from collections import namedtuple

# Define the NamedTuple structure similar to the PyTorch example
class MyNamedTuple(namedtuple('MyNamedTupleBase', ['first', 'second'])):
    pass

# Subclass to allow __setattr__ (mimicking the PyTorch example's behavior)
class MyNamedTupleSubclass(MyNamedTuple):
    pass

# Create the object with TensorFlow tensors
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))

# Define the computation function
# Note: tf.compat.v1.tpu.batch_parallel does not pass arguments to the computation function
# directly in the same way torch.compile does. We capture the tuple in the closure.
def computation():
    extra_info = tf.constant(4.0)
    # Attempt to add dynamic attribute
    extended_tup.extra_info = extra_info
    return extended_tup

print("\nTesting NamedTuple with batch_parallel:")

try:
    # Attempt to run the computation using batch_parallel
    # Note: This requires a TPU environment to execute successfully.
    # We use inputs=None as we are not sharding tensor inputs, but testing object mutation.
    setattr_result = tf.compat.v1.tpu.batch_parallel(computation, inputs=None)
    
    # Verify if the dynamic attribute persists
    print(f"batch_parallel result: {setattr_result}")
    print(f"Extra info: {setattr_result.extra_info}")

except AttributeError as e:
    print(f"AttributeError caught (Bug reproduced): {e}")
except Exception as e:
    # Handling cases where TPU is not available or other runtime errors
    print(f"Execution failed (Environment/Setup issue): {e}")
```
