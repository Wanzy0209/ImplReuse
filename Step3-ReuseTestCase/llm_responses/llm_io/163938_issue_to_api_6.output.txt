import warnings
import sys

def test_optimization_options_no_future_warnings():
    """
    Test that importing and using tf.data.experimental.OptimizationOptions 
    does not trigger FutureWarnings related to functools.partial, 
    similar to the issue reported in torch.distributed.algorithms.ddp_comm_hooks.
    
    The original bug occurs when functools.partial is used directly within 
    an Enum class definition in Python 3.13+. This test checks if the 
    OptimizationOptions class (which defines configuration options) 
    exhibits similar behavior.
    """
    # The specific FutureWarning regarding functools.partial is introduced in Python 3.13
    if sys.version_info < (3, 13):
        print("Skipping test: Python version < 3.13 (FutureWarning not applicable).")
        return

    with warnings.catch_warnings(record=True) as w:
        # Ensure all warnings are captured
        warnings.simplefilter("always")

        # Reproduce the logic: Import the module containing the API
        # In the original bug, the warning appeared immediately upon import.
        import tensorflow as tf
        from tensorflow.data.experimental import OptimizationOptions

        # Instantiate to ensure any class-level definitions are evaluated
        options = OptimizationOptions()

        # Filter for the specific FutureWarning mentioned in the bug report
        partial_warnings = [
            warning for warning in w 
            if issubclass(warning.category, FutureWarning) 
            and "functools.partial" in str(warning.message)
            and "method descriptor" in str(warning.message)
        ]

        # Assert that no such warnings were raised
        assert len(partial_warnings) == 0, (
            f"Found {len(partial_warnings)} FutureWarning(s) related to functools.partial "
            f"when importing/using OptimizationOptions: "
            f"{[str(warning.message) for warning in partial_warnings]}"
        )

if __name__ == "__main__":
    test_optimization_options_no_future_warnings()
    print("Test passed: No FutureWarnings detected.")
