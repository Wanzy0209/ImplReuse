import tensorflow as tf

def test_queue_runner_duplicate_addition():
    """
    Adapts the PyTorch joint_graph.py merge mistake logic to TensorFlow.
    
    The original bug involved a code block being duplicated during a merge,
    causing a specific operation (joint_custom_pre_pass) to execute twice.
    
    This test simulates that scenario by calling tf.compat.v1.train.add_queue_runner
    twice (mimicking the duplicate code blocks) and verifies that the runner
    is added to the collection twice, reproducing the logic error.
    """
    
    # QueueRunners are not compatible with eager execution
    tf.compat.v1.disable_eager_execution()

    with tf.Graph().as_default():
        # Setup: Create a simple queue and a QueueRunner
        queue = tf.queue.FIFOQueue(capacity=10, dtypes=[tf.int32], shapes=[])
        enqueue_op = queue.enqueue([1])
        qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op])

        # --- Simulating the Merge Mistake Logic ---
        
        # Block 1: Original code location
        if qr is not None:
            tf.compat.v1.train.add_queue_runner(qr)

        # ... (Simulating other code like remove_noop_ops) ...

        # Block 2: The duplicate code added by mistake (from the merge conflict)
        if qr is not None:
            tf.compat.v1.train.add_queue_runner(qr)

        # --- Verification ---
        
        # Retrieve the collection to check for duplicates
        runners = tf.compat.v1.get_collection(tf.compat.v1.GraphKeys.QUEUE_RUNNERS)
        
        # The bug results in the operation happening twice (duplicate entry)
        assert len(runners) == 2, f"Expected 2 runners due to duplicate logic, found {len(runners)}"
        assert runners[0] is qr
        assert runners[1] is qr
        
        print("Test passed: The merge mistake logic resulted in duplicate queue runners as expected.")

if __name__ == "__main__":
    test_queue_runner_duplicate_addition()
