```python
import torch

# Create input tensors (1D)
a = torch.tensor([1, 2, 3])
b = torch.tensor([4, 5])

# Call the similar API: torch.cartesian_prod
result = torch.cartesian_prod(a, b)

# Expected output based on itertools.product logic
expected = torch.tensor([[1, 4],
                         [1, 5],
                         [2, 4],
                         [2, 5],
                         [3, 4],
                         [3, 5]])

# Assertion to verify correctness
assert torch.equal(result, expected)
```
