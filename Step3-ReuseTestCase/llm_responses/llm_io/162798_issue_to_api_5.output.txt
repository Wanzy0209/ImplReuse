```python
import torch
import math

def test_nanmedian_empty_tensor():
    """
    Test case for Issue 162798: [MPS] torch.nanmedian returns 0 for empty input.
    
    This test verifies that torch.nanmedian returns NaN for empty tensors across 
    CPU, CUDA, and MPS backends. It leverages the availability check pattern 
    (similar to torch.distributed.is_nccl_available) to conditionally execute 
    backend-specific tests.
    """
    # Create an empty tensor
    x = torch.empty((0,), dtype=torch.float32)

    # Test on CPU (Always available)
    cpu_result = torch.nanmedian(x)
    assert torch.isnan(cpu_result), "CPU nanmedian should return NaN for empty input"

    # Test on CUDA (if available)
    # This mirrors the pattern of checking availability before execution
    if torch.cuda.is_available():
        cuda_x = x.to('cuda')
        cuda_result = torch.nanmedian(cuda_x)
        assert torch.isnan(cuda_result), "CUDA nanmedian should return NaN for empty input"

    # Test on MPS (if available)
    # This is the specific case reported in the bug where it returned 0 instead of NaN
    if torch.backends.mps.is_available():
        mps_x = x.to('mps')
        mps_result = torch.nanmedian(mps_x)
        assert torch.isnan(mps_result), "MPS nanmedian should return NaN for empty input, not 0"

if __name__ == "__main__":
    test_nanmedian_empty_tensor()
    print("Test passed successfully.")
```
