import tensorflow as tf
import numpy as np

def test_triu_gradient_behavior():
    """
    Adapted test case to verify gradient behavior of tf.keras.ops.triu.
    Similar to the torch.min issue, we verify how gradients flow through the operation.
    For triu, gradients should flow through the upper triangle (values kept) 
    and be zeroed out for the lower triangle (values masked to 0).
    """
    
    # Case 1: Default behavior (k=0)
    # Create a 3x3 matrix of ones
    a = tf.ones((3, 3))
    
    with tf.GradientTape() as tape:
        tape.watch(a)
        # Apply triu (upper triangle including diagonal)
        out = tf.keras.ops.triu(a)
    
    # Compute gradients
    grads = tape.gradient(out, a)
    
    # Expected: Upper triangle is 1.0 (gradient flows), Lower is 0.0 (masked)
    # [[1, 1, 1],
    #  [0, 1, 1],
    #  [0, 0, 1]]
    expected_grads_k0 = np.array([[1., 1., 1.],
                                  [0., 1., 1.],
                                  [0., 0., 1.]])
    
    print(">>> # tf.keras.ops.triu -> default k=0")
    print(f">>> a.grad (approximated)\n{grads.numpy()}")
    np.testing.assert_allclose(grads.numpy(), expected_grads_k0)

    # Case 2: Offset behavior (k=1)
    # Verify gradient behavior changes with the 'k' parameter (similar to checking 'dim' in torch.min)
    with tf.GradientTape() as tape:
        tape.watch(a)
        # Apply triu with offset 1 (upper triangle excluding diagonal)
        out = tf.keras.ops.triu(a, k=1)
    
    grads = tape.gradient(out, a)
    
    # Expected: Strict upper triangle is 1.0, Diagonal and Lower are 0.0
    # [[0, 1, 1],
    #  [0, 0, 1],
    #  [0, 0, 0]]
    expected_grads_k1 = np.array([[0., 1., 1.],
                                  [0., 0., 1.],
                                  [0., 0., 0.]])
    
    print("...")
    print(">>> # tf.keras.ops.triu(a, k=1) -> offset diagonal")
    print(f">>> a.grad (approximated)\n{grads.numpy()}")
    np.testing.assert_allclose(grads.numpy(), expected_grads_k1)

if __name__ == "__main__":
    test_triu_gradient_behavior()
