```python
import torch
import torch._dynamo
import pytest
from torch.utils._python_dispatch import TorchDispatchMode

# Leveraging the class-based structure pattern from the similar API 
# (tf.keras.initializers.HeUniform) to define a custom dispatch mode.
# This mimics the inheritance and initialization pattern found in the similar API.
class CustomDispatchMode(TorchDispatchMode):
    def __init__(self):
        super().__init__()

    def __torch_dispatch__(self, func, types, args, kwargs):
        # This mode acts as a "non-infra torch dispatch mode" which triggers the bug
        return func(*args, **kwargs)

def test_compile_with_non_infra_dispatch_mode():
    """
    Test that torch.compile raises an error when a non-infra dispatch mode is present,
    rather than silently skipping compilation.
    """
    def simple_fn(x):
        return x + 1

    # The bug report indicates that torch.compile silently skips compilation
    # when a non-infra dispatch mode (like DebugMode or custom modes) is active.
    # The expected behavior (fix) is to raise an error.
    with pytest.raises(RuntimeError, match="non-infra torch dispatch mode"):
        with CustomDispatchMode():
            compiled_fn = torch.compile(simple_fn)
            compiled_fn(torch.randn(2))
```
