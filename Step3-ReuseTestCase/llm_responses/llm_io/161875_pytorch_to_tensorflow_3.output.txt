import tensorflow as tf

# Reproduce the logic: passing an extremely large integer value to a dimension parameter.
# The original PyTorch bug was triggered by setting padding=9223372036854775803.
# We apply this extreme value to the window_length parameter of tf.signal.hann_window.

extreme_value = 9223372036854775803

try:
    # Attempt to generate a Hann window with an extreme length
    window = tf.signal.hann_window(window_length=extreme_value)
    
    # If the API handles it gracefully (e.g., by raising an error), this line might not be reached.
    # If it causes a segmentation fault, the program will crash here.
    print(f"Window created successfully. Shape: {window.shape}")

except Exception as e:
    # We expect a resource error or invalid argument error, not a segmentation fault.
    print(f"Caught exception: {type(e).__name__}: {e}")
