import torch
from torch.nn import Conv3d

# Adapt input data for Conv3d (Batch, Channel, Depth, Height, Width)
input_data = torch.randn(1, 16, 10, 10, 10)


class MyConv3dModule(torch.nn.Module):

    def __init__(self):
        super(MyConv3dModule, self).__init__()
        # Use the same problematic padding value as in the Conv1d bug report
        self.conv1 = Conv3d(in_channels=16, out_channels=32, kernel_size=3,
            stride=1, padding=9223372036854775803)
        self.add_module(name='conv1', module=self.conv1)


model = MyConv3dModule()
# This call is expected to trigger the error similar to the Conv1d case
output = model.conv1(input_data)
