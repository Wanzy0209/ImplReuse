import torch

# Similar API pattern extracted from the issue context
def log(a):
    return (3 * a).log()

def test_log_return_type():
    """
    Test case to verify that the similar API 'log' returns a valid Tensor
    and does not exhibit the 'return None' behavior described in the original bug
    (where _jit_compile was typed as None but returned a value).
    """
    # Setup input data
    input_tensor = torch.tensor([1.0, 2.0, 3.0])
    
    # Execute the function
    result = log(input_tensor)
    
    # Assertions
    # The original bug involved a function returning a value (Module/str) 
    # while being typed as None. We verify here that the result is indeed 
    # a value (Tensor) and not None.
    assert result is not None, "Function returned None, but expected a Tensor"
    assert isinstance(result, torch.Tensor), f"Expected torch.Tensor, got {type(result)}"

    expected = (3 * input_tensor).log()
    assert torch.equal(result, expected), "Calculated result does not match expected value"
