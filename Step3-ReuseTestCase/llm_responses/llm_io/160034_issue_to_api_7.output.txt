import torch
import pytest

def test_index_put_mps_complex64_accumulate():
    """
    Test case for Issue 160034.
    
    This test reproduces the bug where torch.index_put_ fails on the MPS backend
    when using complex64 dtype with accumulate=True. The error message is currently
    generic and should ideally be improved.
    
    The structure of this test leverages the pattern from the similar API
    (tf.compat.v1.global_variables_initializer) by checking the execution
    environment (MPS availability) before attempting the operation, similar to
    how the TF API checks context.executing_eagerly().
    """
    
    # Pattern from similar API: Check execution context/environment
    # analogous to 'if context.executing_eagerly():'
    if not torch.backends.mps.is_available():
        pytest.skip("MPS backend is not available (requires Apple Silicon)")

    device = torch.device("mps")
    
    # Setup tensors as per the bug report
    image = torch.zeros(10, dtype=torch.complex64, device=device)
    data = torch.ones(3, dtype=torch.complex64, device=device)
    indices = torch.tensor([1, 3, 5], device=device)

    # This operation is expected to fail on MPS with complex64
    # We assert that a RuntimeError is raised
    with pytest.raises(RuntimeError) as exc_info:
        image.index_put_(indices=(indices,), values=data, accumulate=True)

    # Verify the error message matches the reported failure mode
    # (Checking for the specific generic error mentioned in the issue)
    error_message = str(exc_info.value)
    assert "Expected supportedFloatingType" in error_message or \
           "scalar_type == kInt" in error_message, \
           f"Error message does not match the reported failure mode. Got: {error_message}"

    # Note: Once the bug is fixed, this test should be updated to verify
    # that the operation succeeds OR that the error message explicitly mentions
    # the lack of complex64 support on MPS.
