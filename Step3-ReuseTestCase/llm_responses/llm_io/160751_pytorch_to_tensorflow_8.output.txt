import tensorflow as tf

def func():
    # Create a tensor inside a name scope
    # This is analogous to the tensor operations in the PyTorch bug report
    with tf.name_scope("test_scope"):
        a = tf.constant([1.0, -2.0])
    
    # Check if the scope name is present in the tensor name
    # This mimics the assertion in the original bug report.
    # The original bug checks if an assertion is caught; here we check if the 
    # naming side-effect (analogous to synchronization) is preserved.
    scope_name = "test_scope"
    is_scoped = scope_name in a.name
    
    # In the PyTorch bug, the synchronize call is removed in aot_eager mode.
    # Here we verify if the scope is applied in eager vs graph modes.
    assert is_scoped, f"Scope '{scope_name}' was removed (similar to synchronize removal bug). Tensor name: {a.name}"
    
    print(f"Assertion passed. Tensor name: {a.name}")

def test_fn():
    # Test 1: Eager Mode
    # This corresponds to the standard eager execution in PyTorch
    print("--- Testing Eager Mode ---")
    try:
        func()
    except AssertionError as e:
        print(f"Caught Exception: {e}")

    # Test 2: Graph Mode (tf.function)
    # This corresponds to torch.compile (backend="aot_eager") in the PyTorch bug
    print("\n--- Testing Graph Mode (tf.function) ---")
    try:
        # tf.function compiles the logic into a graph
        compiled_func = tf.function(func)
        compiled_func()
    except AssertionError as e:
        print(f"Caught Exception: {e}")

if __name__ == "__main__":
    test_fn()
