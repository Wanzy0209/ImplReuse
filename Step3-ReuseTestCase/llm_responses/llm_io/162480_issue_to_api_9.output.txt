import torch
import unittest

class TestAOTInductorFloatHandling(unittest.TestCase):
    def test_square_rebind_unbacked_float(self):
        """
        Test case for Issue #162480.
        Verifies that torch.compile (AOTInductor) handles float values
        correctly in rebind_unbacked logic when using torch.square.
        """
        # Define the function using the similar API (torch.square)
        # and a pattern that interacts with shapes.
        def model(x):
            # Use torch.square
            y = torch.square(x)
            # Perform a view operation with a size derived from a float calculation.
            # This attempts to trigger the symbolic shape logic where
            # a float might be encountered in rebind_unbacked.
            # Note: We cast to int to make it a valid size, but the
            # intermediate float interaction is the key.
            return y.view(int(y.size(0) * 1.0))

        # Compile the model
        compiled_model = torch.compile(model)

        # Run with different input sizes to trigger re-specialization
        # and the rebind_unbacked path.
        inp1 = torch.randn(10)
        out1 = compiled_model(inp1)

        inp2 = torch.randn(20)
        out2 = compiled_model(inp2)

        # Assertions to ensure correctness
        self.assertEqual(out1.shape, (10,))
        self.assertEqual(out2.shape, (20,))
        self.assertTrue(torch.allclose(out1, torch.square(inp1)))
        self.assertTrue(torch.allclose(out2, torch.square(inp2)))

if __name__ == "__main__":
    unittest.main()
