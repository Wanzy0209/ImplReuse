import tensorflow as tf
import unittest

class TestFloat64Matmul(unittest.TestCase):
    def test_float64_matmul_gpu(self):
        """
        Test case to reproduce float64 matmul logic similar to the PyTorch issue.
        Note: The similar API provided (tf.compat.v1.tpu.PaddingSpec) is an Enum 
        for TPU padding policies and is not functionally equivalent to torch.matmul.
        This test uses the functional equivalent tf.matmul to preserve the bug reproduction logic.
        """
        # Check for GPU availability (ROCm or CUDA)
        gpus = tf.config.list_physical_devices('GPU')
        if not gpus:
            self.skipTest("No GPU available to test device-specific matmul behavior")
        
        # Configure memory growth to avoid OOM
        try:
            for gpu in gpus:
                tf.config.experimental.set_memory_growth(gpu, True)
        except RuntimeError as e:
            print(f"Memory growth setting failed: {e}")

        with tf.device('/GPU:0'):
            # Reproduce the logic: float64 matmul
            # PyTorch: torch.randn(1000, 1000, device="cuda", dtype=torch.float64)
            x = tf.random.normal((1000, 1000), dtype=tf.float64)
            y = tf.random.normal((1000, 1000), dtype=tf.float64)
            
            # PyTorch: z = x @ y
            z = tf.matmul(x, y)
            
            # PyTorch: print(z[0, 0])
            # We assert that the operation completes and produces a finite value
            result = z[0, 0]
            
            self.assertIsNotNone(result)
            self.assertTrue(tf.math.is_finite(result), "Result should be finite (not NaN or Inf)")
            self.assertEqual(z.shape, (1000, 1000))

if __name__ == '__main__':
    unittest.main()
