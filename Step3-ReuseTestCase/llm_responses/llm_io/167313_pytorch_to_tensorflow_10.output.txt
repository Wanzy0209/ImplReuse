import tensorflow as tf
import numpy as np

def test_convert_to_tensor_dtype_consistency():
    """
    Adapted test case for tf.keras.ops.convert_to_tensor based on the 
    torch.addmm bug report (Issue ID: 167313).
    
    Original Bug: torch.compile ignored alpha/beta parameters in addmm.
    Adapted Logic: Verify that tf.function (compilation) respects the 'dtype' 
    parameter in convert_to_tensor, similar to how alpha/beta should be respected.
    """
    
    # 1. Setup inputs
    # Using a numpy array to ensure a type conversion is required
    x = np.random.rand(2, 3).astype(np.float32)

    # 2. Define the function using the target API
    # We specify a target dtype (analogous to specifying alpha/beta in the original bug)
    f = lambda x: tf.keras.ops.convert_to_tensor(x, dtype=tf.float16)

    # 3. Compile the function (TensorFlow equivalent of torch.compile)
    fc = tf.function(f)

    # 4. Run Eager
    eager_result = f(x)
    print(f"Eager Result dtype: {eager_result.dtype}")

    # 5. Run Compiled
    compiled_result = fc(x)
    print(f"Compiled Result dtype: {compiled_result.dtype}")

    # 6. Verify behavior
    # The original bug showed a mismatch between eager and compiled results.
    # We assert that the parameter (dtype) is respected in both modes.
    assert eager_result.dtype == compiled_result.dtype, \
        f"Bug detected: 'dtype' parameter ignored in compiled mode. Eager: {eager_result.dtype}, Compiled: {compiled_result.dtype}"

if __name__ == "__main__":
    test_convert_to_tensor_dtype_consistency()
