import tensorflow as tf
import numpy as np

def test_temporary_tensorspec_pattern():
    """
    Test case adapted from PyTorch MemPool issue (ID: 167745).
    
    The original issue highlights failures when temporary MemPool objects 
    are used within context managers. This test verifies that the similar 
    TensorFlow API (tf.TensorSpec) handles temporary objects correctly 
    when used to define function signatures (which act as a scope for tracing).
    """
    
    # Setup: Create a base tensor to derive a spec from (analogous to make_custom_pool)
    base_tensor = tf.constant([[1.0, 2.0], [3.0, 4.0]])

    # Pattern: Using a temporary object created inline to configure behavior.
    # PyTorch: with torch.cuda.use_mem_pool(torch.cuda.MemPool(pool1)):
    # TensorFlow: @tf.function(input_signature=[tf.TensorSpec.from_tensor(...)])
    
    # We create the TensorSpec directly in the decorator argument. 
    # If the API does not hold a reference to this temporary object properly,
    # tracing or execution might fail (similar to the MemPool bug).
    @tf.function(input_signature=[tf.TensorSpec.from_tensor(base_tensor)])
    def process_in_temporary_scope(x):
        return x + 1.0

    # Execution
    input_data = tf.constant([[5.0, 6.0], [7.0, 8.0]])
    result = process_in_temporary_scope(input_data)

    # Assertion
    expected = tf.constant([[6.0, 7.0], [8.0, 9.0]])
    assert tf.reduce_all(tf.equal(result, expected)).numpy(), "Temporary TensorSpec pattern failed"
    
    print("Test passed: Temporary TensorSpec handled correctly in function signature.")

if __name__ == "__main__":
    test_temporary_tensorspec_pattern()
