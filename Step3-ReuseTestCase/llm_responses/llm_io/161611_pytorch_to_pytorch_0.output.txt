import torch

def test_torch_zeros_dtype_redundancy():
    """
    Test that torch.zeros correctly initializes a tensor with the specified dtype,
    and verifies that calling .to() without assignment is a no-op (redundant).
    Based on Issue #161611.
    """
    # Setup: Define a query tensor to mimic the context of the bug report
    query = torch.empty(4, 8, dtype=torch.float16)
    
    # Action: Create attn_bias using torch.zeros with the same dtype as query
    # This is the call site identified in the bug report.
    L, S = 4, 8
    attn_bias = torch.zeros(L, S, dtype=query.dtype, device=query.device)
    
    # Assertion 1: Verify torch.zeros respects the dtype argument immediately
    assert attn_bias.dtype == torch.float16, \
        f"Expected dtype torch.float16, but got {attn_bias.dtype}"
    
    # Action: Mimic the redundant line from the bug report
    # The bug report noted: attn_bias.to(query.dtype) is redundant and result not assigned.
    # We verify that calling .to() without assignment does not modify the tensor.
    attn_bias.to(torch.float32) # Attempt to change dtype without assignment
    
    # Assertion 2: Verify the tensor remains unchanged
    assert attn_bias.dtype == torch.float16, \
        "Tensor dtype changed despite .to() result not being assigned"
    
    print("Test passed: torch.zeros initializes correctly and redundant .to() is a no-op.")

if __name__ == "__main__":
    test_torch_zeros_dtype_redundancy()
