```python
import tensorflow as tf
import numpy as np

# Ensure reproducibility
tf.random.set_seed(42)

# Define the shape (analogous to the input tensor size in the original test)
shape = (8192,)

# 1. Eager execution (float32)
# We use a specific seed to ensure determinism across calls
init_eager = tf.keras.initializers.LecunNormal(seed=42)
out1 = init_eager(shape, dtype=tf.float32)

# 2. Compiled execution (float32)
# We wrap the initialization in tf.function (TensorFlow's compilation equivalent)
@tf.function
def compiled_init(initializer):
    return initializer(shape, dtype=tf.float32)

# We must use a fresh initializer with the same seed to generate the same random values
init_compiled = tf.keras.initializers.LecunNormal(seed=42)
out2 = compiled_init(init_compiled)

# 3. High precision reference (float64)
# Fresh initializer with same seed to match the random values
init_high = tf.keras.initializers.LecunNormal(seed=42)
out3_high = init_high(shape, dtype=tf.float64)

# Calculate differences
# Cast to float64 for accurate difference calculation against the reference
diff_eager = tf.reduce_max(tf.abs(tf.cast(out3_high, tf.float64) - tf.cast(out1, tf.float64)))
diff_compiled = tf.reduce_max(tf.abs(tf.cast(out3_high, tf.float64) - tf.cast(out2, tf.float64)))

print("Max diff (High vs Eager FP32):", diff_eager.numpy())
print("Max diff (High vs Compiled FP32):", diff_compiled.numpy())
```
