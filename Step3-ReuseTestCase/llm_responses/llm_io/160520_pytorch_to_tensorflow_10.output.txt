import tensorflow as tf
import numpy as np

# Ensure we are using TF 1.x behavior as requested by the API name (compat.v1)
tf.compat.v1.disable_eager_execution()

def test_range_input_producer_device_context():
    """
    Adapted from PyTorch Issue 160520.
    
    Original Bug: Inductor w/ cpp_wrapper introduces redundant H2D-D2H memcpy 
    under DeviceContext for non-tensor arguments.
    
    Adaptation: Verifies the behavior of tf.compat.v1.train.range_input_producer
    when running inside a specific device context. It checks if the data pipeline
    handles tensor arguments and device placement correctly without introducing
    implicit data transfer errors or inefficiencies during the loop.
    """
    
    # Determine device to use (mimicking the .cuda() setup)
    # Using CPU if GPU is not available to ensure the test is runnable everywhere
    device_name = '/device:GPU:0' if tf.test.is_gpu_available() else '/cpu:0'
    
    with tf.compat.v1.Session() as sess:
        # Mimic the 'with torch.device("cuda"):' context from the original bug report
        with tf.device(device_name):
            # In the PyTorch bug, non-tensor arguments were redundantly converted to tensors.
            # Here we explicitly pass a tensor as 'limit' to test argument handling 
            # within the device context.
            limit_val = 10
            limit_tensor = tf.constant(limit_val, dtype=tf.int32, name="limit_tensor")
            
            # Original API: torch.compile (model execution)
            # Similar API: tf.compat.v1.train.range_input_producer (data generation)
            # We set num_epochs=1 to allow the loop to finish (or raise OutOfRange)
            producer = tf.compat.v1.train.range_input_producer(
                limit_tensor, 
                num_epochs=1, 
                shuffle=False,
                capacity=32
            )
            
            # Dequeue operation to consume the data (analogous to mlp(x))
            # This simulates the forward pass and data retrieval
            batch_size = 2
            batch = producer.dequeue_many(batch_size)
            
            # Add a simple operation to ensure computation happens on the device
            # This mimics the processing done in SimpleMLP
            result = batch * 2

        # Initialize local variables (required for range_input_producer epochs counter)
        sess.run(tf.compat.v1.local_variables_initializer())
        sess.run(tf.compat.v1.global_variables_initializer())
        
        # Coordinator for queue runners (mimics the execution loop)
        coord = tf.train.Coordinator()
        threads = tf.train.start_queue_runners(sess=sess, coord=coord)
        
        try:
            # Mimic the loop in the original repro code: for i in range(10)
            # We run a few iterations to check for redundant data handling
            for i in range(5):
                # Run the operation
                output = sess.run(result)
                
                # Basic assertion to ensure correctness and data integrity
                assert output.shape == (batch_size,)
                
                # Verify values are expected (0, 2, 4, 6... since we multiplied by 2)
                # range_input_producer produces 0 to limit-1
                expected_base = np.arange(i * batch_size, (i + 1) * batch_size)
                expected = expected_base * 2
                np.testing.assert_array_equal(output, expected)
                
        except tf.errors.OutOfRangeError:
            # Expected when num_epochs is exhausted
            pass
        finally:
            coord.request_stop()
            coord.join(threads)

if __name__ == "__main__":
    test_range_input_producer_device_context()
