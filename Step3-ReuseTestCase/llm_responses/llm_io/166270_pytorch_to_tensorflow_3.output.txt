```python
import tensorflow as tf
import numpy as np

# Set seed for reproducibility
tf.random.set_seed(1061983224)

# Sentinel variable to ensure gradient computation is active
sentinel = tf.Variable(1.0, dtype=tf.float32)

def fuzzed_program(arg_0):
    # Adaptation logic:
    # The original PyTorch test involved reducing a tensor to a scalar (squeeze) 
    # and expanding it back. 
    # Since tf.experimental.numpy.tril requires a matrix (rank >= 2), 
    # we adapt the shape manipulation flow to produce a 2D tensor.
    
    # arg_0: size=(4,), dtype=bool
    # Reshape to (1, 2, 2) to introduce a dimension that can be squeezed
    var_node_1 = tf.reshape(arg_0, [1, 2, 2]) 
    
    # Squeeze the dimension of size 1, resulting in (2, 2)
    # This mimics the squeeze operation in the original bug report
    var_node_2 = tf.squeeze(var_node_1)
    
    # Call the target API: tf.experimental.numpy.tril
    # This operates on the 2D tensor
    var_node_0 = tf.experimental.numpy.tril(var_node_2)
    
    # Ensure gradient computation by multiplying with sentinel
    result = var_node_0 * sentinel
    return result

# Create input tensor
# Mimicking the original bool dtype and size
arg_0 = tf.cast(tf.random.uniform((4,), minval=0, maxval=2, dtype=tf.int32), tf.bool)

# 1. Test Eager Execution
print("Testing Eager Execution...")
with tf.GradientTape() as tape:
    tape.watch(arg_0)
    result_original = fuzzed_program(arg_0)
print('✅ eager success')

# 2. Test Compiled Execution (tf.function)
print("Testing Compiled Execution...")
compiled_program = tf.function(fuzzed_program)

with tf.GradientTape() as tape:
    tape.watch(arg_0)
    result_compiled = compiled_program(arg_0)
print('✅ compile success')

# 3. Verify Results
# Check if the outputs match
if tf.reduce_all(tf.equal(result_original, result_compiled)).numpy():
    print('✅ results match')
else:
    print('❌ results diverge')
    print("Eager result:", result_original.numpy())
    print("Compiled result:", result_compiled.numpy())
```
