import tensorflow as tf

def test_fixed_shards_partitioner():
    """
    Adapted test case for tf.distribute.experimental.partitioners.FixedShardsPartitioner.
    The original PyTorch bug involved splitting a dataset into 3 parts (train, val, test).
    This test verifies that the TensorFlow partitioner can handle a similar shape and partition count.
    """
    # Setup: Define the partitioner with 3 shards (analogous to the 3 splits in the PyTorch example)
    num_shards = 3
    partitioner = tf.distribute.experimental.partitioners.FixedShardsPartitioner(num_shards=num_shards)

    # Data: Define a tensor shape similar to the PyTorch example (100 samples, 3 features)
    # In PyTorch: x = torch.randn(100, 3)
    shape = tf.TensorShape([100, 3])
    dtype = tf.float32

    # Action: Call the partitioner to determine how to split the tensor
    # This is analogous to random_split(dataset, lengths)
    # Note: FixedShardsPartitioner calculates partition dimensions, it does not split the data values directly.
    partitions = partitioner(shape, dtype, axis=0)

    # Verification:
    # FixedShardsPartitioner logic: result[axis] = min(num_shards, shape.dims[axis].value)
    # Since num_shards (3) < shape[0] (100), we expect the 0th dimension to be split into 3 partitions.
    # The result list represents the partition count per dimension.
    expected_partitions = [1] * len(shape)
    expected_partitions[0] = num_shards

    assert partitions == expected_partitions, (
        f"Expected partitions {expected_partitions}, but got {partitions}. "
        f"Partitioner failed to split shape {shape} into {num_shards} shards."
    )

    print(f"Partitioner worked for shape {shape} with {num_shards} shards.")

if __name__ == "__main__":
    test_fixed_shards_partitioner()
