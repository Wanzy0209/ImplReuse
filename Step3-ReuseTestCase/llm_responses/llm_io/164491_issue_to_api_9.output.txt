import tensorflow as tf
import numpy as np

def test_matmul_row_major_rhs_with_async_scope():
    """
    Test case to verify matrix multiplication with a row-major RHS matrix
    works correctly within tf.experimental.async_scope.
    
    This test is derived from a PyTorch issue (164491) where internal functions
    _scaled_mm and _int_mm were slow or raised errors with row-major RHS matrices.
    The test leverages the tf.experimental.async_scope pattern to group the
    operation and handle potential execution errors, similar to the provided
    TensorFlow example.
    """
    # Setup dimensions
    M, K, N = 128, 64, 32
    
    # Create LHS matrix (Row-major)
    lhs = tf.random.normal((M, K), dtype=tf.float32)
    
    # Create RHS matrix (Row-major)
    # The bug report highlights that _scaled_mm fails/slows with row-major RHS.
    # In TensorFlow, we verify that standard matmul handles this layout correctly.
    rhs = tf.random.normal((K, N), dtype=tf.float32)

    # Leverage the similar API: tf.experimental.async_scope
    # The provided example uses a try-except block to handle errors during async execution.
    try:
        with tf.experimental.async_scope():
            # Perform the matrix multiplication.
            # This corresponds to the _int_mm or _scaled_mm operation in the bug report.
            # We run it inside the scope to ensure it completes asynchronously but synchronizes on exit.
            result = tf.matmul(lhs, rhs)
            
            # Accessing .numpy() forces synchronization, ensuring the op is complete
            # before the scope exits, mimicking the barrier behavior.
            _ = result.numpy()

    except Exception as e:
        # The bug report mentions _scaled_mm raises an error.
        # We catch exceptions here to mimic the robust error handling shown in the
        # tf.experimental.async_scope documentation.
        print(f"Error during async matmul execution: {e}")
        raise

    # Assertions to verify correctness
    assert result.shape == (M, N), f"Expected shape ({M}, {N}), got {result.shape}"
    
    # Verify against numpy to ensure the row-major calculation is correct
    np_lhs = lhs.numpy()
    np_rhs = rhs.numpy()
    expected = np.dot(np_lhs, np_rhs)
    
    np.testing.assert_allclose(result.numpy(), expected, rtol=1e-5, atol=1e-5)

if __name__ == "__main__":
    test_matmul_row_major_rhs_with_async_scope()
