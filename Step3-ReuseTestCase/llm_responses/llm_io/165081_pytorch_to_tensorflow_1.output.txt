import tensorflow as tf

# The original issue involves Eager/Compile divergence with data-dependent expressions.
# We adapt the test to use tf.keras.backend.random_uniform (the similar API)
# inside a tf.function (the compiled context) to verify behavior.

@tf.function
def fuzzed_program_tf():
    # Define dtype as float64 to match the original test case
    dtype = tf.float64

    # Replacing arg_0...arg_18 and torch.full calls with tf.keras.backend.random_uniform.
    # This tests the similar API's ability to generate tensors for the computation graph.

    # var_node_6 = arg_0 (9, 9, 9)
    var_node_6 = tf.keras.backend.random_uniform((9, 9, 9), minval=-1.0, maxval=1.0, dtype=dtype)

    var_node_5 = tf.matmul(var_node_6, var_node_7)

    # var_node_9 = torch.full((9, 11, 12), ...)

    # var_node_10 = arg_2 (9, 12, 8)

    var_node_8 = tf.matmul(var_node_9, var_node_10)
    var_node_4 = tf.matmul(var_node_5, var_node_8)

    # var_node_13 = arg_3 (9, 8, 13)
    var_node_13 = tf.keras.backend.random_uniform((9, 8, 13), minval=-1.0, maxval=1.0, dtype=dtype)
    # var_node_14 = arg_4 (9, 13, 7)
    var_node_14 = tf.keras.backend.random_uniform((9, 13, 7), minval=-1.0, maxval=1.0, dtype=dtype)

    var_node_12 = tf.matmul(var_node_13, var_node_14)

    # var_node_15 = arg_5 (9, 7, 16)
    var_node_15 = tf.keras.backend.random_uniform((9, 7, 16), minval=-1.0, maxval=1.0, dtype=dtype)

    var_node_11 = tf.matmul(var_node_12, var_node_15)
    var_node_3 = tf.matmul(var_node_4, var_node_11)

if __name__ == "__main__":
