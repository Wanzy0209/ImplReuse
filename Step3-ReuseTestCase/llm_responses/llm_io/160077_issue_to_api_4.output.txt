import tensorflow as tf

def test_lecun_normal_in_graph_mode_with_device():
    """
    Test case for tf.keras.initializers.LecunNormal based on the PyTorch bug report:
    'torch._tensor' has no attribute 'split' while using torch.compile under torch.device context.
    
    This test translates the scenario to TensorFlow:
    - Using tf.function (analogous to torch.compile)
    - Using tf.device context
    - Calling the API (LecunNormal) inside these contexts.
    """
    
    # Define the function to be compiled (analogous to 'f' in the bug report)
    @tf.function
    def initialize_tensor(shape):
        # Using the similar API: tf.keras.initializers.LecunNormal
        initializer = tf.keras.initializers.LecunNormal(seed=42)
        return initializer(shape)

    # Reproduce the logic: Device context + Compiled execution
    # Note: In the bug report, 'cuda' is used. Here we use '/CPU:0' for general compatibility,
    # but '/GPU:0' would be the direct translation if available.
    with tf.device("/CPU:0"):
        # Execute the compiled function
        weights = initialize_tensor((2, 2))

    # Assertions to verify the operation succeeded
    assert weights.shape == (2, 2)
    assert weights.dtype == tf.float32
    # LecunNormal variance is scale/fan_in. Default scale=1. fan_in=2. Var ~ 0.5.
    # We just check it's not zero or NaN to ensure execution.
    assert not tf.math.reduce_any(tf.math.is_nan(weights))

if __name__ == "__main__":
    test_lecun_normal_in_graph_mode_with_device()
    print("Test passed.")
