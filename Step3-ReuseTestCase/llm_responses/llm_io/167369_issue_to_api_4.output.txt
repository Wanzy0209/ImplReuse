import tensorflow as tf

# Define a custom class similar to the Config class in the original issue
class CustomLayer(tf.keras.layers.Layer):
    def __init__(self, name="custom_layer", **kwargs):
        super().__init__(name=name, **kwargs)
        self.value = 42

    def get_config(self):
        config = super().get_config()
        config.update({"value": self.value})
        return config

# Define the forward function
# Original logic: return x * len(repr(config))
# Adapted logic: return x * tf.strings.length(tf.constant(tf.keras.layers.serialize(layer)))
def forward(x, layer):
    # Use the similar API: tf.keras.layers.serialize
    # This converts the layer object to a JSON string representation
    serialized_str = tf.keras.layers.serialize(layer)
    
    # Convert the Python string to a TensorFlow tensor for graph operations
    serialized_tensor = tf.constant(serialized_str)
    
    # Calculate the length of the serialized string
    length = tf.strings.length(serialized_tensor)
    
    # Perform the multiplication
    return x * tf.cast(length, dtype=x.dtype)

# Setup inputs
layer = CustomLayer()
x = tf.constant([[1.0, 2.0], [3.0, 4.0]])

# Compile using tf.function (TensorFlow equivalent to torch.compile)
# This attempts to trace the function similar to Dynamo
compiled_forward = tf.function(forward)

# Execute the compiled function
result = compiled_forward(x, layer)

# Assertions to verify correctness
# 1. Check shape
assert result.shape == (2, 2), "Output shape mismatch"

# 2. Check values
# Calculate expected length manually
expected_len = len(tf.keras.layers.serialize(layer))
expected_result = x * expected_len

# Assert that the compiled result matches the expected result
assert tf.reduce_all(tf.equal(result, expected_result)), "Output values mismatch"

print("Test passed: tf.keras.layers.serialize works inside tf.function.")
