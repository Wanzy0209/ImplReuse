import torch
import torch.nn as nn

def test_margin_ranking_loss_mps():
    """
    Test case for torch.nn.MarginRankingLoss on the MPS device.
    
    Context: Based on Issue 160237, where 'aten::grid_sampler_3d' was not 
    implemented for MPS. This test verifies if the similar API, 
    torch.nn.MarginRankingLoss, is implemented and functions correctly 
    on the MPS device.
    """
    # Check if MPS is available to replicate the environment of the bug report
    if not torch.backends.mps.is_available():
        print("MPS device is not available. Skipping MPS-specific test.")
        return

    device = torch.device("mps")

    # Create sample inputs
    # x1, x2: 1D mini-batch Tensors
    # target: label Tensor (containing 1 or -1)
    input1 = torch.randn(10, requires_grad=True, device=device)
    input2 = torch.randn(10, requires_grad=True, device=device)
    target = torch.ones(10, device=device)

    # Initialize the MarginRankingLoss module
    margin_ranking_loss = nn.MarginRankingLoss()

    # Attempt to compute the loss on MPS
    try:
        output = margin_ranking_loss(input1, input2, target)
        
        # Verify the output properties
        assert output.dim() == 0, "Loss output should be a scalar (0-dim tensor)"
        assert not torch.isnan(output), "Loss output should not be NaN"
        assert output.item() >= 0, "MarginRankingLoss value should be non-negative"
        
        print(f"Test Passed: MarginRankingLoss works on MPS. Loss value: {output.item()}")

    except NotImplementedError as e:
        print(f"Test Failed: The operator is not currently implemented for the MPS device.")
        print(f"Error: {e}")
        raise

if __name__ == "__main__":
    test_margin_ranking_loss_mps()
