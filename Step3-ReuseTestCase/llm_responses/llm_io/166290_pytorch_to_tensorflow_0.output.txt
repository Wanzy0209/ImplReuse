import tensorflow as tf

class TestSqueezeDivergence(tf.test.TestCase):
    def test_squeeze_logic(self):
        """
        Adapts the PyTorch fuzzer test case to TensorFlow.
        Verifies that tf.squeeze behaves consistently between eager and graph (tf.function) modes.
        """
        # Set seed for reproducibility
        tf.random.set_seed(974450504)

        # Define inputs
        # arg_0: size=(17, 30, 17, 3), dtype=bool
        arg_0 = tf.cast(tf.random.uniform((17, 30, 17, 3), minval=0, maxval=2, dtype=tf.int32), tf.bool)
        # arg_1: size=(15,), dtype=int64
        arg_1 = tf.cast(tf.random.uniform((15,), minval=5, maxval=30, dtype=tf.int32), tf.int64)

        # Sentinel tensor to ensure gradient computation logic is preserved
        sentinel = tf.constant(1.0)

        def fuzzed_program(arg_0, arg_1, sentinel):
            var_node_3 = arg_0
            
            # torch.chunk(var_node_3, 3, dim=3)[0]
            # Splits the tensor into 3 parts along dimension 3 and takes the first part.
            var_node_2 = tf.split(var_node_3, 3, axis=3)[0] # shape (17, 30, 17, 1)

            var_node_5 = tf.fill([17], 3) # shape (17,)

            # Gather logic equivalent to torch.gather
            _input_size_var_node_4 = tf.shape(var_node_5)[0]
            _index_var_node_4 = tf.random.uniform((15,), minval=0, maxval=_input_size_var_node_4, dtype=tf.int32)
            var_node_4 = tf.gather(var_node_5, _index_var_node_4, axis=0)

            # Index select logic equivalent to torch.index_select
            _input_size_var_node_1 = tf.shape(var_node_2)[0]
            _index_var_node_1 = tf.random.uniform((15,), minval=0, maxval=_input_size_var_node_1, dtype=tf.int32)
            var_node_1 = tf.gather(var_node_2, _index_var_node_1, axis=0) # shape (15, 30, 17, 1)

            # Original API Under Test: torch.squeeze -> tf.squeeze
            var_node_0 = tf.squeeze(var_node_1) # shape (15, 30, 17)

            # Ensure gradient computation by multiplying with sentinel
            result = var_node_0 * sentinel
            return result

        # 1. Run Eager
        result_eager = fuzzed_program(arg_0, arg_1, sentinel)
        print('✅ eager success')

        # 2. Run Compiled (tf.function)
        compiled_program = tf.function(fuzzed_program)
        result_compiled = compiled_program(arg_0, arg_1, sentinel)
        print('✅ compile success')

        # Verify results match
        self.assertAllEqual(result_eager, result_compiled)
        
        # Verify shape matches expected output (15, 30, 17)
        self.assertShapeEqual(result_eager.numpy(), (15, 30, 17))

if __name__ == '__main__':
    tf.test.main()
