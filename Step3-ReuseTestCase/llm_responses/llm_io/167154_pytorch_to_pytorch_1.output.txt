from torch.nn import functional as F

# Check for MPS availability to ensure the test runs on the correct backend
if not torch.backends.mps.is_available():
    print("MPS is not available. Skipping test.")
else:
    # Setup weight and bias (consistent with the original bug report)
    weight = torch.rand((768, 768), device="mps", dtype=torch.float32)
    bias = torch.rand((768), device="mps", dtype=torch.float32)

    # Create a list of tensors with varying sequence lengths (ragged data)
    # This structure is intended to be handled by torch.nested.nested_tensor
    # and exercises non-contiguous memory layout logic on the backend.
    tensor_list = [
        torch.rand((10, 768), device="mps"),
        torch.rand((15, 768), device="mps"),
        torch.rand((5, 768), device="mps")
    ]

    # Use the similar API: torch.nested.nested_tensor
    # This replaces the torch.as_strided call from the original bug report
    input = torch.nested.nested_tensor(tensor_list, device="mps")

    # Attempt the operation that triggered the buffer allocation regression
    try:
        output = F.linear(input, weight, bias)
        print("Test passed: MPS buffer allocation handled correctly for nested_tensor.")
    except RuntimeError as e:
        # Check for the specific assertion error mentioned in the bug report
        if "buffer is not large enough" in str(e):
            print(f"Bug reproduced: {e}")
