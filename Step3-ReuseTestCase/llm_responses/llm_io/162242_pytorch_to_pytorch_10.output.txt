import numpy as np
import torch


def test_cholesky_determinism():
    # Setup a batch of positive-definite matrices for Cholesky decomposition
    # We create random matrices A and compute A @ A.T + I to ensure positive-definiteness
    torch.manual_seed(42)
    batch_size = 2
    size = 3
    
    # Generate inputs on CPU first
    A = torch.randn(batch_size, size, size, dtype=torch.float32)
    inputs = A @ A.transpose(-1, -2) + torch.eye(size).unsqueeze(0).float()

    # Move to CUDA if available to mimic the original test's environment
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    inputs = inputs.to(device)

    gt_res = None
    gt_grad = None

    for i in range(1000):
        if device.type == 'cuda':
            torch.cuda.empty_cache()

        x = inputs.clone().detach()
        x.requires_grad = True

        # Adapted call site: torch.cholesky instead of torch.scatter
        # Note: cholesky expects square matrices, inputs are [2, 3, 3]
        res = torch.cholesky(x, upper=False)
        
        # Check gradients
        res.backward(torch.ones_like(res))

        print(f"Test {i + 1}/{1000}")
        
        # Capture the first run as the ground truth for comparison
        if i == 0:
            gt_res = res.cpu().detach().numpy()
            gt_grad = x.grad.cpu().numpy()
        else:
            # Assert determinism of the forward pass
            np.testing.assert_allclose(res.cpu().detach().numpy(), gt_res, rtol=1e-5, atol=1e-5)
            # Assert determinism of the backward pass
            np.testing.assert_allclose(x.grad.cpu().numpy(), gt_grad, rtol=1e-5, atol=1e-5)


if __name__ == "__main__":
    test_cholesky_determinism()
