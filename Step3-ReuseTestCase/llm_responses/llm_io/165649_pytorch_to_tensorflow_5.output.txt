import tensorflow as tf
import numpy as np

def test_tf_log10_edge_cases():
    """
    Adapted test case based on PyTorch Issue 165649.
    Original Issue: torch.fmod crashes on int64 tensors when dividend is INT64_MIN and divisor is -1.
    Target API: tf.experimental.numpy.log10
    
    This test verifies that tf.experimental.numpy.log10 handles extreme integer values
    (specifically INT64_MIN and -1) gracefully without crashing, even though log10 is 
    mathematically distinct from fmod. The goal is to check robustness against the 
    specific input values that caused a crash in the original API.
    """
    
    # Recreate the inputs from the original PyTorch bug report
    # Dividend is INT64_MIN
    dividend = np.full((2, 3), np.iinfo(np.int64).min, dtype=np.int64)
    # Divisor is -1
    divisor = np.full((3,), -1, dtype=np.int64)

    print("Dividend tensor (INT64_MIN):", dividend)
    print("Divisor tensor (-1):", divisor)

    # Test 1: Apply log10 to the dividend (INT64_MIN)
    # Expected behavior: Since log10 of a negative number is undefined for real numbers,
    # and the implementation promotes to float, we expect NaNs, not a crash.
    try:
        result_dividend = tf.experimental.numpy.log10(dividend)
        print("Result (log10 of dividend):", result_dividend)
        # Assert that the operation completed and returned a tensor of NaNs
        assert result_dividend.shape == dividend.shape
        assert np.all(np.isnan(result_dividend.numpy()))
        print("Test 1 Passed: log10 handled INT64_MIN gracefully (returned NaNs).")
    except Exception as e:
        print(f"Test 1 Failed: Exception occurred with dividend: {e}")

    # Test 2: Apply log10 to the divisor (-1)
    # Expected behavior: log10(-1) is NaN.
    try:
        result_divisor = tf.experimental.numpy.log10(divisor)
        print("Result (log10 of divisor):", result_divisor)
        # Assert that the operation completed and returned a tensor of NaNs
        assert result_divisor.shape == divisor.shape
        assert np.all(np.isnan(result_divisor.numpy()))
        print("Test 2 Passed: log10 handled -1 gracefully (returned NaNs).")
    except Exception as e:
        print(f"Test 2 Failed: Exception occurred with divisor: {e}")

if __name__ == "__main__":
    test_tf_log10_edge_cases()
