import tensorflow as tf

def test_bessel_j1_non_contiguous():
    """
    Adapted test case for Issue ID: 167154.
    
    Original Bug: MPS backend in PyTorch failed to allocate buffers of the right size 
    for non-contiguous tensors (specifically those with 0-strides/broadcasting) 
    passed to linear layers.
    
    Adaptation: We recreate the non-contiguous tensor structure using TensorFlow's 
    broadcasting (equivalent to 0-stride in PyTorch) and pass it to 
    tf.compat.v1.math.special.bessel_j1 to verify buffer handling.
    """
    
    # Recreate the tensor structure from the original bug report:
    # Shape: (5, 499, 768)
    # Stride: (0, 768, 1) -> The first dimension is broadcasted.
    
    # 1. Create the base tensor.
    # In PyTorch, the base size was calculated based on the stride.
    # Here, the underlying data corresponds to the shape (499, 768).
    base_shape = (499, 768)
    # Create a tensor with enough elements to fill the base shape
    base = tf.range(tf.reduce_prod(tf.constant(base_shape)), dtype=tf.float32)
    base = tf.reshape(base, base_shape)

    # 2. Create the non-contiguous (broadcasted) input tensor.
    # tf.broadcast_to is the TensorFlow equivalent of a tensor with 0-stride.
    # This mimics torch.as_strided(base, size=(5, 499, 768), stride=(0, 768, 1))
    input_tensor = tf.broadcast_to(base, (5, 499, 768))

    # 3. Call the target API.
    # We use tf.compat.v1.math.special.bessel_j1 as requested.
    try:
        result = tf.compat.v1.math.special.bessel_j1(input_tensor)
        
        # 4. Assertions.
        # Verify the operation completed and the output shape is correct.
        assert result.shape == (5, 499, 768), f"Expected shape (5, 499, 768), got {result.shape}"
        assert result.dtype == tf.float32, f"Expected dtype float32, got {result.dtype}"
        
        print("Test passed. API handled non-contiguous (broadcasted) input correctly.")
        return True
        
    except Exception as e:
        print(f"Test failed with error: {e}")
        return False

if __name__ == "__main__":
    test_bessel_j1_non_contiguous()
