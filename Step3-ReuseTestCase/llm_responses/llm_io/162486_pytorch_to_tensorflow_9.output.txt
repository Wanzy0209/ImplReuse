import tensorflow as tf

def test_sparse_slice_split(device: str = '/GPU:0'):
    """
    Adapts the logic of torch.utils.data.random_split to tf.sparse.slice.
    Instead of random splitting, we perform positional slicing to partition 
    the sparse tensor into train/val/test sets.
    """
    try:
        with tf.device(device):
            # Create a SparseTensor to mimic the dataset structure
            # Original: x = torch.randn(100, 3), y = torch.randn(100, 2)
            # We create a sparse tensor with 100 rows
            num_samples = 100
            indices = [[i, 0] for i in range(num_samples)]
            values = tf.random.normal([num_samples]) # Random values
            sparse_dataset = tf.SparseTensor(
                indices=indices, 
                values=values, 
                dense_shape=[num_samples, 1]
            )

            # Calculate split sizes based on ratios [0.7, 0.2, 0.1]
            split_1_size = int(num_samples * 0.7) # 70
            split_2_size = int(num_samples * 0.2) # 20
            split_3_size = num_samples - split_1_size - split_2_size # 10

            # Perform slicing using tf.sparse.slice
            # This replaces random_split by partitioning the tensor sequentially
            train_dataset = tf.sparse.slice(
                sparse_dataset, 
                start=[0, 0], 
                size=[split_1_size, 1]
            )
            
            val_dataset = tf.sparse.slice(
                sparse_dataset, 
                start=[split_1_size, 0], 
                size=[split_2_size, 1]
            )
            
            test_dataset = tf.sparse.slice(
                sparse_dataset, 
                start=[split_1_size + split_2_size, 0], 
                size=[split_3_size, 1]
            )

            # Verify the splits
            assert int(train_dataset.dense_shape[0]) == 70
            assert int(val_dataset.dense_shape[0]) == 20
            assert int(test_dataset.dense_shape[0]) == 10

            print(f"Device {device} worked.")

    except Exception as e:
        print(f"Device {device} failed with error: {e}")

# Test on CPU (should always work)
test_sparse_slice_split(device='/CPU:0')

# Test on GPU (will attempt to run, similar to the original bug's intent)
# Note: This requires a CUDA-enabled TensorFlow installation.
test_sparse_slice_split(device='/GPU:0')
