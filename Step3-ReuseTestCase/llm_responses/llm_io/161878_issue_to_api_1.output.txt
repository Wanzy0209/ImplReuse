```python
import torch
import torch.nn as nn
import time
import torch._inductor

# This test case is derived from Issue #161878 regarding a performance regression
# in PyTorch Inductor (CPU) with AMP static shapes and the C++ wrapper.
#
# The structure of this test leverages the logic pattern found in the similar API:
# tf.compat.v1.local_variables_initializer. That API checks `context.executing_eagerly()`
# to decide between returning a no-op or an initializer op.
#
# Here, we adapt that pattern to PyTorch by explicitly branching between an "Eager"
# execution path and a "Compiled" (Inductor) execution path to measure and compare
# their performance, ensuring the regression (slowdown in the compiled path) is captured.

class SimpleBERTLayer(nn.Module):
    """
    A minimal model mimicking the structure of BERT_pytorch to reproduce the workload
    without requiring the full TorchBench dependency.
    """
    def __init__(self, embed_dim=128, num_heads=8):
        super().__init__()
        self.attention = nn.MultiheadAttention(embed_dim, num_heads, batch_first=True)
        self.norm1 = nn.LayerNorm(embed_dim)
        self.ffn = nn.Sequential(
            nn.Linear(embed_dim, embed_dim * 4),
            nn.GELU(),
            nn.Linear(embed_dim * 4, embed_dim)
        )
        self.norm2 = nn.LayerNorm(embed_dim)

    def forward(self, x):
        # Self-attention block
        attn_out, _ = self.attention(x, x, x)
        x = self.norm1(x + attn_out)
        # Feed-forward block
        ffn_out = self.ffn(x)
        x = self.norm2(x + ffn_out)
        return x

def run_execution_path(model, inputs, use_inductor, amp_enabled):
    """
    Executes the model in either Eager or Inductor mode.
    
    This function mirrors the logic of tf.compat.v1.local_variables_initializer:
    It checks the execution context (use_inductor) and executes the corresponding
    code path, allowing us to isolate the performance characteristics of each.
    """
    if use_inductor:
        # Inductor path (The regression occurred here)
        # The bug report specifically mentions 'cpp wrapper' and 'static shape'
        with torch.no_grad():
            if amp_enabled:
                with torch.autocast(device_type="cpu", dtype=torch.bfloat16):
                    return model(inputs)
            else:
                return model(inputs)
    else:
        # Eager path (Baseline)
        with torch.no_grad():
            if amp_enabled:
                with torch.autocast(device_type="cpu", dtype=torch.bfloat16):
                    return model(inputs)
            else:
                return model(inputs)

def test_inductor_amp_static_shape_regression():
    device = torch.device("cpu")
    
    # Configuration matching the bug report:
    # - batch_size: 2
    # - AMP: Enabled
    # - Static Shape: Fixed input size
    batch_size = 2
    seq_len = 128
    embed_dim = 128
    
    model = SimpleBERTLayer(embed_dim=embed_dim).to(device)
    inputs = torch.randn(batch_size, seq_len, embed_dim, device=device)
    
    # Enable C++ wrapper as mentioned in the bug report
    torch._inductor.config.cpp_wrapper = True
    torch._inductor.config.triton.enabled = False # Ensure CPU path is used

    # 1. Benchmark Eager Mode
    # Warmup
    for _ in range(10):
        run_execution_path(model, inputs, use_inductor=False, amp_enabled=True)
    
    start_time = time.time()
    for _ in range(100):
        run_execution_path(model, inputs, use_inductor=False, amp_enabled=True)
    eager_latency = time.time() - start_time

    # 2. Benchmark Inductor Mode
    # Compile the model
    compiled_model = torch.compile(model, mode="reduce-overhead")
    
    # Warmup (Compilation happens here)
    for _ in range(10):
        run_execution_path(compiled_model, inputs, use_inductor=True, amp_enabled=True)
    
    start_time = time.time()
    for _ in range(100):
        run_execution_path(compiled_model, inputs, use_inductor=True, amp_enabled=True)
    inductor_latency = time.time() - start_time

    speedup = eager_latency / inductor_latency
    
    print(f"Eager Latency: {eager_latency:.4f}s")
    print(f"Inductor Latency: {inductor_latency:.4f}s")
    print(f"Speedup: {speedup:.2f}x")

    # The bug report showed a regression where speedup dropped from ~2.59x to ~1.57x.
    # We assert that the speedup is significantly greater than 1.0 to catch severe regressions.
    # A threshold of 1.2 is chosen as a safe lower bound for a simple model on generic hardware.
    assert speedup > 1.2, (
        f"Performance regression detected. "
        f"Expected speedup > 1.2, but got {speedup:.2f}x. "
        f"This matches the pattern of Issue #161878."
    )

if __name__ == "__main__":
    test_inductor_amp_static_shape_regression()
```
