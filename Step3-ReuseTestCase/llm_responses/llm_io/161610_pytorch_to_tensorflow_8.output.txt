```python
import tensorflow as tf
from typing import NamedTuple

# Define the NamedTuple structure similar to the PyTorch example
class MyNamedTuple(NamedTuple):
    first: tf.Tensor
    second: tf.Tensor

class MyNamedTupleSubclass(MyNamedTuple):
    pass

# Define the function that adds a dynamic attribute
# We use tf.keras.name_scope here as requested by the prompt to adapt to this API
def fn(tup: MyNamedTuple) -> MyNamedTuple:
    with tf.keras.name_scope("dynamic_attr_scope"):
        extra_info = tf.constant(4.0)
        tup.extra_info = extra_info  # Add dynamic attribute
    return tup

# Wrap the function with tf.function to simulate the compilation/graph mode behavior
# This is the TensorFlow equivalent of torch.compile
compiled_fn = tf.function(fn)

print("\nTesting NamedTuple with __setattr__ (Eager):")
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))
setattr_result = fn(extended_tup)
try:
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
except AttributeError as e:
    print(f"AttributeError: {e}")

print("\nTesting NamedTuple with __setattr__ (Compiled/Graph):")
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))
setattr_result = compiled_fn(extended_tup)
try:
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
except AttributeError as e:
    print(f"AttributeError: {e}")
```
