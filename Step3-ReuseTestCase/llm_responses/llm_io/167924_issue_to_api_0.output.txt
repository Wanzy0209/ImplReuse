import torch
import unittest

class TestMPSRepeatInterleave(unittest.TestCase):
    """
    Test case for Issue 167924: Crash on MPS when using repeat_interleave with sliced tensor.
    
    This test leverages the pattern from tf.compat.v1.global_variables_initializer,
    which checks the execution context (eager vs graph) before performing initialization.
    Here, we check the device context (MPS availability) before running the operation.
    """

    def test_repeat_interleave_sliced_tensor(self):
        # Pattern from tf.compat.v1.global_variables_initializer:
        # Check context/environment before execution.
        # if context.executing_eagerly(): ...
        if not torch.backends.mps.is_available():
            self.skipTest("MPS backend is not available")

        # Original Bug Reproduction Logic
        # Original API Under Test: torch.arange
        counts = torch.tensor([0, 1, 0], device="mps")
        data = torch.arange(2, device="mps")

        # The operation that caused the segfault
        # Using a non-prefix slice counts[1:3]
        # counts[1:3] results in [1, 0]
        result = data.repeat_interleave(counts[1:3], dim=0)

        # Verify the result is correct to ensure no silent corruption
        # data is [0, 1]. Repeats are [1, 0].
        # 0 repeated 1 time -> [0]
        # 1 repeated 0 times -> []
        # Expected: [0]
        expected = torch.tensor([0], device="mps")
        self.assertTrue(torch.equal(result, expected))

if __name__ == "__main__":
    unittest.main()
