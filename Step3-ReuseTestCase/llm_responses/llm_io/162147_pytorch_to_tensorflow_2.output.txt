import tensorflow as tf
import numpy as np
import math

class FastLearnedCellX3(tf.keras.layers.Layer):
    """
    Adapted from the PyTorch FastLearnedCellX3 to test tf.experimental.numpy.tril.
    The original module uses complex indexing (torch.index_select) which was reported as slow.
    This adaptation replaces the indexing logic with the similar API (tril) to verify
    its behavior on the complex tensor shapes defined in the original issue.
    """
    def __init__(self, D_in, H, D_out,
                 L_w1=12, L_w2=12, L_b2=12,
                 k1=3, k2=3, k3=3, tau1=1.0, tau2=1.0, tau3=1.0,
                 d_addr=64,           # address bottleneck
                 learn_addr=False,
                 learn_tape_w1=True, learn_tape_w2=True, learn_tape_b2=True):
        super().__init__()
        self.D_in, self.H, self.D_out = D_in, H, D_out

        # keep sizes as plain Python ints (compile-friendly)
        self.L_w1, self.L_w2, self.L_b2 = int(L_w1), int(L_w2), int(L_b2)
        self.k1, self.k2, self.k3 = int(k1), int(k2), int(k3)
        self.t1, self.t2, self.t3 = float(tau1), float(tau2), float(tau3)

        # Initialize value tapes (Weights) similar to the original PyTorch code
        # W1: [L_w1, H, D_in]
        self.W1 = tf.Variable(
            tf.linalg.l2_normalize(tf.random.normal([self.L_w1, H, D_in]), axis=(1, 2)),
            trainable=learn_tape_w1,
            name='W1'
        )
        # W2: [L_w2, D_out, H]
        self.W2 = tf.Variable(
            tf.linalg.l2_normalize(tf.random.normal([self.L_w2, D_out, H]), axis=(1, 2)),
            trainable=learn_tape_w2,
            name='W2'
        )
        # b2: [L_b2, D_out]
        self.b2 = tf.Variable(
            tf.linalg.l2_normalize(tf.random.normal([self.L_b2, D_out]), axis=1),
            trainable=learn_tape_b2,
            name='b2'
        )

    def call(self, inputs):
        """
        Forward pass adapted to use tf.experimental.numpy.tril.
        Instead of selecting indices via index_select, we apply a lower triangular mask
        to the weight tensors to test the API's broadcasting capabilities over the
        leading dimensions (the 'L' dimension).
        """
        # Apply tril to W1 [L, H, D_in]. 
        # The mask [H, D_in] is broadcasted across L.
        masked_W1 = tf.experimental.numpy.tril(self.W1)

        # Apply tril to W2 [L, D_out, H].
        masked_W2 = tf.experimental.numpy.tril(self.W2)

        # Apply tril to b2 [L, D_out].
        masked_b2 = tf.experimental.numpy.tril(self.b2)

        return masked_W1, masked_W2, masked_b2

def test_fast_learned_cell_tril():
    # Setup parameters matching the original bug report context
    D_in, H, D_out = 128, 256, 10
    L_w1, L_w2, L_b2 = 12, 12, 12
    
    # Instantiate the layer
    layer = FastLearnedCellX3(D_in, H, D_out, L_w1, L_w2, L_b2)

    # Create a dummy input tensor [Batch, D_in]
    # Note: The original logic was complex, but here we just need to trigger the layer call
    x = tf.random.normal([32, D_in])

    # Execute the forward pass
    w1_out, w2_out, b2_out = layer(x)

    # --- Verification of tf.experimental.numpy.tril behavior ---

    # 1. Check output shapes are preserved
    assert w1_out.shape == (L_w1, H, D_in), f"Shape mismatch for W1: {w1_out.shape}"
    assert w2_out.shape == (L_w2, D_out, H), f"Shape mismatch for W2: {w2_out.shape}"
    assert b2_out.shape == (L_b2, D_out),   f"Shape mismatch for b2: {b2_out.shape}"

    # 2. Verify that the upper triangle is zeroed out
    # We check the element at [0, 1] in the last two dimensions for all leading dimensions.
    # Since k=0 (default), the diagonal (0,0) is kept, but (0,1) is strictly upper.
    
    # Check W1 upper triangle
    assert tf.reduce_all(w1_out[:, 0, 1] == 0.0), "Upper triangle of W1 should be zero"
    
    # Check W2 upper triangle
    assert tf.reduce_all(w2_out[:, 0, 1] == 0.0), "Upper triangle of W2 should be zero"
    
    # Check b2 upper triangle
    assert tf.reduce_all(b2_out[:, 0, 1] == 0.0), "Upper triangle of b2 should be zero"

    # 3. Verify that the diagonal (lower triangle) is preserved (non-zero)
    # We check the element at [0, 0] in the last two dimensions.
    # Note: Random init implies these are non-zero with high probability.
    
    assert tf.reduce_any(w1_out[:, 0, 0] != 0.0), "Diagonal of W1 should be preserved"
    assert tf.reduce_any(w2_out[:, 0, 0] != 0.0), "Diagonal of W2 should be preserved"
    assert tf.reduce_any(b2_out[:, 0, 0] != 0.0), "Diagonal of b2 should be preserved"

    print("Test passed: tf.experimental.numpy.tril behaves correctly on complex tensor shapes.")

if __name__ == "__main__":
    test_fast_learned_cell_tril()
