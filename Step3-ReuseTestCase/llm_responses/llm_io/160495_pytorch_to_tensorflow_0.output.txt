import tensorflow as tf
import numpy as np

def test_tf_strings_as_string_complex():
    """
    Adapted test case based on PyTorch Issue 160495.
    Original Issue: torch.add (inductor) crashes with 0-dim complex tensors.
    Target API: tf.strings.as_string
    
    This test verifies that tf.strings.as_string handles 0-dimensional 
    complex tensors correctly in both eager and graph (compiled) modes.
    """
    
    # Recreate the tensors from the PyTorch bug report
    # x = torch.randn(1, dtype=torch.complex64)
    x = tf.constant([1.5 + 2.5j], dtype=tf.complex64)

    # y = torch.empty((), dtype=torch.complex64)
    # Using a specific value for reproducibility, but matching shape (0-dim) and dtype
    y = tf.constant(0.0 + 0.0j, dtype=tf.complex64)

    # Define eager function
    def eager_fn(tensor):
        return tf.strings.as_string(tensor)

    # Define compiled function (analogous to torch.compile)
    @tf.function
    def compiled_fn(tensor):
        return tf.strings.as_string(tensor)

    print("Testing tf.strings.as_string with 0-dim complex tensor (y)...")
    try:
        res_eager = eager_fn(y)
        print(f"Eager success: {res_eager.numpy()}")
        assert isinstance(res_eager, tf.Tensor)
        assert res_eager.dtype == tf.string
    except Exception as e:
        print(f"Eager failed: {e}")

    try:
        res_compiled = compiled_fn(y)
        print(f"Compiled success: {res_compiled.numpy()}")
        assert isinstance(res_compiled, tf.Tensor)
        assert res_compiled.dtype == tf.string
    except Exception as e:
        print(f"Compiled failed: {e}")

    print("\nTesting tf.strings.as_string with 1-dim complex tensor (x)...")
    try:
        res_eager = eager_fn(x)
        print(f"Eager success: {res_eager.numpy()}")
        assert isinstance(res_eager, tf.Tensor)
        assert res_eager.dtype == tf.string
    except Exception as e:
        print(f"Eager failed: {e}")

    try:
        res_compiled = compiled_fn(x)
        print(f"Compiled success: {res_compiled.numpy()}")
        assert isinstance(res_compiled, tf.Tensor)
        assert res_compiled.dtype == tf.string
    except Exception as e:
        print(f"Compiled failed: {e}")

if __name__ == "__main__":
    test_tf_strings_as_string_complex()
