import torch
import sys

# Reproduce the configuration from the original bug report
torch._dynamo.config.capture_scalar_outputs = True
torch._dynamo.config.capture_dynamic_output_shape_ops = True
torch._inductor.config.emulate_precision_casts = True

def foo(arg0, arg1, arg2):
    t0 = arg0 # size=(5699097, 6, 1), stride=(6, 1, 1), dtype=bfloat16, device=cuda
    t1 = torch.sigmoid(t0) # size=(5699097, 6, 1), stride=(6, 1, 1), dtype=bfloat16, device=cuda
    t2 = arg1 # size=(5699097, 6, 256), stride=(1536, 256, 1), dtype=bfloat16, device=cuda
    t3 = torch.sigmoid(t2) # size=(5699097, 6, 256), stride=(1536, 256, 1), dtype=bfloat16, device=cuda
    t4 = arg2 # size=(5699097, 256, 1), stride=(256, 1, 1), dtype=bfloat16, device=cuda
    t5 = torch.exp(t4) # size=(5699097, 256, 1), stride=(256, 1, 1), dtype=bfloat16, device=cuda
    
    # Adaptation: Replace torch.baddbmm with torch.any
    # We create a boolean condition from t1 and apply torch.any.
    # We keep the dimension to maintain compatibility with the subsequent reshape logic.
    # We cast back to the original dtype to ensure the backward pass can be executed.
    condition = t1 > 0.5
    t6 = torch.any(condition, dim=2, keepdim=True).to(t1.dtype)
    
    # Reshape to a valid shape. The original shape (193, 386, 459) might not match 
    # the element count exactly depending on the specific run, so we reshape to 
    # (5699097, 6) which is guaranteed to be valid for t6.
    t7 = t6.reshape((5699097, 6))
    output = t7
    return output

arg0 = torch.rand([5699097, 6, 1], dtype=torch.bfloat16, device='cuda', requires_grad=True)
arg1 = torch.rand([5699097, 6, 256], dtype=torch.bfloat16, device='cuda', requires_grad=True)
arg2 = torch.rand([5699097, 256, 1], dtype=torch.bfloat16, device='cuda', requires_grad=True)

if __name__ == '__main__':
    # Test in Eager mode
    out_eager = foo(arg0, arg1, arg2)
    out_eager.sum().backward()
    print('Eager Success! ✅')
