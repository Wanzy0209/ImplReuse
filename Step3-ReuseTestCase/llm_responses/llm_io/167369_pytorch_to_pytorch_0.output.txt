```python
import torch

class Config:
    def __repr__(self):
        return "Config()"

def forward(x, config):
    # Calling repr() on non-constant user object
    # This previously triggered the bug
    return x * len(repr(config))

def test_compile_repr():
    config = Config()
    x = torch.randn(2, 2)

    # Compile the function with fullgraph=True as specified in the bug report
    compiled = torch.compile(forward, fullgraph=True)
    
    # Execute the compiled function
    result = compiled(x, config)
    
    # Execute the original (eager) function for comparison
    expected = forward(x, config)
    
    # Verify that the compiled function produces the same result
    assert torch.allclose(result, expected), "Compiled output does not match eager output"
    
    print("Test passed: repr() is handled correctly by torch.compile")

if __name__ == "__main__":
    test_compile_repr()
```
