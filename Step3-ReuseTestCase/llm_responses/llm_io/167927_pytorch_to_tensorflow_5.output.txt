import tensorflow as tf
import unittest

class TestAddQueueRunnerModeConflict(unittest.TestCase):
    """
    Adapted from PyTorch Issue 167927:
    torch.compile(fullgraph=True) should accept torch.compiler.disable.

    The PyTorch issue describes a conflict between a strict compilation mode 
    (fullgraph=True) and an explicit exclusion mechanism (torch.compiler.disable),
    resulting in an Unsupported error.

    This test case adapts that logic to the TensorFlow API 
    tf.compat.v1.train.add_queue_runner. This API is designed for Graph mode 
    (TF1 style) and is documented as incompatible with Eager execution (TF2 default).
    
    We verify that calling this API in the default "strict" (Eager) environment 
    raises an error, mirroring the behavior of the PyTorch bug where a specific 
    mode configuration causes an operation to fail.
    """

    def test_add_queue_runner_in_eager_mode(self):
        # Ensure we are in eager mode (default in TF2)
        self.assertTrue(tf.executing_eagerly(), "Test requires eager execution")

        # Setup: Create a simple queue and queue runner
        # This mimics the setup of a function to be compiled/managed
        queue = tf.queue.FIFOQueue(capacity=10, dtypes=tf.float32)
        enqueue_op = queue.enqueue(1.0)
        runner = tf.compat.v1.train.QueueRunner(queue, [enqueue_op])

        # Action: Attempt to add the queue runner
        # In PyTorch: torch.compile(..., fullgraph=True) raises error on disable
        # In TF: add_queue_runner raises error in Eager mode
        with self.assertRaises(RuntimeError) as context:
            tf.compat.v1.train.add_queue_runner(runner)

        # Verification: Check that the error is related to the mode conflict
        # The PyTorch error was "Unsupported: Skip calling..."
        # The TF error should mention incompatibility with eager execution
        self.assertIn("eager", str(context.exception).lower())

if __name__ == "__main__":
    unittest.main()
