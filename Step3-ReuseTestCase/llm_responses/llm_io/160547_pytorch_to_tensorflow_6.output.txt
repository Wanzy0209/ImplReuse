import tensorflow as tf
from collections import namedtuple

def test_queue_runner_namedtuple():
    """
    Adapts the PyTorch NamedTuple export test to tf.compat.v1.train.QueueRunner.
    
    Original Bug Logic:
    - PyTorch export fails when tracing with a NamedTuple input in non-strict mode.
    - It succeeds when inputs are converted to kwargs.
    
    Adaptation for TensorFlow:
    - Since QueueRunner does not have a 'strict' mode or automatic input unpacking 
      like PyTorch's export, we test the robustness of the API when inputs are 
      provided via a NamedTuple structure (simulating the 'workaround' or 
      structured input scenario).
    """
    
    # Define a NamedTuple to hold the QueueRunner arguments
    RunnerArgs = namedtuple('RunnerArgs', ['queue', 'enqueue_ops', 'close_op'])

    # Setup TensorFlow graph components
    # Note: QueueRunner is TF1 specific, usually runs in graph mode or compat.v1
    q = tf.queue.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[])
    elem = tf.placeholder(tf.float32, shape=[])
    enqueue_op = q.enqueue(elem)
    close_op = q.close()

    # Create the NamedTuple input
    # This mirrors the 'inp = Point(...)' in the original test
    args = RunnerArgs(queue=q, enqueue_ops=[enqueue_op], close_op=close_op)

    # Test 1: Initialize QueueRunner using fields from the NamedTuple
    # This corresponds to the 'workaround' logic in the original bug report
    # (converting namedtuple to kwargs/fields)
    try:
        qr = tf.compat.v1.train.QueueRunner(
            queue=args.queue,
            enqueue_ops=args.enqueue_ops,
            close_op=args.close_op
        )
        print("Test 1 (NamedTuple fields): Success")
        assert qr is not None
        assert len(qr.enqueue_ops) == 1
    except Exception as e:
        print(f"Test 1 (NamedTuple fields): Failed - {e}")
        raise

    # Test 2: Initialize QueueRunner using direct arguments (Control group)
    # This corresponds to the 'succeeds' case in the original test
    try:
        qr_direct = tf.compat.v1.train.QueueRunner(
            queue=q,
            enqueue_ops=[enqueue_op],
            close_op=close_op
        )
        print("Test 2 (Direct args): Success")
        assert qr_direct is not None
    except Exception as e:
        print(f"Test 2 (Direct args): Failed - {e}")
        raise

    print("All tests passed.")

if __name__ == "__main__":
    test_queue_runner_namedtuple()
