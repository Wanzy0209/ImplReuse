import tensorflow as tf
import numpy as np

def test_vdot():
    """
    Adapted test case based on the PyTorch index_select bug.
    The original bug involved a failure when handling 0-dimensional (scalar) tensors.
    This test verifies if tf.keras.ops.vdot can handle scalar inputs correctly.
    """
    
    # Test Case 1: Real scalar inputs
    # Creating 0-dimensional tensors (scalars)
    a = tf.constant(2.0)
    b = tf.constant(3.0)

    try:
        output = tf.keras.ops.vdot(a, b)
        print(f"vdot test succeeds with scalar inputs. output: {output}")
        # Verify correctness: 2.0 * 3.0 = 6.0
        assert np.isclose(output.numpy(), 6.0), f"Expected 6.0, got {output.numpy()}"
    except Exception as e:
        print(f"vdot test fails with scalar inputs: {e}")

    # Test Case 2: Complex scalar inputs
    # The provided implementation details for vdot show specific handling for complex numbers.
    a_complex = tf.constant(1.0 + 2.0j)
    b_complex = tf.constant(3.0 + 4.0j)

    try:
        output_complex = tf.keras.ops.vdot(a_complex, b_complex)
        print(f"vdot test succeeds with complex scalar inputs. output: {output_complex}")
        # Verify correctness: dot(conj(1+2j), 3+4j) = dot(1-2j, 3+4j) = 11 - 2j
        assert np.isclose(output_complex.numpy(), 11.0 - 2.0j), f"Expected 11-2j, got {output_complex.numpy()}"
    except Exception as e:
        print(f"vdot test fails with complex scalar inputs: {e}")

if __name__ == "__main__":
    test_vdot()
