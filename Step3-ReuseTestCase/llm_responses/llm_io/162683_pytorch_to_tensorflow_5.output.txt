```python
import tensorflow as tf
import time

# Adapted from the original PyTorch test case to benchmark tf.dtypes.saturate_cast
# Original shapes were pairs for matmul (A, B). saturate_cast operates on a single tensor.
# We use the first shape from each pair to define the input tensor.
shapes = [
     (1, 12, 10, 64),
     (1, 12, 10, 10),
]

def benchmark_saturate_cast(shape, input_dtype=tf.float16, target_dtype=tf.int8, repeat=500):
    # Create tensor with values in range [-1, 1] similar to the original PyTorch setup
    # (torch.empty(...).uniform_(0,1) * 2 - 1)
    A = tf.random.uniform(shape, minval=-1.0, maxval=1.0, dtype=input_dtype)
    
    # warm up
    for _ in range(5000):
        _ = tf.dtypes.saturate_cast(A, target_dtype)
        
    # run
    times = []
    for i in range(repeat):
        start = time.time()
        _ = tf.dtypes.saturate_cast(A, target_dtype)
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    for shape in shapes:
        t = benchmark_saturate_cast(shape)
        print(f"{shape} saturate_cast(float16 -> int8) -> {t:.3f} us")
```
