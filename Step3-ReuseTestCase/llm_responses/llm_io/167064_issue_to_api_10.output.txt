import torch
import torch.distributions

def test_broadcast_to_with_distribution_validation_disabled():
    """
    Test that torch.broadcast_to works correctly even if the global
    distribution validation state is set to False, which is a side effect
    reported in the bug (Issue 167064) related to torch.compile.
    """
    # Save the original state of distribution validation
    original_validate = torch.distributions.Distribution.get_default_validate_args()

    # Reproduce the side effect mentioned in the bug report.
    # The bug indicates that torch.compile (triggered by redundant global code)
    # calls set_default_validate_args(False).
    torch.distributions.Distribution.set_default_validate_args(False)

    try:
        # Test the similar API: torch.broadcast_to
        # We verify that the API functions correctly under the conditions
        # created by the bug's side effect.
        x = torch.tensor([1.0, 2.0, 3.0])
        shape = (2, 3)
        
        # Call torch.broadcast_to
        result = torch.broadcast_to(x, shape)

        # Assertions to ensure correctness
        assert result.shape == shape, f"Expected shape {shape}, but got {result.shape}"
        assert torch.equal(result[0], x), "First row should match input tensor"
        assert torch.equal(result[1], x), "Second row should match input tensor"
        
    finally:
        # Restore the original state to avoid affecting other tests
        torch.distributions.Distribution.set_default_validate_args(original_validate)

if __name__ == "__main__":
    test_broadcast_to_with_distribution_validation_disabled()
    print("Test passed.")
