import tensorflow as tf

def test_bessel_y0_non_contiguous():
    """
    Adapted test case for Issue ID: 167154.
    Verifies that tf.math.special.bessel_y0 handles non-contiguous tensors
    (simulating the stride=0 broadcast behavior) correctly on the MPS backend.
    """
    
    # Check for GPU availability (MPS on macOS is mapped to GPU in TensorFlow)
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("Warning: No GPU found. This test is intended for MPS (GPU) execution.")
    
    # Execute on the GPU device to target the MPS backend
    with tf.device('/gpu:0'):
        # Recreate the non-contiguous tensor scenario from the PyTorch bug.
        # PyTorch used shape=(5, 499, 768) with stride=(0, 768, 1).
        # The stride of 0 in the first dimension implies broadcasting.
        
        # Create a base tensor with shape (1, 499, 768)
        # This mimics the underlying storage of the PyTorch tensor
        base_shape = (1, 499, 768)
        
        # Use tf.range to mimic torch.arange, cast to float32
        # Calculate total elements for the base to match the "storage" size
        base_numel = 1 * 499 * 768
        base = tf.cast(tf.range(base_numel, dtype=tf.int32), tf.float32)
        base = tf.reshape(base, base_shape)
        
        # Broadcast to (5, 499, 768) to create a non-contiguous view
        # This effectively creates a tensor with stride 0 in the first dimension
        target_shape = (5, 499, 768)
        input_tensor = tf.broadcast_to(base, target_shape)
        
        # Call the target API: tf.math.special.bessel_y0
        # The original bug resulted in a buffer allocation error.
        # We assert that this operation completes without crashing.
        result = tf.math.special.bessel_y0(input_tensor)
        
        # Verify the output properties
        assert result.shape == target_shape, f"Shape mismatch: expected {target_shape}, got {result.shape}"
        assert result.dtype == tf.float32, f"Dtype mismatch: expected float32, got {result.dtype}"
        
        print("Test passed: tf.math.special.bessel_y0 handled non-contiguous input successfully.")

if __name__ == "__main__":
    test_bessel_y0_non_contiguous()
