import tensorflow as tf
import numpy as np

def test_tf_queue_runner_with_tensor_ops():
    """
    Adapted test case based on PyTorch issue 164465.
    Verifies the behavior of tf.compat.v1.train.add_queue_runner 
    alongside tensor operations similar to the original bug report 
    (embedding, view, iota, max).
    """
    # Disable eager execution to use tf.compat.v1.train.add_queue_runner
    tf.compat.v1.disable_eager_execution()

    # Define placeholders for inputs
    # PyTorch: x = torch.ones(1, 64, device='cuda', dtype=torch.int64)
    # PyTorch: y = torch.randn(64, 3072, device='cuda', dtype=torch.bfloat16)
    x = tf.compat.v1.placeholder(tf.int64, shape=[1, 64], name='x')
    y = tf.compat.v1.placeholder(tf.float16, shape=[64, 3072], name='y')

    # Cast input to bfloat16 to match the original bug report's dtype
    y_bf16 = tf.cast(y, tf.bfloat16)

    # Replicate the logic from the PyTorch function
    # embedding = torch.ops.aten.embedding.default(arg1_1, arg0_1)
    embedding = tf.nn.embedding_lookup(y_bf16, x)

    # view = torch.ops.aten.view.default(embedding, [64, 3072])
    view = tf.reshape(embedding, [64, 3072])

    # unsqueeze = torch.ops.aten.unsqueeze.default(view, 0)
    unsqueeze = tf.expand_dims(view, 0)

    # expand = torch.ops.aten.expand.default(unsqueeze, [576, -1, -1])
    # Using broadcast_to as the semantic equivalent of expand
    expand = tf.broadcast_to(unsqueeze, [576, 64, 3072])

    # view_1 = torch.ops.aten.view.default(expand, [2, 8, 36, 64, 3072])
    view_1 = tf.reshape(expand, [2, 8, 36, 64, 3072])

    # permute = torch.ops.aten.permute.default(view_1, [0, 1, 3, 2, 4])
    permute = tf.transpose(view_1, [0, 1, 3, 2, 4])

    # clone = torch.ops.aten.clone.default(permute, memory_format = torch.contiguous_format)
    clone = tf.identity(permute)

    # view_2 = torch.ops.aten.view.default(clone, [2, 18432, 3072])
    view_2 = tf.reshape(clone, [2, 18432, 3072])

    # iota = torch.ops.prims.iota.default(36, start = 0, step = 1, dtype = torch.int64, ...)
    iota = tf.range(0, 36, dtype=tf.int64)

    # view_3 = torch.ops.aten.view.default(iota, [1, 36])
    view_3 = tf.reshape(iota, [1, 36])

    # max_1 = torch.ops.aten.max.default(view_3)
    max_1 = tf.reduce_max(view_3)

    # --- Integration of Target API: tf.compat.v1.train.add_queue_runner ---
    # Create a FIFOQueue to manage inputs (simulating a data pipeline)
    queue = tf.queue.FIFOQueue(capacity=10, 
                               dtypes=[tf.int64, tf.float16], 
                               shapes=[[1, 64], [64, 3072]])
    
    # Enqueue operation
    enqueue_op = queue.enqueue([x, y])

    # Create a QueueRunner
    # The QueueRunner will manage the threads that run the enqueue_op
    qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op])

    # Use the specific API requested: add_queue_runner
    # This adds the QueueRunner to the graph's QUEUE_RUNNERS collection
    tf.compat.v1.train.add_queue_runner(qr)

    # Execution
    with tf.compat.v1.Session() as sess:
        # Initialize variables
        sess.run(tf.compat.v1.global_variables_initializer())
        sess.run(tf.compat.v1.local_variables_initializer())

        # Start the queue runners
        coord = tf.train.Coordinator()
        threads = tf.compat.v1.train.start_queue_runners(coord=coord, sess=sess)

        try:
            # Prepare dummy data
            x_val = np.ones((1, 64), dtype=np.int64)
            y_val = np.random.randn(64, 3072).astype(np.float16)

            # Run the session
            # Note: We feed the placeholders directly. The QueueRunner is active 
            # as per the API requirement, though we are not explicitly dequeuing 
            # in this minimal logic reproduction.
            result = sess.run(max_1, feed_dict={x: x_val, y: y_val})

            # Verify the result
            # The max of range(0, 36) is 35.
            assert result == 35, f"Expected max to be 35, but got {result}"
            print("Test Passed. Result:", result)

        except Exception as e:
            print(f"Test Failed: {e}")
            raise
        finally:
            # Stop the queue runners
            coord.request_stop()
            coord.join(threads)

if __name__ == "__main__":
    test_tf_queue_runner_with_tensor_ops()
