```python
import tensorflow as tf

def addcmul_func(x, y, z):
    return x + (y * z)

# Create tensors
# Note: To strictly mimic the XPU hardware aspect of the original bug, 
# one would need intel_extension_for_tensorflow. This test case uses 
# standard TensorFlow to ensure runnability while preserving the logic flow.
x = tf.random.normal((128,))
y = tf.random.normal((128,))
z = tf.random.normal((128,))

# 1. Eager mode execution
out = addcmul_func(x, y, z)
print("eager mode passed")

# 2. Execution using the similar API (tf.compat.v1.name_scope)
# The original bug occurred when wrapping the function execution.
# Here we wrap the execution in the name_scope context.
with tf.compat.v1.name_scope("addcmul_scope"):
    out_scoped = addcmul_func(x, y, z)

print("name_scope passed")

# Verify results to ensure no crash and correct output
assert out_scoped is not None
assert out_scoped.shape == (128,)
```
