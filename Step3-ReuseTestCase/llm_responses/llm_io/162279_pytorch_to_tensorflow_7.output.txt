```python
import tensorflow as tf
import numpy as np
import shutil
import os

def cleanup(path):
    if os.path.exists(path):
        shutil.rmtree(path)

def create_model_empty():
    """
    Analogous to AnyDimsModelEmpty.
    Uses shape=() to represent an empty dimension list (scalar).
    """
    # shape=() implies a scalar input (empty dimensions)
    inputs = tf.keras.Input(shape=(), dtype=tf.bool, name="input_empty")
    # Mimic the operation logic (pass-through for shape verification)
    outputs = inputs
    return tf.keras.Model(inputs=inputs, outputs=outputs, name="ModelEmpty")

def create_model_null():
    """
    Analogous to AnyDimsModelNull.
    Uses shape=None to represent null/unknown dimensions.
    """
    # shape=None implies unknown shape (null dimensions)
    inputs = tf.keras.Input(shape=None, dtype=tf.bool, name="input_null")
    outputs = inputs
    return tf.keras.Model(inputs=inputs, outputs=outputs, name="ModelNull")

def process(model, x):
    print(f"Model: {model.name}")
    print("Running eager mode...")
    y = model(x)
    print(f"Input shape: {x.shape}, Output shape: {y.shape}")
    
    print("Exporting...")
    export_path = f"/tmp/export_{model.name}"
    cleanup(export_path)
    
    # Export the model (analogous to torch.export.export)
    tf.saved_model.save(model, export_path)
    
    # Verify the export by reloading and checking the signature
    loaded = tf.saved_model.load(export_path)
    print(f"Exported successfully to {export_path}")
    print()

# Test inputs
# Scalar input for the empty model
x_empty = np.array(True)
# Vector input for the null model
x_null = np.array([True, False, True])

# Sequential processing to check for state pollution or caching issues
# The original bug manifested when processing both models one after the other.
process(create_model_empty(), x_empty)
process(create_model_null(), x_null)

# Reverse order to verify independence
process(create_model_null(), x_null)
process(create_model_empty(), x_empty)
```
