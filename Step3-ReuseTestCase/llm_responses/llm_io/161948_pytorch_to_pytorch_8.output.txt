```python
import torch

# List of configurations to test (rows, cols, offset)
# Adapted from threads_list to test different matrix shapes and offsets
test_configs = [
    (5, 5, 0),   # Square matrix, main diagonal
    (5, 5, 1),   # Square matrix, offset 1
    (5, 5, -1),  # Square matrix, offset -1
    (4, 6, 0),   # Rectangular matrix
    (10, 10, 2)  # Larger matrix
]

print("Verifying torch.tril_indices...")

for n, m, k in test_configs:
    # Call the similar API
    # torch.tril_indices(row, col, offset=0, ...)
    indices = torch.tril_indices(row=n, col=m, offset=k)

    # Verify correctness by reconstructing the matrix from indices
    # Create a zero matrix
    matrix = torch.zeros(n, m)
    # Fill the indices returned by tril_indices with 1
    matrix[indices[0], indices[1]] = 1

    # Compare with the standard torch.tril function to ensure correctness
    expected_matrix = torch.tril(torch.ones(n, m), diagonal=k)

    # Assert that the reconstructed matrix matches the expected lower triangle
    assert torch.equal(matrix, expected_matrix), \
        f"Test failed for n={n}, m={m}, k={k}"

    print(f"Passed: n={n}, m={m}, k={k}")

print("All tests passed.")
```
