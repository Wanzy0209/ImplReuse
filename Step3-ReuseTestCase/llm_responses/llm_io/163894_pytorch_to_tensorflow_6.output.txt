import tensorflow as tf
import numpy as np

def fuzzed_program(arg_0, sentinel):
    # var_node_1: size=(1, 2), stride=(2, 1), dtype=int64
    var_node_1 = arg_0

    # var_node_5: size=(1, 2), dtype=int32
    var_node_5 = tf.fill([1, 2], tf.cast(-66, tf.int32))
    
    # var_node_6: size=(1, 2), dtype=int64
    var_node_6 = tf.fill([1, 2], tf.cast(77, tf.int64))
    
    # var_node_4: add(int32, int64) -> int32 (PyTorch behavior)
    # TensorFlow promotes to int64, so we cast to match the intended flow
    var_node_6_int32 = tf.cast(var_node_6, tf.int32)
    var_node_4 = tf.add(var_node_5, var_node_6_int32)

    # var_node_7: size=(1, 2), dtype=int32
    var_node_7 = tf.fill([1, 2], tf.cast(-64, tf.int32))
    
    # var_node_3: mul(int32, int32) -> int32
    var_node_3 = tf.multiply(var_node_4, var_node_7)

    # --- Similar API Step ---
    # Original: var_node_9 = torch.full((3, 4), False) -> var_node_8 = torch.nonzero(var_node_9)
    # Adaptation: Use tf.keras.ops.convert_to_tensor to convert a value into the tensor flow.
    # We create a numpy array to simulate the data that would be processed.
    # The original comment for var_node_8 indicates size=(1, 2), dtype=int64.
    input_data = np.array([[0, 0]], dtype=np.int64)
    var_node_8 = tf.keras.ops.convert_to_tensor(input_data, dtype=tf.int64)
    # -----------------------

    # var_node_2: add(int32, int64) -> int32
    # Cast var_node_8 to int32 to maintain type consistency with the original logic
    var_node_8_int32 = tf.cast(var_node_8, tf.int32)
    var_node_2 = tf.add(var_node_3, var_node_8_int32)

    # var_node_0: div(int64, int32)
    # Using float division to ensure compatibility with the sentinel multiplication
    var_node_0 = tf.math.divide(tf.cast(var_node_1, tf.float32), tf.cast(var_node_2, tf.float32))

    # Ensure gradient computation by multiplying with sentinel
    result = var_node_0 * sentinel
    
    # Handle complex check (TensorFlow tensors don't have .is_complex() method in the same way, 
    # but we can check dtype)
    if result.dtype.is_complex:
        result = tf.math.real(result)
        
    return result

# Sentinel tensor
sentinel = tf.constant(1.0, dtype=tf.float32)

# Argument 0
arg_0 = tf.constant([[1, 2]], dtype=tf.int64)

# Test Eager Mode
print('Testing Eager Mode...')
try:
    result_eager = fuzzed_program(arg_0, sentinel)
    print(f'✅ eager success: {result_eager.numpy()}')
except Exception as e:
    print(f'❌ eager failure: {e}')

# Test Compiled Mode (tf.function)
print('Testing Compiled Mode (tf.function)...')
try:
    compiled_program = tf.function(fuzzed_program)
    result_compiled = compiled_program(arg_0, sentinel)
    print(f'✅ compile success: {result_compiled.numpy()}')
except Exception as e:
    print(f'❌ compile failure: {e}')

# Compare results
if 'result_eager' in locals() and 'result_compiled' in locals():
    if tf.reduce_all(tf.equal(result_eager, result_compiled)):
        print('✅ Eager and Compiled results match.')
    else:
        print('❌ Divergence detected between Eager and Compiled modes.')
        print(f'Eager: {result_eager}')
