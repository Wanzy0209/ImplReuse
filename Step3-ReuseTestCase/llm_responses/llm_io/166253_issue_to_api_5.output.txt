```python
import torch
import unittest

class TestTorchFullCompile(unittest.TestCase):
    def test_full_with_compile_float64(self):
        """
        Test that torch.full updates the fill value correctly
        when used with torch.compile and dtype=torch.float64.
        
        This test addresses the bug where the value argument seems to be fixed
        as the value in the first call after torch.compile.
        """
        def func(x):
            return torch.full((2, ), x, dtype=torch.float64)

        # Compile the function
        compiled_func = torch.compile(func)

        # First call with value 5.0
        x1 = torch.tensor(5.0, dtype=torch.float64)
        result1 = compiled_func(x1)
        expected1 = torch.tensor([5.0, 5.0], dtype=torch.float64)
        self.assertTrue(torch.equal(result1, expected1), 
                        f"First call failed: expected {expected1}, got {result1}")

        # Second call with value 10.0
        # Bug: In the reported issue, this returns [5.0, 5.0] instead of [10.0, 10.0]
        x2 = torch.tensor(10.0, dtype=torch.float64)
        result2 = compiled_func(x2)
        expected2 = torch.tensor([10.0, 10.0], dtype=torch.float64)
        self.assertTrue(torch.equal(result2, expected2), 
                        f"Second call failed: expected {expected2}, got {result2}")

if __name__ == "__main__":
    unittest.main()
```
