import torch

def test_quantized_cholesky_solve():
    # Create standard float tensors for the linear system Ax = b
    # A is the Cholesky factor (lower triangular)
    A = torch.tensor([[1.0, 0.0], [1.0, 1.0]])
    # b is the right-hand side
    b = torch.tensor([[1.0], [2.0]])

    # Quantize the inputs to QuantizedCPU
    # Note: Similar to the zeros_like issue, linear algebra ops may not support quantized tensors.
    quant_A = torch.quantize_per_tensor(A, scale=1.0, zero_point=0, dtype=torch.quint8)
    quant_b = torch.quantize_per_tensor(b, scale=1.0, zero_point=0, dtype=torch.quint8)

    print("Quantized A:", quant_A)
    print("Quantized b:", quant_b)

    try:
        # Attempt to solve the linear system using quantized tensors
        # This replaces the original torch.zeros_like call
        out_tensor = torch.cholesky_solve(quant_b, quant_A, upper=False)
        
        print("Output tensor:", out_tensor)
    except (NotImplementedError, RuntimeError) as e:
        # Catching expected errors if the backend is not implemented
        print(f"Expected error caught: {type(e).__name__}: {e}")

if __name__ == "__main__":
    test_quantized_cholesky_solve()
