import tensorflow as tf
import pytest

def test_keras_ops_arange_with_issue_dtypes():
    """
    Test tf.keras.ops.arange with dtypes mentioned in Issue 161518.
    The issue highlights memory throughput for float8_e4m3fn, bfloat16, and float32.
    We verify that tf.keras.ops.arange correctly handles these dtypes, 
    preserving the logic of iterating over specific data types.
    """
    # Dtypes from the bug report: torch.float8_e4m3fn, torch.bfloat16, torch.float32
    # Mapping to TensorFlow equivalents
    dtypes = [tf.bfloat16, tf.float32]
    
    # float8_e4m3fn is experimental in TF, check availability
    if hasattr(tf.dtypes, 'float8_e4m3fn'):
        dtypes.insert(0, tf.dtypes.float8_e4m3fn)

    # Try to use GPU if available, mirroring the bug report's context (device="cuda")
    gpus = tf.config.list_physical_devices('GPU')
    device = "/GPU:0" if gpus else "/CPU:0"

    with tf.device(device):
        for dtype in dtypes:
            # Mimic the logic of creating tensors of specific sizes based on element size
            # The bug report calculates size as (128 // element_size) * 1024 * 1024
            # We use a smaller representative size for the unit test
            element_size = tf.dtypes.dtype(dtype).size
            num_elements = (1024 // element_size) * 1024 

            # Use the similar API: tf.keras.ops.arange
            # Note: arange generates a sequence, unlike zeros/cat, but we test the dtype handling
            tensor = tf.keras.ops.arange(0, num_elements, 1, dtype=dtype)

            # Assertions to verify correct generation
            assert tensor.dtype == dtype, f"Expected dtype {dtype}, got {tensor.dtype}"
            assert tensor.shape == (num_elements,), f"Expected shape ({num_elements},), got {tensor.shape}"
            
            # Verify content
            assert tensor[0].numpy() == 0, "Start value incorrect"
            assert tensor[-1].numpy() == num_elements - 1, "End value incorrect"

if __name__ == "__main__":
    test_keras_ops_arange_with_issue_dtypes()
