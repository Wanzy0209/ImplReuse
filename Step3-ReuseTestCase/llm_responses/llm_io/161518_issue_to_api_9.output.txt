import tensorflow as tf
import time

def test_exp_memory_throughput():
    """
    Test case adapted from Issue 161518 (Memory Throughput on B200 for copy_).
    
    The original issue benchmarks memory throughput for elementwise operations 
    (like copy_/cat) in PyTorch. This test adapts that logic to benchmark the 
    similar API tf.keras.backend.exp, which is also an elementwise operation 
    that should be memory-bandwidth bound for large tensors.
    """
    
    # Check for GPU availability
    gpus = tf.config.list_physical_devices('GPU')
    if not gpus:
        print("Test skipped: No GPU found. This benchmark requires a GPU.")
        return

    print(f"Running on GPU: {gpus[0].name}")
    print("Dtype\tTime (ms)\tBandwidth (GiB/s)")

    # Test with different data types, similar to the original issue
    # Note: float8_e4m3fn is supported in newer TF versions, using float32/bfloat16 for compatibility
    dtypes = [tf.float32, tf.bfloat16]

    for dtype in dtypes:
        element_size = tf.dtypes.as_dtype(dtype).size

        # Create a large tensor (approx 128 MiB)
        # 128 * 1024 * 1024 bytes / element_size
        num_elements = (128 * 1024 * 1024) // element_size
        
        with tf.device('/GPU:0'):
            x = tf.zeros(num_elements, dtype=dtype)

            # Warmup runs to avoid initialization overhead
            for _ in range(10):
                _ = tf.keras.backend.exp(x)

            # Benchmark loop
            iterations = 100
            start_time = time.time()
            
            for _ in range(iterations):
                y = tf.keras.backend.exp(x)
            
            # Force synchronization to ensure accurate timing
            _ = y.numpy()
            
            end_time = time.time()

            avg_time_ms = (end_time - start_time) / iterations * 1000

            # Calculate Bandwidth
            # For exp(x): Read x (128 MiB) + Write y (128 MiB) = 256 MiB total IO
            total_io_mib = 256
            # Convert MiB to GiB (1 GiB = 1024 MiB)
            bandwidth_gib_s = (total_io_mib / 1024) / (avg_time_ms / 1000)

            print("\t".join([str(dtype), f"{avg_time_ms:.2f}", f"{bandwidth_gib_s:.2f}"]))

            # Assertion to verify correctness of the operation
            # exp(0) should equal 1 for all elements
            assert tf.reduce_all(tf.equal(y, 1.0)).numpy(), f"Test failed for {dtype}: exp(0) != 1"

if __name__ == "__main__":
    test_exp_memory_throughput()
