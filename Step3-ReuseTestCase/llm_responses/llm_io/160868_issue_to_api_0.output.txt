```python
import tensorflow as tf
import numpy as np
import os

# Force CPU to minimize unrelated noise (similar to the original repro)
os.environ.setdefault("CUDA_VISIBLE_DEVICES", "")

class HistoryTestModel:
    """
    A wrapper to test tf.keras.callbacks.History behavior, 
    mirroring the structure of the original PyTorch SliceHugeStepModel.
    """
    def __init__(self, epochs=0):
        super().__init__()
        self.epochs = epochs

    def get_model(self):
        # Simple model similar to the original minimal reproduction
        return tf.keras.models.Sequential([tf.keras.layers.Dense(10)])

    def run(self):
        model = self.get_model()
        model.compile(optimizer='sgd', loss='mse')
        
        # Instantiate the History callback
        history_cb = tf.keras.callbacks.History()
        
        # Generate dummy data
        x_train = np.arange(100).reshape(5, 20).astype('float32')
        y_train = np.zeros(5).astype('float32')
        
        # Run fit
        # Using epochs=0 mimics the "empty" result of the huge step in the original bug
        print(f"[History] running with epochs={self.epochs}...")
        model.fit(x_train, y_train, epochs=self.epochs, callbacks=[history_cb], verbose=0)
        
        return history_cb

def run_history_empty():
    """
    Test History with epochs=0 (analogous to the empty tensor result 
    from the huge step in the original bug).
    """
    m = HistoryTestModel(epochs=0)
    history = m.run()
    
    # Assertions to ensure no crash and correct empty state
    assert hasattr(history, 'history'), "History object missing 'history' attribute"
    # In TF, history keys might exist but values should be empty lists if epochs=0
    # or the dict might be empty depending on TF version behavior.
    # We check that it doesn't crash and returns a valid object.
    print("[History] OK with epochs=0, history keys:", list(history.history.keys()))

def run_history_normal():
    """
    Test History with normal epochs to ensure standard functionality.
    """
    m = HistoryTestModel(epochs=2)
    history = m.run()
    
    assert 'loss' in history.history, "Loss not recorded in history"
    assert len(history.history['loss']) == 2, f"Expected 2 loss entries, got {len(history.history['loss'])}"
    print("[History] OK with epochs=2, output shape:", (len(history.history['loss']),))

if __name__ == "__main__":
    run_history_empty()
    run_history_normal()
```
