import torch
import triton
import torch._inductor.config as inductor_config
import warnings

def test_inductor_triton_addmm():
    """
    Test case for Issue 161618: Inductor fails on triton main branch.
    
    This test reproduces the logic of the bug report. It leverages the semantic
    of the similar API (tf.compat.v1.logging.error) by capturing and handling
    warnings/errors generated during the autotuning process using Python's
    warnings module.
    """
    if not torch.cuda.is_available():
        print("CUDA is not available. Skipping test.")
        return

    # Setup dimensions and tensors from the bug report
    m = 20120
    k = 1536
    n = 512

    a = torch.randn((m, n)).requires_grad_(False).cuda()
    mat1 = torch.randn((m, k)).requires_grad_(False).cuda()
    mat2 = torch.randn((k, n)).requires_grad_(False).cuda()
    
    # The function under test
    f = lambda a, mat1, mat2: torch.addmm(a, mat1, mat2)

    # Capture warnings to mirror the logging behavior of the similar API
    # (tf.compat.v1.logging.error) which handles outputting messages.
    with warnings.catch_warnings(record=True) as w:
        warnings.simplefilter("always")
        
        with inductor_config.patch(
            max_autotune=True,
            max_autotune_gemm_backends="TRITON",
            autotune_fallback_to_aten=False,
        ):
            compiled = torch.compile(f, dynamic=False)
            
            # Execute the compiled function
            result = compiled(a, mat1, mat2)

        # Verify the result shape is correct
        assert result.shape == (m, n), f"Expected shape ({m}, {n}), got {result.shape}"
        
        # Optional: Check if warnings were raised, as seen in the original bug log
        # if w:
        #     print(f"Captured {len(w)} warnings during execution.")

if __name__ == "__main__":
    test_inductor_triton_addmm()
