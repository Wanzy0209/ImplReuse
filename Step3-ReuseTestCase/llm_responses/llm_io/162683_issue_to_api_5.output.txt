```python
import tensorflow as tf
import time

# Set seed for reproducibility
tf.random.set_seed(0)

# Adapted shapes for banded_triangular_solve based on the original matmul shapes.
# Original matmul shapes: ((1, 12, 10, 64), (1, 12, 64, 10))
# For banded_triangular_solve(bands, rhs):
# rhs shape is [..., M, N]. We map (1, 12, 64, 10) to rhs (M=64, N=10).
# bands shape is [..., K, M]. We use K=2 bands, so shape is (1, 12, 2, 64).
shapes = [
    ((1, 12, 2, 64), (1, 12, 64, 10)),
    ((1, 12, 2, 10), (1, 12, 10, 64)),
]

def benchmark_banded_solve(bands_shape, rhs_shape, dtype=tf.float16, repeat=500):
    # Initialize tensors with random values
    # Using positive values for bands to ensure stability in this simple benchmark
    bands = tf.random.uniform(bands_shape, minval=0.1, maxval=1.0, dtype=dtype)
    rhs = tf.random.uniform(rhs_shape, minval=-1.0, maxval=1.0, dtype=dtype)
    
    # Warm up
    for _ in range(5000):
        _ = tf.linalg.banded_triangular_solve(bands, rhs)
    
    # Run benchmark
    times = []
    for i in range(repeat):
        start = time.time()
        _ = tf.linalg.banded_triangular_solve(bands, rhs)
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
            
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    for bands_shape, rhs_shape in shapes:
        t = benchmark_banded_solve(bands_shape, rhs_shape)
        print(f"{bands_shape} x {rhs_shape}  ->  {t:.3f} us")
```
