import tensorflow as tf

# Adapted from: torch.ones(1, device="cuda").item()
# The original test creates a tensor of size 1 on CUDA and retrieves its value.
# tf.SparseTensorSpec is a type specification for a SparseTensor.
# We adapt the test by creating a spec for a tensor of size 1 and verifying its properties.

def test_sparse_tensor_spec_creation():
    # Create a SparseTensorSpec with shape (1,) corresponding to the size 1 in the original test
    # Note: tf.SparseTensorSpec does not take a 'device' argument, as it is metadata-only.
    spec = tf.SparseTensorSpec(shape=(1,), dtype=tf.float32)

    # Verify the properties (analogous to checking the tensor state)
    assert spec.shape == (1,), f"Expected shape (1,), got {spec.shape}"
    assert spec.dtype == tf.float32, f"Expected dtype float32, got {spec.dtype}"
    
    # Verify the value_type property
    assert spec.value_type == tf.sparse.SparseTensor, "Expected value_type to be tf.sparse.SparseTensor"

    print("Test passed: tf.SparseTensorSpec created and properties verified.")

if __name__ == "__main__":
    test_sparse_tensor_spec_creation()
