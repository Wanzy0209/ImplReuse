```python
import tensorflow as tf

# Enable eager execution (The target API)
tf.compat.v1.enable_eager_execution()

def foo(arg0, arg1):
    # t0 = arg0
    # t2 = t0.clone()
    t2 = tf.identity(arg0)
    
    # t2.fill_diagonal_(t1.item())
    # TensorFlow equivalent: tf.linalg.set_diag
    # arg1 is a scalar (0-d tensor). 
    # We need to expand it to match the diagonal length of t2.
    # t2 shape is (1, 1), so diagonal length is 1.
    diagonal_values = tf.fill([tf.shape(t2)[0]], arg1)
    t2 = tf.linalg.set_diag(t2, diagonal_values)
    
    return t2

# Setup inputs
# PyTorch: arg0 = torch.empty([1, 1], ... requires_grad=True)
# TF: Variable with initial value (zeros is safer than uninitialized for testing)
arg0 = tf.Variable(tf.zeros([1, 1], dtype=tf.float32))
arg1 = tf.Variable(0.0, dtype=tf.float32)

# Run Eager
with tf.GradientTape() as tape:
    out_eager = foo(arg0, arg1)
    loss = tf.reduce_sum(out_eager)

grads = tape.gradient(loss, [arg0, arg1])

# Verify output
assert out_eager.shape == (1, 1)
# Verify gradients exist
assert grads[0] is not None
assert grads[1] is not None

print('Eager Execution Success! ✅')
```
