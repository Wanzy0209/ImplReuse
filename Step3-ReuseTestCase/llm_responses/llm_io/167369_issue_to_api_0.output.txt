import tensorflow as tf
import tf.keras.activations as activations

# Define a custom activation to mimic the user-defined object in the bug report
class CustomActivation:
    def __init__(self, name):
        self.name = name

    def __call__(self, x):
        return x * 2

    def get_config(self):
        return {"name": self.name}

# Register it or just use a simple function for the test
def my_activation(x):
    return tf.nn.relu(x)

def forward(x, activation_fn):
    # The logic: Serialize the object (similar to repr(config) in the bug)
    # This tests if the serialization operation is supported inside the traced context
    serialized_name = activations.serialize(activation_fn)
    
    # Use the result in a calculation to ensure it's traced/executed
    # PyTorch: x * len(repr(config))
    # TensorFlow: x * length of serialized string
    length = tf.strings.length(serialized_name)
    return x * tf.cast(length, dtype=x.dtype)

# Compile the function (equivalent to torch.compile)
# This is the context where the bug occurred in PyTorch
compiled_forward = tf.function(forward)

# Setup inputs
x = tf.random.normal((2, 2))

# Run the compiled function
# This verifies if the similar API (serialize) handles the tracing
# where the original API (repr) failed.
try:
    result = compiled_forward(x, my_activation)
    # Assertion to verify execution and correctness
    assert result.shape == (2, 2)
    print("Test passed: Serialization handled inside tf.function")
except Exception as e:
    print(f"Test failed: {e}")
