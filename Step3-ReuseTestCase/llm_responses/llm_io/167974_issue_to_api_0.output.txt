import torch
import tensorflow as tf
import tf.experimental.numpy as tnp

def test_embeddingbag_2d_include_last_offset():
    """
    Test case for Issue 167974: EmbeddingBag sum has incorrect offsets 
    when input is 2D and include_last_offset is set to True.
    
    This test leverages tf.experimental.numpy.ndim to verify the input 
    dimensionality, which is the specific condition triggering the bug.
    """
    # Setup input as described in the bug report
    input_tensor = torch.tensor([[1, 2, 4, 5], [4, 3, 2, 9]], dtype=torch.long)

    # Leverage the similar API: tf.experimental.numpy.ndim
    # We verify the input is 2D, as the bug specifically occurs with 2D inputs.
    # Convert torch tensor to tensorflow tensor to use the API.
    tf_input = tf.constant(input_tensor.numpy())
    input_rank = tnp.ndim(tf_input)
    
    assert input_rank == 2, f"Expected 2D input to trigger bug condition, got ndim={input_rank}"

    # Initialize EmbeddingBag with include_last_offset=True
    # According to the bug, when input is 2D, offsets are generated as [0, 4]
    # instead of the expected [0, 4, 8].
    embedding_sum = torch.nn.EmbeddingBag(10, 3, mode='sum', include_last_offset=True)

    # Execute the forward pass
    # The bug manifests internally in the generated offsets.
    # We assert the output shape is correct for the 2D input (2 bags).
    output = embedding_sum(input_tensor)
    
    # The input has 2 rows, so we expect 2 output embeddings
    assert output.shape == (2, 3), f"Expected output shape (2, 3), got {output.shape}"

    print("Test passed. Input ndim verified as 2. EmbeddingBag executed with include_last_offset=True.")

if __name__ == "__main__":
    test_embeddingbag_2d_include_last_offset()
