import tensorflow as tf

def test_vdot():
    # Create a multi-dimensional tensor (analogous to 'x' in the original bug)
    a = tf.ones([2, 3])
    # Create a zero-dimensional (scalar) tensor (analogous to 'index' in the original bug)
