import torch
import torch.nn as nn
import torch.nn.functional as F

class VAE(nn.Module):

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super(VAE, self).__init__()
        self.input_dim = input_dim
        self.latent_dim = latent_dim
        self.encoder = nn.Sequential(nn.Linear(input_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim), nn.ReLU())
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_var = nn.Linear(hidden_dim, latent_dim)
        self.decoder = nn.Sequential(nn.Linear(latent_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim), nn.Sigmoid())

    def encode(self, x):
        h = self.encoder(x)
        mu = self.fc_mu(h)
        log_var = self.fc_var(h)
        return (mu, log_var)

    def reparameterize(self, mu, log_var):
        std = torch.exp(0.5 * log_var)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        return self.decoder(z)

    def forward(self, x):
        x = x.view(-1, self.input_dim)
        (mu, log_var) = self.encode(x)
        z = self.reparameterize(mu, log_var)
        reconstruction = self.decode(z)
        return (reconstruction, mu, log_var)

def get_default_model():
    input_dim = 784
    hidden_dim = 400
    latent_dim = 20
    model = VAE(input_dim=input_dim, hidden_dim=hidden_dim, latent_dim=latent_dim)
    return model

def get_sample_inputs():
    batch_size = 32
    input_dim = 784
    x = torch.randn(batch_size, input_dim)
    return (x,)

def main():
    model = get_default_model()
    model.eval()
    inputs = get_sample_inputs()
    
    # Eager mode execution
    with torch.no_grad():
        (reconstruction, mu, log_var) = model(*inputs)
    
    print('Model executed successfully!')
    print(f'Input shape: {inputs[0].shape}')
    print(f'Reconstruction shape: {reconstruction.shape}')
    print(f'Mu shape: {mu.shape}')
    print(f'Log_var shape: {log_var.shape}')
    
    # Test torch.any on eager output
    # The original bug was about accessing .shape on a tuple. 
    # Here we test torch.any on the tensor elements of the tuple.
    any_reconstruction = torch.any(reconstruction > 0.5)
    any_mu = torch.any(mu > 0)
    print(f'Eager any(reconstruction > 0.5): {any_reconstruction}')
    print(f'Eager any(mu > 0): {any_mu}')

    # Compile the model
    compiled_model = torch.compile(model)
    
    with torch.no_grad():
        output_compile = compiled_model(*inputs)
    
    # output_compile is a tuple (reconstruction, mu, log_var)
    # We adapt the test to use torch.any on the elements of the tuple
    # instead of accessing .shape on the tuple itself which causes the bug.
    
    # Unpack the tuple to access tensors
    compiled_reconstruction, compiled_mu, compiled_log_var = output_compile
    
    # Test torch.any on compiled output
    any_compiled_reconstruction = torch.any(compiled_reconstruction > 0.5)
    any_compiled_mu = torch.any(compiled_mu > 0)
    
    print(f'Compile any(reconstruction > 0.5): {any_compiled_reconstruction}')
    print(f'Compile any(mu > 0): {any_compiled_mu}')
    
    # Verify that torch.any works correctly on the compiled outputs
    assert isinstance(any_compiled_reconstruction, torch.Tensor), "torch.any should return a Tensor"
    assert any_compiled_reconstruction.shape == torch.Size([]), "Result of torch.any should be a scalar"
    
    print('Test passed: torch.any works correctly on compiled model outputs.')

if __name__ == '__main__':
    main()
