import tensorflow as tf
import numpy as np

def test_negative_redundant_dtype_conversion():
    """
    Adapted test case based on Issue 161611.
    
    The original bug highlighted a redundant dtype conversion in a PyTorch docstring example
    where a tensor created with a specific dtype was converted back to that same dtype 
    without assignment.
    
    This test verifies the behavior of the similar API (tf.keras.ops.negative) regarding
    dtype preservation and the redundancy of explicit casting.
    """
    # Setup: Create a tensor with a specific dtype (mimicking 'query' in the original bug)
    # We use float32 as it is a standard dtype for such operations.
    query = tf.constant([1.0, 2.0, 3.0], dtype=tf.float32)

    # Operation: Apply the similar API (tf.keras.ops.negative)
    # In the original bug, this corresponds to the operations on attn_bias.
    result = tf.keras.ops.negative(query)

    # Check 1: Verify the operation preserves the dtype.
    # The original bug noted that attn_bias was already query.dtype.
    assert result.dtype == query.dtype, \
        f"Expected output dtype {query.dtype}, but got {result.dtype}"

    # Check 2: Verify that a redundant cast is unnecessary.
    # The original bug had: attn_bias.to(query.dtype) which was redundant.
    # In TensorFlow, the equivalent is tf.cast.
    redundant_cast = tf.cast(result, query.dtype)

    # Since dtypes match, the cast should be a no-op regarding dtype.
    assert redundant_cast.dtype == result.dtype
    assert tf.reduce_all(tf.equal(result, redundant_cast)).numpy()

    # Check 3: Verify the "ineffective assignment" aspect of the original bug.
    # The original bug line `attn_bias.to(query.dtype)` was not assigned, so it did nothing.
    # We verify that calling tf.cast without assignment does not alter the original tensor.
    original_values = result.numpy()
    
    # Perform a cast to a different dtype but do not assign it back to 'result'
    # This mimics the ineffective line in the bug report.
    tf.cast(result, tf.float64) 
    
    # Assert that 'result' remains unchanged
    assert np.array_equal(result.numpy(), original_values), \
        "Unassigned cast should not modify the tensor"
    assert result.dtype == query.dtype, \
        "Unassigned cast should not modify the tensor dtype"

    print("Test passed: Redundant dtype conversion logic verified for tf.keras.ops.negative.")

if __name__ == "__main__":
    test_negative_redundant_dtype_conversion()
