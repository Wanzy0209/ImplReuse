import tensorflow as tf
import numpy as np

# Ensure reproducibility
tf.random.set_seed(9)
np.random.seed(9)

def fuzzed_program(arg_0, sentinel):
    # Use the specified API to convert inputs to tensors
    arg_0 = tf.compat.v1.convert_to_tensor(arg_0)
    sentinel = tf.compat.v1.convert_to_tensor(sentinel)

    var_node_1 = arg_0

    # var_node_5: torch.full((1, 2), -66, dtype=torch.int32)
    var_node_5 = tf.fill((1, 2), tf.cast(-66, tf.int32))

    # var_node_6: torch.full((1, 2), 77, dtype=torch.int64)
    var_node_6 = tf.fill((1, 2), tf.cast(77, tf.int64))
    # PyTorch add(int32, int64) promotes to int32. Cast var_node_6 to int32.
    var_node_6 = tf.cast(var_node_6, tf.int32)

    # var_node_4: add(var_node_5, var_node_6)
    var_node_4 = tf.add(var_node_5, var_node_6)

    # var_node_7: torch.full((1, 2), -64, dtype=torch.int32)
    var_node_7 = tf.fill((1, 2), tf.cast(-64, tf.int32))

    # var_node_3: mul(var_node_4, var_node_7)
    var_node_3 = tf.multiply(var_node_4, var_node_7)

    # var_node_9: torch.full((3, 4), False, dtype=torch.bool)
    var_node_9 = tf.fill((3, 4), False)

    # var_node_8: torch.nonzero(var_node_9)
    # In TensorFlow, tf.where returns indices of True elements.
    # Since var_node_9 is all False, this returns an empty tensor of shape (0, 2).
    var_node_8 = tf.where(var_node_9)

    # var_node_2: add(var_node_3, var_node_8)
    # Broadcasting (1, 2) with (0, 2) results in (0, 2).
    var_node_2 = tf.add(var_node_3, var_node_8)

    # var_node_0: div(var_node_1, var_node_2)
    # PyTorch div on ints results in float, but the comment says dtype=int64.
    # We perform float division and cast to int64 to match the observed behavior.
    # Note: If var_node_2 is empty, the result is empty.
    var_node_0 = tf.cast(tf.math.divide(tf.cast(var_node_1, tf.float32), tf.cast(var_node_2, tf.float32)), tf.int64)

    # Ensure gradient computation by multiplying with sentinel
    result = var_node_0 * sentinel

    # Check for complex (TF doesn't have is_complex() on tensor directly in the same way, but we can check dtype)
    if result.dtype == tf.complex64 or result.dtype == tf.complex128:
        result = tf.math.real(result)

    return result

# Sentinel tensor
sentinel = tf.Variable(1.0, dtype=tf.float32)

# arg_0: torch.randint(0, 3, (1, 2), dtype=torch.int64)
arg_0 = tf.random.uniform((1, 2), minval=0, maxval=3, dtype=tf.int64)

# Test Eager
print("Testing Eager Mode...")
result_eager = fuzzed_program(arg_0, sentinel)
print(f"✅ Eager success. Shape: {result_eager.shape}, Dtype: {result_eager.dtype}")

# Test Compiled (tf.function)
print("\nTesting Compiled Mode (tf.function)...")
compiled_program = tf.function(fuzzed_program)
result_compiled = compiled_program(arg_0, sentinel)
print(f"✅ Compile success. Shape: {result_compiled.shape}, Dtype: {result_compiled.dtype}")

# Compare results
print("\nComparing results...")
# Use numpy for comparison
if np.array_equal(result_eager.numpy(), result_compiled.numpy()):
    print("✅ Results match between Eager and Compiled modes.")
else:
    print("❌ Divergence detected between Eager and Compiled modes!")
    print(f"Eager result: {result_eager.numpy()}")
    print(f"Compiled result: {result_compiled.numpy()}")
