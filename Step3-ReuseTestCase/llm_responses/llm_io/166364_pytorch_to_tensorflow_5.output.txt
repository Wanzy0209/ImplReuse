import tensorflow as tf

def test_add_queue_runner_with_learnable_scalar():
    """
    Test function to verify that tf.compat.v1.train.add_queue_runner
    handles operations involving learnable scalars (Variables) correctly.
    
    This adapts the PyTorch logic of using a learnable scalar (nn.Parameter)
    inside a modification function (score_mod) to the TensorFlow context
    of using a Variable inside an enqueue operation managed by a QueueRunner.
    """
    
    # Queue runners require graph mode execution
    tf.compat.v1.disable_eager_execution()
    
    # Adaptation: nn.Parameter(torch.tensor(0.0)) -> tf.Variable
    # Creating a learnable scalar
    temp = tf.Variable(0.0, dtype=tf.float32, name="learnable_scalar")

    # Adaptation: score_mod function -> Enqueue operation
    # In PyTorch, the logic was 'score = score + temp'.
    # Here, we define an operation that uses the variable to enqueue data.
    queue = tf.compat.v1.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[])
    
    # Define an operation that interacts with the learnable scalar
    # This mimics the dependency on the parameter found in the original bug report
    enqueue_data = temp + 1.0
    enqueue_op = queue.enqueue(enqueue_data)

    # Create the QueueRunner with the operation
    qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op])

    # API Call: tf.compat.v1.train.add_queue_runner
    # This corresponds to the step where the configuration is applied to the system
    try:
        tf.compat.v1.train.add_queue_runner(qr)
    except Exception as e:
        print(f"Failed to add queue runner: {e}")
        raise

    # Verification: Check if the QueueRunner was successfully added to the collection
    runners = tf.compat.v1.get_collection(tf.compat.v1.GraphKeys.QUEUE_RUNNERS)
    assert qr in runners, "QueueRunner was not added to the default collection"

    # Verification: Ensure the graph is valid and the variable interaction works
    # This mimics checking if "Forward works well"
    with tf.compat.v1.Session() as sess:
        sess.run(tf.compat.v1.global_variables_initializer())
        
        # Verify the variable exists and has the initial value
        val = sess.run(temp)
        assert val == 0.0, f"Expected variable value 0.0, got {val}"
        
        print("Test passed: QueueRunner added successfully with learnable scalar.")

if __name__ == "__main__":
    test_add_queue_runner_with_learnable_scalar()
