import tensorflow as tf

def test_softmax_cross_entropy_with_logits():
    """
    Adapted test case for tf.compat.v1.nn.softmax_cross_entropy_with_logits
    based on the structure of the PyTorch DataParallel custom backend issue.
    
    The original issue tests parallel execution on a custom backend.
    This adaptation tests the TensorFlow loss function execution on the available GPU backend.
    """
    
    # Check for GPU availability (mimicking torch.<mybackend>.is_available())
    gpus = tf.config.list_physical_devices('GPU')
    
    if gpus:
        print(f"Detected {len(gpus)} GPU(s)")

        # Original: batch_size = 20, input_size = 10, output_size = 5
        # We adapt this to logits and labels for the loss function.
        batch_size = 20
        num_classes = 5

        # Execute on the first GPU (mimicking model.<mybackend>())
        with tf.device('/GPU:0'):
            # Generate random logits (unscaled log probabilities)
            # Shape: (batch_size, num_classes)
            logits = tf.random.normal([batch_size, num_classes])
            
            # Generate valid probability labels (sum to 1 along the class axis)
            # Shape: (batch_size, num_classes)
            labels = tf.nn.softmax(tf.random.normal([batch_size, num_classes]))

            # Call the API under test
            # Original API: torch.nn.DataParallel (wrapping a model)
            # Similar API: tf.compat.v1.nn.softmax_cross_entropy_with_logits (computing loss)
            loss = tf.compat.v1.nn.softmax_cross_entropy_with_logits(
                labels=labels,
                logits=logits
            )

            # Verify the output shape
            # The loss is computed per sample in the batch
            assert loss.shape == (batch_size,), f"Expected shape ({batch_size},), got {loss.shape}"
            
            # Verify the output type
            assert loss.dtype == tf.float32, f"Expected dtype float32, got {loss.dtype}"

            print("success")
    else:
        # Mimic the 'else: raise' block from the original code
        raise RuntimeError("No GPUs available to run the test")

if __name__ == "__main__":
    test_softmax_cross_entropy_with_logits()
