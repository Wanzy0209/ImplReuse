import torch
import torch.special
import unittest

class TestTorchCompileSimilarAPI(unittest.TestCase):
    """
    Test case generated based on Issue 164301 (torch.compile regression).
    
    The original issue reported a performance regression in torch.compile for 
    mxfp8 quantization operations. This test applies the same reproduction 
    logic (compiling a specific tensor operation) to the similar API 
    torch.special.airy_ai to ensure no functional or compilation regressions 
    exist for this function.
    """

    @unittest.skipIf(not torch.cuda.is_available(), "CUDA not available, skipping GPU test")
    def test_airy_ai_compile_correctness(self):
        # Setup: Create a tensor similar to the scale used in the bug report (M x K)
        # The original bug used 16384x16384, we use a smaller size for unit test efficiency
        M, K = 2048, 2048
        x = torch.randn(M, K, dtype=torch.float32, device='cuda')

        # Define the function using the similar API
        def airy_func(input_tensor):
            return torch.special.airy_ai(input_tensor)

        # Apply torch.compile (the API under test in the original issue)
        compiled_airy_func = torch.compile(airy_func)

        # Run eager execution
        expected = airy_func(x)

        # Run compiled execution
        # Note: The first run triggers compilation, subsequent runs use the compiled kernel
        actual = compiled_airy_func(x)

        # Assertion: Verify that the compiled version produces correct results.
        # This catches potential codegen regressions that might affect correctness
        # or cause crashes, analogous to the performance regression reported.
        self.assertTrue(
            torch.allclose(expected, actual, rtol=1e-4, atol=1e-4),
            "torch.compile produced incorrect results for torch.special.airy_ai"
        )

if __name__ == '__main__':
    unittest.main()
