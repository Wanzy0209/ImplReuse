import tensorflow as tf
import pytest

# Disable TensorFlow v2 behavior to ensure compat.v1 components work as expected
tf.compat.v1.disable_v2_behavior()

class TestReparameterizationTypeConstraints:
    """
    Adapted from the PyTorch issue where torch.export.export failed due to 
    'Constraints violated (seq)'.
    
    In the PyTorch case, the constraint was a dynamic shape check (seq <= 512).
    Here, we adapt the logic to verify the 'constraint' of the 
    ReparameterizationType property on distributions. We check if the 
    distribution's reparameterization type matches the expected static constraint.
    """

    def test_reparameterization_type_fully_reparameterized(self):
        """
        Test that a Normal distribution adheres to the FULLY_REPARAMETERIZED constraint.
        """
        from tf.compat.v1.distributions import Normal, ReparameterizationType

        # Define the distribution
        dist = Normal(loc=0.0, scale=1.0)

        # Define the expected constraint (Type)
        expected_type = ReparameterizationType.FULLY_REPARAMETERIZED

        # Verify the constraint is met
        # This mimics the guard check in the PyTorch error:
        # "Not all values of seq ... in the specified range ... satisfy the generated guard"
        assert dist.reparameterization_type == expected_type, (
            f"Reparameterization constraint violated! "
            f"Expected {expected_type}, got {dist.reparameterization_type}"
        )

    def test_reparameterization_type_not_reparameterized(self):
        """
        Test that a Bernoulli distribution adheres to the NOT_REPARAMETERIZED constraint.
        """
        from tf.compat.v1.distributions import Bernoulli, ReparameterizationType

        # Define the distribution
        dist = Bernoulli(probs=0.5, dtype=tf.float32)

        # Define the expected constraint (Type)
        expected_type = ReparameterizationType.NOT_REPARAMETERIZED

        # Verify the constraint is met
        assert dist.reparameterization_type == expected_type, (
            f"Reparameterization constraint violated! "
            f"Expected {expected_type}, got {dist.reparameterization_type}"
        )

    def test_reparameterization_type_consistency(self):
        """
        Test that the ReparameterizationType instances are consistent across multiple calls,
        ensuring no unexpected state changes (similar to multiple runs in the original bug).
        """
        from tf.compat.v1.distributions import Normal, ReparameterizationType

        # Run 1
        dist1 = Normal(loc=0.0, scale=1.0)
        type1 = dist1.reparameterization_type

        # Run 2 (Simulating the 'multiple_run' aspect of the original test case)
        dist2 = Normal(loc=1.0, scale=2.0)
        type2 = dist2.reparameterization_type

        # Assert consistency
        assert type1 is type2, (
            "ReparameterizationType constraint consistency violated between runs. "
            f"Run 1: {type1}, Run 2: {type2}"
        )
        assert type1 is ReparameterizationType.FULLY_REPARAMETERIZED

if __name__ == "__main__":
    pytest.main([__file__])
