```python
import tensorflow as tf
import os

# Setup: Disable eager execution to use QueueRunners (TF1 style)
tf.compat.v1.disable_eager_execution()

# Define a simple queue and enqueue operation
# Mimicking the "Module" definition
with tf.device("/gpu:0"):  # Mimic torch.device("cuda")
    queue = tf.compat.v1.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[()])
    enqueue_op = queue.enqueue(1.0)
    
    # Create the QueueRunner
    qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op] * 2)
    
    # API Under Test: add_queue_runner
    tf.compat.v1.train.add_queue_runner(qr)
    
    # Operation to fetch data
    dequeue_op = queue.dequeue()

# Profiling setup (mimicking the original test's intent to observe overhead)
log_dir = "./tf_log"
if not os.path.exists(log_dir):
    os.makedirs(log_dir)

with tf.compat.v1.Session() as sess:
    # Initialize
    sess.run(tf.compat.v1.global_variables_initializer())
    
    # Coordinator for threads
    coord = tf.compat.v1.train.Coordinator()
    
    # Start the queue runners (which includes the one we added)
    threads = tf.compat.v1.train.start_queue_runners(sess=sess, coord=coord)
    
    # Run loop mimicking the original test
    # Note: TF Profiler in graph mode is complex, we focus on the execution logic
    try:
        for i in range(10):
            # Fetch data
            val = sess.run(dequeue_op)
            # In the original bug, redundant H2D/D2H happened here.
            # Here we just verify the runner works within the device context.
            assert val == 1.0
    finally:
        coord.request_stop()
        coord.join(threads)
        
print("Test passed: QueueRunner added and executed successfully.")
```
