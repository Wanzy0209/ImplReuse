import tensorflow as tf
import sys

def test_max_pool1d_4d_input():
    """
    Adapted test case based on PyTorch Issue 160053.
    
    Original Issue: torch.nn.functional.pad with mode="circular" fails for 4D input 
    despite the error message claiming 4D is supported.
    
    Target API: tf.nn.max_pool1d
    Similarity: Both APIs handle spatial operations on tensors with specific dimensionality requirements.
    
    This test verifies how tf.nn.max_pool1d handles a 4D input tensor, checking if the 
    error message (if any) accurately reflects the API's dimensionality constraints.
    """
    
    # Create a 4D input tensor, similar to the PyTorch bug report: torch.empty(2,2,2,2)
    # tf.nn.max_pool1d expects a 3D tensor (Batch, Width, Channels) by default.
    input_tensor = tf.zeros((2, 2, 2, 2), dtype=tf.float32)
    
    print(f"Input shape: {input_tensor.shape}")
    
    try:
        # Attempt to apply max_pool1d.
        # 'SAME' padding is used to mirror the 'mode' argument in the original PyTorch code,
        # ensuring padding logic is involved.
        output = tf.nn.max_pool1d(
            input=input_tensor, 
            ksize=2, 
            strides=1, 
            padding='SAME'
        )
        print("Test Result: Operation succeeded (API might support 4D or broadcast it).")
        print(f"Output shape: {output.shape}")
        
    except Exception as e:
        print(f"Test Result: Operation failed with error.")
        print(f"Error Message: {e}")
        
        # In the PyTorch bug, the error message claimed 4D was supported ("Only 2D, 3D, 4D, 5D..."),
        # but the implementation failed.
        # For TensorFlow, we verify that the error message correctly identifies the rank mismatch
        # (expecting 3D, got 4D) rather than being misleading.
        error_str = str(e).lower()
        if "rank" in error_str or "shape" in error_str or "must be rank 3" in error_str:
            print("Assertion Passed: Error message correctly identifies the dimensionality mismatch.")
        else:
            print("Assertion Warning: Error message might be unclear about dimensionality requirements.")
            # We raise an assertion if the error is completely unrelated to shape/rank
            # to ensure we are testing the right aspect.
            assert False, f"Unexpected error type: {e}"

if __name__ == "__main__":
    test_max_pool1d_4d_input()
