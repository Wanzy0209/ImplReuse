import tensorflow as tf
import numpy as np

def test_exp_scalar_input():
    """
    Test case adapted from the logic of the PyTorch index_select bug report.
    The original bug highlighted that index_select failed on MPS when given a 
    zero-dimensional (scalar) index tensor. This test verifies that 
    tf.experimental.numpy.exp correctly handles a zero-dimensional input tensor.
    """
    # Create a zero-dimensional (scalar) tensor, analogous to the index tensor in the bug report
    x = tf.constant(1.0) 
    
    try:
        output = tf.experimental.numpy.exp(x)
        print(f"exp test succeeds for scalar input. output shape: {output.shape}")
        
        # Assertions to verify correct behavior
        assert output.shape == (), f"Expected scalar output shape (), got {output.shape}"
        assert np.isclose(output.numpy(), np.e), f"Expected value {np.e}, got {output.numpy()}"
        
    except Exception as e:
        print(f"exp test fails for scalar input: {e}")
        raise

def test_exp_multi_dim_input():
    """
    Standard test case with multi-dimensional input to ensure general functionality.
    """
    x = tf.ones([2, 3])
    
    try:
        output = tf.experimental.numpy.exp(x)
        print(f"exp test succeeds for multi-dim input. output shape: {output.shape}")
        
        # Assertions
        assert output.shape == (2, 3), f"Expected shape (2, 3), got {output.shape}"
        assert np.allclose(output.numpy(), np.e), f"Expected all values to be {np.e}"
        
    except Exception as e:
        print(f"exp test fails for multi-dim input: {e}")
        raise

if __name__ == "__main__":
    test_exp_scalar_input()
    test_exp_multi_dim_input()
