```python
import torch
from torch.distributions import constraints

# Define the constraint: [0, 10)
lower_bound = 0.0
upper_bound = 10.0
interval = constraints.half_open_interval(lower_bound, upper_bound)

# Test values: -1 (outside), 0 (lower bound), 5 (inside), 10 (upper bound), 11 (outside)
test_values = torch.tensor([-1.0, 0.0, 5.0, 10.0, 11.0])
# Expected results: False, True, True, False, False
expected_results = torch.tensor([False, True, True, False, False])

# Check the constraint
results = interval.check(test_values)

# Verify results
assert torch.equal(results, expected_results), f"Expected {expected_results}, but got {results}"

print(f"Constraint: {interval}")
print(f"Test Values: {test_values}")
print(f"Results: {results}")
print("Test passed.")
```
