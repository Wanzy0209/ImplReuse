import tensorflow as tf
import numpy as np

def test_tf_image_rot90_adapted():
    # Setup similar to the PyTorch environment
    # Note: TensorFlow does not have a direct equivalent to torch._dynamo.config
    # We set seeds for reproducibility
    tf.random.set_seed(751735337)
    np.random.seed(751735337)

    # Define the adapted program using TensorFlow operations
    def adapted_program(arg_0, arg_1, arg_2, arg_3, arg_4, arg_5, arg_6, sentinel):
        # var_node_4 = arg_0 # size=(15, 108, 4), dtype=int16
        var_node_4 = arg_0
        
        # var_node_3 = torch.chunk(var_node_4, 4, dim=1)[0] # size=(15, 27, 4)
        # tf.split returns a list, we take the first element
        var_node_3 = tf.split(var_node_4, 4, axis=1)[0]
        
        # var_node_2 = torch.chunk(var_node_3, 4, dim=2)[0] # size=(15, 27, 1)
        var_node_2 = tf.split(var_node_3, 4, axis=2)[0]
        
        # var_node_1 = torch.squeeze(var_node_2) # size=(15, 27)
        var_node_1 = tf.squeeze(var_node_2)
        
        # var_node_8 = torch.full((13, 27), 3, dtype=torch.int16) # size=(13, 27)
        var_node_8 = tf.fill((13, 27), tf.cast(3, tf.int16))
        
        # var_node_9 = arg_1 # size=(11,), dtype=int64
        var_node_9 = arg_1
        
        # _input_size_var_node_7 = var_node_8.size(0)
        _input_size_var_node_7 = tf.shape(var_node_8)[0]
        
        # _index_var_node_7 = torch.randint(0, _input_size_var_node_7, (11,), device=var_node_8.device)
        # Using tf.random.uniform to mimic randint
        _index_var_node_7 = tf.random.uniform((11,), minval=0, maxval=_input_size_var_node_7, dtype=tf.int64)
        
        # var_node_7 = torch.index_select(var_node_8, 0, _index_var_node_7) # size=(11, 27)
        # tf.gather is the equivalent of index_select
        var_node_7 = tf.gather(var_node_8, _index_var_node_7, axis=0)
        
        # var_node_6 = torch.clamp(var_node_7, min=-1.0, max=1.0)
        var_node_6 = tf.clip_by_value(var_node_7, clip_value_min=-1.0, clip_value_max=1.0)
        
        # var_node_12 = arg_2 # size=(3, 27)
        var_node_12 = arg_2
        
        # var_node_11 = torch.clamp(var_node_12, min=-1.0, max=1.0)
        var_node_11 = tf.clip_by_value(var_node_12, clip_value_min=-1.0, clip_value_max=1.0)
        
        # --- Target API Test ---
        # The original API under test was torch.gather, but we are adapting to tf.image.rot90.
        # We apply tf.image.rot90 to one of the generated 3D tensors (var_node_3).
        # var_node_3 has shape (15, 27, 4). 
        # tf.image.rot90 expects 3D [height, width, channels] or 4D [batch, height, width, channels].
        # We treat (15, 27, 4) as [height, width, channels].
        
        # Test k=1 (90 degrees counter-clockwise)
        # Shape should change from (15, 27, 4) to (27, 15, 4)
        rotated_tensor = tf.image.rot90(var_node_3, k=1)
        
        return var_node_1, var_node_6, var_node_11, rotated_tensor

    # Prepare inputs
    # arg_0: size=(15, 108, 4), dtype=int16
    arg_0 = tf.random.uniform((15, 108, 4), minval=-10, maxval=10, dtype=tf.int16)
    # arg_1: size=(11,), dtype=int64
    arg_1 = tf.random.uniform((11,), minval=0, maxval=100, dtype=tf.int64)
    # arg_2: size=(3, 27), dtype=int16
    arg_2 = tf.random.uniform((3, 27), minval=-10, maxval=10, dtype=tf.int16)
    # arg_3 to arg_6: placeholders for unused arguments in this snippet
    arg_3 = tf.constant([], dtype=tf.int16)
    arg_4 = tf.constant([], dtype=tf.int16)
    arg_5 = tf.constant([], dtype=tf.int16)
    arg_6 = tf.constant([], dtype=tf.int64)
    sentinel = None

    # Execute the adapted program
    v1, v6, v11, rotated = adapted_program(arg_0, arg_1, arg_2, arg_3, arg_4, arg_5, arg_6, sentinel)

    # Verify the behavior of the similar API (tf.image.rot90)
    # Check shape transformation
    assert rotated.shape == (27, 15, 4), f"Expected shape (27, 15, 4), got {rotated.shape}"
    
    # Verify intermediate results to ensure logic preservation
    assert v1.shape == (15, 27)
    assert v6.shape == (11, 27)
    assert v11.shape == (3, 27)
    
    print("Test passed. tf.image.rot90 executed successfully with adapted inputs.")

if __name__ == "__main__":
    test_tf_image_rot90_adapted()
