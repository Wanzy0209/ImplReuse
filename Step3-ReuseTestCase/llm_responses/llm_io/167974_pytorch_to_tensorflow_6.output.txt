```python
import tensorflow as tf
import numpy as np

def test_categorical_2d_input():
    """
    Adapted test case for tf.keras.random.categorical based on the 
    PyTorch EmbeddingBag bug report (Issue ID: 167974).
    
    Original Bug: EmbeddingBag with include_last_offset=True generated 
    incorrect offsets [0, 4] instead of [0, 4, 8] for a 2D input of shape [2, 4].
    
    Adaptation: 
    - tf.keras.random.categorical does not have an 'include_last_offset' parameter 
      nor does it perform embedding lookups. It generates samples from a categorical distribution.
    - We preserve the '2D input' aspect of the bug report. The original input was 
      [[1, 2, 4, 5], [4, 3, 2, 9]] (shape [2, 4]).
    - We use this shape for the 'logits' parameter in the TensorFlow API.
    """
    
    # Original PyTorch input shape was [2, 4]. 
    # For tf.keras.random.categorical, input is logits with shape [batch_size, num_classes].
    # We use the same shape [2, 4] to test handling of 2D inputs.
    logits = tf.constant([[1.0, 2.0, 4.0, 5.0], [4.0, 3.0, 2.0, 9.0]], dtype=tf.float32)
    
    # PyTorch parameter 'include_last_offset' has no equivalent in tf.keras.random.categorical.
    # We define a standard number of samples to draw.
    num_samples = 5
    
    # Execute the function
    # Note: We set a seed for reproducibility, similar to how one might expect deterministic behavior
    # when testing specific offset logic in the original bug.
    samples = tf.keras.random.categorical(logits, num_samples=num_samples, seed=42)
    
    # Verify output shape
    # Expected shape: [batch_size, num_samples] -> [2, 5]
    assert samples.shape == (2, 5), f"Expected shape (2, 5), got {samples.shape}"
    
    # Verify output type (default is int64)
    assert samples.dtype == tf.int64, f"Expected dtype int64, got {samples.dtype}"
    
    print("Test passed: tf.keras.random.categorical handles 2D input correctly.")

if __name__ == "__main__":
    test_categorical_2d_input()
```
