```python
import tensorflow as tf

# Ensure reproducibility
tf.random.set_seed(42)

# Define shape similar to the original test
shape = (8192,)

# 1. Eager execution (float32)
# HeUniform samples from U[-limit, limit] where limit = sqrt(6 / fan_in)
initializer_eager = tf.keras.initializers.HeUniform(seed=42)
out1 = initializer_eager(shape, dtype=tf.float32)

# 2. Compiled/Graph execution (float32)
# Using tf.function to simulate the compilation/graph optimization step
@tf.function
def run_initializer(initializer, shape, dtype):
    return initializer(shape, dtype=dtype)

initializer_compiled = tf.keras.initializers.HeUniform(seed=42)
out2 = run_initializer(initializer_compiled, shape, tf.float32)

# 3. High precision reference (float64)
initializer_high = tf.keras.initializers.HeUniform(seed=42)
out3_high = initializer_high(shape, dtype=tf.float64)

# Calculate differences
# Compare float32 eager vs float64 reference
diff_eager_high = tf.reduce_max(tf.abs(tf.cast(out3_high, tf.float32) - out1))

# Compare float32 compiled vs float64 reference
diff_compiled_high = tf.reduce_max(tf.abs(tf.cast(out3_high, tf.float32) - out2))

print(f"Max diff (Eager FP32 vs High Precision): {diff_eager_high.numpy()}")
print(f"Max diff (Compiled FP32 vs High Precision): {diff_compiled_high.numpy()}")

# Check if compiled version deviates from eager version
diff_eager_compiled = tf.reduce_max(tf.abs(out1 - out2))
print(f"Max diff (Eager vs Compiled): {diff_eager_compiled.numpy()}")

# Verify behavior: With the same seed, eager and graph mode should produce identical results
assert diff_eager_compiled == 0.0, "Eager and Compiled results should match with the same seed"
```
