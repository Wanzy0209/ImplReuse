```python
import torch

# Setup inputs for torch.nn.functional.embedding_bag
# input: LongTensor containing bags of indices
input_indices = torch.tensor([0, 1, 2, 3, 4, 5, 6, 7], dtype=torch.long)
# offsets: Determines the starting index position of each bag
offsets = torch.tensor([0, 4])
# weight: The embedding matrix
weight = torch.randn(10, 3)

# The problematic value from the original bug report (INT64_MAX - 2)
huge_padding_idx = 9223372036854775803

try:
    # Call the similar API with the extreme value adapted to padding_idx
    output = torch.nn.functional.embedding_bag(
        input_indices,
        weight,
        offsets=offsets,
        padding_idx=huge_padding_idx
    )
    print("Test completed without crash.")
except Exception as e:
    print(f"Exception raised: {e}")
```
