import torch
import torch.nn as nn
import onnx
import os

def test_dynamo_pixel_shuffle_export():
    """
    Test case for Issue 162061: Dynamo-based ONNX exporter adds seemingly 
    unnecessary operations to the graph around pixel shuffle.
    
    This test preserves the original bug reproduction logic while leveraging 
    the 'zeros' pattern from the similar API (tf.keras.backend.zeros) 
    translated to PyTorch (torch.zeros).
    """
    
    class Model(nn.Module):
        def __init__(self, kernel_size=3, upscale_factor=2):
            super(Model, self).__init__()
            self.conv = nn.Conv2d(1, 4, kernel_size=kernel_size, padding="same")
            self.pixel_shuffle = nn.PixelShuffle(upscale_factor)

        def forward(self, input):
            x = self.conv(input)
            x = self.pixel_shuffle(x)
            return x

    model = Model()
    model.eval()
    
    # Leveraging the 'zeros' API pattern (similar to tf.keras.backend.zeros)
    # for input tensor creation instead of torch.ones.
    x = torch.zeros((1, 1, 128, 128))
    
    onnx_path = "dynamo_pixel_shuffle.onnx"
    
    try:
        # Export the model using Dynamo
        torch.onnx.export(
            model,
            x,
            onnx_path,
            dynamo=True,
            external_data=False,
            opset_version=20,
            input_names=["input"],
            output_names=["output"],
            dynamic_shapes={"input": {0: "batch", 2: "width", 3: "height"}},
        )
        
        # Verify the ONNX model is valid
        onnx_model = onnx.load(onnx_path)
        onnx.checker.check_model(onnx_model)
        
        # Basic assertion to ensure file was created
        assert os.path.exists(onnx_path)
        
        # Note: A full regression test would check the number of nodes or 
        # specific presence/absence of ops (e.g., Reshape, Transpose) 
        # to ensure no unnecessary operations were added.
        
    finally:
        # Clean up
        if os.path.exists(onnx_path):
            os.remove(onnx_path)

if __name__ == "__main__":
    test_dynamo_pixel_shuffle_export()
