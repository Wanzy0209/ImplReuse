import torch
import torch.nn.functional as F

def test_grid_sample_3d_on_mps():
    """
    Test case to reproduce the NotImplementedError for 'aten::grid_sampler_3d'
    on the MPS device (Issue ID: 160237).
    
    The bug occurs when using torch.nn.functional.grid_sample with 5D inputs
    (volumetric data) on the MPS backend.
    """
    # Check if MPS is available
    if not torch.backends.mps.is_available():
        print("MPS device is not available. Skipping test.")
        return

    # Create a 5D input tensor (Batch, Channel, Depth, Height, Width)
    # This shape triggers the 3D grid sampler logic (aten::grid_sampler_3d)
    input_tensor = torch.randn(1, 3, 8, 8, 8)

    # Create a 5D grid tensor (Batch, Depth, Height, Width, 3)
    # The last dimension must be 3 for (x, y, z) coordinates
    grid_tensor = torch.rand(1, 8, 8, 8, 3) * 2 - 1  # Normalize to [-1, 1]

    # Move tensors to MPS device
    input_mps = input_tensor.to('mps')
    grid_mps = grid_tensor.to('mps')

    try:
        # Call grid_sample. With 5D inputs, this calls aten::grid_sampler_3d internally.
        output = F.grid_sample(
            input_mps,
            grid_mps,
            mode='bilinear',
            padding_mode='zeros',
            align_corners=True
        )
        print("Test Passed: grid_sample_3d executed successfully on MPS.")
        print(f"Output shape: {output.shape}")

    except NotImplementedError as e:
        print(f"Test Failed (Bug Reproduced): {e}")
        raise

if __name__ == "__main__":
    test_grid_sample_3d_on_mps()
