```python
import tensorflow as tf
import numpy as np

# Set seed for reproducibility, mirroring the original test case
tf.random.set_seed(1014698)

# The problematic shape identified in the PyTorch bug report: (20, 0)
# This shape caused a divergence between eager and compiled execution in PyTorch.
target_shape = (20, 0)

def test_lecun_normal_initializer():
    # Initialize the API under test
    # LecunNormal is a variance scaling initializer.
    initializer = tf.keras.initializers.LecunNormal(seed=1014698)

    print("--- Testing Eager Execution ---")
    try:
        # Call the initializer eagerly with the problematic shape
        result_eager = initializer(shape=target_shape)
        print(f"✅ Eager success. Shape: {result_eager.shape}, Dtype: {result_eager.dtype}")
    except Exception as e:
        print(f"❌ Eager failed: {e}")
        return

    print("\n--- Testing Compiled Execution (tf.function) ---")
    
    # Wrap the initialization in tf.function to simulate compilation
    # This is the TensorFlow equivalent to torch.compile
    @tf.function
    def compiled_initializer(shape):
        return initializer(shape)

    try:
        result_compiled = compiled_initializer(target_shape)
        print(f"✅ Compile success. Shape: {result_compiled.shape}, Dtype: {result_compiled.dtype}")
    except Exception as e:
        print(f"❌ Compile failed: {e}")
        return

    print("\n--- Verification ---")
    # Verify that the behavior is consistent between eager and compiled modes
    # Specifically checking for shape divergence which was the core issue in the original bug.
    assert result_eager.shape == result_compiled.shape, \
        f"Shape divergence detected: Eager {result_eager.shape} vs Compiled {result_compiled.shape}"
    
    # Verify the tensor is empty (since one dimension is 0)
    assert tf.size(result_eager) == 0, "Expected tensor size to be 0 based on shape (20, 0)"
    
    print("✅ All checks passed. Eager and Compiled execution are consistent.")

if __name__ == "__main__":
    test_lecun_normal_initializer()
```
