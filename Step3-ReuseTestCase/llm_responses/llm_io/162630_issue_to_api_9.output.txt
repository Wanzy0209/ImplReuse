import tensorflow as tf
import tf.experimental.numpy as tnp

def test_ndim_frequent_metadata_access():
    """
    Test case derived from Issue 162630 (Intrusive Caching DLPack for Fast Conversion).
    
    The original issue highlights the overhead of frequent metadata access (shape, strides)
    during tensor exchanges. This test verifies that tf.experimental.numpy.ndim, which 
    accesses tensor rank metadata, behaves correctly and consistently under frequent access,
    mirroring the scenario described in the bug report.
    """
    # Create a tensor representing a model weight or intermediate value
    # Shape: (Batch, Sequence, Hidden) -> Rank 3
    tensor_data = tf.zeros((10, 20, 30))
    expected_rank = 3

    # Simulate frequent tensor exchanges/metadata access.
    # In the original bug, this was DLPack conversion accumulating overhead.
    # Here we use the similar API: tf.experimental.numpy.ndim.
    # We loop to ensure the API handles repeated calls robustly.
    for _ in range(1000):
        rank = tnp.ndim(tensor_data)
        # The fix in the original issue ensures metadata is populated correctly and efficiently.
        # We assert correctness of the metadata access here.
        assert rank == expected_rank, f"Expected rank {expected_rank}, but got {rank}"

    # Verify with different shapes to ensure metadata population is robust across dimensions
    # Scalar (Rank 0)
    scalar_tensor = tf.constant(1.0)
    assert tnp.ndim(scalar_tensor) == 0
    
    # 1D Tensor (Rank 1)
    vector_tensor = tf.constant([1, 2, 3, 4])
    assert tnp.ndim(vector_tensor) == 1

    # 4D Tensor (Rank 4) - e.g., an image batch
    image_batch = tf.ones((32, 224, 224, 3))
    assert tnp.ndim(image_batch) == 4

if __name__ == "__main__":
    test_ndim_frequent_metadata_access()
    print("Test passed.")
