```python
import tensorflow as tf

# Disable eager execution to use TF v1 graph features
tf.compat.v1.disable_eager_execution()

# Define a simple function to mimic the PyTorch 'fn'
def fn(x):
    return x * 2.0

# 1. First execution (Eager-like)
# Define the graph and placeholders
a1 = tf.compat.v1.placeholder(tf.float32)
b1 = fn(a1)
# Calculate gradients to match the PyTorch logic
grad1 = tf.gradients(b1, a1)[0]

with tf.compat.v1.Session() as sess:
    # Initialize variables if any (none in this simple case)
    sess.run(tf.compat.v1.global_variables_initializer())
    result1, grad_val1 = sess.run([b1, grad1], feed_dict={a1: 5.0})

# 2. Use the Similar API: tf.compat.v1.reset_default_graph
# This corresponds to the "checkpoint" step in the original logic,
# acting as the state transition point.
tf.compat.v1.reset_default_graph()

# 3. Second execution (Graph Replay-like)
# Recreate the graph to simulate a fresh state or replay
a2 = tf.compat.v1.placeholder(tf.float32)
b2 = fn(a2)
grad2 = tf.gradients(b2, a2)[0]

with tf.compat.v1.Session() as sess:
    sess.run(tf.compat.v1.global_variables_initializer())
    result2, grad_val2 = sess.run([b2, grad2], feed_dict={a2: 5.0})

# 4. Assert
# In the original, we check if gradients match.
# Here, we check if the operation still works after reset.
assert result1 == result2, "Mismatch in outputs"
assert grad_val1 == grad_val2, "Mismatch in gradients"

print("Output 1:", result1)
print("Output 2:", result2)
print("Gradient 1:", grad_val1)
print("Gradient 2:", grad_val2)
```
