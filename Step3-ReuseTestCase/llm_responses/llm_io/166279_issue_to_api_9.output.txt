import torch
import torch._dynamo

# Leverage the similar API to check the CUDA environment context.
# The bug report involves CUDA tensors and operations, so checking 
# specific CUDA capabilities (like Flash Attention) is relevant 
# to ensure the test runs in a compatible environment.
is_flash_available = torch.backends.cuda.is_flash_attention_available()

if not torch.cuda.is_available():
    print("CUDA is not available. Skipping test as it requires CUDA context.")
else:
    print(f"Running test on CUDA. Flash Attention Available: {is_flash_available}")
    
    torch._dynamo.config.capture_scalar_outputs = True
    torch.manual_seed(1166094474)

    def fuzzed_program(arg_0, arg_1, arg_2, arg_3, sentinel):
        # Reproduce the logic from the bug report
        # Ensure tensors are on CUDA as indicated in the bug description comments
        var_node_3 = torch.full((12,), False, dtype=torch.bool, device='cuda')
        var_node_2 = torch.chunk(var_node_3, 4, dim=0)[0]
        
        var_node_6 = arg_0
        var_node_7 = arg_1
        _input_size_var_node_5 = var_node_6.size(0)
        _index_var_node_5 = torch.randint(0, _input_size_var_node_5, (10,), device=var_node_6.device)
        var_node_5 = torch.gather(var_node_6, 0, _index_var_node_5)
        var_node_4 = torch.chunk(var_node_5, 2, dim=0)[0]
        
        var_node_10 = arg_2
        var_node_9 = torch.chunk(var_node_10, 4, dim=1)[0]
        var_node_8 = torch.squeeze(var_node_9)
        
        var_node_1 = torch.cat([var_node_2, var_node_4, var_node_8], dim=0)
        var_node_11 = arg_3
        var_node_0 = torch.cat([var_node_1, var_node_11], dim=0)
        
        # Ensure gradient computation
        result = var_node_0 * sentinel
        if result.is_complex():
            result = result.real
        return result

    # Sentinel tensor to ensure gradient computation
    sentinel = torch.tensor(1.0, requires_grad=True, device='cuda')

    # Initialize arguments on CUDA
    arg_0 = torch.as_strided(torch.randint(0, 2, (12,), dtype=torch.int8).bool(), (12,), (1,)).to('cuda')
    arg_1 = torch.as_strided(torch.randint(5, 30, (10,)).to(torch.int64), (10,), (1,)).to('cuda')
    arg_2 = torch.as_strided(torch.randint(0, 2, (24,), dtype=torch.int8).bool(), (6, 4), (4, 1)).to('cuda')
    arg_3 = torch.as_strided(torch.randint(0, 2, (2,), dtype=torch.int8).bool(), (2,), (1,)).to('cuda')

    args = (arg_0, arg_1, arg_2, arg_3, sentinel)
    
    try:
        # Test Eager Mode
        result_original = fuzzed_program(*args)
        print('✅ eager success')

        # Test Compiled Mode
        compiled_program = torch.compile(fuzzed_program, fullgraph=True, dynamic=True)
        result_compiled = compiled_program(*args)
        print('✅ compile success')
        
        # Verify consistency
        assert torch.allclose(result_original, result_compiled)
        print("✅ Test Passed: Eager and Compiled results match.")
        
    except AssertionError as e:
        print(f"❌ Test Failed: {e}")
        raise
    except Exception as e:
        print(f"❌ Runtime Error: {e}")
        raise
