import torch

def test_index_select_scalar_index(device):
    """
    Tests that index_select can handle a zero-dimensional (scalar) index tensor.
    This mirrors the behavior of tf.keras.backend.exp which accepts scalar tensors.
    """
    # Setup input tensor
    x = torch.ones([2, 3], device=device)
    
    # Setup scalar index tensor (zero-dimensional)
    # Similar to how tf.keras.backend.exp accepts tf.constant(2.0)
    index = torch.tensor(1, device=device)
    
    # Perform index_select
    # Bug: MPS throws "Dimension specified as -1 but tensor has no dimensions"
    output = torch.index_select(x, dim=0, index=index)
    
    # Assertions
    # Expected shape: [1, 3] because we selected one row (index 1)
    assert output.shape == torch.Size([1, 3]), f"Expected shape [1, 3], got {output.shape}"
    assert torch.all(output == 1.0), "Expected all ones"

if __name__ == "__main__":
    # Test on CPU (baseline)
    test_index_select_scalar_index("cpu")
    print("index_select test succeeds for device: cpu")

    # Test on MPS if available
    if torch.backends.mps.is_available():
        test_index_select_scalar_index("mps")
        print("index_select test succeeds for device: mps")
    else:
        print("MPS device not available, skipping MPS test.")
