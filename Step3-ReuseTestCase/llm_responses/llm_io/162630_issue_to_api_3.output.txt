import torch
import torch.utils.dlpack

def test_dlpack_repeated_conversion_with_backend_check():
    """
    Test case for Issue 162630: Intrusive Caching DLPack for Fast Conversion.
    
    This test verifies that repeated DLPack conversions (the scenario described 
    in the bug report) work correctly. It leverages the similar API 
    torch.backends.mps.is_built to determine the appropriate device context,
    ensuring the test covers the MPS backend path if supported.
    """
    # Leverage the similar API: torch.backends.mps.is_built
    # We use this to check if the MPS backend is compiled in, similar to how
    # one might check capabilities before utilizing specific optimizations.
    if torch.backends.mps.is_built():
        device = torch.device("mps")
    else:
        device = torch.device("cpu")

    # Create a tensor on the selected device
    # The bug report mentions overhead in "frequent tensor exchanges"
    # such as model weights or intermediate values.
    tensor = torch.randn(10, 10, device=device)

    # Original bug reproduction logic: Frequent tensor exchanges.
    # The RFC proposes caching to reduce the overhead of these repeated calls.
    # We perform the conversion multiple times to exercise the caching path
    # (or the lack thereof) and ensure correctness is maintained.
    for _ in range(100):
        # Convert to DLPack
        dlpack_capsule = torch.utils.dlpack.to_dlpack(tensor)
        
        # Convert back from DLPack
        tensor_reconstructed = torch.utils.dlpack.from_dlpack(dlpack_capsule)
        
        # Assert that the data remains consistent after the round trip
        assert torch.equal(tensor, tensor_reconstructed), \
            "Tensor data mismatch detected during repeated DLPack conversion"

if __name__ == "__main__":
    test_dlpack_repeated_conversion_with_backend_check()
