import torch
from torch.distributed.tensor import distribute_tensor, Replicate, DTensor
from torch.distributed.tensor.experimental import register_sharding
from torch.testing._internal.distributed._tensor.common_dtensor import DTensorTestBase, with_comms

class TestTensorKwargsSharding(DTensorTestBase):
    """
    Test case for Issue 167977: Sharding propagation failed for custom operation 
    with Tensor in kwargs.
    
    This test verifies that DTensor correctly handles sharding propagation when
    an operation (like aten::min.dim_min) receives DTensor arguments via kwargs.
    """
    
    @with_comms
    def test_sharding_propagation_tensor_kwargs(self):
        # 1. Setup Device Mesh
        mesh = self.build_device_mesh()

        # 2. Create and Distribute Tensors
        # Input tensor
        x = torch.randn(4, 4, device="cuda")
        x_dt = distribute_tensor(x, mesh, [Replicate()])

        # Output tensors (passed as kwargs via 'out')
        value = torch.randn(4, 1, device="cuda")
        indices = torch.randn(4, 1, device="cuda").long()
        value_dt = distribute_tensor(value, mesh, [Replicate()])
        indices_dt = distribute_tensor(indices, mesh, [Replicate()])

        # 3. Register Sharding Strategy
        # We target aten::min.dim_min which has the signature:
        # (Tensor self, int dim, bool keepdim=False, *, Tensor(a!) min, Tensor(b!) min_indices)
        # The 'min' and 'min_indices' are keyword-only arguments.
        aten = torch.ops.aten

        @register_sharding(aten.min.dim_min)
        def custom_strategy(mesh, args, kwargs):
            # The strategy must account for DTensors in both args and kwargs.
            # args: [x_dt]
            # kwargs: {'dim': 1, 'keepdim': True, 'min': value_dt, 'min_indices': indices_dt}
            
            # Define sharding for inputs: x, min, min_indices
            input_strategy = [Replicate(), Replicate(), Replicate()]
            
            # Define sharding for outputs: values, indices
            output_strategy = [Replicate(), Replicate()]
            
            return [(input_strategy, output_strategy)]

        # 4. Execute Operation
        # This triggers the sharding propagation logic.
        # Prior to the fix, this would raise an AssertionError in 
        # expand_to_full_mesh_op_strategy due to mismatched lengths between
        # input specs (args + kwargs) and input args strategy (args only).
        try:
            torch.min(x_dt, dim=1, keepdim=True, out=(value_dt, indices_dt))
        except AssertionError as e:
            self.fail(f"Sharding propagation failed for tensor kwargs: {e}")

        # 5. Basic Validation
        self.assertIsInstance(value_dt, DTensor)
        self.assertIsInstance(indices_dt, DTensor)

if __name__ == "__main__":
    from torch.testing._internal.common_utils import run_tests
    run_tests()
