```python
import tensorflow as tf

class Config:
    def __repr__(self):
        return "Config()"

# In TensorFlow, graph tracing is typically handled by @tf.function.
# We use tf.name_scope inside to organize the graph, similar to how
# torch.compile organizes the execution graph.
@tf.function
def forward(x, config):
    with tf.name_scope("repr_scope"):
        # Calling repr() on non-constant user object
        # This tests if the graph tracer (AutoGraph) handles repr() correctly
        return x * len(repr(config))

config = Config()
x = tf.random.normal((2, 2))

# Execute the traced function
result = forward(x, config)

# Verify the result
assert result.shape == (2, 2)
print("Test passed.")
```
