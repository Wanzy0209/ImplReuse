import torch

@torch.compile()
def test(x, y):
    # Adapt the logic to use torch.gather instead of flex_attention
    # to verify gradient flow through a similar tensor operation.
    B, L, D = x.shape
    device = x.device

    # Create an index tensor for gathering
    # We want to gather values from y based on some indices
    # y is (B, L), let's expand it to (B, L, D) to match x's dimensions for the operation
    y_expanded = y.unsqueeze(-1).expand(-1, -1, D)

    # Create indices (e.g., reverse order or some pattern)
    indices = torch.arange(L - 1, -1, -1, device=device).view(1, L, 1).expand(B, L, D)

    # Use torch.gather to select values from y_expanded
    gathered_y = torch.gather(y_expanded, 1, indices)

    # Combine with x
    # torch._dynamo.graph_break()
    return x + gathered_y

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
B, L, D = 2, 16, 64

x = torch.randn(B, L, D, device=DEVICE, requires_grad=True)
y = torch.randn(B, L, device=DEVICE, requires_grad=True)

out = test(x, y).mean().backward()

print(torch.__version__)
print(f"x: {(x.grad is not None) and (x.grad.norm() > 0)}, y: {(y.grad is not None) and (y.grad.norm() > 0)}")
assert x.grad is not None and x.grad.norm() > 0
assert y.grad is not None and y.grad.norm() > 0
