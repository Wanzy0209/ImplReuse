import tensorflow as tf
from unittest.mock import patch

def test_tf_experimental_numpy_add_single_execution():
    """
    Test that tf.experimental.numpy.add invokes the underlying operation exactly once.
    
    This test is derived from a PyTorch issue (ID: 165624) where a merge mistake 
    caused a graph pass to be executed twice. This test verifies that the similar 
    TensorFlow API does not suffer from a similar duplicate execution pattern.
    """
    
    # We mock the underlying _math_ops.add to track how many times it is called.
    # The path is derived from the provided similar API information.
    with patch('tensorflow.python.framework.experimental._math_ops.add') as mock_underlying_add:
        # Configure the mock to return a valid tensor
        mock_result = tf.constant(5.0)
        mock_underlying_add.return_value = mock_result

        # Inputs
        a = tf.constant(2.0)
        b = tf.constant(3.0)

        # Call the API under test
        result = tf.experimental.numpy.add(a, b)

        # Assertion 1: The result is the mocked return value
        assert result is mock_result, "The function did not return the expected mocked result."

        # Assertion 2: The underlying add was called exactly once.
        # In the PyTorch bug, the pass was called twice (count += 1 happened twice).
        # Here we ensure no such duplication exists in the TensorFlow implementation.
        assert mock_underlying_add.call_count == 1, \
            f"Expected underlying add to be called once, but it was called {mock_underlying_add.call_count} times."

        # Assertion 3: Verify the arguments passed to the underlying function match expectations
        # The implementation signature is: add(a, b, name=None) -> _math_ops.add(ctx, a, b, name)
        args, kwargs = mock_underlying_add.call_args
        # args[0] is ctx, args[1] is a, args[2] is b
        assert args[1] is a, "First argument passed to underlying add does not match input 'a'."
        assert args[2] is b, "Second argument passed to underlying add does not match input 'b'."
