import tensorflow as tf
import os
import shutil

# Mimicking the SimpleModel class structure from the original test case
class TextSummaryModel:
    def __init__(self, log_dir):
        self.log_dir = log_dir
        # Clean up previous logs if any to ensure a clean test environment
        if os.path.exists(self.log_dir):
            shutil.rmtree(self.log_dir)
        self.writer = tf.summary.create_file_writer(self.log_dir)

    def write_summary(self, name, tensor, step):
        """
        Mimicking the forward pass.
        This method wraps the tf.summary.text API call.
        """
        with self.writer.as_default():
            # The API under test: tf.summary.text
            tf.summary.text(name, tensor, step=step)
        self.writer.flush()

# Check availability (Mimicking the torch.<mybackend>.is_available check)
# We check if TensorFlow is accessible and if we can create a summary writer.
try:
    print(f"TensorFlow Version: {tf.__version__}")
    # Mimicking the device count check logic loosely by checking GPU availability
    gpus = tf.config.list_physical_devices('GPU')
    print(f"检测到 {len(gpus)} 个GPU" if gpus else "检测到 0 个GPU (使用CPU)")

    # Instantiate the model (Mimicking model = SimpleModel())
    model = TextSummaryModel('./logs/text_summary_test')

    # Input Data (Mimicking input_data = torch.randn(...))
    # The API description mentions 1d and 2d tensors. We use a 2D string tensor.
    input_data = tf.constant([["Header 1", "Header 2"], ["Row 1 Col 1", "Row 1 Col 2"]])

    # Run the model (Mimicking output = model(input_data))
    model.write_summary("custom_text_summary", input_data, step=0)

    # Verification (Mimicking print("success") but adding an assertion as requested)
    # Check if the log file was created successfully
    log_files = os.listdir(model.log_dir)
    assert len(log_files) > 0, "Log file was not created."
    
    print("success")

except Exception as e:
    print(f"Test failed with error: {e}")
    raise
