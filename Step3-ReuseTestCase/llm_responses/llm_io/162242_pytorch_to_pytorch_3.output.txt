```python
import numpy as np
import torch


def test_trunc_deterministic():
    # Define ground truth for torch.trunc
    # Input will be arange(24) + 0.5, so output is arange(24)
    gt_res = np.arange(24, dtype=np.float32).reshape([2, 3, 4])
    
    # Gradient of torch.trunc is 0 everywhere
    gt_input_grad = np.zeros((2, 3, 4), dtype=np.float32)

    for i in range(100):
        torch.cuda.empty_cache()
        
        # Create input with fractional parts to test truncation
        inputs = (torch.arange(24, dtype=torch.float32).reshape([2, 3, 4]) + 0.5).cuda()
        
        inputs.requires_grad = True

        # Adapted call site: torch.trunc
        # torch.trunc takes only the input tensor and returns the truncated values
        res = torch.trunc(inputs)
        
        res.backward(torch.ones_like(res))

        print(f"Test {i + 1}")
        np.testing.assert_allclose(res.cpu().detach().numpy(), gt_res)
        np.testing.assert_allclose(inputs.grad.cpu().numpy(), gt_input_grad)


if __name__ == "__main__":
    test_trunc_deterministic()
```
