```python
import tensorflow as tf

# Mimic the setup: manual seed for reproducibility
tf.random.set_seed(0)

# Mimic the input: random tensor
# Original input shape was (4, 6, 7). 
# Multinomial expects logits of shape [batch_size, num_classes].
# We adapt the input to (4, 6) to represent batch_size=4 and num_classes=6.
logits = tf.random.normal((4, 6))

# Define parameters for Multinomial
# We use 7 as num_samples to loosely correspond to the last dimension of the original input.
num_samples = 7

# Run on CPU
with tf.device('/CPU:0'):
    out_cpu = tf.raw_ops.Multinomial(logits=logits, num_samples=num_samples, seed=0, seed2=0)

# Run on GPU (if available) to check for consistency, similar to the original MPS check
try:
    with tf.device('/GPU:0'):
        out_gpu = tf.raw_ops.Multinomial(logits=logits, num_samples=num_samples, seed=0, seed2=0)

    # Check if outputs match between CPU and GPU
    if not tf.reduce_all(out_cpu == out_gpu):
        print("Output does not match!")
        print("CPU Output:")
        print(out_cpu.numpy())
        print("GPU Output:")
        print(out_gpu.numpy())
    else:
        print("Outputs match.")
except RuntimeError as e:
    # Handle cases where GPU is not available
    print(f"GPU not available: {e}")
    print("CPU Output:")
    print(out_cpu.numpy())
```
