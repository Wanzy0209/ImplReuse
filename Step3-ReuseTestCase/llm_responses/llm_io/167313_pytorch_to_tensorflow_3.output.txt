```python
import tensorflow as tf

# Setup data
x = tf.random.uniform((2,))

# Define function
# Original PyTorch: lambda x, a, b: torch.nn.functional.relu(torch.addmm(x, a, b, alpha=0.5, beta=0.5))
# Adapted TF: lambda x: tf.nn.relu(tf.stop_gradient(x))
# Note: tf.stop_gradient does not have alpha/beta parameters. 
# We test the core functionality (gradient stopping) under compilation (tf.function).
f = lambda x: tf.nn.relu(tf.stop_gradient(x))

# Compile
fc = tf.function(f)

# Forward pass check (Values should be identical)
out_eager = f(x)
out_compiled = fc(x)
assert tf.reduce_all(tf.equal(out_eager, out_compiled)).numpy(), "Forward pass mismatch"

# Backward pass check (Gradients should be None/0)
# Eager
with tf.GradientTape() as tape:
    y = f(x)
grad_eager = tape.gradient(y, x)

# Compiled
with tf.GradientTape() as tape:
    y = fc(x)
grad_compiled = tape.gradient(y, x)

# Assert gradients are None (stopped)
assert grad_eager is None, "Eager gradient should be None"
assert grad_compiled is None, "Compiled gradient should be None"
```
