import torch
import unittest

class TestNestedTensorShareMemory(unittest.TestCase):
    """
    Test case for Issue #161915: share_memory_support for NestedTensor.
    
    This test verifies that calling share_memory_() on a NestedTensor
    with jagged layout does not cause a segmentation fault.
    """

    def test_nested_tensor_share_memory_jagged(self):
        """
        Reproduces the original bug logic:
        1. Create random tensors.
        2. Construct a NestedTensor with jagged layout.
        3. Call share_memory_().
        
        Expected: The operation completes without crashing.
        """
        # Setup data as per the original issue
        a = torch.randn(3)
        b = torch.randn(5)
        
        # Create the NestedTensor
        nt = torch.nested.nested_tensor([a, b], layout=torch.jagged)
        
        # The original bug caused a Segmentation fault here.
        # We assert that the method executes and returns the tensor object.
        try:
            result = nt.share_memory_()
            self.assertIs(result, nt, "share_memory_() should return self")
        except Exception as e:
            self.fail(f"share_memory_() raised an unexpected exception: {e}")

if __name__ == '__main__':
    unittest.main()
