```python
import torch

def test_compile_xpu_regression():
    """
    Test case to verify torch.compile functionality on XPU devices.
    This reproduces the segmentation fault issue reported in PyTorch 2.8.0+.
    """
    # Check for XPU availability
    if not torch.xpu.is_available():
        print("XPU device not found. Skipping test.")
        return

    # Define the function to be compiled
    def addcmul_func(x, y, z):
        return x + (y * z)

    # Create tensors on XPU
    x = torch.randn(128).to("xpu")
    y = torch.randn(128).to("xpu")
    z = torch.randn(128).to("xpu")

    # Test eager mode
    out_eager = addcmul_func(x, y, z)
    print("Eager mode passed")

    # Test torch.compile (Similar API)
    # This call site is the focus of the regression test
    addcmul_func_compiled = torch.compile(addcmul_func)
    out_compiled = addcmul_func_compiled(x, y, z)
    print("torch.compile passed")

    # Verify correctness
    assert torch.allclose(out_eager, out_compiled), "Compiled output differs from eager output"

if __name__ == "__main__":
    test_compile_xpu_regression()
```
