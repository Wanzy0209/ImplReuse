import tensorflow as tf

def test_bessel_y1_non_contiguous():
    """
    Adapted test case for tf.compat.v1.math.special.bessel_y1 based on 
    PyTorch MPS buffer allocation regression with non-contiguous tensors.
    
    The original bug involved a tensor with shape (5, 499, 768) and stride (0, 768, 1).
    In TensorFlow, a stride of 0 in a dimension is semantically equivalent to broadcasting.
    We recreate this scenario to verify the API's behavior with such inputs.
    """
    
    # Create a base tensor representing the underlying storage
    # The PyTorch logic implies a base size of 499 * 768 given the strides
    base_size = 499 * 768
    base = tf.range(base_size, dtype=tf.float32)
    base = tf.reshape(base, (1, 499, 768))

    # Mimic the non-contiguous input using broadcasting
    # PyTorch stride (0, 768, 1) for shape (5, 499, 768) broadcasts the first dimension
    input_tensor = tf.broadcast_to(base, (5, 499, 768))

    # Execute the target API
    try:
        result = tf.compat.v1.math.special.bessel_y1(input_tensor)
        
        # Verify the output shape matches the input shape
        assert result.shape == (5, 499, 768), \
            f"Shape mismatch: expected (5, 499, 768), got {result.shape}"
            
        # Verify the operation ran successfully (check for NaNs if appropriate, 
        # though bessel_y1 is defined for positive inputs like arange)
        print("Test passed: API handled non-contiguous/broadcasted input correctly.")
        
    except Exception as e:
        print(f"Test failed with error: {e}")
        raise

if __name__ == "__main__":
    test_bessel_y1_non_contiguous()
