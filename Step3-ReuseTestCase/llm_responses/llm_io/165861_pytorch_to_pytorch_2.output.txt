import torch
import torch.nn as nn

# Check for CUDA availability as the bug is specific to CUDA
if torch.cuda.is_available():
    # Define the pooling and unpooling layers
    kernel_size = 2
    stride = 2
    pool = nn.MaxPool3d(kernel_size, stride=stride, return_indices=True)
    unpool = nn.MaxUnpool3d(kernel_size, stride=stride)

    # Create input tensor with a batch dimension larger than uint16 max (2**16)
    # Shape: (Batch, Channels, Depth, Height, Width)
    # We keep spatial dimensions small to focus on the batch dimension size.
    x = torch.rand(2**16, 1, 2, 2, 2, device="cuda")

    # Forward pass through MaxPool3d
    output, indices = pool(x)

    # Attempt the inverse pass
    # This may trigger a similar CUDA error if MaxUnpool3d uses uint16 for dimension indexing
    try:
        reconstructed = unpool(output, indices)
        print("MaxUnpool3d with large batch dimension succeeded")
    except RuntimeError as e:
        print(f"MaxUnpool3d failed: {e}")
else:
    print("CUDA not available, skipping test")
