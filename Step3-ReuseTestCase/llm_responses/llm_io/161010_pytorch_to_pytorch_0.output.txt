import torch
import unittest

class TestCompileStridePreservation(unittest.TestCase):
    def test_compile_preserve_stride_with_clone(self):
        """
        Test that torch.compile preserves stride behavior when using
        clone(memory_format=torch.preserve_format).
        """
        device = "cuda" if torch.cuda.is_available() else "cpu"
        A = torch.rand(5, 5, device=device)

        def f(A, count):
            Q, R = torch.linalg.qr(A)
            rhs = torch.ones(Q.shape[0], 1, device=A.device)
            a = torch.linalg.solve_triangular(R, Q.T @ rhs, upper=True)
            
            # Check if the stride is preserved when cloning with preserve_format
            if a.stride() == a.clone(memory_format=torch.preserve_format).stride():
                return count + 1
            return count

        # Eager execution
        res_eager = f(A, torch.zeros(1))
        
        # Compiled execution
        compiled_f = torch.compile(f)
        res_compiled = compiled_f(A, torch.zeros(1))

        # Assert that compiled mode produces the same result as eager mode
        self.assertTrue(torch.equal(res_eager, res_compiled), 
                        f"torch.compile failed to preserve stride. Eager: {res_eager.item()}, Compiled: {res_compiled.item()}")

if __name__ == "__main__":
    unittest.main()
