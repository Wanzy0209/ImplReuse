import tensorflow as tf
from collections import namedtuple
import unittest

class TestNamedTupleNumerics(unittest.TestCase):
    def test_namedtuple_input_with_numerics_check(self):
        """
        Adapted from PyTorch issue #160547.
        Verifies that tf.debugging.enable_check_numerics handles inputs 
        structured as Python namedtuples correctly.
        """
        # Define the NamedTuple structure similar to the PyTorch bug
        Point = namedtuple('Point', ['x', 'y'])

        # Define a TensorFlow function that mimics the PyTorch module's forward pass
        @tf.function
        def model_forward(p):
            return p.x + p.y

        # Enable the numerics check. 
        # In the original bug, strict=False failed. Here we verify the debugging 
        # mechanism works correctly with the structured input.
        tf.debugging.enable_check_numerics()

        # Case 1: Valid inputs (Should succeed)
        # Corresponds to the 'succeeds' scenario in the original bug
        valid_inp = Point(tf.ones(3), tf.ones(3))
        
        # This should run without raising an error
        result = model_forward(valid_inp)
        self.assertEqual(result.shape, (3,))
        print("Valid input test passed.")

        # Case 2: Inputs with NaN (Should raise InvalidArgumentError)
        # This verifies the API correctly processes the NamedTuple structure
        # and identifies the bad numerics, analogous to verifying the export 
        # handles the input structure correctly.
        nan_inp = Point(tf.constant([1.0, float('nan'), 1.0]), tf.ones(3))
        
        with self.assertRaises(tf.errors.InvalidArgumentError) as context:
            model_forward(nan_inp)
        
        self.assertIn("NaN", str(context.exception))
        print("NaN input test passed: Error caught as expected.")

if __name__ == '__main__':
    unittest.main()
