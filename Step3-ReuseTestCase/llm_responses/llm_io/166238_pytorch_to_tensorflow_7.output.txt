import tensorflow as tf
import collections

def test_name_scope_with_defaultdict():
    """
    Test that creating a collections.defaultdict inside a tf.keras.backend.name_scope
    context works correctly. This adapts the PyTorch Dynamo regression test where
    torch.compile failed to trace defaultdict creation.
    """
    # Define a simple default factory
    def default_factory():
        return 0

    # Use the similar API: tf.keras.backend.name_scope
    # This acts as the context manager, similar to how torch.compile wraps execution.
    with tf.keras.backend.name_scope("test_scope"):
        # This is the core logic from the original bug report:
        # Creating a defaultdict inside the specific API context.
        d = collections.defaultdict(default_factory)
        
        # Perform operations to ensure it is functional
        d['a'] = 1
        d['b'] = 2

    # Verify the behavior
    assert isinstance(d, collections.defaultdict)
    assert d['a'] == 1
    assert d['b'] == 2
    assert d['non_existent_key'] == 0  # Verify default factory works

if __name__ == "__main__":
    test_name_scope_with_defaultdict()
    print("Test passed successfully.")
