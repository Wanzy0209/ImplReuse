import tensorflow as tf

# Set seed for reproducibility
tf.random.set_seed(1215252001)

# Define inputs
# Note: TensorFlow tensors are immutable and contiguous. 
# We cannot replicate the exact zero-stride memory layout of torch.as_strided,
# but we replicate the shapes and data types to preserve the computational logic.
arg_0 = tf.random.uniform((9, 1, 15, 4), minval=5, maxval=30, dtype=tf.int32)
arg_1 = tf.random.uniform((20, 15), minval=5, maxval=30, dtype=tf.int64)
arg_2 = tf.random.uniform((18, 15), minval=5, maxval=30, dtype=tf.int64)

# Sentinel tensor
sentinel = tf.constant(1.0)

def fuzzed_program(arg_0, arg_1, arg_2, sentinel):
    var_node_4 = arg_0
    var_node_3 = tf.squeeze(var_node_4, axis=1)
    var_node_2 = tf.split(var_node_3, 4, axis=2)[0]
    var_node_1 = tf.squeeze(var_node_2, axis=-1)

    var_node_7 = arg_1
    var_node_8 = arg_2 # Unused in logic but present in signature

    _input_size_var_node_6 = tf.shape(var_node_7)[0]
    _index_var_node_6 = tf.random.uniform((18, 15), minval=0, maxval=_input_size_var_node_6, dtype=tf.int64)

    var_node_6 = tf.gather(var_node_7, _index_var_node_6, axis=0)
    var_node_5 = tf.split(var_node_6, 2, axis=0)[0]

    var_node_0 = tf.multiply(var_node_1, var_node_5)

    result = var_node_0 * sentinel
    if result.dtype.is_complex:
        result = tf.math.real(result)
    return result

# Adaptation: Use tf.compat.v1.name_scope to wrap the logic
# This verifies that the similar API (name_scope) correctly handles the graph construction
# for these operations.
with tf.compat.v1.name_scope("fuzzed_program_scope"):
    result = fuzzed_program(arg_0, arg_1, arg_2, sentinel)

print('✅ name_scope execution success')
