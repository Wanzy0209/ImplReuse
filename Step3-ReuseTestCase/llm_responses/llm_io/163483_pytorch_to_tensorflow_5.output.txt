import tensorflow as tf
import numpy as np

def test_ones_initializer():
    """
    Adapted test case for tf.keras.initializers.Ones based on the 
    PyTorch all_gather memory ordering bug.
    
    Original Bug Logic: Input tensor has specific memory format -> Operation -> 
    Output tensor should preserve that format.
    
    Adapted Logic: Input arguments define specific shape/dtype -> Initializer -> 
    Output tensor should preserve those properties.
    """
    
    # Setup: Define the expected properties (analogous to the input tensor in the original test)
    # Using a 4D shape similar to the original test case (2, 2, 2, 2)
    target_shape = (2, 2, 2, 2)
    target_dtype = tf.float32
    
    # Initialize the API
    initializer = tf.keras.initializers.Ones()
    
    # Action: Call the initializer to generate the tensor
    # This corresponds to the 'all_gather' operation in the original test
    tensor = initializer(shape=target_shape, dtype=target_dtype)
    
    # Verification: Check if the output properties match the input specifications
    # 1. Check Shape Preservation
    shape_match = tensor.shape == target_shape
    
    # 2. Check Value Correctness (Functional correctness)
    values_match = tf.reduce_all(tensor == 1.0).numpy()
    
    # 3. Check Dtype Preservation
    dtype_match = tensor.dtype == target_dtype
    
    # Output results similar to the original test format
    print(f"Shape Match: {shape_match}")
    print(f"Values Match (All Ones): {values_match}")
    print(f"Dtype Match: {dtype_match}")
    print(f"Generated Tensor:\n{tensor.numpy()}")
    
    # Assertions to ensure the test fails if behavior is unexpected
    assert shape_match, f"Shape mismatch: expected {target_shape}, got {tensor.shape}"
    assert values_match, "Values are not all ones"
    assert dtype_match, f"Dtype mismatch: expected {target_dtype}, got {tensor.dtype}"
    
    print("\nTest passed: tf.keras.initializers.Ones preserves shape, dtype, and values.")

if __name__ == "__main__":
    test_ones_initializer()
