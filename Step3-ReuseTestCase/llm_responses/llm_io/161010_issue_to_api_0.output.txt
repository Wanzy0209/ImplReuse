import tensorflow as tf

def check_format_persistence(expected_format):
    """
    Checks if the current image data format matches the expected format.
    This mirrors the logic in the PyTorch issue where stride preservation is checked.
    """
    current_format = tf.keras.backend.image_data_format()
    if current_format == expected_format:
        return True
    return False

# Save original format to restore later
original_format = tf.keras.backend.image_data_format()

try:
    # Test Case 1: Check 'channels_first' format
    tf.keras.backend.set_image_data_format('channels_first')
    
    # Eager execution check
    res_eager = check_format_persistence('channels_first')
    assert res_eager, "Eager mode: Failed to preserve 'channels_first' format"
    
    # Compiled execution check (analogous to torch.compile)
    # We verify that the API behavior is consistent when traced/compiled
    compiled_check = tf.function(check_format_persistence)
    res_compiled = compiled_check('channels_first')
    assert res_compiled, "Compiled mode: Failed to preserve 'channels_first' format"

    # Test Case 2: Check 'channels_last' format
    tf.keras.backend.set_image_data_format('channels_last')
    
    # Verify with the compiled function again
    res_compiled_2 = compiled_check('channels_last')
    assert res_compiled_2, "Compiled mode: Failed to preserve 'channels_last' format"

    print("All tests passed: image_data_format preserved correctly in both eager and compiled modes.")

finally:
    # Restore original format
    tf.keras.backend.set_image_data_format(original_format)
