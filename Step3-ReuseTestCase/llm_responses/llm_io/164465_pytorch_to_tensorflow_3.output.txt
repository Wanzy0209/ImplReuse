import tensorflow as tf

# Enable eager execution as requested by the target API context
tf.compat.v1.enable_eager_execution()

def test_int64_arange_max():
    # Define inputs matching the original PyTorch test case
    # PyTorch: x = torch.ones(1, 64, device='cuda', dtype=torch.int64)
    x = tf.ones((1, 64), dtype=tf.int64)
    
    # PyTorch: y = torch.randn(64, 3072, device='cuda', dtype=torch.bfloat16)
    y = tf.random.normal((64, 3072), dtype=tf.bfloat16)

    # PyTorch: embedding = torch.ops.aten.embedding.default(arg1_1, arg0_1)
    # TensorFlow equivalent: tf.nn.embedding_lookup(params, ids)
    embedding = tf.nn.embedding_lookup(y, x)

    # PyTorch: view = torch.ops.aten.view.default(embedding, [64, 3072])
    view = tf.reshape(embedding, [64, 3072])

    # PyTorch: unsqueeze = torch.ops.aten.unsqueeze.default(view, 0)
    unsqueeze = tf.expand_dims(view, 0)

    # PyTorch: expand = torch.ops.aten.expand.default(unsqueeze, [576, -1, -1])
    expand = tf.broadcast_to(unsqueeze, [576, -1, -1])

    # PyTorch: view_1 = torch.ops.aten.view.default(expand, [2, 8, 36, 64, 3072])
    view_1 = tf.reshape(expand, [2, 8, 36, 64, 3072])

    # PyTorch: permute = torch.ops.aten.permute.default(view_1, [0, 1, 3, 2, 4])
    permute = tf.transpose(view_1, [0, 1, 3, 2, 4])

    # PyTorch: clone = torch.ops.aten.clone.default(permute, memory_format = torch.contiguous_format)
    # TensorFlow tensors are immutable; identity serves as an explicit copy/pass-through
    clone = tf.identity(permute)

    # PyTorch: view_2 = torch.ops.aten.view.default(clone, [2, 18432, 3072])
    view_2 = tf.reshape(clone, [2, 18432, 3072])

    # PyTorch: iota = torch.ops.prims.iota.default(36, start = 0, step = 1, dtype = torch.int64, device = 'cuda', requires_grad = False)
    # TensorFlow equivalent: tf.range
    iota = tf.range(0, 36, dtype=tf.int64)

    # PyTorch: view_3 = torch.ops.aten.view.default(iota, [1, 36])
    view_3 = tf.reshape(iota, [1, 36])

    # PyTorch: max_1 = torch.ops.aten.max.default(view_3)
    # This is the critical operation from the bug report (int64 + max)
    max_1 = tf.reduce_max(view_3)

    # Verify the result
    # The range is 0 to 35, so the max should be 35
    assert max_1.numpy() == 35, f"Expected max to be 35, got {max_1.numpy()}"
    
    print("Test passed successfully.")

if __name__ == "__main__":
    test_int64_arange_max()
