import tensorflow as tf
import numpy as np

def test_rot90_negative_k_zero_dim():
    """
    Test case for tf.experimental.numpy.rot90 inspired by Issue 161014.
    
    The original bug in PyTorch's constant_pad_nd involved inconsistent behavior
    and crashes when negative padding values resulted in a dimension of size 0.
    
    This test adapts that logic to tf.experimental.numpy.rot90 by:
    1. Using an input tensor with a dimension of size 0 (analogous to the result of negative padding).
    2. Using a negative parameter 'k' (analogous to negative padding values).
    3. Verifying that the operation handles these edge cases consistently without crashing.
    """
    
    # Create a 3-D tensor [Height, Width, Channels] with Width = 0
    # This mirrors the state where the PyTorch operation resulted in a size 0 dimension.
    # Input shape: (5, 0, 3)
    input_tensor = tf.constant(np.zeros((5, 0, 3), dtype=np.float32))

    # Test 1: Negative rotation (k=-1)
    # In the original bug, negative values caused the issue. Here we test negative k.
    # Expected behavior: Rotate 270 degrees (or -90). 
    # Shape should swap: (5, 0, 3) -> (0, 5, 3)
    try:
        result = tf.experimental.numpy.rot90(input_tensor, k=-1)
        assert result.shape == (0, 5, 3), \
            f"Test 1 Failed: Expected shape (0, 5, 3) but got {result.shape}"
    except Exception as e:
        print(f"Test 1 Raised Exception: {e}")
        raise

    # Test 2: Negative rotation (k=-2)
    # Rotate 180 degrees.
    # Shape should swap back: (5, 0, 3) -> (5, 0, 3) (but reversed axes)
    try:
        result = tf.experimental.numpy.rot90(input_tensor, k=-2)
        assert result.shape == (5, 0, 3), \
            f"Test 2 Failed: Expected shape (5, 0, 3) but got {result.shape}"
    except Exception as e:
        print(f"Test 2 Raised Exception: {e}")
        raise

    # Test 3: Mixed scenario (similar to the failing case in the bug report)
    # The bug report failed when mixing padding that resulted in size 0 with other padding.
    # Here we mix a zero-dimension input with a standard positive rotation to ensure consistency.
    try:
        result = tf.experimental.numpy.rot90(input_tensor, k=1)
        assert result.shape == (0, 5, 3), \
            f"Test 3 Failed: Expected shape (0, 5, 3) but got {result.shape}"
    except Exception as e:
        print(f"Test 3 Raised Exception: {e}")
        raise

if __name__ == "__main__":
    test_rot90_negative_k_zero_dim()
    print("All tests passed.")
