import tensorflow as tf
from tensorflow.python.distribute.input_lib import DistributedIterator

# The malicious value from the original bug report (INT64_MAX - 4)
# This value caused a crash (realloc invalid pointer) in PyTorch Conv1d
INVALID_CARDINALITY = 9223372036854775803

# Mock input_workers to satisfy the constructor's initial validation checks
# (The API requires input_workers to be non-None)
class MockInputWorkers:
    pass

# Prepare dummy data to satisfy other constructor requirements
# The API requires either 'iterators' or both 'components' and 'element_spec'
dummy_components = [tf.constant([1.0, 2.0, 3.0])]
dummy_element_spec = tf.TensorSpec(shape=(None,), dtype=tf.float32)

# Attempt to create the DistributedIterator with the huge cardinality value.
# This tests if the TensorFlow API handles the large integer gracefully
# or if it suffers from a similar overflow/memory allocation crash.
try:
    iterator = DistributedIterator(
        input_workers=MockInputWorkers(),
        components=dummy_components,
        element_spec=dummy_element_spec,
        cardinality=INVALID_CARDINALITY
    )
    print("Iterator created successfully.")
except Exception as e:
    print(f"Exception caught: {e}")
