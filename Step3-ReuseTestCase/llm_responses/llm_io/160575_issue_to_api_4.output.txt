import torch
import warnings

def test_get_arch_list_maxwell_support():
    """
    Test case for Issue 160575: Verifying that torch.cuda.get_arch_list
    correctly reflects the absence of Maxwell (sm_52) support.
    
    This test adapts the logic pattern from tf.debugging.enable_traceback_filtering,
    which checks a version constraint (sys.version_info) and raises an error/warning.
    Here, we check the architecture capability constraint (sm_52 vs sm_61).
    """
    # Get the list of supported architectures
    arch_list = torch.cuda.get_arch_list()

    # Define the capabilities based on the bug report
    # The bug report states: "Minimum and Maximum cuda capability supported... is (6.1) - (9.0)"
    # and "The current PyTorch install supports CUDA capabilities sm_61 sm_70 sm_75 sm_80 sm_86 sm_90."
    min_capability = "sm_61"
    deprecated_capability = "sm_52"

    # Logic adapted from tf.debugging.enable_traceback_filtering:
    # if sys.version_info < (3, 7): raise RuntimeError
    # Here we verify the capability constraints.

    # Check if the minimum capability is supported
    # Equivalent to checking Python version >= 3.7
    if min_capability not in arch_list:
        raise AssertionError(
            f"Minimum capability {min_capability} is missing from arch_list: {arch_list}. "
            "This contradicts the warning message in the bug report."
        )

    # Check for the absence of the deprecated capability (sm_52)
    # The bug report confirms it is missing in the 2.8.0+cu126 build.
    if deprecated_capability in arch_list:
        warnings.warn(
            UserWarning(
                f"Found deprecated capability {deprecated_capability} in arch_list: {arch_list}. "
                "Issue 160575 suggests this should be missing for the current build."
            )
        )

    # Explicit assertion to confirm the bug report's observation
    assert deprecated_capability not in arch_list, \
        f"sm_52 should not be in the supported architecture list according to Issue 160575. Found: {arch_list}"
