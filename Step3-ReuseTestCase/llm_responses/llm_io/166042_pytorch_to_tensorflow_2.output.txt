import tensorflow as tf

def test_fuzzed_program():
    # Enable eager execution to match the "Eager" context in the bug title
    # (Note: Eager is default in TF 2.x)
    
    # Initialize inputs with dummy data to satisfy shape requirements
    # In the original fuzzer, these were specific values, but we use ones/zeros for reproducibility
    arg_0 = tf.ones((4, 8), dtype=tf.bfloat16)
    arg_1 = tf.ones((7, 12), dtype=tf.bfloat16)
    arg_2 = tf.ones((12, 2), dtype=tf.bfloat16)
    arg_3 = tf.ones((14, 9), dtype=tf.bfloat16)
    arg_4 = tf.ones((2,), dtype=tf.bfloat16)

    # Replicate the tensor generation logic from the PyTorch fuzzer
    # var_node_4 = arg_0
    var_node_5 = tf.constant(-0.80078125, shape=(8, 7), dtype=tf.bfloat16)
    var_node_3 = tf.matmul(arg_0, var_node_5) # (4, 7)

    # var_node_7 = arg_1
    # var_node_8 = arg_2
    var_node_6 = tf.matmul(arg_1, arg_2) # (7, 2)

    var_node_2 = tf.matmul(var_node_3, var_node_6) # (4, 2)

    var_node_11 = tf.constant(1.515625, shape=(2, 3), dtype=tf.bfloat16)
    var_node_12 = tf.constant(0.2353515625, shape=(3, 16), dtype=tf.bfloat16)
    var_node_10 = tf.matmul(var_node_11, var_node_12) # (2, 16)

    var_node_14 = tf.constant(2.21875, shape=(16, 4), dtype=tf.bfloat16)
    var_node_15 = tf.constant(-1.7421875, shape=(4, 9), dtype=tf.bfloat16)
    var_node_13 = tf.matmul(var_node_14, var_node_15) # (16, 9)

    var_node_9 = tf.matmul(var_node_10, var_node_13) # (2, 9)

    var_node_1 = tf.matmul(var_node_2, var_node_9) # (4, 9)

    var_node_19 = arg_3
    var_node_20 = tf.constant(0.8203125, shape=(9, 2), dtype=tf.bfloat16)
    var_node_18 = tf.matmul(var_node_19, var_node_20) # (14, 2)

    var_node_22 = arg_4
    var_node_23 = tf.constant(-0.7421875, shape=(2,), dtype=tf.bfloat16)
    var_node_21 = tf.add(var_node_22, var_node_23) # (2,)

    var_node_17 = tf.matmul(var_node_18, var_node_21) # (14,)

    # Adaptation to tf.compat.v1.losses.hinge_loss
    # The PyTorch bug involved passing these tensors to an embedding function.
    # Here we test the similar API (hinge_loss) with the generated tensors.
    # hinge_loss expects labels and logits of the same shape.
    # We use var_node_1 as logits and generate a dummy labels tensor.
    
    logits = var_node_1 # Shape (4, 9)
    labels = tf.zeros((4, 9), dtype=tf.bfloat16) # Dummy labels matching shape

    try:
        loss = tf.compat.v1.losses.hinge_loss(labels=labels, logits=logits)
        print("Test passed. Loss computed:", loss)
        assert loss.dtype == tf.bfloat16 or loss.dtype == tf.float32 # Check output dtype validity
    except Exception as e:
        print(f"Test failed with exception: {e}")
        raise

if __name__ == "__main__":
    test_fuzzed_program()
