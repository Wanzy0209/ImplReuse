import tensorflow as tf
import numpy as np

def test_geomspace_with_repeat_slice():
    """
    Adapted test case for Issue 167924.
    Original PyTorch bug: Crash on MPS when using repeat_interleave with sliced tensor.
    This test uses tf.experimental.numpy.geomspace (similar to torch.arange) 
    and tf.repeat (similar to torch.repeat_interleave) to verify behavior.
    """
    
    # Check for GPU availability (MPS on Mac is usually mapped to GPU in TensorFlow)
    gpus = tf.config.list_physical_devices('GPU')
    device_name = '/GPU:0' if gpus else '/CPU:0'
    
    print(f"Running test on device: {device_name}")

    with tf.device(device_name):
        # Original: counts = torch.tensor([0, 1, 0], device="mps")
        counts = tf.constant([0, 1, 0], dtype=tf.int32)

        # Original: data = torch.arange(2, device="mps")
        # Adaptation: Use tf.experimental.numpy.geomspace to generate the data tensor.
        # torch.arange(2) creates a sequence of length 2. 
        # geomspace(1, 2, num=2) creates [1.0, 2.0].
        data = tf.experimental.numpy.geomspace(1, 2, num=2)

        # Original: data.repeat_interleave(counts[1:3], dim=0)
        # The core bug logic involves passing a non-prefix slice of a tensor (counts[1:3])
        # to the repeat operation.
        # counts[1:3] extracts [1, 0].
        sliced_counts = counts[1:3]

        # TensorFlow equivalent of torch.repeat_interleave is tf.repeat
        # We verify if the operation completes without crashing (segfault).
        try:
            result = tf.repeat(data, sliced_counts, axis=0)
            
            # Verify the result is correct based on the inputs
            # data = [1.0, 2.0], repeats = [1, 0] -> result should be [1.0]
            expected = tf.constant([1.0])
            
            # Assert to check correctness
            assert tf.reduce_all(tf.equal(result, expected)).numpy(), \
                f"Expected {expected.numpy()}, but got {result.numpy()}"
            
            print("Test Passed. Operation completed successfully.")
            print(f"Input data: {data.numpy()}")
            print(f"Sliced counts: {sliced_counts.numpy()}")
            print(f"Result: {result.numpy()}")

        except Exception as e:
            print(f"Test Failed with exception: {e}")
            raise

if __name__ == "__main__":
    test_geomspace_with_repeat_slice()
