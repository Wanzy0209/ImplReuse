import tensorflow as tf
import numpy as np

def test_tf_math_real():
    """
    Adapted test case for tf.math.real based on the torch.nn.functional.normalize bug report.
    
    Original Bug Logic:
    - Input: [[3.799999, 0.0, 0.0]]
    - Issue: torch.compile + cuda resulted in norm > 1 (precision error).
    - Check: Compare compiled vs uncompiled output.
    
    Adapted Logic for tf.math.real:
    - Input: [[3.799999, 0.0, 0.0]] (Real) and [[3.799999+1j, 0j, 0j]] (Complex).
    - Issue to check: Does tf.function (compilation) introduce precision errors or incorrect behavior?
    - Check: Verify compiled vs uncompiled outputs match expected real parts.
    """
    
    # Setup input data similar to the PyTorch issue
    input_data_real = [[3.799999, 0.0, 0.0]]
    input_data_complex = [[3.799999 + 1.5j, 0.0 + 0.0j, 0.0 + 0.0j]]

    # Define compiled and uncompiled functions
    @tf.function
    def get_real_compiled(input_tensor):
        return tf.math.real(input_tensor)

    def get_real_uncompiled(input_tensor):
        return tf.math.real(input_tensor)

    # --- Test 1: Real Input ---
    # tf.math.real on a real tensor acts as an identity function.
    print("--- Test 1: Real Input ---")
    tensor_real = tf.constant(input_data_real, dtype=tf.float32)
    print("Input vector:", tensor_real.numpy()[0])

    res_compiled = get_real_compiled(tensor_real)
    res_uncompiled = get_real_uncompiled(tensor_real)

    print("Result (compiled):", res_compiled.numpy()[0])
    print("Result (uncompiled):", res_uncompiled.numpy()[0])

    # Assert compiled and uncompiled are identical
    assert np.allclose(res_compiled.numpy(), res_uncompiled.numpy()), \
        "Mismatch between compiled and uncompiled results for real input."
    
    # Assert result matches input (identity check)
    assert np.allclose(res_compiled.numpy(), input_data_real), \
        "Real input was modified unexpectedly."

    # --- Test 2: Complex Input ---
    # This is the primary use case for tf.math.real.
    print("\n--- Test 2: Complex Input ---")
    tensor_complex = tf.constant(input_data_complex, dtype=tf.complex64)
    print("Input vector (complex):", tensor_complex.numpy()[0])

    res_complex_compiled = get_real_compiled(tensor_complex)
    res_complex_uncompiled = get_real_uncompiled(tensor_complex)

    print("Real part (compiled):", res_complex_compiled.numpy()[0])
    print("Real part (uncompiled):", res_complex_uncompiled.numpy()[0])

    # Assert compiled and uncompiled are identical
    assert np.allclose(res_complex_compiled.numpy(), res_complex_uncompiled.numpy()), \
        "Mismatch between compiled and uncompiled results for complex input."

    # Assert the extracted real part is correct
    expected_real = np.array([[3.799999, 0.0, 0.0]])
    assert np.allclose(res_complex_compiled.numpy(), expected_real), \
        f"Extracted real part {res_complex_compiled.numpy()} does not match expected {expected_real}."

    print("\nAll assertions passed.")

if __name__ == "__main__":
    test_tf_math_real()
