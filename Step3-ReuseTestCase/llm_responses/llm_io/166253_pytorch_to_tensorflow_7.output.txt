```python
import tensorflow as tf
import tempfile
import os

# Setup a temporary directory for logs to ensure the summary ops have a target
log_dir = tempfile.mkdtemp()

def func_nojit(x):
    # We use tf.print to verify the value being processed inside the function.
    # This mimics the print statement in the original PyTorch test case.
    tf.print("Processing value:", x)
    return tf.summary.scalar("test_summary", x)

# tf.function is the TensorFlow equivalent of torch.compile
func_jit = tf.function(func_nojit)

# Create a summary writer context
writer = tf.summary.create_file_writer(log_dir)

# Run the test
with writer.as_default():
    for func in [func_nojit, func_jit]:
        print(f"--- Running {func.__name__} ---")
        
        # Define inputs matching the original test case (float64)
        x1 = tf.constant(5.0, dtype=tf.float64)
        x2 = tf.constant(10.0, dtype=tf.float64)
        
        # Call the function with different values
        # In the original bug, the compiled version would print 5.0 for both calls
        func(x1)
        func(x2)
        
        print("-" * 30)

print(f"Logs written to: {log_dir}")
```
