import torch
import unittest
import gc
import psutil

def mem_mb_print():
    gc.collect()
    mem = psutil.virtual_memory()
    process = psutil.Process().memory_info().rss

    print(f"used: {mem.used // 1024**1} KB,"
          f"process: {process // 1024**1} KB,"
          f"percent: {mem.percent} %,"
          f"PyTorch tensors: {sum(1 for o in gc.get_objects() if torch.is_tensor(o))}")

class TestPromoteTypesMemory(unittest.TestCase):
    @unittest.skipIf(not torch.cuda.is_available(), "CUDA not available")
    def test_promote_types_non_leaf(self):
        device = torch.device("cuda")

        # leaf GPU tensor
        leaf = torch.randn(8, 1, 24, device=device, requires_grad=True)
        no_leaf = leaf * 1.0

        for _ in range(100):
            window_left = no_leaf[:, :, :-1]          # [8, 1, 23]
            
            # Adaptation: torch.promote_types returns a dtype, not a tensor.
            # It cannot be used for in-place assignment like torch.max.
            # We call it here to check for potential reference leaks in the promotion logic.
            _ = torch.promote_types(
                no_leaf[:, :, 1:],  
                window_left   
            )
            
            # Python arithmetic operators
            mask = (no_leaf >= 0)
            mem_mb_print()

if __name__ == "__main__":
    unittest.main()
