import torch
import torch.nn as nn
import unittest

class TestMPSConvTranspose3DAutocast(unittest.TestCase):
    """
    Test case for Issue 160332: Use FP32 for ConvTranspose3D when using autocast on MPS.
    
    The bug report indicates that `torch.amp.autocast` attempts to use FP16 for 
    ConvTranspose3d on MPS, which is not supported, leading to a RuntimeError.
    This test verifies that the operation executes successfully within the autocast context,
    implying the correct dtype fallback or handling is in place.
    """

    @unittest.skipIf(not torch.backends.mps.is_available(), "MPS backend is not available")
    def test_convtranspose3d_autocast_mps(self):
        device = torch.device('mps')

        # Reproduce the logic from the bug report
        with torch.amp.autocast(device_type=device.type):
            m = nn.ConvTranspose3d(16, 33, 3, stride=2)
            m.to(device)
            x = torch.randn(20, 16, 10, 50, 100).to(device)
            
            # This line previously raised:
            # RuntimeError: ConvTranspose 3D with BF16 or FP16 types is not supported on MPS
            # We expect this to run without error.
            u = m(x)

        # Verify the output shape is correct
        # Input: (N=20, C=16, D=10, H=50, W=100)
        # Output: (N=20, C=33, D=21, H=101, W=201) based on kernel=3, stride=2
        self.assertEqual(u.shape, (20, 33, 21, 101, 201))

if __name__ == '__main__':
    unittest.main()
