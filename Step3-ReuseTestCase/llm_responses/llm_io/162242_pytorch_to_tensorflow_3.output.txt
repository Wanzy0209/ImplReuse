import tensorflow as tf
import numpy as np

# Disable eager execution to ensure tf.compat.v1 operations behave as in graph mode
tf.compat.v1.disable_eager_execution()

def test_assert_rank():
    """
    Adapted test case for tf.compat.v1.assert_rank based on the tensor shapes
    from the original torch.scatter issue.
    """
    # Reconstruct tensors from the original PyTorch test case
    # inputs: shape [2, 3, 4] -> Rank 3
    # src: shape [2, 2, 3] -> Rank 3
    # index: shape [2, 2, 3] -> Rank 3
    
    inputs = tf.constant(np.arange(24, dtype=np.float32).reshape([2, 3, 4]))
    src = tf.constant(np.arange(-99, -99 + (2 * 2 * 3), dtype=np.float32).reshape([2, 2, 3]))
    index = tf.constant(
        [
            [
                [0, 1, 0],
                [1, 1, 0],
            ],
            [
                [1, 0, 1],
                [0, 0, 1],
            ],
        ],
        dtype=tf.int64,
    )

    with tf.compat.v1.Session() as sess:
        # Test 1: Verify that the tensors have the expected rank (3)
        print("Test 1: Verifying rank 3 for inputs, src, and index...")
        try:
            # These assertions should pass as all tensors are 3D
            sess.run(tf.compat.v1.assert_rank(inputs, 3))
            sess.run(tf.compat.v1.assert_rank(src, 3))
            sess.run(tf.compat.v1.assert_rank(index, 3))
            print("Test 1 Passed: All tensors have rank 3.")
        except Exception as e:
            print(f"Test 1 Failed: {e}")

        # Test 2: Verify that the assertion raises an error for incorrect rank
        # Create a tensor with Rank 2
        tensor_rank_2 = tf.constant([[1.0, 2.0], [3.0, 4.0]])
        print("Test 2: Verifying assertion failure for rank 2 tensor (expecting rank 3)...")
        try:
            sess.run(tf.compat.v1.assert_rank(tensor_rank_2, 3))
            print("Test 2 Failed: Expected InvalidArgumentError but none was raised.")
        except tf.errors.InvalidArgumentError as e:
            print(f"Test 2 Passed: Correctly raised InvalidArgumentError.")

if __name__ == "__main__":
    test_assert_rank()
