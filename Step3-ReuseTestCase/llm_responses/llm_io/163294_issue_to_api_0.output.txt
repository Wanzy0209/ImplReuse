import torch
import torch.nn.functional as F

class HardtanhNoGradCase(torch.nn.Module):
    def forward(self, x):
        with torch.no_grad():
            # Leveraging the similar API (torch.nn.functional.hardtanh_) 
            # inside the context manager where the bug occurs.
            F.hardtanh_(x)
        return x

def test_retracing_set_grad_with_hardtanh():
    # First export
    ep = torch.export.export(
        HardtanhNoGradCase(),
        (torch.randn(6),),
        strict=False,
    )
    print("First Export:")
    print(ep)

    # Second export (retracing)
    # The original bug manifests here, creating an empty submodule (submod_1)
    # when handling the wrap_with_set_grad_enabled operator.
    ep2 = torch.export.export(ep.module(), (torch.randn(6),))
    print("\nSecond Export (Retracing):")
    print(ep2)

    # Check if the operation is present in the graph
    # If the bug (empty submodule) is present, the operation might be missing or malformed.
    graph_str = str(ep2)
    assert "hardtanh" in graph_str, "Expected hardtanh operation in the re-exported graph"

if __name__ == "__main__":
    test_retracing_set_grad_with_hardtanh()
