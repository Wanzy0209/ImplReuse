```python
import torch
import torch.distributed as dist
import torch.multiprocessing as mp
import os

def setup(rank, world_size):
    os.environ['MASTER_ADDR'] = 'localhost'
    os.environ['MASTER_PORT'] = '12355'
    # Initialize the process group
    dist.init_process_group("gloo", rank=rank, world_size=world_size)

def cleanup():
    dist.destroy_process_group()

def test_batch_isend_irecv_views(rank, world_size):
    setup(rank, world_size)

    batch_size = 4
    total_columns = 10
    # Define offsets to create two non-contiguous views (slices)
    # Slice 1: columns 0 to 4
    # Slice 2: columns 6 to 10
    # Gap: columns 4 to 6 (should remain zero on receiver)
    split_offsets = [0, 4, 6, 10]

    # Use a fixed seed to ensure both ranks generate the same "expected" data
    torch.manual_seed(42)
    
    if rank == 0:
        # Sender rank
        # Create a random tensor
        local_tensor = torch.randn(batch_size, total_columns)
        
        # Create views (slices)
        dst_t1 = local_tensor[:, split_offsets[0]:split_offsets[1]]
        dst_t2 = local_tensor[:, split_offsets[2]:split_offsets[3]]

        process_group = dist.group.WORLD
        send_op1 = dist.P2POp(dist.isend, dst_t1, 1, process_group)
        send_op2 = dist.P2POp(dist.isend, dst_t2, 1, process_group)

        # Batch send operations
        reqs = dist.batch_isend_irecv([send_op1, send_op2])
        
        # Wait for completion
        for req in reqs:
            req.wait()
            
        print(f"Rank {rank} sent tensors with shapes {dst_t1.shape} and {dst_t2.shape}")

    elif rank == 1:
        # Receiver rank
        # Initialize with zeros to verify data transfer clearly
        local_tensor_dst = torch.zeros(batch_size, total_columns)
        
        # Create views matching the sender's slices
        receiving_tensor_view1 = local_tensor_dst[:, split_offsets[0]:split_offsets[1]]
        receiving_tensor_view2 = local_tensor_dst[:, split_offsets[2]:split_offsets[3]]

        process_group = dist.group.WORLD
        receive_op1 = dist.P2POp(dist.irecv, receiving_tensor_view1, 0, process_group)
        receive_op2 = dist.P2POp(dist.irecv, receiving_tensor_view2, 0, process_group)

        # Batch receive operations
        reqs = dist.batch_isend_irecv([receive_op1, receive_op2])
        
        # Wait for completion
        for req in reqs:
            req.wait()

        # --- Verification ---
        # Generate the expected tensor locally using the same seed
        expected_tensor = torch.randn(batch_size, total_columns)
        expected_view1 = expected_tensor[:, split_offsets[0]:split_offsets[1]]
        expected_view2 = expected_tensor[:, split_offsets[2]:split_offsets[3]]

        # Check if the received data matches the expected data
        # This assertion will fail if the bug (data inconsistency) is present
        assert torch.equal(receiving_tensor_view1, expected_view1), \
            "Data inconsistency in receiving_tensor_view1"
        assert torch.equal(receiving_tensor_view2, expected_view2), \
            "Data inconsistency in receiving_tensor_view2"
        
        # Ensure the untouched region remains zero
        untouched_region = local_tensor_dst[:, split_offsets[1]:split_offsets[2]]
        assert torch.all(untouched_region == 0), \
            "Untouched region was modified"

        print(f"Rank {rank} received data correctly and verified consistency.")

    cleanup()

def main():
    world_size = 2
    # Spawn 2 processes
    mp.spawn(test_batch_isend_irecv_views,
             args=(world_size,),
             nprocs=world_size,
             join=True)

if __name__ == "__main__":
    main()
```
