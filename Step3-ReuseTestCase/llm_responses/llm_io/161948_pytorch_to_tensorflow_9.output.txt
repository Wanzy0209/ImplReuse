```python
import tensorflow as tf
import time

# Use compat.v1 to match the target API context
tf.compat.v1.disable_eager_execution()

# List of threads to test
threads_list = [1, 2, 4, 8, 16, 32, 48]

# Size of the large tensors
tensor_size = (10000, 10000)

# Store results
times = []

# Define the computation graph
# We use Variables to hold the large tensors
a = tf.Variable(tf.random.uniform(tensor_size))
b = tf.Variable(tf.random.uniform(tensor_size))

# Define the operation (Matrix Multiplication)
c = tf.matmul(a, b)

# Define the loss (sum of elements)
loss = tf.reduce_sum(c)

# Target API: Calculate gradients
grads = tf.compat.v1.gradients(loss, [a, b])

# Initialize variables
init = tf.compat.v1.global_variables_initializer()

# Benchmark for each thread count
for threads in threads_list:
    # Configure TensorFlow session threads
    # This is the TF equivalent of torch.set_num_threads
    config = tf.compat.v1.ConfigProto()
    config.intra_op_parallelism_threads = threads
    config.inter_op_parallelism_threads = threads

    with tf.compat.v1.Session(config=config) as sess:
        sess.run(init)

        # Warm up
        _ = sess.run(grads)

        # Time the gradient computation
        start_time = time.time()
        _ = sess.run(grads)
        elapsed_time = time.time() - start_time

        print(f"Threads: {threads}, Time: {elapsed_time:.4f} s")
        times.append(elapsed_time)
```
