import torch
import torch._dynamo.functional_export
from typing import Callable, TypeVar

# Helper functions from the original bug report
def compute(x, w):
    return torch.nn.functional.linear(x, w)

def nop(x, w):
    torch._check(x.shape[0] == 0)
    return torch.empty_like(x)

def chunked_compute(x, w):
    sz = x.shape[0]
    torch._check(sz <= 8)
    out0 = torch.cond(sz > 0, compute, nop, (x[0:2], w))
    out1 = torch.cond(sz > 2, compute, nop, (x[2:4], w))
    out2 = torch.cond(sz > 4, compute, nop, (x[4:6], w))
    out3 = torch.cond(sz > 6, compute, nop, (x[6:8], w))
    return torch.cat([out0, out1, out2, out3])

class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(16, 16)

    def forward(self, x):
        return chunked_compute(x, self.linear.weight)

# Reusing the pattern of the similar API: tf.compat.v1.disable_control_flow_v2
# The TF API is a decorator factory that takes a message and returns a wrapper.
# We adapt this pattern to wrap the PyTorch model for export.
def setup_dynamo_export(unused_msg: str) -> Callable:
    """
    Mimics the structure of tf.compat.v1.disable_control_flow_v2.
    Wraps the model instantiation with the dynamo export logic.
    """
    def wrapper(model: torch.nn.Module):
        # Instead of setting an attribute like TF, we call the PyTorch export API
        return torch._dynamo.functional_export._dynamo_graph_capture_for_export(model)
    return wrapper

# Test execution
if __name__ == "__main__":
    x, w = torch.randn(4, 16, requires_grad=True), torch.randn(16, 16, requires_grad=True)
    
    # Verify logic correctness before export
    assert torch.equal(compute(x, w), chunked_compute(x, w))

    # Apply the wrapper pattern similar to the TF API
    export_decorator = setup_dynamo_export("Testing stack trace with control flow")
    
    # Instantiate model and apply export
    model_instance = Model()
    mod = export_decorator(model_instance)
    
    # Run the exported module
    result = mod(x)
    
    # Verify the result matches expected output
    expected = chunked_compute(x, model_instance.linear.weight)
    assert torch.allclose(result, expected), "Exported graph output does not match expected"
    
    print("Test passed.")
