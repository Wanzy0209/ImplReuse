import torch
import unittest

class TestTorchAnyCompile(unittest.TestCase):
    """
    Test case for torch.any based on the torch.compile regression report.
    The original bug (Issue 164301) describes a performance regression in 
    torch.compile for operations along rows (dim0). This test verifies that 
    torch.any, which shares torch._dynamo compilation paths, functions 
    correctly when compiled.
    """

    def test_any_compile_correctness(self):
        """
        Verify that torch.any produces correct results when run through 
        torch.compile, matching the eager execution result.
        """
        def fn(x):
            return torch.any(x)

        # Use dimensions similar to the bug report (M=16384, K=16384)
        # The bug involved 'dim0' (rows), so we test reduction along dim 0 as well.
        x = torch.randn(16384, 16384, device='cpu') > 0.5

        # Eager execution
        expected = fn(x)
        
        # Compiled execution
        compiled_fn = torch.compile(fn)
        actual = compiled_fn(x)

        self.assertEqual(expected, actual)

    def test_any_dim0_compile(self):
        """
        Verify torch.any reduction along dimension 0 (rows) under torch.compile.
        This mirrors the 'dim0_mxfp8_floor' mode from the original bug report.
        """
        def fn(x):
            return torch.any(x, dim=0)

        x = torch.randn(16384, 16384, device='cpu') > 0.5

        expected = fn(x)
        compiled_fn = torch.compile(fn)
        actual = compiled_fn(x)

        self.assertEqual(expected, actual)

if __name__ == '__main__':
    unittest.main()
