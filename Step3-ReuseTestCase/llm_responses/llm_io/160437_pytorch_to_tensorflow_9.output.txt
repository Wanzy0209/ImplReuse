import tensorflow as tf

# Adapted test case for tf.name_scope
# The original bug involves graph compilation and empty graphs.
# We use @tf.function to trigger graph compilation and tf.name_scope to test the API.

@tf.function
def fn(x, i):
    # Using tf.name_scope as the context manager for the operation
    with tf.name_scope("my_scope"):
        # Conditional logic similar to the original test
        if i == 1:
            # In PyTorch, this triggers a graph break.
            # In TensorFlow, we check if the scope handles the control flow correctly.
            pass
        return x + 1

inp = tf.random.normal((3,))

# Execute with different inputs to trigger tracing
fn(inp, 0)
fn(inp, 1)
fn(inp, 2)

# Verify the graph is not empty (checking for the bug condition)
# We inspect the concrete function generated by tf.function
concrete_fn = fn.get_concrete_function(tf.TensorSpec(shape=(3,), dtype=tf.float32), tf.TensorSpec(shape=(), dtype=tf.int32))
graph = concrete_fn.graph

# Assert that the graph contains operations (is not empty)
ops = list(graph.get_operations())
assert len(ops) > 0, "Graph should not be empty"
