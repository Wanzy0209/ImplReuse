import numpy as np
import torch


def test_transpose_deterministic():
    # Ground truth for torch.transpose(inputs, 0, 1)
    # inputs shape: [2, 3, 4]
    # output shape: [3, 2, 4]
    gt_res = np.array(
        [
            [
                [0.0, 1.0, 2.0, 3.0],
                [12.0, 13.0, 14.0, 15.0],
            ],
            [
                [4.0, 5.0, 6.0, 7.0],
                [16.0, 17.0, 18.0, 19.0],
            ],
            [
                [8.0, 9.0, 10.0, 11.0],
                [20.0, 21.0, 22.0, 23.0],
            ],
        ],
        dtype=np.float32,
    )
    
    # Ground truth for gradients
    # Transpose is a permutation, so gradient is just the upstream gradient permuted back.
    # Since upstream is ones, input grad is ones.
    gt_input_grad = np.ones((2, 3, 4), dtype=np.float32)

    for i in range(1000):
        torch.cuda.empty_cache()
        inputs = torch.arange(24, dtype=torch.float32).reshape([2, 3, 4]).cuda()
        
        inputs.requires_grad = True

        # Adapted call site: torch.transpose instead of torch.scatter
        # torch.transpose(input, dim0, dim1)
        res = torch.transpose(inputs, 0, 1)
        
        res.backward(torch.ones_like(res))

        print(f"Test {i + 1}/{1000}")
        np.testing.assert_allclose(res.cpu().detach().numpy(), gt_res)
        np.testing.assert_allclose(inputs.grad.cpu().numpy(), gt_input_grad)


if __name__ == "__main__":
    test_transpose_deterministic()
