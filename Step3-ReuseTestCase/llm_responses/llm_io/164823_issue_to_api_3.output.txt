import torch
import torch.nn as nn
import unittest

class TestSparseCompile(unittest.TestCase):
    def test_sparse_to_dense_compile(self):
        """
        Test that torch.compile works with models using to_sparse() and to_dense().
        This test reproduces the logic from Issue 164823.
        It also leverages torch.backends.cudnn.version to check the backend environment,
        similar to the pattern found in the similar API code.
        """
        
        # Leverage the similar API: torch.backends.cudnn.version
        # We check the backend version/status if CUDA is available, 
        # mirroring the initialization check pattern in the similar API.
        if torch.cuda.is_available():
            cudnn_version = torch.backends.cudnn.version()
            self.assertIsNotNone(cudnn_version, "cuDNN version should be available if CUDA is present")
            print(f"Running test with cuDNN version: {cudnn_version}")

        class TestModel(nn.Module):
            def forward(self, x):
                x_sparse = x.to_sparse()
                result = x_sparse * 2
                return result.to_dense()

        x = torch.randn(10, 10)
        model = TestModel()

        # Run in eager mode
        eager_output = model(x)

        # Run in compiled mode (defaults to inductor backend)
        # The bug report specifically mentions the 'inductor' backend failing.
        compiled_model = torch.compile(model, backend="inductor")
        
        # This line should not raise NotImplementedError
        compiled_output = compiled_model(x)

        # Verify outputs match
        self.assertTrue(torch.allclose(eager_output, compiled_output))

if __name__ == "__main__":
    unittest.main()
