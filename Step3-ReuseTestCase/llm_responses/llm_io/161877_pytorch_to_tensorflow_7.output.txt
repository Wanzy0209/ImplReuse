import tensorflow as tf
import numpy as np

def test_tf_keras_ops_arctan2_large_values():
    """
    Adapted test case for tf.keras.ops.arctan2 based on PyTorch Conv1d issue 161877.
    
    The original bug involved passing an extremely large integer (9223372036854775803)
    to the 'padding' parameter of Conv1d, causing a crash (realloc(): invalid pointer).
    
    Since tf.keras.ops.arctan2 is a mathematical operation that accepts tensor inputs
    rather than configuration parameters like 'padding', we adapt the test by passing
    the problematic large integer value as data input to verify the API's robustness
    against extreme numerical values.
    """
    
    # 1. Setup input data similar to the original PyTorch test case
    # PyTorch: torch.randn(1, 16, 100)
    input_data = tf.random.normal((1, 16, 100))
    
    # The large integer value from the original bug report
    large_int_value = 9223372036854775803
    
    # 2. Test Case: Standard operation with random data
    # Verifies basic functionality
    x1 = input_data
    x2 = input_data
    output = tf.keras.ops.arctan2(x1, x2)
    
    assert output.shape == (1, 16, 100), "Output shape mismatch for standard inputs"
    
    # 3. Test Case: Operation with the large integer value (as data)
    # This attempts to preserve the core logic of testing with the extreme value
    # that caused the crash in the original API.
    # We use float64 to accommodate the large integer precision.
    x1_large = tf.constant(large_int_value, shape=(1, 16, 100), dtype=tf.float64)
    x2_large = tf.constant(1.0, shape=(1, 16, 100), dtype=tf.float64)
    
    # Execute the operation
    output_large = tf.keras.ops.arctan2(x1_large, x2_large)
    
    # Assertions
    assert output_large.shape == (1, 16, 100), "Output shape mismatch for large value inputs"
    
    # Verify the mathematical result is correct (arctan2(x, 1) for large x approaches pi/2)
    # We check the first element to ensure calculation occurred correctly without crashing
    expected_val = np.arctan2(large_int_value, 1.0)
    assert np.allclose(output_large.numpy().flatten()[0], expected_val), \
        "Output value mismatch for large value inputs"

    print("Test passed: tf.keras.ops.arctan2 handled large values gracefully.")

if __name__ == "__main__":
    test_tf_keras_ops_arctan2_large_values()
