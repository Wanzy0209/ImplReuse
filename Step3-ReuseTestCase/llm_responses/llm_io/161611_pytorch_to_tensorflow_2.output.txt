import tensorflow as tf
import numpy as np

def test_vdot_redundant_dtype_handling():
    """
    Adapted from PyTorch Issue 161611.
    
    Original Issue: A redundant dtype conversion (attn_bias.to(query.dtype)) 
    was performed on a tensor that already had the correct dtype, and the 
    result was not assigned.
    
    Adaptation for tf.keras.ops.vdot:
    Verify that vdot handles dtype promotion internally. This ensures that 
    manual casting of inputs (which could be redundant if types match, or 
    ineffective if not assigned) is not required for the operation to succeed.
    """
    
    # Setup inputs
    # Case 1: Inputs are already the same dtype (float32)
    # This mirrors the scenario where a manual cast would be redundant.
    a = tf.constant([1.0, 2.0], dtype=tf.float32)
    b = tf.constant([3.0, 4.0], dtype=tf.float32)
    
    # Call vdot. It should handle the operation without needing manual casting.
    result_same_dtype = tf.keras.ops.vdot(a, b)
    
    # Case 2: Inputs are different dtypes (int32 and float32)
    # vdot should promote int32 to float32 automatically.
    a_int = tf.constant([1, 2], dtype=tf.int32)
    b_float = tf.constant([3.0, 4.0], dtype=tf.float32)
    
    # Call vdot. Internal promotion handles the type mismatch.
    result_mixed_dtype = tf.keras.ops.vdot(a_int, b_float)
    
    # Expected result: 1*3 + 2*4 = 11
    expected_val = 11.0
    
    # Assertions
    # Check that the calculation is correct for both cases
    assert np.isclose(result_same_dtype, expected_val), \
        f"Expected {expected_val}, got {result_same_dtype}"
    assert np.isclose(result_mixed_dtype, expected_val), \
        f"Expected {expected_val}, got {result_mixed_dtype}"
        
    # Check that the output dtype is consistent (float32)
    assert result_same_dtype.dtype == tf.float32, \
        f"Expected float32, got {result_same_dtype.dtype}"
    assert result_mixed_dtype.dtype == tf.float32, \
        f"Expected float32, got {result_mixed_dtype.dtype}"
    
    print("Test passed: vdot handles dtype promotion correctly, making manual casting redundant.")

if __name__ == "__main__":
    test_vdot_redundant_dtype_handling()
