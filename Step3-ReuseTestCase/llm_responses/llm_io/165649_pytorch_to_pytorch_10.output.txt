```python
import torch
import torch.distributed as dist
import torch.multiprocessing as mp
import os

def setup(rank, world_size):
    # Initialize the process group
    os.environ['MASTER_ADDR'] = 'localhost'
    os.environ['MASTER_PORT'] = '12355'
    dist.init_process_group("gloo", rank=rank, world_size=world_size)

def cleanup():
    dist.destroy_process_group()

def run_test(rank, world_size):
    setup(rank, world_size)

    # Adapted from original test case: Create tensor with INT64_MIN
    # The original bug involved torch.fmod crashing with INT64_MIN.
    # We test if torch.distributed.broadcast handles this extreme value correctly.
    if rank == 0:
        tensor = torch.full((2, 3), torch.iinfo(torch.int64).min, dtype=torch.int64, device='cpu')
    else:
        tensor = torch.zeros((2, 3), dtype=torch.int64, device='cpu')

    print(f"Rank {rank} before broadcast: {tensor[0, 0].item()}")

    # Call the similar API: torch.distributed.broadcast
    dist.broadcast(tensor, src=0)

    print(f"Rank {rank} after broadcast: {tensor[0, 0].item()}")

    # Verify the value was broadcast correctly
    expected_val = torch.iinfo(torch.int64).min
    assert tensor[0, 0].item() == expected_val, f"Rank {rank} failed to receive correct value."

    cleanup()

if __name__ == "__main__":
    world_size = 2
    mp.spawn(run_test, args=(world_size,), nprocs=world_size, join=True)
```
