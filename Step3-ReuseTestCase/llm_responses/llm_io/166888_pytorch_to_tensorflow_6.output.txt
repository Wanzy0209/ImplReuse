```python
import tensorflow as tf

# Adapt the function to use tf.compat.v1.name_scope
# We use tf.function to mimic the compilation/graph aspect of torch.compile
@tf.function
def f(x, max_val):
    # The target API: tf.compat.v1.name_scope
    # This context manager groups operations under a specific name prefix
    with tf.compat.v1.name_scope("clamp_scope"):
        # In TensorFlow, clip_by_value accepts tensors directly.
        # Unlike the PyTorch case where .item() was used (causing the bug),
        # here we pass the tensor max_val directly to maintain graph integrity.
        y = tf.clip_by_value(x, 0.0, max_val)
    return y

# Setup inputs similar to the original test case
x = tf.random.normal((10, 20, 30))
max_val = tf.constant(5.0)

# Execute the function
result = f(x, max_val)

# Verify the result
assert result.shape == (10, 20, 30)
# Verify clamping logic (values should be <= 5.0)
assert tf.reduce_all(result <= 5.0)
```
