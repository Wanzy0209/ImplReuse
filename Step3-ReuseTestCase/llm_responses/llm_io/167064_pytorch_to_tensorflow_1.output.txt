import tensorflow as tf

# Global state to monitor for side effects, mimicking torch.distributions settings
GLOBAL_VALIDATE_ARGS = True

def test_batch_parallel_global_state():
    """
    Adapted test case for Issue 167064.
    
    Original Bug: torch.compile in PyTorch was redundantly executing global code
    that set torch.distributions.Distribution.set_default_validate_args(False),
    causing unwanted global side effects.
    
    This test verifies that tf.compat.v1.tpu.batch_parallel does not similarly
    pollute global state during its setup or execution.
    """
    
    # Define a simple computation to be parallelized
    def computation():
        return tf.constant([1.0, 2.0, 3.0])

    # Ensure initial state
    assert GLOBAL_VALIDATE_ARGS is True, "Initial global state check failed"

    # Note: tf.compat.v1.tpu.batch_parallel requires a TPU environment.
    # This test structure assumes the environment is configured or mocks the behavior
    # to check for side effects in the compilation/execution graph construction.
    
    try:
        # Attempt to initialize TPU system if available
        resolver = tf.distribute.cluster_resolver.TPUClusterResolver()
        tf.config.experimental_connect_to_cluster(resolver)
        tf.tpu.experimental.initialize_tpu_system(resolver)
        strategy = tf.distribute.TPUStrategy(resolver)

        with strategy.scope():
            # Prepare inputs (batch dimension sharding)
            inputs = [[tf.constant([1.0, 2.0, 3.0])]]
            
            # Call the API similar to torch.compile
            # In PyTorch, the bug occurred during the call to compile.
            result = tf.compat.v1.tpu.batch_parallel(
                computation, 
                inputs=inputs,
                num_shards=1
            )
            
            # Force execution to ensure any graph compilation code runs
            _ = tf.identity(result)

    except (tf.errors.NotFoundError, ValueError) as e:
        # Handle cases where TPU is not present (common in local dev)
        # We still want to check if the API call construction itself affects globals
        # even if it fails to execute on hardware.
        print(f"TPU not found or initialization failed (expected in non-TPU env): {e}")
        pass
    except Exception as e:
        print(f"Unexpected error: {e}")
        pass

    # Core Assertion: Verify the global state has not been modified
    # In the PyTorch bug, this would have been flipped to False.
    assert GLOBAL_VALIDATE_ARGS is True, \
        "Global state was modified by batch_parallel, indicating redundant global code execution."

if __name__ == "__main__":
    test_batch_parallel_global_state()
