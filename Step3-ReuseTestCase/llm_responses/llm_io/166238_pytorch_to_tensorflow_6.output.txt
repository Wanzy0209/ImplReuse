import collections
import tensorflow as tf

def test_defaultdict_creation_in_name_scope():
    """
    Test that collections.defaultdict can be created within 
    tf.compat.v1.name_scope without causing tracing or graph errors.
    This addresses a similar regression found in PyTorch Dynamo where
    defaultdict creation was unsupported during tracing.
    """
    
    # We use tf.function to simulate the graph tracing environment
    # similar to torch.compile.
    @tf.function
    def create_defaultdict_in_scope():
        with tf.compat.v1.name_scope("defaultdict_scope"):
            # The operation that caused the bug in PyTorch Dynamo
            dd = collections.defaultdict(list)
            dd["a"].append(1)
            dd["b"].append(2)
            return dd

    # Execute the function to ensure it traces correctly
    result = create_defaultdict_in_scope()
    
    # Verify the result
    assert result["a"] == [1]
    assert result["b"] == [2]
    assert isinstance(result, collections.defaultdict)

if __name__ == "__main__":
    test_defaultdict_creation_in_name_scope()
    print("Test passed.")
