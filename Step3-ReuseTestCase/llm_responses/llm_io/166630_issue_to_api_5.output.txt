import unittest
import logging
import tensorflow as tf

class TestVariableAggregationLogging(unittest.TestCase):
    """
    Test case to reproduce the logging error similar to PyTorch issue #166630.
    The bug occurs when a format string expects a key (e.g., 'aggregation')
    but the provided dictionary is missing that key, causing a KeyError.
    """

    def setUp(self):
        # Configure a logger to mimic the environment in the bug report
        self.logger = logging.getLogger("test_tf_logger")
        self.logger.setLevel(logging.INFO)
        # Add a handler to ensure the logging path is exercised
        if not self.logger.handlers:
            handler = logging.StreamHandler()
            self.logger.addHandler(handler)

    def test_logging_missing_aggregation_arg(self):
        """
        Reproduces the bug: Format string expects 'aggregation', but dict is missing it.
        This corresponds to 'signals_to_handle' missing in the PyTorch issue.
        """
        # Format string containing the placeholder for 'aggregation'
        log_format = (
            "Starting variable configuration:\n"
            "  name               : %(name)s\n"
            "  aggregation        : %(aggregation)s\n"
            "  dtype              : %(dtype)s\n"
        )

        # Dictionary missing the 'aggregation' key
        # This mimics the missing 'signals_to_handle' in the original bug
        config_dict = {
            "name": "my_variable",
            "dtype": "float32",
            # "aggregation": tf.compat.v1.VariableAggregation.MEAN # Missing!
        }

        # Expecting a KeyError because the format string argument is missing
        with self.assertRaises(KeyError):
            self.logger.info(log_format, config_dict)

    def test_logging_with_all_args(self):
        """
        Verifies that logging works correctly when all arguments are present.
        """
        log_format = (
            "Starting variable configuration:\n"
            "  name               : %(name)s\n"
            "  aggregation        : %(aggregation)s\n"
            "  dtype              : %(dtype)s\n"
        )

        config_dict = {
            "name": "my_variable",
            "dtype": "float32",
            "aggregation": tf.compat.v1.VariableAggregation.MEAN
        }

        # This should not raise an error
        try:
            self.logger.info(log_format, config_dict)
        except Exception as e:
            self.fail(f"Logging failed unexpectedly: {e}")

if __name__ == "__main__":
    unittest.main()
