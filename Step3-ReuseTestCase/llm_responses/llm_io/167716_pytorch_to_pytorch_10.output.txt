import torch

# Adapted test case for torch.symeig based on the torch.sparse.mm bug report.
# The original issue involved a Segmentation fault when converting the result of a 
# sparse operation to dense. Here we verify that torch.symeig produces valid 
# output without corruption, using a symmetric matrix with gradients enabled.

def test_torch_symeig():
    # Set seed for reproducibility
    torch.manual_seed(42)

    # Create a symmetric matrix (required for symeig)
    # We enable requires_grad to mimic the conditions of the original bug report
    A = torch.tensor([[1., 0, 2], [0, 3, 0], [2, 0, 1]], requires_grad=True)

    # Call the similar API: torch.symeig
    # This computes eigenvalues and eigenvectors
    eigenvalues, eigenvectors = torch.symeig(A, eigenvectors=True)

    # Verify the output is valid (no NaNs or Infs which would indicate corruption)
    assert torch.isfinite(eigenvalues).all(), "Eigenvalues contain NaN or Inf"
    assert torch.isfinite(eigenvectors).all(), "Eigenvectors contain NaN or Inf"

    # Verify the mathematical property: A = V * diag(E) * V^T
    # This ensures the operation returned correct data and not garbage memory
    diag_e = torch.diag(eigenvalues)
    reconstructed = eigenvectors @ diag_e @ eigenvectors.t()
    
    # Allow for small floating point errors
    assert torch.allclose(A, reconstructed, atol=1e-5), "Matrix reconstruction failed"

    print("Test passed: torch.symeig returned valid results.")

if __name__ == "__main__":
    test_torch_symeig()
