```python
import pytest
import torch
from torch.optim import AdamW
from omegaconf import OmegaConf

def test_adamw_betas_type_validation():
    """
    Test that AdamW validates the 'betas' parameter type.
    It should raise a TypeError if an OmegaConf ListConfig is passed
    to prevent unexpected serialization behavior (Issue 167319).
    """
    # Create a simple model
    model = torch.nn.Linear(10, 1)

    # Create an OmegaConf config containing the betas
    cfg = OmegaConf.create({
        'model': {'betas': [0.9, 0.999]},
        'data': {'batch_size': 32}
    })

    # Test 1: Passing OmegaConf ListConfig should raise TypeError
    with pytest.raises(TypeError):
        optimizer = AdamW(model.parameters(), lr=1e-3, betas=cfg.model.betas)

    # Test 2: Valid types (tuple and list) should still work
    # This ensures the fix doesn't break standard usage
    optimizer_tuple = AdamW(model.parameters(), lr=1e-3, betas=(0.9, 0.999))
    assert optimizer_tuple is not None

    optimizer_list = AdamW(model.parameters(), lr=1e-3, betas=[0.9, 0.999])
    assert optimizer_list is not None
```
