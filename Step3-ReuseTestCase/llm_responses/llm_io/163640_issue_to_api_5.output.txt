import torch
import torch.nn as nn
import unittest

class TinyEnc(nn.Module):
    def __init__(self, d_model=512, nhead=8, num_layers=1):
        super().__init__()
        layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=nhead, batch_first=True, dropout=0.1
        )
        self.enc = nn.TransformerEncoder(layer, num_layers=num_layers)
        self.proj = nn.Linear(d_model, 10)

    def forward(self, x, pad_mask):
        # Passing src_key_padding_mask triggers torch._nested_tensor_from_mask_left_aligned
        y = self.enc(x, mask=None, src_key_padding_mask=pad_mask)
        return self.proj(y)

class TestTransformerCompileWithSDP(unittest.TestCase):
    def test_compile_with_padding_mask_and_sdp_check(self):
        """
        Test that TransformerEncoder compiles with src_key_padding_mask.
        This test leverages torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed
        to check the backend state, as the bug involves internal SDPA/NestedTensor interactions.
        """
        # Leverage the similar API to check the SDP backend state
        # This relates to the math backend which might utilize the nested tensor logic
        # mentioned in the bug report.
        try:
            sdp_reduction_allowed = torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed()
        except AttributeError:
            # Handle cases where CUDA is not available or the API is missing
            sdp_reduction_allowed = False
            print("Warning: torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed not available (likely non-CUDA build).")

        torch.manual_seed(0)
        m = TinyEnc().eval()

        B, T, C = 1, 41, 512
        x = torch.randn(B, T, C, dtype=torch.float32)
        pad_mask = (torch.rand(B, T) > 0.5)
        pad_mask[..., 0] = True

        # Eager execution
        with torch.inference_mode():
            y_eager = m(x, pad_mask)
        
        self.assertIsNotNone(y_eager)
        self.assertEqual(y_eager.shape, (B, T, 10))

        # Compile execution (fullgraph=True required to reproduce the specific bug path)
        # The bug report indicates this raises torch._dynamo.exc.Unsupported
        cm = torch.compile(m, backend="inductor", fullgraph=True)
        
        try:
            with torch.inference_mode():
                y_compiled = cm(x, pad_mask)
            
            self.assertIsNotNone(y_compiled)
            self.assertEqual(y_compiled.shape, y_eager.shape)
            print(f"Test passed. SDP reduction allowed: {sdp_reduction_allowed}")
        except Exception as e:
            self.fail(f"torch.compile failed with src_key_padding_mask. Error: {e}")

if __name__ == "__main__":
    unittest.main()
