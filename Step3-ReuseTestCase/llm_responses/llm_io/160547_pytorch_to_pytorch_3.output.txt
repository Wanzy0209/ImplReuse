import torch
import torch.distributed as dist
from collections import namedtuple

def test_new_group_namedtuple(self):
    # Define a NamedTuple for the arguments of new_group
    # We select 'ranks' and 'backend' as representative arguments
    GroupArgs = namedtuple('GroupArgs', 'ranks backend')
    
    # Create input using the NamedTuple
    # Assuming a world size of at least 2 for ranks [0, 1]
    inp = GroupArgs([0, 1], 'gloo')
    
    # Check if distributed is initialized to ensure the test is runnable
    if not dist.is_initialized():
        print("torch.distributed is not initialized. Skipping test.")
        return

    # Attempt to call new_group by unpacking the NamedTuple
    # This mimics the structure of the original bug report's direct call
    try:
        group = dist.new_group(*inp)
        print(f"Success with unpacked NamedTuple: {group}")
    except Exception as e:
        print(f"Error with unpacked NamedTuple: {e}")

    # Workaround: convert to kwargs (as suggested in the original bug)
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    try:
        group = dist.new_group(**inp_kwargs)
        print(f"Success with kwargs: {group}")
    except Exception as e:
        print(f"Error with kwargs: {e}")
