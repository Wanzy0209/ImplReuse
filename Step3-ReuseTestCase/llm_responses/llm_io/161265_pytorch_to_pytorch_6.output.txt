```python
import torch

# This test verifies torch.argmin behavior on large tensors (>4GB) on MPS.
# It is adapted from a bug report where torch.full (used by torch.ones) failed
# to correctly initialize memory beyond the 4GB boundary.
# If the initialization fails (zeros appear at the end), argmin will return
# an index pointing to the end instead of 0.

if torch.backends.mps.is_available():
    # Create a tensor slightly larger than 4GB (2^32 + 10 bytes)
    # Shape: (2, 2^31 + 5), dtype: int8 (1 byte per element)
    a = torch.ones(2, (1 << 31) + 5, dtype=torch.int8, device='mps')

    # Find the index of the minimum value.
    # Since the tensor is filled with 1s, the minimum is 1 and the first index is 0.
    # If the bug is present, values at the end are 0, so argmin will return a large index.
    min_index = torch.argmin(a)

    # Assert that the minimum value is found at the start of the tensor.
    assert min_index.item() == 0, f"Expected argmin index 0, got {min_index.item()}. Memory initialization might be corrupted."
else:
    print("MPS device not available. Skipping test.")
```
