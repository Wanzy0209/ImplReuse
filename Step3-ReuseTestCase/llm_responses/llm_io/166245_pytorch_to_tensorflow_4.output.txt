import tensorflow as tf
import numpy as np

def test_tf_rot90_adapted():
    """
    Adapted test case for tf.experimental.numpy.rot90 based on PyTorch fuzzer issue.
    The original issue involved complex tensor manipulations (chunk, squeeze, index_select)
    leading to a compiler assertion. This test replicates the tensor shapes and data flow
    in TensorFlow to verify the stability of tf.experimental.numpy.rot90 with similar inputs.
    """
    
    # Set seed for reproducibility
    tf.random.set_seed(751735337)

    # --- Input Generation (mimicking PyTorch args) ---
    # arg_0: size=(15, 108, 4), dtype=int16
    arg_0 = tf.random.uniform((15, 108, 4), minval=-100, maxval=100, dtype=tf.int16)
    
    # arg_1: size=(11,), dtype=int64
    arg_1 = tf.random.uniform((11,), minval=0, maxval=100, dtype=tf.int64)
    
    # arg_2: size=(3, 27), dtype=int16
    arg_2 = tf.random.uniform((3, 27), minval=-100, maxval=100, dtype=tf.int16)
    
    # arg_3, arg_4, arg_5: size=(1, 27), dtype=int16
    arg_3 = tf.random.uniform((1, 27), minval=-100, maxval=100, dtype=tf.int16)
    arg_4 = tf.random.uniform((1, 27), minval=-100, maxval=100, dtype=tf.int16)
    arg_5 = tf.random.uniform((1, 27), minval=-100, maxval=100, dtype=tf.int16)
    
    # arg_6: size=(1,), dtype=int64
    arg_6 = tf.random.uniform((1,), minval=0, maxval=100, dtype=tf.int64)

    # --- Logic Translation ---
    
    # var_node_4 = arg_0
    var_node_4 = arg_0
    
    # var_node_3 = torch.chunk(var_node_4, 4, dim=1)[0]
    # tf.split returns a list, we take the first element
    var_node_3 = tf.split(var_node_4, 4, axis=1)[0]
    
    # var_node_2 = torch.chunk(var_node_3, 4, dim=2)[0]
    var_node_2 = tf.split(var_node_3, 4, axis=2)[0]
    
    # var_node_1 = torch.squeeze(var_node_2)
    var_node_1 = tf.squeeze(var_node_2)
    
    # var_node_8 = torch.full((13, 27), 3, dtype=torch.int16)
    var_node_8 = tf.fill((13, 27), tf.cast(3, tf.int16))
    
    # _input_size_var_node_7 = var_node_8.size(0)
    _input_size_var_node_7 = tf.shape(var_node_8)[0]
    
    # _index_var_node_7 = torch.randint(0, _input_size_var_node_7, (11,), device=var_node_8.device)
    _index_var_node_7 = tf.random.uniform((11,), minval=0, maxval=_input_size_var_node_7, dtype=tf.int64)
    
    # var_node_7 = torch.index_select(var_node_8, 0, _index_var_node_7)
    var_node_7 = tf.gather(var_node_8, _index_var_node_7, axis=0)
    
    # var_node_6 = torch.clamp(var_node_7, min=-1.0, max=1.0)
    var_node_6 = tf.clip_by_value(var_node_7, clip_value_min=-1.0, clip_value_max=1.0)
    
    # var_node_12 = arg_2
    var_node_12 = arg_2
    
    # var_node_11 = torch.clamp(var_node_12, min=-1.0, max=1.0)
    var_node_11 = tf.clip_by_value(var_node_12, clip_value_min=-1.0, clip_value_max=1.0)
    
    # var_node_13 = torch.full((1,), 3, dtype=torch.int64)
    var_node_13 = tf.fill((1,), tf.cast(3, tf.int64))
    
    # _input_size_var_node_10 = var_node_11.size(0)
    _input_size_var_node_10 = tf.shape(var_node_11)[0]
    
    # _index_var_node_10 = torch.randint(0, _input_size_var_node_10, (1,), device=var_node_11.device)
    _index_var_node_10 = tf.random.uniform((1,), minval=0, maxval=_input_size_var_node_10, dtype=tf.int64)
    
    # var_node_10 = torch.index_select(var_node_11, 0, _index_var_node_10)
    var_node_10 = tf.gather(var_node_11, _index_var_node_10, axis=0)
    
    # var_node_16 = arg_3
    # var_node_17 = arg_4
    # var_node_18 = arg_5
    # var_node_15 = torch.cat([var_node_16, var_node_17, var_node_18], dim=0)
    var_node_15 = tf.concat([arg_3, arg_4, arg_5], axis=0)
    
    # var_node_20 = arg_6
    # var_node_19 = torch.clamp(var_node_20, min=None, max=1.0)
    # PyTorch clamp with min=None is equivalent to tf.minimum
    var_node_19 = tf.minimum(arg_6, tf.cast(1.0, tf.int64))
    
    # _input_size_var_node_14 = var_node_15.size(0)
    _input_size_var_node_14 = tf.shape(var_node_15)[0]
    
    # _index_var_node_14 = torch.randint(0, _input_size_var_node_14, (1,), device=var_node_15.device)
    _index_var_node_14 = tf.random.uniform((1,), minval=0, maxval=_input_size_var_node_14, dtype=tf.int64)
    
    # var_node_14 = torch.index_select(var_node_15, 0, _index_var_node_14)
    var_node_14 = tf.gather(var_node_15, _index_var_node_14, axis=0)
    
    # var_node_22 = torch.full((4, 27), 3, dtype=torch.int16)
    var_node_22 = tf.fill((4, 27), tf.cast(3, tf.int16))
    
    # var_node_24 = torch.full((4,), 3, dtype=torch.int64)
    var_node_24 = tf.fill((4,), tf.cast(3, tf.int64))
    
    # var_node_25 = torch.full((2,), 3, dtype=torch.int64)
    var_node_25 = tf.fill((2,), tf.cast(3, tf.int64))

    # --- Target API Verification: tf.experimental.numpy.rot90 ---
    
    # Test 1: Apply rot90 to the 3D tensor (var_node_4)
    # Original shape: (15, 108, 4). Expected rotation on last two dims: (15, 4, 108)
    try:
        result_3d = tf.experimental.numpy.rot90(var_node_4, k=1)
        assert result_3d.shape == (15, 4, 108), f"Expected shape (15, 4, 108), got {result_3d.shape}"
        print("Test 1 Passed: tf.experimental.numpy.rot90 on 3D tensor.")
    except Exception as e:
        print(f"Test 1 Failed: {e}")

    # Test 2: Apply rot90 to the 2D tensor (var_node_1)
    # Original shape: (15, 27). Expected rotation: (27, 15)
    try:
        result_2d = tf.experimental.numpy.rot90(var_node_1, k=1)
        assert result_2d.shape == (27, 15), f"Expected shape (27, 15), got {result_2d.shape}"
        print("Test 2 Passed: tf.experimental.numpy.rot90 on 2D tensor.")
    except Exception as e:
        print(f"Test 2 Failed: {e}")

    # Test 3: Apply rot90 to a tensor created via index_select (var_node_7)
    # Original shape: (11, 27). Expected rotation: (27, 11)
    try:
        result_indexed = tf.experimental.numpy.rot90(var_node_7, k=1)
        assert result_indexed.shape == (27, 11), f"Expected shape (27, 11), got {result_indexed.shape}"
        print("Test 3 Passed: tf.experimental.numpy.rot90 on indexed tensor.")
    except Exception as e:
        print(f"Test 3 Failed: {e}")

if __name__ == "__main__":
    test_tf_rot90_adapted()
