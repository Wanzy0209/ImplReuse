```python
import tensorflow as tf
import numpy as np

# Set seeds for reproducibility
tf.random.set_seed(0)
np.random.seed(0)

# Original PyTorch input: (4, 6, 7) interpreted as (Channels, Height, Width)
# TensorFlow fractional_max_pool expects (Batch, Height, Width, Channels)
# We adapt the input to (1, 6, 7, 4)
x = np.random.randn(1, 6, 7, 4).astype(np.float32)

# Original PyTorch parameters:
# kernel_size=[1, 6], stride=[4, 9], ceil_mode=True
# This results in an output shape of (Channels, 2, 1) -> (4, 2, 1)
# Height: 6 -> 2 (Ratio 3.0)
# Width: 7 -> 1 (Ratio 7.0)
pooling_ratio = [1.0, 3.0, 7.0, 1.0]

# Execute the similar API
# deterministic=True is used to mimic the fixed seed behavior of the original test
output, row_seq, col_seq = tf.nn.fractional_max_pool(
    value=x,
    pooling_ratio=pooling_ratio,
    deterministic=True
)

# Verify the output shape matches the expected dimensions derived from the original bug report
# Expected shape: (Batch, H_out, W_out, Channels) = (1, 2, 1, 4)
expected_shape = (1, 2, 1, 4)

if output.shape != expected_shape:
    print(f"Shape mismatch! Expected: {expected_shape}, Got: {output.shape}")
else:
    print("Test passed. Output shape matches expected dimensions.")
    print("Output:", output.numpy())
```
