import torch
import sys

def test_fmod_int64_min_div_neg_one():
    """
    Test case for Issue 165649.
    Verifies that torch.fmod does not crash (segfault) when calculating
    the remainder of INT64_MIN divided by -1.
    """
    # Setup tensors based on the bug report
    dividend = torch.full((2, 3), torch.iinfo(torch.int64).min, dtype=torch.int64, device='cpu')
    divisor = torch.full((3,), -1, dtype=torch.int64, device='cpu')

    try:
        # This call previously caused a segfault / arithmetic exception
        result = torch.fmod(dividend, divisor)
    except Exception as e:
        print(f"Test FAILED: Exception occurred during torch.fmod call: {e}")
        sys.exit(1)

    # Verify the result. Mathematically, x % -1 is 0.
    # Even with INT64_MIN, the remainder should be 0.
    expected = torch.zeros((2, 3), dtype=torch.int64, device='cpu')
    
    if not torch.equal(result, expected):
        print(f"Test FAILED: Expected result to be 0, but got:\n{result}")
        sys.exit(1)

    print("Test PASSED: torch.fmod handled INT64_MIN / -1 correctly without crashing.")

if __name__ == "__main__":
    test_fmod_int64_min_div_neg_one()
