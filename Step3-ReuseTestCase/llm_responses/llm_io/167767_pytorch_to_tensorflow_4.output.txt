import tensorflow as tf
import numpy as np

def test_random_normal_initialization():
    # Test 1: Basic initialization with mean=0.0 (analogous to min=0.0 in the bug report)
    # The original bug involved a clamp operation that failed to update values.
    # Here we verify that the initializer correctly respects the mean parameter.
    initializer = tf.keras.initializers.RandomNormal(mean=0.0, stddev=1.0)
    values = initializer(shape=(1000,))
    print(f"Mean with mean=0.0: {np.mean(values)}")
    # Assert that the mean is statistically close to 0.0
    assert abs(np.mean(values) - 0.0) < 0.1, "Mean is not close to 0.0"

    # Test 2: Initialization with mean=1e-7 (analogous to min=1e-7)
    # Verifying if the API handles small float values correctly.
    initializer = tf.keras.initializers.RandomNormal(mean=1e-7, stddev=1.0)
    values = initializer(shape=(1000,))
    print(f"Mean with mean=1e-7: {np.mean(values)}")
    # Assert that the mean is statistically close to 1e-7
    assert abs(np.mean(values) - 1e-7) < 0.1, "Mean is not close to 1e-7"

    # Test 3: Dtype handling (analogous to device handling in the original test)
    # Ensuring the initializer respects the dtype argument.
    initializer = tf.keras.initializers.RandomNormal(mean=0.0, stddev=1.0, dtype=tf.float32)
    values = initializer(shape=(1,))
    print(f"Dtype: {values.dtype}")
    assert values.dtype == tf.float32, "Dtype mismatch"

    print("All tests passed.")

if __name__ == "__main__":
    test_random_normal_initialization()
