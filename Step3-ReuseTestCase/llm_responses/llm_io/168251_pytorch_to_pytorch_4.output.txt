import torch
import torch.nn as nn
import torch.nn.functional as F
from torch._dynamo.polyfills.builtins import all as dynamo_all

class VAE(nn.Module):

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super(VAE, self).__init__()
        self.input_dim = input_dim
        self.latent_dim = latent_dim
        self.encoder = nn.Sequential(nn.Linear(input_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim), nn.ReLU())
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_var = nn.Linear(hidden_dim, latent_dim)
        self.decoder = nn.Sequential(nn.Linear(latent_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim), nn.Sigmoid())

    def encode(self, x):
        h = self.encoder(x)
        mu = self.fc_mu(h)
        log_var = self.fc_var(h)
        return (mu, log_var)

    def reparameterize(self, mu, log_var):
        std = torch.exp(0.5 * log_var)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        return self.decoder(z)

    def forward(self, x):
        x = x.view(-1, self.input_dim)
        (mu, log_var) = self.encode(x)
        z = self.reparameterize(mu, log_var)
        reconstruction = self.decode(z)
        return (reconstruction, mu, log_var)

def get_default_model():
    input_dim = 784
    hidden_dim = 400
    latent_dim = 20
    model = VAE(input_dim=input_dim, hidden_dim=hidden_dim, latent_dim=latent_dim)
    return model

def get_sample_inputs():
    batch_size = 32
    input_dim = 784
    x = torch.randn(batch_size, input_dim)
    return (x,)

def main():
    model = get_default_model()
    model.eval()
    inputs = get_sample_inputs()
    
    # Test eager mode
    with torch.no_grad():
        output_eager = model(*inputs)
    
    # Verify eager output is a tuple
    assert isinstance(output_eager, tuple)
    print(f'Eager output type: {type(output_eager)}')

    # Test compiled mode
    compiled_model = torch.compile(model)
    with torch.no_grad():
        output_compile = compiled_model(*inputs)
    
    # Verify compiled output is a tuple
    assert isinstance(output_compile, tuple)
    print(f'Compiled output type: {type(output_compile)}')

    # Adapt the test case to use the similar API: torch.all (via its Dynamo polyfill)
    # The original bug report showed that accessing .shape on the tuple failed.
    # Here we verify that torch.all (polyfill) can handle the tuple of tensors correctly.
    
    # Test torch.all polyfill on the tuple output from eager mode
    result_eager = dynamo_all(output_eager)
    print(f'Eager torch.all (polyfill) result: {result_eager}')
    assert isinstance(result_eager, bool)

    # Test torch.all polyfill on the tuple output from compiled mode
    result_compile = dynamo_all(output_compile)
    print(f'Compiled torch.all (polyfill) result: {result_compile}')
    assert isinstance(result_compile, bool)
    
    print('Test passed successfully!')

if __name__ == '__main__':
    main()
