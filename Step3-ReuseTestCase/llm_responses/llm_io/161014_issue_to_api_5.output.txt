import tensorflow as tf
import numpy as np

def test_multiply_edge_cases():
    """
    Test case for tf.keras.layers.multiply inspired by the inconsistent behavior
    of torch.ops.aten.constant_pad_nd with negative padding.
    
    The original issue highlights edge cases where operations result in zero-sized 
    dimensions or involve negative parameters. This test checks if 
    tf.keras.layers.multiply handles similar edge cases (negative values, 
    zero-sized dimensions) consistently.
    """
    
    # 1. Standard Case: Basic element-wise multiplication
    # Mirrors the standard usage of constant_pad_nd
    x1 = tf.constant([1.0, 2.0, 3.0])
    x2 = tf.constant([4.0, 5.0, 6.0])
    result = tf.keras.layers.multiply([x1, x2])
    assert result.shape == (3,)
    assert np.allclose(result.numpy(), [4.0, 10.0, 18.0])

    # 2. Negative Values Case
    # Mirrors the "negative padding" aspect of the bug report.
    # We verify that negative values in the tensors are handled correctly.
    y1 = tf.constant([1.0, -2.0, 3.0])
    y2 = tf.constant([-1.0, 2.0, -3.0])
    result_neg = tf.keras.layers.multiply([y1, y2])
    assert result_neg.shape == (3,)
    assert np.allclose(result_neg.numpy(), [-1.0, -4.0, -9.0])

    # 3. Zero-Sized Dimension Case
    # Mirrors the torch.Size([5, 0]) result in the bug report.
    # We verify that multiplying tensors with a dimension of size 0 works correctly.
    z1 = tf.ones([5, 0])
    z2 = tf.ones([5, 0])
    result_zero = tf.keras.layers.multiply([z1, z2])
    assert result_zero.shape == (5, 0)
    # Ensure the operation doesn't crash and returns an empty tensor of correct shape

    # 4. Broadcasting Edge Case
    # Mirrors the shape manipulation logic. Ensuring broadcasting rules are consistent.
    a = tf.ones([3, 1])
    b = tf.ones([1, 3])
    result_broadcast = tf.keras.layers.multiply([a, b])
    assert result_broadcast.shape == (3, 3)
    assert np.allclose(result_broadcast.numpy(), np.ones((3, 3)))

if __name__ == "__main__":
    test_multiply_edge_cases()
    print("Test passed.")
