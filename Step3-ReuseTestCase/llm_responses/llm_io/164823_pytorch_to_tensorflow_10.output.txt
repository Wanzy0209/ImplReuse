import tensorflow as tf

# Disable eager execution to properly test v1 queue-based APIs like range_input_producer
tf.compat.v1.disable_eager_execution()

def test_range_input_producer():
    """
    Adapted test case for tf.compat.v1.train.range_input_producer.
    
    The original PyTorch test case verified the behavior of torch.compile 
    with sparse/dense tensor conversions. Since range_input_producer is a 
    data input pipeline utility that generates integer ranges, we adapt the 
    test to verify its core functionality: producing a sequence of integers 
    and retrieving them via a queue.
    """
    limit = 10
    num_epochs = 1

    # Define the graph operations
    # range_input_producer creates a queue of integers from 0 to limit-1
    queue = tf.compat.v1.train.range_input_producer(
        limit=limit, 
        num_epochs=num_epochs, 
        shuffle=False,
        capacity=32
    )

    # Dequeue an element from the queue
    dequeue_op = queue.dequeue()

    with tf.compat.v1.Session() as sess:
        # Initialize local variables (required for the epochs counter)
        sess.run(tf.compat.v1.local_variables_initializer())
        sess.run(tf.compat.v1.global_variables_initializer())

        # Coordinator and QueueRunners are required to start the queue threads
        coord = tf.compat.v1.train.Coordinator()
        threads = tf.compat.v1.train.start_queue_runners(sess=sess, coord=coord)

        results = []
        try:
            # Run the dequeue op 'limit' times to retrieve the full range
            for _ in range(limit):
                val = sess.run(dequeue_op)
                results.append(val)
        except Exception as e:
            print(f"Error during execution: {e}")
            raise
        finally:
            # Stop the queue threads
            coord.request_stop()
            coord.join(threads)

    print("Output:", results)
    
    # Assertions to verify correct behavior
    assert len(results) == limit, f"Expected {limit} results, got {len(results)}"
    assert sorted(results) == list(range(limit)), "Output does not match expected range"
    print("Test passed.")

if __name__ == "__main__":
    test_range_input_producer()
