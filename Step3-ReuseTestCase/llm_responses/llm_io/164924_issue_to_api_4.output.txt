```python
import tensorflow as tf
import tempfile
import os

class GraphSummary(tf.Module):
    def __init__(self):
        super().__init__()
        # Create a simple graph to use as input data
        @tf.function
        def simple_func(x):
            return x * 2
        self.graph_data = simple_func.get_concrete_function(tf.TensorSpec([], tf.float32)).graph

    def forward(self):
        # Use a temporary directory for logs
        logdir = os.path.join(tempfile.gettempdir(), "tf_logs")
        with tf.summary.create_file_writer(logdir).as_default():
            # Call the similar API: tf.summary.graph
            # This mimics the structure of calling torch.isin in the original issue
            tf.summary.graph(self.graph_data)
        return {'out': True}

# Test Eager
model = GraphSummary()
print("Eager:", model.forward())

# Test Compiled (tf.function is the equivalent of torch.compile)
# The original bug was that torch.isin failed in torch.compile (inductor).
# Here we test if tf.summary.graph fails in tf.function.
compiled_model = tf.function(model.forward)
print("Compiled:", compiled_model())

# Assertions to ensure behavior is consistent
eager_result = model.forward()
compiled_result = compiled_model()
assert eager_result == compiled_result
```
