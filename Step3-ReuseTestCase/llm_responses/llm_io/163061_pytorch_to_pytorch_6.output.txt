import torch

def lobpcg_op(A: torch.Tensor, k: int = 10):
    """
    Computes the k largest eigenvalues and corresponding eigenvectors
    of a symmetric positive definite matrix A.
    """
    return torch.lobpcg(A, k=k)

def main():
    # Setup data for lobpcg
    # Create a symmetric positive definite matrix
    # Using float64 for better numerical stability in lobpcg
    N = 1024
    A = torch.randn(N, N, device='cuda', dtype=torch.float64)
    A = A @ A.T

    # Run the operation multiple times to observe GIL behavior
    # (e.g., using py-spy or similar tools)
    for _ in range(10):
        eigenvalues, eigenvectors = lobpcg_op(A, k=10)

if __name__ == "__main__":
    main()
