import torch
import unittest

class TestLobpcgCompile(unittest.TestCase):
    def test_lobpcg_with_compile(self):
        """
        Test that torch.lobpcg works correctly with torch.compile.
        This verifies the fix for missing float handling in rebind_unbacked
        which affected AOTInductor compilation.
        """
        # Create a symmetric positive definite matrix required by lobpcg
        A = torch.randn(10, 10, dtype=torch.float64)
        A = A @ A.T + torch.eye(10) * 0.1

        # Define a function utilizing the torch.lobpcg API
        def eigen_solver(A):
            # Compute the k largest eigenvalues and corresponding eigenvectors
            eigenvalues, eigenvectors = torch.lobpcg(A, k=2)
            return eigenvalues, eigenvectors

        # Compile the function using torch.compile (which triggers AOTInductor)
        # This path exercises the symbolic_shapes logic where the bug occurred.
        compiled_solver = torch.compile(eigen_solver)

        # Execute the compiled function
        evals, evecs = compiled_solver(A)

        # Basic assertions to verify correctness and execution
        self.assertEqual(evals.shape, (2, 1))
        self.assertEqual(evecs.shape, (10, 2))
        
        # Verify eigenvalues are real (since A is symmetric)
        self.assertTrue(torch.all(torch.isreal(evals)))

if __name__ == '__main__':
    unittest.main()
