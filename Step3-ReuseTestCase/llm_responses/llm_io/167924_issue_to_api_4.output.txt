import tensorflow as tf
import numpy as np

def test_tf_exp_with_sliced_tensor():
    """
    Test case adapted from Issue 167924 (PyTorch MPS crash with repeat_interleave).
    This test verifies that tf.experimental.numpy.exp handles sliced tensors correctly,
    preserving the reproduction logic of creating a tensor, slicing it to a non-prefix,
    and applying an operation.
    """
    # Reproduce the tensor creation logic from the bug report
    # Original: counts = torch.tensor([0, 1, 0], device="mps")
    # Note: exp requires floating point inputs, so we use float32.
    counts = tf.constant([0.0, 1.0, 0.0], dtype=tf.float32)

    # Reproduce the slicing logic that caused the crash in the original issue
    # Original: counts[1:3]
    sliced_counts = counts[1:3]

    # Apply the similar API: tf.experimental.numpy.exp
    # This checks if the API handles the non-contiguous memory layout of the slice correctly.
    result = tf.experimental.numpy.exp(sliced_counts)

    # Assertions to ensure correctness
    # exp(1.0) is approx 2.71828, exp(0.0) is 1.0
    expected_values = np.exp([1.0, 0.0])
    np.testing.assert_allclose(result.numpy(), expected_values)
    print("Test passed: tf.experimental.numpy.exp handled sliced tensor correctly.")

if __name__ == "__main__":
    test_tf_exp_with_sliced_tensor()
