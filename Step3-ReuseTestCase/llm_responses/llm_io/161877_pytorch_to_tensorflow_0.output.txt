```python
import tensorflow as tf

# Adapted test case for tf.signal.kaiser_bessel_derived_window
# based on PyTorch Conv1d issue with extreme padding values.

# The original bug was triggered by passing an extremely large integer (9223372036854775803)
# to the 'padding' parameter, causing a memory allocation crash (realloc(): invalid pointer).
# We adapt this logic by passing the same extreme value to the 'window_length' parameter
# of the TensorFlow API to verify its behavior under similar memory stress conditions.

def test_extreme_window_length():
    extreme_length = 9223372036854775803

    try:
        # Attempt to generate a window with an impossibly large length.
        # This attempts to allocate a tensor of size ~9 quintillion elements.
        output = tf.signal.kaiser_bessel_derived_window(window_length=extreme_length)
        
        # If execution reaches here, the API did not crash but might have handled it internally
        # or the environment has infinite memory (unlikely).
        print(f"API returned output. Shape: {output.shape}")
        
    except Exception as e:
        # We expect a clean exception (e.g., ResourceExhaustedError, InvalidArgumentError)
        # rather than a process abort (Aborted / Segfault).
        print(f"Caught expected exception: {type(e).__name__}")
        print(f"Message: {e}")

if __name__ == "__main__":
    test_extreme_window_length()
```
