import torch
import unittest

class TestTorchComplexCompile(unittest.TestCase):
    """
    Test case for Issue 160882: torch.compile crashes when torch.ops.aten.complex.default
    gets different inputs than when it was compiled.
    
    This test preserves the original bug reproduction logic while adapting the structure
    to be a formal unit test.
    """

    def test_complex_with_dynamic_shapes(self):
        """
        Verifies that torch.compile handles torch.complex correctly when
        input shapes change between calls.
        """
        # Define the function to be compiled, leveraging type hints and clear structure
        # similar to the provided similar API example.
        def f(real: torch.Tensor, imag: torch.Tensor) -> torch.Tensor:
            z = torch.complex(real, imag)
            return torch.fft.irfft(z, dim=1)

        B, F, T = 1, 641, 39

        # Setup source tensors
        r_src = torch.randn(B, F, T)
        i_src = torch.randn(B, F, T)
        
        # Setup tensors with permuted dimensions to simulate dynamic shape change
        r_mismatch = r_src.permute(0, 2, 1)
        i_mismatch = i_src.permute(0, 2, 1)

        # Compile with fullgraph=True to trigger the specific code path
        compiled = torch.compile(f, fullgraph=True)

        # First call with original shapes
        _ = compiled(r_src, i_src)
        
        # Second call with mismatched shapes
        # In the buggy version, this raises an AssertionError.
        # We assert that the operation completes successfully.
        try:
            _ = compiled(r_mismatch, i_mismatch)
        except AssertionError:
            self.fail("torch.compile crashed with AssertionError on dynamic shape input (Issue 160882)")

if __name__ == "__main__":
    unittest.main()
