```python
import tensorflow as tf
import time

# Set seed for reproducibility, mimicking the original PyTorch script
tf.random.set_seed(0)

# Shapes from the original bug report to define variable sizes
shapes = [
    ((1, 12, 10, 64), (1, 12, 64, 10)),
    ((1, 12, 10, 10), (1, 12, 10, 64)),
]

def benchmark_initializer(a_shape, b_shape, repeat=500):
    """
    Benchmarks the execution time of tf.compat.v1.local_variables_initializer.
    This mimics the structure of the original torch.matmul benchmark.
    """
    # Create local variables to provide context for the initializer,
    # analogous to creating tensors A and B in the original issue.
    var_a = tf.Variable(tf.random.uniform(a_shape, minval=-1, maxval=1))
    var_b = tf.Variable(tf.random.uniform(b_shape, minval=-1, maxval=1))

    # Get the initializer op
    init_op = tf.compat.v1.local_variables_initializer()

    # Warm up
    for _ in range(5000):
        # In TF 2.x eager mode, this is a no-op, but we execute it to mimic 
        # the original benchmark's warmup loop structure.
        init_op()

    # Run benchmark
    times = []
    for i in range(repeat):
        start = time.time()
        init_op()
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
    
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    print("Benchmarking tf.compat.v1.local_variables_initializer...")
    for a_shape, b_shape in shapes:
        t = benchmark_initializer(a_shape, b_shape)
        print(f"{a_shape} x {b_shape} -> {t:.3f} us")
        
        # Assertion to detect performance regression.
        # Since local_variables_initializer is a no-op in eager mode (TF2 default),
        # it should be extremely fast. A significant increase would indicate a regression.
        assert t < 100, f"Performance regression detected: {t:.3f} us (expected < 100 us)"
```
