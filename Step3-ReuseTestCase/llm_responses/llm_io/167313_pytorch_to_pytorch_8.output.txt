```python
import torch

# Check for CUDA availability to match the original test environment
if not torch.cuda.is_available():
    print("CUDA not available, skipping test.")
    exit()

# Create a boolean tensor to trigger the specific decomposition path
# mentioned in the context (amax -> any for bool dtype)
x = torch.tensor([[True, False, True], [False, False, True]], device="cuda")

# Define a function using torch.amax with specific parameters.
# We use keepdim=True (non-default) to verify that this parameter is not ignored
# during the compilation/decomposition process, similar to alpha/beta in addmm.
f = lambda x: torch.amax(x, dim=1, keepdim=True)

# Compile the function
fc = torch.compile(f)

# Execute eager and compiled versions
res_eager = f(x)
res_compiled = fc(x)

# Verify that the results match
assert torch.equal(res_eager, res_compiled), f"Values mismatch:\nEager:\n{res_eager}\nCompiled:\n{res_compiled}"

# Verify that the shape is preserved (keepdim=True should keep the dimension)
assert res_eager.shape == res_compiled.shape, f"Shape mismatch:\nEager shape: {res_eager.shape}\nCompiled shape: {res_compiled.shape}"

print("Test passed.")
```
