import tensorflow as tf

def test_linear_activation_preserves_tensor_properties():
    """
    Test case adapted from the PyTorch all_gather memory ordering bug (Issue 163483).
    
    The original bug demonstrates that an operation (all_gather) failed to preserve
    the memory format (channels_last) of the input tensor. 
    
    This test applies the same logic to the similar API, tf.keras.activations.linear.
    Since linear is an identity function, it is expected to perfectly preserve the
    input tensor's properties (values, shape, and implicitly, layout).
    """
    
    # Setup: Create a tensor with a specific shape.
    # In the PyTorch bug, the tensor was reshaped to (2, 2, 2, 2) and set to channels_last.
    # In TensorFlow, the default memory layout for 4D tensors is NHWC (channels_last).
    x = tf.reshape(tf.range(16, dtype=tf.float32), (2, 2, 2, 2))

    # Operation: Apply the linear activation function
    y = tf.keras.activations.linear(x)

    # Verification:
    # 1. Check if values are equal (analogous to torch.equal in the bug report)
    # 2. Check if shape is preserved (analogous to memory format preservation)
    # 3. Check if dtype is preserved
    
    values_equal = tf.reduce_all(tf.equal(x, y)).numpy()
    shape_equal = (x.shape == y.shape)
    dtype_equal = (x.dtype == y.dtype)

    assert values_equal, f"Linear activation changed tensor values. Input: {x}, Output: {y}"
    assert shape_equal, f"Linear activation changed tensor shape. Input: {x.shape}, Output: {y.shape}"
    assert dtype_equal, f"Linear activation changed tensor dtype. Input: {x.dtype}, Output: {y.dtype}"

    print("Test passed: tf.keras.activations.linear preserves tensor properties.")

if __name__ == "__main__":
    test_linear_activation_preserves_tensor_properties()
