```python
import tensorflow as tf
from typing import NamedTuple

# Define the NamedTuple structure similar to the original PyTorch bug report
class MyNamedTuple(NamedTuple):
    first: tf.Tensor
    second: tf.Tensor

class MyNamedTupleSubclass(MyNamedTuple):
    pass

def fn(tup: MyNamedTuple) -> MyNamedTuple:
    # Use tf.constant to mimic torch.tensor
    extra_info = tf.constant(4.0)
    tup.extra_info = extra_info  # Add dynamic attribute
    return tup

print("\nTesting NamedTuple with __setattr__ (Baseline):")
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))
setattr_result = fn(extended_tup)
# Verify baseline behavior
try:
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
except AttributeError as e:
    print(f"Baseline Error: {e}")

print("\nTesting NamedTuple with __setattr__ inside tf.compat.v1.name_scope:")
extended_tup = MyNamedTupleSubclass(first=tf.constant([2.0]), second=tf.constant(1.0))

# Adapt the test to use the similar API: tf.compat.v1.name_scope
# This checks if the context manager interferes with dynamic attributes
with tf.compat.v1.name_scope("test_scope"):
    setattr_result = fn(extended_tup)

# Verify behavior after exiting the scope
try:
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
    print("Test Passed: Attribute persisted inside name_scope.")
except AttributeError as e:
    print(f"Test Failed: {e}")
```
