import tensorflow as tf

def test_floatx_state_caching():
    """
    Test case for tf.keras.backend.floatx based on the logic of
    intrusive caching (Issue 162630).

    The issue describes a mechanism to cache a converted object (DLPack)
    inside the tensor to avoid repeated overhead. The similar API,
    tf.keras.backend.floatx, acts as an accessor to a cached global
    configuration.

    This test verifies that the 'cached' value is correctly retrieved
    and that updates to the underlying state are reflected in subsequent
    accesses, mirroring the consistency required in the caching mechanism.
    """
    # 1. Retrieve the initial cached value.
    # Corresponds to: cached = self->cached_dl_managed_tensor_.load(...)
    initial_floatx = tf.keras.backend.floatx()
    assert isinstance(initial_floatx, str), "floatx should return a string"
    assert initial_floatx in ['float16', 'float32', 'float64'], \
        f"Unexpected default floatx: {initial_floatx}"

    # 2. Update the underlying state.
    # Corresponds to: creating/populating the DLManagedTensorVersioned
    # and setting it via atomic exchange.
    tf.keras.backend.set_floatx('float16')

    # 3. Retrieve the value again to verify the state was updated/correctly cached.
    # Corresponds to: returning the cached pointer.
    updated_floatx = tf.keras.backend.floatx()
    assert updated_floatx == 'float16', \
        f"Expected 'float16', but got '{updated_floatx}'"

    # 4. Restore the original state to avoid side effects.
    tf.keras.backend.set_floatx(initial_floatx)

    # 5. Verify restoration.
    restored_floatx = tf.keras.backend.floatx()
    assert restored_floatx == initial_floatx, \
        f"Failed to restore floatx to '{initial_floatx}'"
