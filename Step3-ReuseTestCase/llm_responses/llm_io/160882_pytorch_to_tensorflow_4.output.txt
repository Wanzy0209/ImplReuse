import tensorflow as tf

def f(low: tf.Tensor, high: tf.Tensor) -> tf.Tensor:
    """
    Function using the target API: tf.experimental.numpy.random.uniform.
    This operation broadcasts the 'low' and 'high' tensors to determine the output shape.
    """
    return tf.experimental.numpy.random.uniform(low, high)

# Dimensions from the original bug report
B, F, T = 1, 641, 39

# Create source tensors with Shape A: (B, F, T)
low_src = tf.random.uniform((B, F, T))
high_src = tf.random.uniform((B, F, T))

# Create mismatched tensors with Shape B: (B, T, F) - permuted
# This simulates the scenario where inputs differ from the compilation phase
low_mismatch = tf.transpose(low_src, [0, 2, 1])
high_mismatch = tf.transpose(high_src, [0, 2, 1])

# Compile the function using tf.function (TensorFlow's equivalent to torch.compile)
compiled_f = tf.function(f)

# First call: Tracing/Compilation with Shape A
# This establishes the initial graph
try:
    _ = compiled_f(low_src, high_src)
    print("First execution (Shape A) successful.")
except Exception as e:
    print(f"First execution failed: {e}")

# Second call: Execution/Re-tracing with Shape B
# This tests if the compiled graph handles the shape change correctly or crashes
try:
    _ = compiled_f(low_mismatch, high_mismatch)
    print("Second execution (Shape B) successful.")
except Exception as e:
    print(f"Second execution failed: {e}")
