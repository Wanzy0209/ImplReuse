```python
import torch

def func():
    # Create a tensor where torch.any will return False
    a = torch.tensor([-1.0, -2.0], device="cuda")
    # Check if any element is greater than 0
    result = torch.any(a > 0)
    assert result, "should throw"
    torch.cuda.synchronize()
    print("should not run")


def test_fn():
    torch._dynamo.reset()
    f_c = torch.compile(func, backend="aot_eager")
    f_c()

if __name__ == "__main__":
    test_fn()
```
