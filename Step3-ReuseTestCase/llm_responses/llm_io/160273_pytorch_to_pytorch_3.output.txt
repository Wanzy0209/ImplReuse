```python
import torch

# Test torch.cumsum gradient behavior
# Adapted from torch.min gradient test (Issue 160273)

# torch.cumsum(input, dim=...) -> cumulative sum over dimension
# Note: torch.cumsum does not reduce to a scalar, so we sum the output to test backward pass.
a = torch.ones([5]).cuda()
a.requires_grad = True
cum_val = torch.cumsum(a, dim=0)
# Summing the result to simulate a scalar reduction backward pass
cum_val.sum().backward()
# Gradient of sum(cumsum(ones)) is [5, 4, 3, 2, 1]
expected_grad_1 = torch.tensor([5., 4., 3., 2., 1.], device='cuda:0')
assert torch.allclose(a.grad, expected_grad_1), f"Test 1 Failed: Expected {expected_grad_1}, got {a.grad}"

# torch.cumsum(input, dim=...) -> cumulative sum over specified dimension
a = torch.ones([5]).cuda()
a.requires_grad = True
cum_val = torch.cumsum(a, dim=0)
# Backward with a vector of ones (gradient of the output)
cum_val.backward(torch.ones_like(cum_val))
# Gradient is [5, 4, 3, 2, 1] as each element contributes to all subsequent sums
expected_grad_2 = torch.tensor([5., 4., 3., 2., 1.], device='cuda:0')
assert torch.allclose(a.grad, expected_grad_2), f"Test 2 Failed: Expected {expected_grad_2}, got {a.grad}"
```
