import torch
import unittest

class TestCudaArchList(unittest.TestCase):
    """
    Test case for torch.cuda.get_arch_list based on Issue 160575.
    Verifies that the API correctly reports the supported CUDA architectures,
    specifically checking for the absence of sm_52 (Maxwell) and presence of
    architectures claimed in the warning message (sm_61, etc.).
    """

    def test_get_arch_list_reflects_build_capabilities(self):
        """
        Test that torch.cuda.get_arch_list returns the list of supported
        architectures. This mirrors the logic of checking build capabilities
        (similar to tf.test.is_built_with_rocm) but for specific compute capabilities.
        """
        if not torch.cuda.is_available():
            self.skipTest("CUDA is not available")

        # Get the list of supported architectures
        arch_list = torch.cuda.get_arch_list()

        # The bug report warning states:
        # "The current PyTorch install supports CUDA capabilities sm_61 sm_70 sm_75 sm_80 sm_86 sm_90."
        # We verify that these architectures are present in the list returned by the API.
        expected_archs_in_warning = ["sm_61", "sm_70", "sm_75", "sm_80", "sm_86", "sm_90"]

        for arch in expected_archs_in_warning:
            self.assertIn(arch, arch_list,
                          f"Expected architecture {arch} to be in get_arch_list() based on the warning message.")

        # The bug report highlights that sm_52 (Maxwell) is missing from the build.
        # We assert that sm_52 is NOT in the list to reflect the reported bug state.
        # If the bug is fixed and support is restored, this assertion should be updated.
        self.assertNotIn("sm_52", arch_list,
                          "sm_52 (Maxwell) is reported as missing in the bug issue and should not be in the list.")

if __name__ == '__main__':
    unittest.main()
