import torch
import torch.nn as nn

def test_torch_any_compile_privateuse1():
    """
    Test case to verify torch.any works with torch.compile on PrivateUse1.
    Adapted from Issue 160909 to test the torch.any API.
    """
    
    # Define a model that uses torch.any
    class AnyModel(nn.Module):
        def forward(self, x):
            # Using torch.any as the target API
            return torch.any(x)

    # Mock backend function as described in the issue
    def my_backend(gm, sample_inputs):
        # In a real scenario, this would contain custom compilation logic.
        # Returning gm.forward is sufficient to test the tracing behavior.
        return gm.forward

    # Check if PrivateUse1 is available to ensure the test is runnable 
    # in environments without a custom backend registered.
    try:
        torch.empty(1, device="PrivateUse1")
    except RuntimeError as e:
        if "PrivateUse1 backend is not registered" in str(e):
            print("Skipping test: PrivateUse1 backend is not registered.")
            return
        raise e

    # Setup model and data
    model = AnyModel().to("PrivateUse1")
    compiled_model = torch.compile(model, backend=my_backend)
    
    # Create data on PrivateUse1
    # Using boolean data for torch.any
    data = torch.tensor([True, False, True]).to("PrivateUse1")

    # Execute the compiled model
    # The bug report indicates errors occur here due to meta tensors 
    # being passed to PrivateUse1 ops during tracing.
    try:
        result = compiled_model(data)
        
        # Assertions to verify correctness
        assert isinstance(result, torch.Tensor)
        assert result.item() == True
        print("Test passed: torch.any compiled successfully on PrivateUse1.")
        
    except RuntimeError as e:
        # Check for the specific error mentioned in the bug report
        if "meta" in str(e):
            print(f"Bug reproduced: {e}")
        else:
            raise e

if __name__ == "__main__":
    test_torch_any_compile_privateuse1()
