import tensorflow as tf

def test_reshape_gradient_behavior():
    """
    Test case to verify the gradient behavior of tf.reshape.
    This test adapts the gradient checking pattern from the torch.min issue
    (Issue ID: 160273) to the tf.reshape API.
    """
    # Setup: Create a tensor
    # Mimics: a = torch.ones([5]).cuda()
    a = tf.ones([5])

    with tf.GradientTape() as tape:
        # Enable gradient tracking
        # Mimics: a.requires_grad=True
        tape.watch(a)

        # Operation: Reshape the tensor
        # Mimics: min_val = torch.min(a)
        # Note: torch.min reduces to a scalar. tf.reshape changes shape.
        # To test gradients, we reduce the reshaped tensor to a scalar.
        reshaped_val = tf.reshape(a, [1, 5])
        scalar_val = tf.reduce_sum(reshaped_val)

    # Backward pass
    # Mimics: min_val.backward()
    grads = tape.gradient(scalar_val, a)

    # Inspection
    # Mimics: a.grad
    # For reshape, gradients should flow back 1-to-1 (identity mapping).
    # Since we summed the output, the gradient should be ones.
    expected_grads = tf.ones([5])
    
    # Assert that gradients match the expected identity mapping
    assert tf.reduce_all(tf.equal(grads, expected_grads)).numpy(), \
        f"Gradient mismatch for tf.reshape. Expected {expected_grads}, got {grads}"
    
    print("Test passed: Gradients flow correctly through tf.reshape.")

if __name__ == "__main__":
    test_reshape_gradient_behavior()
