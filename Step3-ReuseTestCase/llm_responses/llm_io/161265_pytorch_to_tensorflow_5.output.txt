import tensorflow as tf
import numpy as np

def test_large_tensor_conversion():
    """
    Adapted test case for Issue 161265: [MPS] On MacOS-26 torch.full fails for 4+Gb tensors.
    
    This test verifies if tf.keras.ops.convert_to_tensor correctly handles 
    the conversion of large numpy arrays (>4GB) to tensors, ensuring values 
    are preserved (specifically checking for buffer filling issues similar 
    to the Metal fillBuffer bug).
    """
    
    # Define shape to exceed 4GB memory limit
    # 2 rows, (2^31 + 5) columns
    # Total elements: ~4.29 billion
    # Size: ~4.29 GB (int8)
    shape = (2, (1 << 31) + 5)
    
    print(f"Creating numpy array of shape {shape} with dtype int8...")
    
    # Create a numpy array filled with ones.
    # Note: This requires sufficient system RAM (~4.5GB+) to allocate the numpy array.
    np_array = np.ones(shape, dtype=np.int8)
    
    print("Converting to tensor using tf.keras.ops.convert_to_tensor...")
    
    # Convert the large numpy array to a TensorFlow tensor
    # This tests the data transfer and buffer handling logic
    tensor = tf.keras.ops.convert_to_tensor(np_array)
    
    # Verify the values at the end of the buffer
    # The original bug showed these values as 0 (uninitialized) instead of 1
    val_single = tensor[1, -2].numpy()
    val_slice = tensor[:, -2].numpy()
    
    print(f"Value at [1, -2]: {val_single}")
    print(f"Value at [:, -2]: {val_slice}")
    
    # Assertions to check correctness
    assert val_single == 1, f"Expected 1 at [1, -2], got {val_single}"
    assert np.all(val_slice == 1), f"Expected [1, 1] at [:, -2], got {val_slice}"
    
    print("Test passed.")

if __name__ == "__main__":
    test_large_tensor_conversion()
