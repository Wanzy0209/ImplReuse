import tempfile
import torch
import torch.special

class MyModule(torch.nn.Module):
    """
    A module that uses torch.special.logsumexp instead of a custom Triton op.
    This test case preserves the AOT compilation logic from the original bug report
    but substitutes the failing custom operator with the similar standard API.
    """
    def __init__(self):
        super().__init__()

    def forward(self, q):
        # Replace the custom triton_op call with torch.special.logsumexp.
        # This tests the AOT compilation pipeline's stability with standard
        # mathematical operations that handle dimensionality.
        return torch.special.logsumexp(q, dim=0)

def test_aot_compile_with_logsumexp():
    # The original bug was specific to CUDA, so we check for availability.
    if not torch.cuda.is_available():
        print("CUDA not available, skipping test.")
        return

    model = MyModule().to("cuda")
    
    with torch.inference_mode():
        q = torch.randn(1024, device="cuda")
        
        # Preserve the dynamic shapes logic from the original bug report.
        # The original code used a dictionary with index 0 to specify dynamic dimensions
        # for the first argument.
        exported_model = torch.export.export(
            model, 
            (q,), 
            dynamic_shapes=({0: torch.export.Dim("dim")},)
        )
    
    with tempfile.TemporaryDirectory() as tmpdir:
        try:
            # This is the API call that was failing in the original issue.
            # We test it here with the similar API to ensure the compilation
            # pipeline works correctly for standard ops.
            torch._inductor.aoti_compile_and_package(
                exported_model,
                package_path=tmpdir + "/package.pt2",
            )
            print("Test passed: AOT compilation successful with torch.special.logsumexp")
        except Exception as e:
            print(f"Test failed: {e}")
            raise

if __name__ == "__main__":
    test_aot_compile_with_logsumexp()
