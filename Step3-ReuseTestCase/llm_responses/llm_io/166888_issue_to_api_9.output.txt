import torch
import pytest

def test_inductor_item_on_float_tensor_arg():
    """
    Test case for Issue 166888.
    Verifies that torch.compile with the inductor backend correctly handles
    calling .item() on a float tensor argument passed to the compiled function.
    """
    # Skip if CUDA is not available, as the bug is specific to the CUDA inductor backend
    if not torch.cuda.is_available():
        pytest.skip("CUDA not available")

    # Define the function to be compiled
    def f(x, max_val):
        # The bug occurs when .item() is called on a tensor argument (max_val)
        # inside the compiled function, specifically used in operations like clamp.
        return torch.clamp(x, 0, max_val.item())

    # Compile with the 'inductor' backend and 'fullgraph=True' as per the bug report
    compiled_func = torch.compile(f, backend='inductor', fullgraph=True)

    # Setup inputs
    x = torch.randn(10, 20, 30, device='cuda')
    max_val = torch.tensor(5.0, device='cuda')

    # Run the compiled function
    # If the bug is present, this will raise:
    # torch._inductor.exc.InductorError: NameError: 'zuf0' is not defined
    result = compiled_func(x, max_val)

    # Verify the output matches the eager execution
    expected = f(x, max_val)
    torch.testing.assert_close(result, expected)

if __name__ == "__main__":
    test_inductor_item_on_float_tensor_arg()
    print("Test passed.")
