import torch
import pytest

def test_torch_backends_cuda_is_built():
    """
    Test case for torch.backends.cuda.is_built.
    
    This test verifies the consistency of the build flag, similar to how
    the CI workflow in the bug report attempts to verify the existence
    of build artifacts before proceeding with documentation generation.
    """
    # Call the API
    is_built = torch.backends.cuda.is_built()

    # The implementation is: return torch._C._has_cuda
    # We verify this direct relationship.
    assert is_built == torch._C._has_cuda

    # The bug report highlights a failure when a required resource (artifacts)
    # is missing. Here, we check that the presence of the 'torch.cuda' module
    # aligns with the build flag.
    if is_built:
        # If CUDA is built, the module should be available.
        assert hasattr(torch, 'cuda')
        # Further check that the internal C++ capability is indeed enabled.
        assert torch._C._has_cuda is True
    else:
        # If CUDA is not built, the internal flag must be false.
        assert torch._C._has_cuda is False
