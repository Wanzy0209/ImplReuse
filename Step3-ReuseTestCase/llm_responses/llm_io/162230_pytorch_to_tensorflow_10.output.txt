import tensorflow as tf
import numpy as np
import sys

def test_scalar_add():
    """
    Adapted from PyTorch test_scalar_multiply.
    Tests the behavior of tf.compat.v1.math.add with scalar inputs
    inside a tf.function graph context to mimic the original FxGraphRunnable scenario.
    """
    
    # Enable v2 behavior but use the specific compat.v1 API requested
    # We use tf.function to ensure graph execution, which is where the original
    # PyTorch test (FxGraphRunnable) encountered issues.
    
    @tf.function
    def add_scalar_to_tensor(x, scalar):
        return tf.compat.v1.math.add(x, scalar)

    # Test Case 1: Float Tensor + Float Scalar
    print("Running Test Case 1: Float Tensor + Float Scalar...")
    tensor_float = tf.constant([1.5, 2.5, 3.5])
    scalar_float = 2.0
    expected_float = [3.5, 4.5, 5.5]
    
    result_float = add_scalar_to_tensor(tensor_float, scalar_float)
    
    # Verify result
    if not np.allclose(result_float.numpy(), expected_float):
        print(f"FAILED: Expected {expected_float}, got {result_float.numpy()}")
        sys.exit(1)
    print("PASSED")

    # Test Case 2: Int Tensor + Int Scalar
    print("Running Test Case 2: Int Tensor + Int Scalar...")
    tensor_int = tf.constant([10, 20, 30])
    scalar_int = 5
    expected_int = [15, 25, 35]
    
    result_int = add_scalar_to_tensor(tensor_int, scalar_int)
    
    if not np.array_equal(result_int.numpy(), expected_int):
        print(f"FAILED: Expected {expected_int}, got {result_int.numpy()}")
        sys.exit(1)
    print("PASSED")

    # Test Case 3: Scalar + Tensor (Reverse order to check commutativity/broadcasting)
    print("Running Test Case 3: Scalar + Tensor...")
    
    @tf.function
    def add_tensor_to_scalar(scalar, x):
        return tf.compat.v1.math.add(scalar, x)

    result_reverse = add_tensor_to_scalar(100, tensor_int)
    expected_reverse = [110, 120, 130]
    
    if not np.array_equal(result_reverse.numpy(), expected_reverse):
        print(f"FAILED: Expected {expected_reverse}, got {result_reverse.numpy()}")
        sys.exit(1)
    print("PASSED")

    print("\nAll test cases passed successfully.")

if __name__ == "__main__":
    test_scalar_add()
