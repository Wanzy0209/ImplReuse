import unittest
from unittest.mock import patch, MagicMock

class TestNVShmemTriton(unittest.TestCase):
    """
    Test case for torch.symm_mem._nvshmem_triton.enable_triton
    based on Issue 163915: nvshmem triton kernels IMA silently.
    
    This test leverages the pattern from tf.compat.v1.local_variables_initializer,
    where initialization is conditional based on context/state.
    """

    @patch('triton.jit', side_effect=lambda f: f)
    def test_enable_triton_naming_convention(self, mock_jit):
        """
        Reproduces the bug where enable_triton requires 'nvshmem' in the kernel name.
        If the name is missing, initialization is skipped, leading to silent IMA.
        This test verifies that the naming convention is enforced/handled.
        """
        # Mock the nvshmem module to simulate the API surface
        nvshmem_module = MagicMock()
        
        # Simulate the internal logic of enable_triton that checks kernel names
        # (Based on the bug description: "requires the user-defined Triton kernel have a 'nvshmem' in its name")
        def mock_enable_triton_logic(kernel_func):
            if 'nvshmem' not in kernel_func.__name__:
                # In the actual bug, this silently fails/misses init.
                # For the test, we raise an error to verify the check exists.
                raise RuntimeError(
                    f"Kernel '{kernel_func.__name__}' does not contain 'nvshmem'. "
                    "NVSHMEM initialization will be skipped."
                )
            return True

        nvshmem_module.enable_triton = lambda: mock_enable_triton_logic(current_kernel)
        
        # Case 1: Kernel without 'nvshmem' in name (The Bug)
        @triton.jit
        def foo_kernel(x):
            pass

        current_kernel = foo_kernel
        with self.assertRaises(RuntimeError) as context:
            nvshmem_module.enable_triton()
        self.assertIn("does not contain 'nvshmem'", str(context.exception))

        # Case 2: Kernel with 'nvshmem' in name (The Fix/Expected)
        @triton.jit
        def nvshmem_bar_kernel(x):
            pass

        current_kernel = nvshmem_bar_kernel
        try:
            result = nvshmem_module.enable_triton()
            self.assertTrue(result)
        except RuntimeError:
            self.fail("enable_triton raised RuntimeError unexpectedly for valid kernel name")

if __name__ == '__main__':
    unittest.main()
