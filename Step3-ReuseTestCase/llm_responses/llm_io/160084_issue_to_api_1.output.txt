import torch
import unittest

# Helper function mirroring the semantic pattern of tf.test.is_built_with_cuda
# to ensure the test only runs in a CUDA environment.
def is_built_with_cuda():
    """Returns whether PyTorch was built with CUDA (GPU) support."""
    return torch.cuda.is_available()

class TestInductorRegression(unittest.TestCase):
    def test_compile_with_inductor_on_cuda(self):
        # Leveraging the similar API pattern to guard the test execution.
        # This mirrors the usage of tf.test.is_built_with_cuda in TensorFlow tests.
        if not is_built_with_cuda():
            self.skipTest("CUDA is not available, skipping test.")

        class RegressionModel(torch.nn.Module):
            def __init__(self, a=0, b=0):
                super().__init__()
                self.a = torch.nn.Parameter(torch.tensor(a).float())
                self.b = torch.nn.Parameter(torch.tensor(b).float())
                self.first_batch = True

            def forward(self, x=None):
                if self.first_batch:
                    # print(f"Model dtype: {self.a.dtype}, {self.b.dtype}. Input dtype: {x.dtype}")
                    self.first_batch = False
                return x * self.a + self.b
        
        model = RegressionModel()
        # The specific configuration causing the regression in torch 2.8
        model.forward = torch.compile(model.forward, backend="inductor")
        
        inputs = torch.randn(4, 10).to("cuda")
        
        # Execute the forward pass. 
        # This should not raise RuntimeError: opt_ready_stream && opt_parent_stream
        output = model(inputs)
        
        # Basic assertion to ensure computation happened
        self.assertIsNotNone(output)
        self.assertEqual(output.shape, torch.Size([4, 10]))

if __name__ == "__main__":
    unittest.main()
