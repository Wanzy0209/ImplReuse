import torch

def test_torch_exp_compile_with_dynamic_shapes():
    """
    Test case for Issue #162480: missing float handling in rebind_unbacked().
    
    This test leverages the similar API (torch.exp) within a torch.compile context
    using dynamic shapes. This setup exercises the symbolic shape logic 
    (specifically rebind_unbacked) where the float handling bug was reported.
    """
    # The function pattern derived from the similar API information
    def exp_model(a):
        return (3 * a).exp()

    # Compile with dynamic shapes to trigger symbolic shape handling.
    # The bug was in AOTInductor, which is engaged by torch.compile.
    compiled_model = torch.compile(exp_model, dynamic=True)

    # Create input tensor
    input_tensor = torch.randn(10, 10)

    # Run the compiled model. 
    # If the bug is present, the symbolic shape engine might crash 
    # when rebind_unbacked encounters a float value during shape inference.
    result = compiled_model(input_tensor)

    # Verify the output is correct
    expected = exp_model(input_tensor)
    assert torch.allclose(result, expected), "Compiled output does not match expected output"

if __name__ == "__main__":
    test_torch_exp_compile_with_dynamic_shapes()
    print("Test passed.")
