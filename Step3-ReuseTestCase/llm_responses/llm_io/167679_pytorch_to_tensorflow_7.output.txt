```bash
conda create -n tf_test python=3.11 -y
conda activate tf_test
python -m pip install tensorflow

python - << 'EOF'
import sys, platform, tensorflow as tf

print("Python:", sys.version)
print("Executable:", sys.executable)
print("TensorFlow:", tf.__version__)
print("macOS:", platform.mac_ver()[0])
print("platform:", platform.platform())
print("arch:", platform.machine())

# Check the backend configuration
print("Keras Backend:", tf.keras.backend.backend())
print("Image Data Format:", tf.keras.backend.image_data_format())

# Verify the API returns a valid value
valid_formats = ['channels_first', 'channels_last']
current_format = tf.keras.backend.image_data_format()
assert current_format in valid_formats, f"Unexpected image data format: {current_format}"
EOF
```
