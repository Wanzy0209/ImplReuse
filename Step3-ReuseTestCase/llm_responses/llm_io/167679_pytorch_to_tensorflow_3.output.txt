import sys, platform, tensorflow as tf

print("Python:", sys.version)
print("Executable:", sys.executable)
print("TensorFlow:", tf.__version__)
print("macOS:", platform.mac_ver()[0])
print("platform:", platform.platform())
print("arch:", platform.machine())

# Check if TensorFlow is built with CUDA (analogous to is_built check)
print("TF built with CUDA?:", tf.test.is_built_with_cuda())

# Test the target API: tf.compat.v1.no_regularizer
# This function should return None to indicate no regularization is applied
dummy_tensor = tf.constant(1.0)
result = tf.compat.v1.no_regularizer(dummy_tensor)

print("no_regularizer result:", result)
print("no_regularizer is None?:", result is None)

# Assertion to verify correct behavior
assert result is None, "tf.compat.v1.no_regularizer should return None"
