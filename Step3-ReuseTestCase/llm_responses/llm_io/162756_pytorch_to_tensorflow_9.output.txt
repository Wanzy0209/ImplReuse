import tensorflow as tf

def test_name_scope_with_reductions():
    """
    Adapted test case for tf.name_scope based on the PyTorch torch.compile bug report.
    The original issue involved helper functions failing during compilation of 
    sum, mean, and cumsum operations. This test verifies that these operations 
    execute correctly when grouped under a TensorFlow name_scope.
    """
    # Adapt inputs from the original PyTorch test case
    # Original: torch.rand(16, 128, device="cuda")
    x = tf.random.uniform((16, 128))
    y = tf.random.uniform((32, 128))
    z = tf.random.uniform((32, 256))

    # Use the similar API: tf.name_scope
    # This groups the operations, analogous to how torch.compile groups ops.
    with tf.name_scope("combo_ops_scope"):
        # Translate operations:
        # x.sum(1) -> tf.reduce_sum(x, axis=1)
        # y.mean(1) -> tf.reduce_mean(y, axis=1)
        # z.cumsum(1) -> tf.cumsum(z, axis=1)
        out_sum = tf.reduce_sum(x, axis=1)
        out_mean = tf.reduce_mean(y, axis=1)
        out_cumsum = tf.cumsum(z, axis=1)

    # Verify behavior and shapes
    # In eager mode (default), these are executed immediately.
    assert out_sum.shape == (16,), f"Expected shape (16,), got {out_sum.shape}"
    assert out_mean.shape == (32,), f"Expected shape (32,), got {out_mean.shape}"
    assert out_cumsum.shape == (32, 256), f"Expected shape (32, 256), got {out_cumsum.shape}"

    print("Test passed: Operations executed successfully within name_scope.")

if __name__ == "__main__":
    test_name_scope_with_reductions()
