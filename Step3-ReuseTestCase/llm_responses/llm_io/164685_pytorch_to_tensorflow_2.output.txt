import tensorflow as tf
import tf.experimental.dtensor as dtensor

# Set seed for reproducibility, matching the original test case
tf.random.set_seed(19989)

def test_copy_to_mesh_with_arithmetic_logic():
    """
    Adapts the arithmetic logic from the PyTorch bug report to test
    tf.experimental.dtensor.copy_to_mesh.
    
    The original bug involved a divergence between eager and compiled modes
    when handling scalar operations, mixed dtypes (int32, int64), and division.
    This test constructs a similar tensor in TensorFlow and attempts to
    distribute it using copy_to_mesh.
    """
    
    # Setup a simple mesh for DTensor operations
    # Using CPU device to ensure the test runs in most environments
    mesh = dtensor.create_mesh(['x'], ['device:CPU:0'])
    layout = dtensor.Layout.replicated(mesh, rank=0)

    # --- Reproduce the arithmetic logic ---
    
    # arg_0 = torch.tensor(torch.randn(()), dtype=torch.int32).item()
    # TF equivalent: random scalar int32
    arg_0 = tf.constant(tf.random.uniform((), minval=-10, maxval=10, dtype=tf.int32), dtype=tf.int32)
    
    # Sentinel tensor
    sentinel = tf.constant(1.0)

    # var_node_2 = -6
    var_node_2 = -6
    
    # var_node_3 = arg_0
    var_node_3 = arg_0
    
    # var_node_1 = var_node_2 * var_node_3
    # Result is int32
    var_node_1 = var_node_2 * var_node_3

    # var_node_5 = torch.full((), 1, dtype=torch.int64)
    var_node_5 = tf.constant(1, dtype=tf.int64)
    
    # var_node_4 = var_node_5.item()
    # In TF, we keep it as a tensor to maintain graph compatibility, 
    # but it acts as a scalar value.
    var_node_4 = var_node_5

    # var_node_0 = var_node_1 / var_node_4
    # Division promotes int32 to float
    var_node_0 = var_node_1 / var_node_4

    # result = var_node_0 * sentinel
    result = var_node_0 * sentinel

    # --- Test the Target API: tf.experimental.dtensor.copy_to_mesh ---
    
    print("Testing tf.experimental.dtensor.copy_to_mesh with generated tensor...")
    
    try:
        # The API copies a regular tensor onto the DTensor device with the given layout.
        # We pass the 'result' tensor generated by the logic above.
        dtensor_result = dtensor.copy_to_mesh(result, layout)
        
        print("✅ copy_to_mesh success")
        
        # Assertions to verify behavior
        assert isinstance(dtensor_result, dtensor.DTensor), "Result should be a DTensor"
        # Verify the layout matches
        assert dtensor_result.layout == layout, "Layout mismatch"
        # Verify the value is preserved (approximate check for float)
        assert tf.abs(dtensor_result.to_tensor() - result) < 1e-5, "Value mismatch"
        
        print("✅ All assertions passed")
        
    except Exception as e:
        print(f"❌ Error during copy_to_mesh: {e}")
        raise

if __name__ == "__main__":
    test_copy_to_mesh_with_arithmetic_logic()
