import torch
import pytest

def test_linalg_solve_mps_backend():
    """
    Test case for Issue 163962: Internal assert failed when using Tensorly on MPS.
    
    This test focuses on the underlying API 'torch.linalg.solve' which was failing
    when executed on the MPS device. It leverages the pattern of checking backend
    availability (similar to torch.backends.cuda.is_built) to ensure the test
    only runs on supported hardware.
    """
    # Leverage the pattern of the similar API (torch.backends.cuda.is_built)
    # to check for the specific backend involved in the bug (MPS).
    if not torch.backends.mps.is_available():
        pytest.skip("MPS backend is not available, skipping test.")

    # Reproduce the logic: Create tensors on MPS and perform linear algebra operations.
    # The original bug occurred during PARAFAC decomposition which relies on solving linear systems.
    # We test the primitive 'torch.linalg.solve' directly on MPS.
    
    # Create a random matrix A and vector b on MPS
    # Using dimensions similar to the scale in the bug report (12)
    A = torch.randn(12, 12, device="mps")
    b = torch.randn(12, device="mps")

    # Ensure A is invertible by making it positive definite
    A = A @ A.T + torch.eye(12, device="mps") * 1e-3

    # Original API Under Test: torch.linalg.solve
    # This call previously triggered an internal assert failure on MPS.
    try:
        x = torch.linalg.solve(A, b)
    except RuntimeError as e:
        pytest.fail(f"torch.linalg.solve failed on MPS with: {e}")

    # Verify the solution is correct
    assert torch.allclose(A @ x, b, atol=1e-4)
