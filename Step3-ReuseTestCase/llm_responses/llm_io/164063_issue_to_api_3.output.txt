import tensorflow as tf
import numpy as np

def foo(arg0, arg1, arg2, arg3, arg4, sentinel):
    # t0 = arg0
    # Adjusting arg0 shape to match the reshape target for a valid test case
    # Original: size=(36, 7112, 1, 1) -> reshape (28, 24, 3, 127) [Size mismatch in original]
    # We use the target shape directly to ensure the graph logic is testable.
    t0 = arg0 
    
    # t1 = t0.reshape((28, 24, 3, 127))
    t1 = tf.reshape(t0, (28, 24, 3, 127))
    
    # t2 = t1.var(dim=2) 
    # Replacing torch.var with the similar API: tf.experimental.numpy.argmax
    # Note: argmax returns int64, while var returns float. 
    # We cast to bfloat16 to maintain the graph flow (concat with t10).
    t2 = tf.experimental.numpy.argmax(t1, axis=2)
    t2 = tf.cast(t2, tf.bfloat16)
    
    # t3 = arg1
    t3 = arg1
    
    # t4 = arg2
    t4 = arg2
    
    # t5 = torch.nn.functional.embedding(torch.clamp(t3, 0, t4.size(0) - 1).to(torch.long), t4)
    # TF equivalent: tf.gather
    indices = tf.clip_by_value(t3, 0, tf.shape(t4)[0] - 1)
    t5 = tf.gather(t4, indices)
    
    # t6 = arg3
    t6 = arg3
    
    # t7 = torch.nn.functional.pad(t6, [0, 1], mode='constant', value=0.0)
    # TF padding format is [[0,0], [0,0], ..., [pad_before, pad_after]] for each dimension
    t7 = tf.pad(t6, [[0, 0], [0, 0], [0, 1]], mode='CONSTANT', constant_values=0.0)
    
    # t8 = arg4
    t8 = arg4
    
    # t9 = t8.sum(dim=1)
    t9 = tf.reduce_sum(t8, axis=1)
    
    # t10 = torch.baddbmm(t5, t7, t9)
    # baddbmm is batch1 + batch2 @ batch3
    # t5: (30, 24, 127), t7: (30, 24, 16), t9: (30, 16, 127)
    # t7 @ t9 -> (30, 24, 127)
    matmul_result = tf.einsum('bij,bjk->bik', t7, t9)
    t10 = t5 + matmul_result
    
    # t11 = torch.cat([t2, t10], dim=0)
    t11 = tf.concat([t2, t10], axis=0)
    
    output = t11 + sentinel
    return output

# Setup inputs
# Using bfloat16 to match the original bug's precision context
arg0 = tf.random.uniform([28, 24, 3, 127], dtype=tf.bfloat16) 
arg1 = tf.random.uniform([30, 24], minval=0, maxval=512, dtype=tf.int64)
arg2 = tf.random.uniform([512, 127], dtype=tf.bfloat16)
arg3 = tf.random.uniform([30, 24, 15], dtype=tf.bfloat16)
arg4 = tf.random.uniform([30, 4, 16, 127], dtype=tf.bfloat16)
sentinel = tf.constant(0.0, dtype=tf.bfloat16)

if __name__ == '__main__':
    # Run Eager
    out_eager = foo(arg0, arg1, arg2, arg3, arg4, sentinel)
    print('Eager Success! ✅')

    # Run Compiled (tf.function is the TF equivalent of torch.compile)
    compiled_foo = tf.function(foo)
    out_compiled = compiled_foo(arg0, arg1, arg2, arg3, arg4, sentinel)
    print('Compile Success! ✅')

    # Check for divergence
    if not np.allclose(out_eager.numpy(), out_compiled.numpy()):
        print("Divergence detected! ❌")
        exit(1)
    else:
        print("Results match. ✅")
