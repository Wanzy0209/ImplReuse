import torch
from torch import nn

class AddmvModel(nn.Module):
    def __init__(
        self,
        mat: torch.Tensor,
        input_tensor: torch.Tensor,
    ) -> None:
        super().__init__()
        # mat: (M, N), input_tensor: (M)
        self.mat = nn.Parameter(mat, requires_grad=False)
        self.input_tensor = nn.Parameter(input_tensor, requires_grad=False)

    def forward(self, vec: torch.Tensor) -> torch.Tensor:
        # vec: (N)
        # Performs: input_tensor + mat @ vec
        return torch.addmv(self.input_tensor, self.mat, vec)

def test_device(device, vec, mat, input_tensor):
    vec = vec.to(device)
    mat = mat.to(device)
    input_tensor = input_tensor.to(device)

    model = AddmvModel(mat, input_tensor).to(device)
    model_compiled = torch.compile(AddmvModel(mat, input_tensor).to(device), fullgraph=True)

    # warm up
    with torch.no_grad():
        y_original = model(vec)
        y_compiled = model_compiled(vec)

    # proper inference
    with torch.no_grad():
        y_original = model(vec)
        y_compiled = model_compiled(vec)

    print(f'device: {device}, diff: {torch.max(torch.abs(y_original - y_compiled)).item()}')
    print('original', y_original[:5])
    print('compiled', y_compiled[:5])

def main():
    # Define dimensions for addmv: mat (M, N), vec (N), input (M)
    M = 10
    N = 5

    torch.manual_seed(42)
    vec = torch.randn(N, dtype=torch.float32)
    mat = torch.randn(M, N, dtype=torch.float32)
    input_tensor = torch.randn(M, dtype=torch.float32)

    test_device('cpu', vec, mat, input_tensor)
    if torch.cuda.is_available():
        test_device('cuda', vec, mat, input_tensor)

if __name__ == '__main__':
    main()
