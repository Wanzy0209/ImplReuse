```python
import tensorflow as tf

# Setup: Create a tensor similar to the PyTorch example.
# PyTorch: torch.arange(0, 16).reshape(2, 2, 2, 2).cuda().to(memory_format=torch.channels_last)
# TensorFlow: Create a 4D tensor. Note that standard TF tensors do not expose a 'channels_last' 
# memory format flag in the same way PyTorch does, but we verify the TypeSpec captures the 
# shape and dtype correctly.
x = tf.reshape(tf.range(16, dtype=tf.int64), (2, 2, 2, 2))

# Action: Call the target API (tf.type_spec_from_value).
# This is analogous to the operation in the bug report.
spec = tf.type_spec_from_value(x)

# Verification: Check if the spec matches the original tensor properties.
# The original bug checked if x == gathered_x (specifically storage/memory ordering).
# Here we check if the TypeSpec preserves the tensor's shape and dtype.
print(f"Original Tensor Shape: {x.shape}, Dtype: {x.dtype}")
print(f"TypeSpec Shape: {spec.shape}, Dtype: {spec.dtype}")

assert spec.shape == x.shape, f"Shape mismatch: {spec.shape} vs {x.shape}"
assert spec.dtype == x.dtype, f"Dtype mismatch: {spec.dtype} vs {x.dtype}"
assert isinstance(spec, tf.TensorSpec), "Output is not a TensorSpec"

print("Test passed: TypeSpec correctly preserves tensor properties.")
```
