```python
import torch
import torch.nn as nn
import torch.nn.utils.prune as prune
from collections import namedtuple

def test_namedtuple_pruning(self):
    # Adapt the NamedTuple concept to the parameters argument of global_unstructured
    PruneTarget = namedtuple('PruneTarget', 'module name')

    class M(nn.Module):
        def __init__(self):
            super().__init__()
            self.linear1 = nn.Linear(4, 4)
            self.linear2 = nn.Linear(4, 4)

        def forward(self, x):
            return self.linear2(self.linear1(x))

    model = M()

    # Create the input to the API using the NamedTuple
    # Original bug used Point(x, y). Here we use PruneTarget(module, name).
    parameters = [
        PruneTarget(model.linear1, 'weight'),
        PruneTarget(model.linear2, 'weight')
    ]

    # Call the similar API
    # global_unstructured does not have a 'strict' argument, so we just call it directly.
    # The goal is to verify it handles the NamedTuple input correctly.
    prune.global_unstructured(
        parameters,
        pruning_method=prune.L1Unstructured,
        amount=0.5
    )

    # Verify the pruning was applied
    assert hasattr(model.linear1, 'weight_mask')
    assert hasattr(model.linear2, 'weight_mask')
    print("Test passed: global_unstructured handles NamedTuple parameters.")
```
