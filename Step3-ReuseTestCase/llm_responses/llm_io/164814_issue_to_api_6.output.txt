import tensorflow as tf

def test_pad_to_cardinality_with_scalar_elements():
    """
    Test case for tf.data.experimental.pad_to_cardinality based on the 
    PyTorch issue regarding dimensionality mismatch with 0-d tensors (scalars).
    
    The original bug involved a sequence of operations (unique -> reshape -> squeeze)
    that produced a 0-d tensor, causing a stride mismatch during compilation.
    This test verifies that the similar TensorFlow API handles 0-d tensor elements
    correctly within the dataset pipeline.
    """
    
    # Mimic the tensor manipulation logic from the PyTorch issue
    # Logic: arange -> unique -> cast -> reshape([1]) -> squeeze -> scalar
    def generate_scalar_element():
        # _inp_unique_wide = torch.arange(1, device=var_node_3.device, dtype=torch.int64)
        inp = tf.range(1, dtype=tf.int64)
        
        # _uniq_wide = torch.unique(_inp_unique_wide)
        uniq, _ = tf.unique(inp)
        
        # var_node_2 = _uniq_wide.to(var_node_3.dtype)
        casted = tf.cast(uniq, tf.int32)
        
        # var_node_1 = torch.reshape(var_node_2, [1])
        reshaped = tf.reshape(casted, [1])
        
        # var_node_0 = torch.squeeze(var_node_1) -> Results in size=() (scalar)
        # This is the critical step that triggered the dimensionality bug in PyTorch.
        scalar = tf.squeeze(reshaped)
        
        return scalar

    # Create a dataset containing the scalar element wrapped in a dictionary.
    # pad_to_cardinality requires dictionary elements.
    # We use a dataset with a single element to test padding behavior.
    initial_data = {'value': generate_scalar_element()}
    ds = tf.data.Dataset.from_tensors(initial_data)

    # Apply the similar API: tf.data.experimental.pad_to_cardinality
    # We pad to a cardinality of 3 to ensure padding logic is triggered.
    target_cardinality = 3
    ds_padded = ds.apply(tf.data.experimental.pad_to_cardinality(target_cardinality))

    # Execute the pipeline to check for runtime errors or shape mismatches.
    results = list(ds_padded.as_numpy_iterator())

    # Assertions
    assert len(results) == 3, f"Expected 3 elements, got {len(results)}"
    
    # Check the first element (real data)
    assert results[0]['valid'] == True, "First element should be valid"
    assert results[0]['value'].shape == (), "Value should be a scalar (0-d tensor)"
    assert results[0]['value'] == 1, "Scalar value should be 1"

    # Check the padding elements
    for i in range(1, 3):
        assert results[i]['valid'] == False, f"Element {i} should be padding"
        # Default padding for int32 is 0
        assert results[i]['value'] == 0, f"Padding value at {i} should be 0"

    print("✅ Test passed: pad_to_cardinality handles 0-d tensor elements correctly.")

if __name__ == "__main__":
    test_pad_to_cardinality_with_scalar_elements()
