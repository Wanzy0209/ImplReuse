import tensorflow as tf

# Adapt the test case to use tf.keras.backend.name_scope
# We use tf.function to mimic the compilation behavior of torch.compile
@tf.function
def fn(x, i):
    # Use the requested similar API: tf.keras.backend.name_scope
    with tf.keras.backend.name_scope("dynamic_scope"):
        # Mimic the conditional logic from the original bug report
        if i == 1:
            # In PyTorch, torch._dynamo.graph_break() stops tracing here.
            # In TensorFlow, we can't explicitly "break" the graph inside tf.function
            # in the same way, but we can simulate a path where the scope
            # might be empty or the graph structure changes.
            pass
        return x + 1

if __name__ == "__main__":
    inp = tf.random.normal((3,))
    
    # Execute with different inputs to trigger different traces
    # tf.function specializes on Python int arguments, similar to how 
    # torch.compile might generate different graphs for different guards.
    fn(inp, 0)
    fn(inp, 1)
    fn(inp, 2)

    # Retrieve the concrete functions generated for the different calls
    # Note: get_concrete_function requires matching the input signature
    concrete_fn_0 = fn.get_concrete_function(tf.TensorSpec(shape=(3,), dtype=tf.float32), 0)
    concrete_fn_1 = fn.get_concrete_function(tf.TensorSpec(shape=(3,), dtype=tf.float32), 1)
    concrete_fn_2 = fn.get_concrete_function(tf.TensorSpec(shape=(3,), dtype=tf.float32), 2)

    # Verify the graph structures
    print("Graph for i=0:")
    print(concrete_fn_0.graph.as_graph_def())
    
    print("\nGraph for i=1:")
    print(concrete_fn_1.graph.as_graph_def())
    
    print("\nGraph for i=2:")
    print(concrete_fn_2.graph.as_graph_def())

    # Check if the scope 'dynamic_scope' exists in the graphs
    # In the original bug, the graph for i=1 was empty.
    # Here we check if the name_scope is preserved or if the graph is empty.
    def get_op_names(graph):
        return [n.name for n in graph.get_operations()]

    ops_0 = get_op_names(concrete_fn_0.graph)
    ops_1 = get_op_names(concrete_fn_1.graph)
    ops_2 = get_op_names(concrete_fn_2.graph)

    print(f"\nOps for i=0: {ops_0}")
    print(f"Ops for i=1: {ops_1}")
    print(f"Ops for i=2: {ops_2}")

    # Assertion to verify behavior (adapted from the bug context)
    # The original bug showed an empty graph for the break case.
    # In TF, the graph for i=1 should still contain the 'add' operation,
    # but we verify it is not empty.
    assert len(ops_1) > 0, "Graph for i=1 should not be empty"
    assert any("dynamic_scope" in op for op in ops_1), "Scope name should be present in ops"
