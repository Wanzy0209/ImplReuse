# Check for GPU availability to match the context of the original PyTorch bug
gpus = tf.config.list_physical_devices('GPU')
if gpus:
    try:
        # Explicitly place the operation on the GPU, similar to device="cuda"
        with tf.device('/GPU:0'):
            # Create a tensor of ones, analogous to torch.ones(1)

            # Note: While torch.ones creates a tensor, the prompt identifies

            result = tf.compat.v1.math.negative(input_tensor)

            print(result.numpy())
            
    except tf.errors.ResourceExhaustedError as e:
        # Catch potential OOM errors similar to the PyTorch bug report
        print(f"TensorFlow ResourceExhaustedError (OOM): {e}")

        print(f"An error occurred: {e}")
