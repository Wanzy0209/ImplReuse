import tensorflow as tf
import unittest

class TestTPURewriteDoubleApplication(unittest.TestCase):
    def test_tpu_rewrite_correctness(self):
        """
        Adapts the logic from PyTorch Issue 165624 to verify tf.compat.v1.tpu.rewrite.
        
        The PyTorch bug involved a 'joint_custom_pre_pass' being executed twice due to 
        a merge mistake (once at the start and once after standard passes).
        
        This test verifies that tf.compat.v1.tpu.rewrite applies its internal 
        transformations correctly and produces the expected result exactly once, 
        ensuring no unintended double-application of logic occurs that would 
        alter the computation result.
        """
        
        # Define a computation
        # If a transformation pass were applied twice incorrectly (e.g., constant folding),
        # the result might differ if the operation wasn't idempotent.
        def computation(x):
            return x * 2 + 1

        inputs = [tf.constant([1.0, 2.0, 3.0])]

        try:
            # Attempt to initialize TPU
            resolver = tf.distribute.cluster_resolver.TPUClusterResolver()
            tf.config.experimental_connect_to_cluster(resolver)
            tf.tpu.experimental.initialize_tpu_system(resolver)
            strategy = tf.distribute.TPUStrategy(resolver)

            with strategy.scope():
                # Apply the rewrite
                # This corresponds to the graph transformation phase in PyTorch.
                # We verify the output is correct.
                result = tf.compat.v1.tpu.rewrite(computation, inputs)
                
                # Expected result: [1*2+1, 2*2+1, 3*2+1] = [3, 5, 7]
                expected = [3.0, 5.0, 7.0]
                
                # Assert correctness
                self.assertTrue(
                    tf.reduce_all(tf.equal(result[0], tf.constant(expected))).numpy(),
                    "The rewrite produced an unexpected result, possibly indicating a logic error."
                )

        except (ValueError, tf.errors.NotFoundError) as e:
            # Gracefully skip if TPU hardware is not available
            self.skipTest(f"TPU not available or configured: {e}")

if __name__ == '__main__':
    unittest.main()
