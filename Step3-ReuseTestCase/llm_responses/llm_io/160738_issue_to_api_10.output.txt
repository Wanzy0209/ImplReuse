```python
import torch
import sys

def test_var_zero_dim_dim_zero():
    """
    Test case to verify that torch.var handles dim=0 on a zero-dimensional tensor
    correctly across CPU and MPS devices.
    
    This addresses the bug where MPS throws:
    "var_mps: reduction dim must be in the range of input shape"
    while CPU returns the expected result (nan).
    """
    # Check if MPS is available
    if not torch.backends.mps.is_available():
        print("MPS device is not available. Skipping MPS specific test.")
        return

    # Create a zero-dimensional tensor
    # Value 3.0 is arbitrary, variance of a single scalar with unbiased=True is NaN
    x_cpu = torch.tensor(3.0)
    x_mps = x_cpu.to("mps")

    # 1. Test CPU behavior (Expected/Reference)
    try:
        output_cpu = torch.var(x_cpu, dim=0)
        print(f"CPU Test: Success. Output: {output_cpu}")
    except Exception as e:
        print(f"CPU Test Failed: {e}")
        sys.exit(1)

    # 2. Test MPS behavior (Bug Fix Verification)
    try:
        output_mps = torch.var(x_mps, dim=0)
        print(f"MPS Test: Success. Output: {output_mps}")
    except Exception as e:
        print(f"MPS Test Failed: {e}")
        print("Error: MPS backend does not handle dim=0 on zero-dimensional tensors correctly.")
        sys.exit(1)

    # 3. Verify consistency
    # Both should be NaN because variance of a single point (unbiased) is undefined (0/0)
    if torch.isnan(output_cpu) and torch.isnan(output_mps):
        print("Assertion Passed: Both CPU and MPS returned NaN as expected.")
    else:
        print(f"Assertion Failed: Outputs do not match. CPU={output_cpu}, MPS={output_mps}")
        sys.exit(1)

if __name__ == "__main__":
    test_var_zero_dim_dim_zero()
```
