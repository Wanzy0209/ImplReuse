import torch
from torch._dynamo.functional_export import _dynamo_graph_capture_for_export
from torch._functorch.aot_autograd import aot_export_joint_with_descriptors
from torch._guards import tracing, TracingContext
from contextlib import ExitStack

# This test case leverages torch.special.zeta (the similar API) 
# within the module to verify that the export pipeline handles kwargs correctly.
class ModuleWithKwargsAndZeta(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x, q=1.0):
        # Using torch.special.zeta as the core operation
        # zeta(input, other, *, out=None) -> Tensor
        return torch.special.zeta(x, q)

def test_aot_export_with_zeta_kwargs():
    model = ModuleWithKwargsAndZeta()

    # Setup inputs and kwargs
    # x corresponds to 'input' in zeta, q corresponds to 'other'
    inputs = (torch.tensor([2., 4.]),)
    kwargs = {"q": torch.tensor([1., 2.])}

    # Helper function to replicate the bug report's capture and export logic
    def graph_capture_and_aot_export_joint_with_descriptors(model, inputs, kwargs=None):
        if kwargs is None:
            kwargs = {}
        with torch._dynamo.config.patch(install_free_tensors=True):
            # Capture the graph with kwargs
            gm = _dynamo_graph_capture_for_export(model)(*inputs, **kwargs)
            fake_mode = gm.meta.get("fake_mode", None)

        with tracing(TracingContext(fake_mode)):
            return aot_export_joint_with_descriptors_alone(gm, inputs, kwargs=kwargs)

    def aot_export_joint_with_descriptors_alone(model, inputs, kwargs=None):
        if kwargs is None:
            kwargs = {}
        with ExitStack() as stack:
            # Export the graph with kwargs
            joint_with_descriptors = aot_export_joint_with_descriptors(
                stack,
                model,
                inputs,
                kwargs=kwargs,
            )
            return joint_with_descriptors.graph_module

    # Execute the test
    # The bug report indicates that passing kwargs here causes issues.
    # This test verifies that the export completes successfully.
    try:
        gm = graph_capture_and_aot_export_joint_with_descriptors(model, inputs, kwargs=kwargs)
        assert gm is not None
        print("Test Passed: Export successful with kwargs using torch.special.zeta")
    except Exception as e:
        print(f"Test Failed: {e}")
        raise

if __name__ == "__main__":
    test_aot_export_with_zeta_kwargs()
