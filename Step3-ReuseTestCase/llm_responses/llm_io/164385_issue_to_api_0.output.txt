import tensorflow as tf
import numpy as np

def test_average_with_complex_integer_inputs():
    """
    Test case for tf.keras.layers.Average based on the logic of Issue 164385.
    
    The original issue involves a complex expression with FloorDiv operations:
    FloorDiv((24*s37 + 672)*((s14//2016)) + 21, 22)
    
    This test mirrors the construction of the numerator using integer operations
    (specifically tf.math.floordiv) and then applies the division using 
    tf.keras.layers.Average. To divide by 22 using Average, we average a list 
    containing the numerator 22 times.
    """
    
    # Create input tensors (mimicking symbolic variables with concrete integer values)
    # Values chosen to ensure non-zero results for the inner floor division
    s14 = tf.constant(4032, dtype=tf.int32)  # 4032 // 2016 = 2
    s37 = tf.constant(10, dtype=tf.int32)

    # Build the numerator expression step by step (mirroring the bug report)
    # inner_expr = FloorDiv(s14 , 2016)
    inner_expr = tf.math.floordiv(s14, 2016)
    
    # middle_expr = (24 * s37 + 672) * inner_expr
    middle_expr = (24 * s37 + 672) * inner_expr
    
    # numerator = middle_expr + 21
    numerator = middle_expr + 21

    # The original bug performs: FloorDiv(numerator, 22)
    # tf.keras.layers.Average computes: sum(inputs) / len(inputs)
    # To simulate division by 22, we create a list of 22 identical numerators.
    inputs = [numerator] * 22
    
    # Create the Average layer
    avg_layer = tf.keras.layers.Average()
    
    # Apply the layer
    result = avg_layer(inputs)

    # Calculate expected value manually
    # inner = 4032 // 2016 = 2
    # middle = (24*10 + 672) * 2 = (240 + 672) * 2 = 912 * 2 = 1824
    # numerator = 1824 + 21 = 1845
    # result = 1845 / 22 = 83.863636...
    expected_value = 1845.0 / 22.0

    # Assert that the result is close to the expected float value.
    # This verifies that Average performs standard division (promoting to float)
    # rather than floor division (which would be 83), addressing the type/precision
    # concern raised in the original bug report regarding division behavior.
    assert np.isclose(result.numpy(), expected_value), \
        f"Expected {expected_value}, but got {result.numpy()}"

    print("Test passed: tf.keras.layers.Average correctly handled the complex integer inputs.")

if __name__ == "__main__":
    test_average_with_complex_integer_inputs()
