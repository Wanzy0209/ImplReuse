import tensorflow as tf
from tensorflow.keras import losses

# Define a custom user-defined class (similar to Config in the bug report)
class CustomLoss(losses.Loss):
    def __init__(self, name='custom_loss'):
        super().__init__(name=name)
        self.factor = 1.0

    def call(self, y_true, y_pred):
        return y_true - y_pred

    def get_config(self):
        config = super().get_config()
        config.update({"factor": self.factor})
        return config

# Define a function to be traced/compiled (similar to torch.compile)
@tf.function
def forward(x, loss_obj):
    # Calling serialize() on non-constant user object
    # This mirrors the repr(config) call in the original bug
    serialized = tf.keras.losses.serialize(loss_obj)
    # Use the result to ensure it's part of the graph
    return x * tf.cast(len(str(serialized)), tf.float32)

# Setup
custom_loss = CustomLoss()
x = tf.random.normal((2, 2))

# Test execution
try:
    result = forward(x, custom_loss)
    # If this runs without error, the tracing handled the serialization
    assert result.shape == (2, 2)
    print("Test Passed: Serialization traced successfully.")
except Exception as e:
    # If it fails, it mimics the original bug behavior
    print(f"Test Failed: {e}")
