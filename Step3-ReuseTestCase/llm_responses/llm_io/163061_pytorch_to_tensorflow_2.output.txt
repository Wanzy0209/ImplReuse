import tensorflow as tf
from tensorflow.experimental import dtensor

def tf_add(x: tf.Tensor, y: tf.Tensor):
    return x + y

def dtensor_copy_to_mesh(x: tf.Tensor, layout):
    """
    Adapted test for the similar API: tf.experimental.dtensor.copy_to_mesh.
    This API copies a tensor onto the DTensor device with the given layout.
    """
    return dtensor.copy_to_mesh(x, layout)

def main():
    # Setup devices
    # Attempt to use GPU if available to match the original CUDA context
    gpus = tf.config.list_physical_devices('GPU')
    device_type = 'GPU' if gpus else 'CPU'
    num_devices = len(gpus) if gpus else 1
    
    # Initialize DTensor Mesh
    mesh = dtensor.create_mesh(
        [('x', num_devices)],
        device_type=device_type
    )
    
    # Define a replicated layout for the copy operation
    layout = dtensor.Layout([dtensor.UNSHARDED], mesh)

    # Create tensors
    # Place on GPU if available to simulate the original environment
    if gpus:
        with tf.device('/GPU:0'):
            x = tf.random.normal((4096, 4096))
            y = tf.random.normal((4096, 4096))
    else:
        x = tf.random.normal((4096, 4096))
        y = tf.random.normal((4096, 4096))

    # Run loop to observe GIL behavior
    for _ in range(10):
        # Standard TensorFlow operation
        tf_add(x, y)
        
        # Similar API operation: copy_to_mesh
        # This corresponds to the torch.compile call in the original issue
        dtensor_copy_to_mesh(x, layout)

if __name__ == "__main__":
    main()
