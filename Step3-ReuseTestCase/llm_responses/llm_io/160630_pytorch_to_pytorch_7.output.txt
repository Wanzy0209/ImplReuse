```python
import torch

def test_quantized_equal():
    # Create quantized tensors
    quant_input1 = torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 3.0]), scale=1.0, zero_point=0, dtype=torch.quint8)
    quant_input2 = torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 3.0]), scale=1.0, zero_point=0, dtype=torch.quint8)
    quant_input3 = torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 4.0]), scale=1.0, zero_point=0, dtype=torch.quint8)

    # Test equality for identical quantized tensors
    is_equal = torch.equal(quant_input1, quant_input2)
    print(f"Equal tensors (quant_input1 vs quant_input2): {is_equal}")
    assert is_equal == True, "Expected quantized tensors with same values to be equal"

    # Test inequality for different quantized tensors
    is_not_equal = torch.equal(quant_input1, quant_input3)
    print(f"Unequal tensors (quant_input1 vs quant_input3): {is_not_equal}")
    assert is_not_equal == False, "Expected quantized tensors with different values to be unequal"

    print("Test passed.")

if __name__ == "__main__":
    test_quantized_equal()
```
