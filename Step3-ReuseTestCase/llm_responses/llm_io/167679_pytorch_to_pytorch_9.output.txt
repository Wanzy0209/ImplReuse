```bash
conda create -n rng_test python=3.11 -y
conda activate rng_test
python -m pip install torch torchvision torchaudio \
  --index-url https://download.pytorch.org/whl/cpu

python - << 'EOF'
import sys, platform, torch

# Environment Information
print("Python:", sys.version)
print("Executable:", sys.executable)
print("PyTorch:", torch.__version__)
print("macOS:", platform.mac_ver()[0])
print("platform:", platform.platform())
print("arch:", platform.machine())

# Test for torch.random.get_rng_state
print("\nTesting torch.random.get_rng_state...")
try:
    state = torch.random.get_rng_state()
    
    # Verify return type and dtype
    assert isinstance(state, torch.Tensor), f"Expected torch.Tensor, got {type(state)}"
    assert state.dtype == torch.uint8, f"Expected torch.uint8, got {state.dtype}"
    
    print("RNG State Type:", type(state))
    print("RNG State Dtype:", state.dtype)
    print("Test passed: torch.random.get_rng_state returned valid ByteTensor.")
    
except Exception as e:
    print(f"Test failed: {e}")
    sys.exit(1)
EOF
```
