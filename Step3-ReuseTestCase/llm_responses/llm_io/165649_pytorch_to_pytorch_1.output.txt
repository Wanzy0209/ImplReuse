```python
import torch

# Adapted test case for torch.baddbmm based on the torch.fmod crash report.
# The original crash involved INT64_MIN and -1.
# We verify if baddbmm handles these edge cases safely when used with integer tensors.

min_val = torch.iinfo(torch.int64).min

# baddbmm(input, batch1, batch2, beta=1, alpha=1)
# Shapes: (B, M, N), (B, M, K), (B, K, N)
# Using small dimensions for minimal reproducibility
input_tensor = torch.full((2, 3, 3), min_val, dtype=torch.int64, device='cpu')
batch1 = torch.full((2, 3, 3), min_val, dtype=torch.int64, device='cpu')
batch2 = torch.full((2, 3, 3), min_val, dtype=torch.int64, device='cpu')

print("Input tensor:", input_tensor)
print("Batch1 tensor:", batch1)
print("Batch2 tensor:", batch2)

# Using -1 for beta and alpha to mimic the 'divisor is -1' scenario.
# This triggers INT64_MIN * -1, which is an arithmetic edge case similar to INT64_MIN / -1.
result = torch.baddbmm(input_tensor, batch1, batch2, beta=-1, alpha=-1)
print("Result:", result)
```
