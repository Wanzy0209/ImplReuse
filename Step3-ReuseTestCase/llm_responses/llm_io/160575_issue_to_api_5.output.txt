import torch
import sys

def test_get_arch_list_maxwell_support():
    """
    Test case for Issue 160575.
    Verifies that torch.cuda.get_arch_list() reflects the removal of
    Maxwell (sm_52) support in PyTorch 2.8.0+.
    
    This test adapts the pattern from tf.keras.config.enable_traceback_filtering,
    which checks the environment version before executing logic.
    """
    
    # Pattern from similar API: Check environment version
    # tf.keras.config.enable_traceback_filtering checks sys.version_info
    # Here we check torch.__version__ to ensure we are testing the affected version
    version_str = torch.__version__.split('+')[0] # Handle local versions like 2.8.0+cu126
    major, minor = map(int, version_str.split('.')[:2])

    # The bug is specific to 2.8.0 and later
    if major < 2 or (major == 2 and minor < 8):
        print(f"Skipping test for PyTorch version {torch.__version__} (< 2.8.0)")
        return

    # Original API Under Test
    arch_list = torch.cuda.get_arch_list()

    # Bug Description: sm_52 (Maxwell) is not compatible with PyTorch 2.8.0+
    # The warning in the bug report states: "The current PyTorch install supports 
    # CUDA capabilities sm_61 sm_70 sm_75 sm_80 sm_86 sm_90."
    
    assert "sm_52" not in arch_list, \
        f"Bug 160575: sm_52 should not be in arch_list for PyTorch {torch.__version__}. List: {arch_list}"

    # Verify the expected architectures mentioned in the bug report are present
    expected_archs = ["sm_61", "sm_70", "sm_75", "sm_80", "sm_86", "sm_90"]
    for arch in expected_archs:
        assert arch in arch_list, f"Expected architecture {arch} missing from {arch_list}"

    print("Test Passed: Architecture list correctly excludes sm_52 (Maxwell).")

if __name__ == "__main__":
    test_get_arch_list_maxwell_support()
