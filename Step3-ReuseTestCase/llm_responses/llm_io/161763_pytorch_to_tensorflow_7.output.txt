import tensorflow as tf

def test_uint_neg_add_in_scope():
    # Set seed for reproducibility
    tf.random.set_seed(0)

    # Define inputs
    # PyTorch: c = torch.tensor(7, dtype=torch.uint8)
    c = tf.constant(7, dtype=tf.uint8)

    # PyTorch: x = torch.randn(2, 2, dtype=torch.float32)
    # Using tf.random.normal to simulate torch.randn
    x = tf.random.normal((2, 2), dtype=tf.float32)

    print(f"input: {x.numpy()}")

    # Use the similar API: tf.keras.backend.name_scope
    # This wraps the operations in a namespace.
    with tf.keras.backend.name_scope("uint_neg_add_scope"):
        # Operation 1: c + x
        # TF promotes uint8 to float32 for addition
        res0 = c + x

        # Operation 2: torch.neg(c) -> tf.negative(c)
        # Note: TensorFlow casts uint8 to int32 before negation, resulting in -7.
        # This differs from PyTorch eager mode (which wraps to 249) but matches 
        # standard signed arithmetic.
        res1 = tf.negative(c)

        # Operation 3: torch.neg(c) + x
        # TF computes -7 + x
        res2 = tf.negative(c) + x

    print(f"res[0]: {res0.numpy()}")
    print(f"res[1]: {res1.numpy()}")
    print(f"res[2]: {res2.numpy()}")

    # Assertions to verify behavior
    # In TF, neg(uint8) promotes to int32, so we expect -7
    assert res1.numpy() == -7, f"Expected -7, got {res1.numpy()}"
    
    # Verify the addition logic
    # res0 should be 7 + x
    expected_res0 = 7.0 + x.numpy()
    assert tf.reduce_all(tf.equal(res0, expected_res0)), "Mismatch in res[0]"

    # res2 should be -7 + x
    expected_res2 = -7.0 + x.numpy()
    assert tf.reduce_all(tf.equal(res2, expected_res2)), "Mismatch in res[2]"

if __name__ == "__main__":
    test_uint_neg_add_in_scope()
