import torch
import torch.nn as nn

def test_load_state_dict_shape_mismatch_scalar_to_1d():
    """
    Reproduces Issue 165873: No error raised despite shape mismatch in load_state_dict.
    
    When a Parameter is initialized as a scalar (0-d tensor) and a state_dict
    containing a 1D tensor is loaded, PyTorch silently truncates the data to the
    first element instead of raising a RuntimeError.
    """
    
    class SimpleModule(nn.Module):
        def __init__(self, threshold_value):
            super().__init__()
            # Initialize as a scalar (0-d tensor)
            self.threshold = nn.Parameter(torch.tensor(threshold_value))
        
        def forward(self, x):
            return x

    # Create a 1D tensor with significantly more than one element
    large_tensor = torch.randn(32000)
    state_dict = {"threshold": large_tensor}

    module = SimpleModule(0.0)
    
    # BUG: load_state_dict should raise a RuntimeError here because 
    # the shapes (scalar vs [32000]) are incompatible.
    # However, it currently succeeds and silently takes the first value.
    module.load_state_dict(state_dict)
    
    # Assertions to verify the buggy behavior
    # The parameter remains a scalar
    assert module.threshold.shape == torch.Size([]), "Shape mismatch: parameter is no longer scalar"
    
    # The parameter value is the first element of the loaded tensor
    assert module.threshold.item() == state_dict['threshold'][0].item(), \
        "Value mismatch: parameter did not take the first value of the 1D tensor"

if __name__ == "__main__":
    test_load_state_dict_shape_mismatch_scalar_to_1d()
    print("Test passed: Bug reproduced successfully.")
