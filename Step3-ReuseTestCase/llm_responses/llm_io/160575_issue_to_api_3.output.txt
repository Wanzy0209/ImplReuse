import tensorflow as tf

class TestCUDAArchSupport(tf.test.TestCase):
    def test_cuda_build_supports_available_hardware(self):
        """
        Test case to verify that the TensorFlow build supports the available GPU hardware.
        
        This mirrors the logic of the PyTorch bug report where `torch.cuda.get_arch_list()` 
        was checked to confirm if specific architectures (like sm_52) were supported. 
        Here, we use `tf.test.is_built_with_cuda` to gate the test and then verify 
        that operations can actually execute on the GPU, ensuring the necessary 
        kernels are present for the detected hardware.
        """
        # Check if the library is built with CUDA support
        # This is the primary usage of the similar API.
        if not tf.test.is_built_with_cuda():
            self.skipTest("TensorFlow not built with CUDA")

        # Identify available GPUs
        gpus = tf.config.list_physical_devices('GPU')
        self.assertNotEmpty(gpus, "No GPUs found to test compatibility")

        # Verify kernel support for the available hardware
        # In the PyTorch issue, the system warned that sm_52 was not compatible.
        # This test attempts to run a kernel; if the build lacks support for the 
        # specific architecture of the GPU, this operation will fail.
        for gpu in gpus:
            with tf.device(gpu.name):
                # Perform a simple computation requiring a CUDA kernel
                a = tf.constant([1.0, 2.0, 3.0])
                b = tf.constant([4.0, 5.0, 6.0])
                c = tf.math.add(a, b)
                
                # Assert the result to ensure the kernel executed successfully
                self.assertAllEqual(c, [5.0, 7.0, 9.0])

if __name__ == '__main__':
    tf.test.main()
