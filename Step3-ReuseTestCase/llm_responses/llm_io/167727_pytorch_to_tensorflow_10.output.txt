import tensorflow as tf
import numpy as np

def test_convert_to_tensor_large_complex():
    """
    Adapted from PyTorch Issue #167727.
    Original Bug: torch.addmm returns incorrect results for large complex tensors on MPS.
    
    This test adapts the scenario to the similar API tf.keras.ops.convert_to_tensor.
    Since convert_to_tensor is a utility for type conversion rather than a math operation,
    we verify that large complex tensors are converted without data corruption.
    """
    
    # Dimensions from the failing case in the original bug report
    # These large dimensions triggered the issue in PyTorch
    dim_a = (64, 300)
    dim_b = (64, 10000)
    dim_c = (10000, 300)

    # Create random complex64 data using numpy to simulate the input
    # The original bug specifically involved complex64 dtype
    a_np = np.random.randn(*dim_a).astype(np.float32) + 1j * np.random.randn(*dim_a).astype(np.float32)
    b_np = np.random.randn(*dim_b).astype(np.float32) + 1j * np.random.randn(*dim_b).astype(np.float32)
    c_np = np.random.randn(*dim_c).astype(np.float32) + 1j * np.random.randn(*dim_c).astype(np.float32)

    # Convert to TensorFlow tensors using the target API
    # We explicitly request complex64 to match the bug report conditions
    a_tf = tf.keras.ops.convert_to_tensor(a_np, dtype=tf.complex64)
    b_tf = tf.keras.ops.convert_to_tensor(b_np, dtype=tf.complex64)
    c_tf = tf.keras.ops.convert_to_tensor(c_np, dtype=tf.complex64)

    # Verify the conversion preserved the data correctly
    # This checks for the type of data corruption seen in the original bug
    np.testing.assert_allclose(a_tf.numpy(), a_np, rtol=1e-6, err_msg="Mismatch in tensor a")
    np.testing.assert_allclose(b_tf.numpy(), b_np, rtol=1e-6, err_msg="Mismatch in tensor b")
    np.testing.assert_allclose(c_tf.numpy(), c_np, rtol=1e-6, err_msg="Mismatch in tensor c")

    print("Test passed: Large complex tensors converted successfully.")

if __name__ == "__main__":
    test_convert_to_tensor_large_complex()
