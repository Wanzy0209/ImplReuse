import tensorflow as tf
import tensorflow.experimental.numpy as tnp

def test_tf_add_respects_stride():
    """
    Test case to verify if tf.compat.v1.math.add respects the input stride,
    analogous to the torch.rand_like stride issue.
    """
    # Create a base tensor
    base = tnp.zeros((10, 10))
    
    # Create a strided view (analogous to torch.empty_strided).
    # Slicing with a step > 1 creates a non-contiguous tensor with specific strides.
    # Here, the stride for the first dimension will be doubled.
    input_tensor = base[::2, :]
    
    # Apply the similar API: tf.compat.v1.math.add
    # We add a scalar to ensure the shape and layout of the input are the primary factors.
    output_tensor = tf.compat.v1.math.add(input_tensor, 1.0)
    
    # Assert that the stride is respected (preserved) in the output
    assert input_tensor.strides == output_tensor.strides, \
        f"Strides mismatch: input {input_tensor.strides} vs output {output_tensor.strides}"

if __name__ == "__main__":
    test_tf_add_respects_stride()
    print("Test passed: tf.compat.v1.math.add respects input stride.")
