import unittest
import torch
import torch.distributed as dist

class TestDistributedRanksSafety(unittest.TestCase):
    """
    Test case adapted from Issue 167480 (OS command injection in torch.utils.cpp_extension).
    
    The original vulnerability allowed OS command injection via user-supplied strings 
    passed to `extra_cflags` in `load_inline`. This test verifies that the similar 
    API `torch.distributed.get_process_group_ranks` handles string input safely 
    (by rejecting it) rather than executing it or crashing unsafely.
    """

    def test_get_process_group_ranks_rejects_malicious_string(self):
        """
        Verifies that passing a string with shell metacharacters to 
        get_process_group_ranks does not result in command execution.
        """
        # Payload similar to the one used in the original bug report
        malicious_input = "; touch /tmp/pwned"
        
        # The API expects a ProcessGroup object, not a string.
        # We assert that it raises an appropriate error (TypeError, AttributeError, etc.)
        # instead of executing the command or crashing the interpreter unexpectedly.
        with self.assertRaises((TypeError, AttributeError, KeyError)):
            dist.get_process_group_ranks(malicious_input)

if __name__ == "__main__":
    unittest.main()
