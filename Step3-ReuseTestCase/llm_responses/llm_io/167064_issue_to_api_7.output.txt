import tensorflow as tf

# Simulating the global state that was affected in the PyTorch bug
# (torch.distributions.Distribution.set_default_validate_args)
# In this test, we verify that calling the similar API (tf.nn.relu)
# does not cause unintended side effects on global state.
GLOBAL_VALIDATION_FLAG = True

def test_tf_nn_relu_no_global_side_effects():
    """
    Test that using tf.nn.relu (the similar API) does not cause unintended
    side effects on global state, similar to the bug where torch.compile
    modified distribution settings.
    """
    # Capture initial global state
    initial_flag = GLOBAL_VALIDATION_FLAG

    # Define a function using the similar API (tf.nn.relu)
    # We wrap it in tf.function to mimic the 'torch.compile' aspect of the bug.
    @tf.function
    def compute_relu(x):
        return tf.nn.relu(x)

    # Execute the operation (simulating the redundant code execution)
    input_tensor = tf.constant([-2.0, 0.0, 2.0])
    
    # Call multiple times to simulate redundancy mentioned in the issue
    _ = compute_relu(input_tensor)
    _ = compute_relu(input_tensor)

    # Assertion: Check that the global flag was not modified
    # The PyTorch bug caused set_default_validate_args(False) to be called.
    # We ensure our similar API usage does not trigger such a change.
    assert GLOBAL_VALIDATION_FLAG == initial_flag, \
        "Global state was modified by tf.nn.relu execution (or its compilation)"

    print("Test passed: tf.nn.relu does not cause unintended global side effects.")

if __name__ == "__main__":
    test_tf_nn_relu_no_global_side_effects()
