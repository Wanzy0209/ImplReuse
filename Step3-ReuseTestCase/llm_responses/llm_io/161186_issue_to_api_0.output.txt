import tensorflow as tf
import sys

# Check for GPU availability to ensure memory tracking works
if not tf.config.list_physical_devices('GPU'):
    print("This test requires a GPU to run. Skipping.")
    sys.exit(0)

# Define the custom op leveraging the similar API pattern (tf.compat.v1.user_ops.my_fact)
# adapted to implement the logic of the PyTorch MyOp (saving tensors for backward).
@tf.custom_gradient
def my_fact(inp):
    """
    TensorFlow equivalent of the PyTorch MyOp.
    Mimics the behavior of saving outputs for backward to test for memory leaks
    when used with checkpointing mechanisms.
    """
    # Forward pass: Create large tensors similar to the PyTorch bug report
    out_0 = tf.zeros(2**20, dtype=tf.float32)
    out_1 = tf.zeros(2**20, dtype=tf.float32)
    
    # Mimic ctx.save_for_backward by capturing tensors in the closure
    saved_tensors = (inp, out_0, out_1)

    def grad(dA, dB):
        # Backward pass: Access saved tensors (necessary to trigger the leak logic in PyTorch)
        _ = saved_tensors
        return None

    return out_0, out_1, grad

# Define a checkpointing wrapper.
# Note: In TensorFlow, tf.recompute_grad (from tensorflow_addons) is the direct equivalent
# to torch.utils.checkpoint. We define a simplified wrapper here to ensure the test
# is runnable without external dependencies, while preserving the structural logic.
def checkpoint(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

# Test setup
dummy_input = tf.Variable(tf.random.normal((2**20,)))

def get_memory_mb():
    """Returns current GPU memory usage in MiB."""
    mem_info = tf.config.experimental.get_memory_info('GPU:0')
    return mem_info['current'] / (1024**2)

print("Starting TensorFlow memory leak test based on Issue 161186...")
try:
    for i in range(1000):
        with tf.GradientTape() as tape:
            # Apply the custom op wrapped in checkpoint logic
            # Equivalent to: torch.utils.checkpoint.checkpoint(op_fn, dummy_input, use_reentrant=False)
            full_out = checkpoint(my_fact)(dummy_input)[0]
            loss = tf.reduce_sum(full_out)
        
        # Compute gradients
        grad = tape.gradient(loss, dummy_input)
        
        # Print memory usage periodically
        if i % 10 == 0:
            print(f"Iteration {i}: {get_memory_mb():.2f} MiB")
            
except Exception as e:
    print(f"Test failed with exception: {e}")

print("Test completed.")
