import torch
import torch.special

def test_scaled_modified_bessel_k0_compile_float64():
    """
    Test case for torch.special.scaled_modified_bessel_k0 based on the 
    torch.full bug (Issue 166253).
    
    The original bug involved torch.compile caching the scalar value argument
    of torch.full when using dtype=torch.float64, causing subsequent calls
    with different values to return the result of the first call.
    
    This test verifies that torch.special.scaled_modified_bessel_k0 does not
    exhibit similar behavior when compiled with float64 inputs.
    """
    
    def func_nojit(x):
        return torch.special.scaled_modified_bessel_k0(x)

    # Compile the function
    func_jit = torch.compile(func_nojit)

    # Create float64 inputs with different values
    x1 = torch.tensor(5.0, dtype=torch.float64)
    x2 = torch.tensor(10.0, dtype=torch.float64)

    # Calculate expected results (non-compiled)
    expected_x1 = func_nojit(x1)
    expected_x2 = func_nojit(x2)

    # Run compiled function with first input
    result_x1 = func_jit(x1)
    
    # Run compiled function with second input
    # In the original torch.full bug, this would return the result for x1
    result_x2 = func_jit(x2)

    # Assert that the first call matches the expected output
    assert torch.allclose(result_x1, expected_x1), \
        f"Compiled function failed on first call. Expected {expected_x1}, got {result_x1}"

    # Assert that the second call matches the expected output for x2
    # (This is the critical check for the "stuck value" bug)
    assert torch.allclose(result_x2, expected_x2), \
        f"Compiled function failed on second call. Expected {expected_x2}, got {result_x2}"

    # Assert that the results are actually different, ensuring the value isn't stuck
    assert not torch.allclose(result_x1, result_x2), \
        "Compiled function returned the same result for different inputs (value might be stuck)."

    print("Test passed: torch.special.scaled_modified_bessel_k0 handles dynamic float64 values correctly under torch.compile.")

if __name__ == "__main__":
    test_scaled_modified_bessel_k0_compile_float64()
