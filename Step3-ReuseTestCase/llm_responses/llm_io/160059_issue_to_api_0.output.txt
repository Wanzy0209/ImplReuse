```python
import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    @torch.compiler.disable(recursive=False)
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Using the similar API (torch.nn.functional.tanhshrink) 
        # instead of the custom inverse_sigmoid from the original issue.
        return F.tanhshrink(x)


def test_jit_script_with_compiler_disable():
    """
    Test that torch.jit.script works correctly when the forward method 
    is decorated with @torch.compiler.disable.
    
    This test leverages torch.nn.functional.tanhshrink to verify the 
    interaction between the decorator, scripting, and functional APIs.
    """
    model = Model()
    
    # This call was failing in PyTorch 2.8.0 with 'undefined value' error
    # when the forward method had the @torch.compiler.disable decorator.
    scripted_model = torch.jit.script(model)
    
    # Verify the scripted model executes correctly
    input_tensor = torch.randn(5, 5)
    output = scripted_model(input_tensor)
    
    # Verify output matches the eager execution of the similar API
    expected = F.tanhshrink(input_tensor)
    assert torch.allclose(output, expected), "Scripted output does not match eager output"


if __name__ == "__main__":
    test_jit_script_with_compiler_disable()
```
