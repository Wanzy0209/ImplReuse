import tensorflow as tf
import numpy as np

def test_concatenate_gradient_distribution():
    """
    Test case to verify gradient behavior of tf.keras.layers.concatenate.
    This mirrors the logic of checking gradients for reduction operations
    (like torch.min) by verifying how gradients flow back to input tensors
    when concatenating along different axes.
    """
    
    # Setup input tensors
    # Using ones to easily verify gradient flow (upstream gradient is 1.0)
    a = tf.ones([2, 2])
    b = tf.ones([2, 2])

    # Case 1: Concatenate along axis 0
    # This parallels checking behavior with a specific dimension argument
    with tf.GradientTape() as tape:
        tape.watch([a, b])
        # Perform concatenation
        concat_val = tf.keras.layers.concatenate([a, b], axis=0)
        # Create a scalar loss to backpropagate (similar to min_val.backward())
        loss = tf.reduce_sum(concat_val)

    grads = tape.gradient(loss, [a, b])

    # Assertion: Gradients should map correctly back to inputs.
    # Since we summed the output, and all inputs are 1.0, gradients should be 1.0
    # for all elements in 'a' and 'b'.
    assert np.array_equal(grads[0].numpy(), np.ones((2, 2))), \
        f"Expected gradient for 'a' to be ones, got {grads[0].numpy()}"
    assert np.array_equal(grads[1].numpy(), np.ones((2, 2))), \
        f"Expected gradient for 'b' to be ones, got {grads[1].numpy()}"

    # Case 2: Concatenate along axis 1
    # Verifies consistency of gradient behavior when the axis argument changes
    with tf.GradientTape() as tape:
        tape.watch([a, b])
        concat_val = tf.keras.layers.concatenate([a, b], axis=1)
        loss = tf.reduce_sum(concat_val)

    grads = tape.gradient(loss, [a, b])

    assert np.array_equal(grads[0].numpy(), np.ones((2, 2))), \
        f"Expected gradient for 'a' to be ones, got {grads[0].numpy()}"
    assert np.array_equal(grads[1].numpy(), np.ones((2, 2))), \
        f"Expected gradient for 'b' to be ones, got {grads[1].numpy()}"

    print("Test passed: Gradients are correctly distributed for concatenate.")

if __name__ == "__main__":
    test_concatenate_gradient_distribution()
