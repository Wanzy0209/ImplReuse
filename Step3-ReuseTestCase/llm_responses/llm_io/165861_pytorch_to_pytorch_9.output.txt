```python
import torch
import torch.nn as nn

# Create tensors with a dimension size equal to uint16 max value (2**16)
# This shape caused the issue in F.pad (reflect mode)
x = torch.rand(2**16, 2, device="cuda")
y = torch.rand(2**16, 2, device="cuda")

# Instantiate MSELoss
criterion = nn.MSELoss()

# Run the loss calculation
# If MSELoss has similar indexing issues with large dimensions, this might crash
loss = criterion(x, y)

print("MSELoss with large batch dimension ok")
```
