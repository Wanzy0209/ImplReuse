import tensorflow as tf
import unittest

class TestAddQueueRunner(unittest.TestCase):
    def test_add_queue_runner_to_collection(self):
        """
        Adapted from PyTorch Issue 160909.
        
        Original Logic:
        1. Setup a custom context (PrivateUse1 device).
        2. Call the API (torch.compile).
        3. Verify behavior (Check for meta tensor errors).

        Adapted Logic for tf.compat.v1.train.add_queue_runner:
        1. Setup a custom context (QueueRunner with specific queue).
        2. Call the API (add_queue_runner).
        3. Verify behavior (Check if runner is added to the graph collection).
        """
        # Ensure we are in graph mode as add_queue_runner is designed for TF1 graph execution
        tf.compat.v1.disable_eager_execution()

        with tf.compat.v1.Graph().as_default():
            # Setup: Create a FIFOQueue and a QueueRunner
            # This mirrors the 'Model' and 'data' setup in the PyTorch example
            queue = tf.compat.v1.queue.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[()])
            enqueue_op = queue.enqueue([1.0])
            qr = tf.compat.v1.train.QueueRunner(queue, [enqueue_op])

            # Action: Add the queue runner to the default collection
            # This mirrors the 'torch.compile' call
            tf.compat.v1.train.add_queue_runner(qr)

            # Verification: Check if the queue runner is correctly added to the collection
            # This mirrors checking for the 'RuntimeError' or result in the PyTorch example
            collection = tf.compat.v1.get_collection(tf.compat.v1.GraphKeys.QUEUE_RUNNERS)
            
            self.assertIn(qr, collection, "QueueRunner was not added to the collection.")
            self.assertEqual(len(collection), 1, "Unexpected number of queue runners in collection.")

if __name__ == '__main__':
    unittest.main()
