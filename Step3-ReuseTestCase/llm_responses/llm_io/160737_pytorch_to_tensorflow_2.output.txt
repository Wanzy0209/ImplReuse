import tensorflow as tf

def test_tril():
    # Test 1: Valid 2D input (Control case)
    # This should succeed as tril expects a matrix (rank >= 2)
    x = tf.ones([2, 3])
    try:
        output = tf.experimental.numpy.tril(x)
        print(f"tril test succeeds for 2D input. output shape: {output.shape}")
    except Exception as e:
        print(f"tril test fails for 2D input: {e}")

    # Test 2: Scalar input (Adapted bug reproduction logic)
    # The original bug involved passing a scalar (0-dimensional) tensor 
    # where a specific dimensionality was expected.
    # For tril, the input 'm' must have rank at least 2. 
    # We test if the API handles this scalar input gracefully (e.g., raises ValueError)
    # rather than crashing with a cryptic dimension error.
    scalar_x = tf.constant(1)  # zero-dimensional tensor
    try:
        output = tf.experimental.numpy.tril(scalar_x)
        print(f"tril test succeeds for scalar input. output shape: {output.shape}")
    except Exception as e:
        print(f"tril test fails for scalar input: {e}")

if __name__ == "__main__":
    test_tril()
