import tensorflow as tf
import time

def test_get_uid_performance():
    """
    Test case adapted from the PyTorch matmul CPU overhead bug report.
    This test measures the CPU overhead of calling tf.keras.backend.get_uid
    in a tight loop to detect performance regressions.
    """
    warmup = 128
    iters = 16384

    # Warmup phase to account for any initialization overhead
    for _ in range(warmup):
        tf.keras.backend.get_uid('dense')

    # Measure execution time
    # Note: Unlike CUDA operations, get_uid is synchronous Python code,
    # so explicit synchronization is not required.
    t0 = time.perf_counter()
    for _ in range(iters):
        tf.keras.backend.get_uid('dense')
    t1 = time.perf_counter()

    # Calculate average time per call in microseconds
    avg_time_us = 1e6 * (t1 - t0) / iters
    print(f"Average time per get_uid call: {avg_time_us} us")

    # Assertion to catch significant performance regressions.
    # get_uid is a simple dictionary operation and should be very fast.
    # We set a generous threshold (e.g., 5 microseconds) to avoid flakiness
    # while ensuring the API hasn't introduced heavy CPU overhead.
    assert avg_time_us < 5.0, (
        f"Performance regression detected: {avg_time_us} us/call. "
        "Expected < 5.0 us/call."
    )

if __name__ == "__main__":
    test_get_uid_performance()
