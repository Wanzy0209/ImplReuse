import tensorflow as tf
import numpy as np

def test_bessel_j1_contiguity():
    """
    Adapted test case for tf.math.special.bessel_j1 based on the 
    PyTorch MPS linear bug (Issue 162730).
    
    Verifies that the operation produces consistent results for 
    contiguous and non-contiguous tensors.
    """
    
    # Create test tensors
    # Using a shape that allows for easy non-contiguous creation via transpose
    shape = (12, 64, 768)
    x = tf.random.normal(shape, seed=42)
    
    # Create non-contiguous tensor via transpose
    # This changes the memory strides without changing the logical data
    x_noncontig = tf.transpose(x, perm=[2, 0, 1])
    
    # Create a contiguous version
    # In TensorFlow, reshaping a transposed tensor to the same shape 
    # typically forces a copy to ensure standard row-major memory layout.
    x_contig = tf.reshape(x_noncontig, tf.shape(x_noncontig))
    
    # Apply the operation to both versions
    result_noncontig = tf.math.special.bessel_j1(x_noncontig)
    result_contig = tf.math.special.bessel_j1(x_contig)
    
    # Verify results match
    # We use numpy for the comparison to get concrete values
    match = np.allclose(result_noncontig.numpy(), result_contig.numpy(), atol=1e-5)
    max_diff = np.max(np.abs(result_noncontig.numpy() - result_contig.numpy()))
    
    print(f"Results match: {match}")
    print(f"Max difference: {max_diff}")
    
    # Assert to ensure test fails if behavior is inconsistent
    assert match, f"tf.math.special.bessel_j1 produced inconsistent results between contiguous and non-contiguous tensors. Max diff: {max_diff}"

if __name__ == "__main__":
    test_bessel_j1_contiguity()
