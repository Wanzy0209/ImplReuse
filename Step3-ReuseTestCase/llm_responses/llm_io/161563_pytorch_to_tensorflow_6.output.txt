import tensorflow as tf

# The QueueRunner API is part of the TensorFlow 1.x compatibility layer and 
# requires graph mode to be enabled (disabling eager execution).
# This is analogous to ensuring the correct mode/context is set in PyTorch.
tf.compat.v1.disable_eager_execution()

def test_queue_runner_basic_execution():
    """
    Adapted test case for tf.compat.v1.train.QueueRunner.
    
    Original Logic:
    1. Load a model (state).
    2. Prepare inputs.
    3. Run export (graph capture/execution).
    
    Adapted Logic:
    1. Create a Queue (state).
    2. Define enqueue operations (inputs/actions).
    3. Run QueueRunner (threaded execution of graph ops).
    """
    
    # 1. Setup: Create a FIFOQueue to act as the model state/storage
    # Analogous to loading the model in the original test.
    queue = tf.compat.v1.FIFOQueue(capacity=10, dtypes=[tf.float32], shapes=[()])
    
    # 2. Inputs: Define the operations to be run (enqueue ops)
    # Analogous to preparing inputs for the model.
    enqueue_op1 = queue.enqueue([1.5])
    enqueue_op2 = queue.enqueue([2.5])
    
    # 3. API Call: Create the QueueRunner
    # Analogous to torch.export.export, this sets up the execution graph/threads.
    # The QueueRunner holds the list of enqueue operations to be run in threads.
    runner = tf.compat.v1.train.QueueRunner(queue, [enqueue_op1, enqueue_op2])
    
    # 4. Execution: Run the graph to verify behavior
    # Analogous to triggering the export process.
    with tf.compat.v1.Session() as sess:
        # Initialize queue variables
        sess.run(tf.compat.v1.global_variables_initializer())
        sess.run(queue.enqueue_many([[0.0]])) # Init with dummy data
        
        # Start the queue runner threads
        coord = tf.compat.v1.train.Coordinator()
        threads = runner.create_threads(sess, coord=coord, start=True)
        
        # Verify that the operations ran by dequeuing data
        # We expect 1.5 and 2.5 to be enqueued by the runner threads
        results = []
        for _ in range(2):
            results.append(sess.run(queue.dequeue()))
            
        # Assertions to verify correct behavior
        assert 1.5 in results
        assert 2.5 in results
        
        # Clean up
        coord.request_stop()
        coord.join(threads)

if __name__ == "__main__":
    test_queue_runner_basic_execution()
    print("Test passed: QueueRunner executed operations successfully.")
