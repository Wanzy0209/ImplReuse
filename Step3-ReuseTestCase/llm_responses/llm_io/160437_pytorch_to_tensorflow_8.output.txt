import tensorflow as tf

# We use tf.function to simulate the graph compilation/tracing context 
# provided by torch.compile in the original issue.
@tf.function
def fn(x, i):
    # Using the target API: tf.keras.name_scope
    # This replaces the implicit graph context of torch.compile
    with tf.keras.name_scope("conditional_scope"):
        # In the original PyTorch code, torch._dynamo.graph_break() is called here.
        # TensorFlow does not have a direct "graph break" equivalent in eager mode 
        # that results in an empty graph, but we preserve the conditional logic 
        # to test how the name_scope handles different execution paths.
        if i == 1:
            # Simulating a condition that might alter graph structure
            # In PyTorch this breaks the graph; here we just pass or perform a no-op
            # to ensure the scope handles the branch correctly.
            pass
        return x + 1

# Input tensor
inp = tf.random.normal((3,))

# Execute the function with different arguments to trigger different graph paths
# (similar to the original report's execution flow)
result_0 = fn(inp, 0)
result_1 = fn(inp, 1)
result_2 = fn(inp, 2)

# Basic assertions to verify the logic holds and the API behaves as expected
# (The original bug report didn't have assertions, but this makes the test case valid)
assert tf.reduce_all(result_0 == inp + 1)
assert tf.reduce_all(result_1 == inp + 1)
assert tf.reduce_all(result_2 == inp + 1)

print("Test passed. tf.keras.name_scope handled conditional execution correctly.")
