import tensorflow as tf

def test_name_scope_global_state_leak():
    """
    Adapted from PyTorch Issue 167064.
    
    The original bug reported that torch.compile would internally call
    torch.distributions.Distribution.set_default_validate_args(False),
    causing an unwanted global side effect that persisted after the call.
    
    This test verifies that the similar TensorFlow API, tf.keras.name_scope,
    does not leak its naming context globally (a similar global state side effect).
    It ensures that the scope context is properly exited and does not affect
    operations defined outside of it.
    """
    # 1. Check global state (name scope) before API usage
    # We create a variable to inspect the current naming context.
    v_before = tf.Variable(1.0, name="v_before")
    # The name should not contain the scope we are about to test
    assert "test_leak_scope" not in v_before.name, \
        f"Pre-condition failed: Variable name '{v_before.name}' unexpectedly contains scope."

    # 2. Execute the API (tf.keras.name_scope)
    # We use it as a context manager, which is the standard usage.
    with tf.keras.name_scope("test_leak_scope"):
        # Inside the scope, names should have the prefix
        v_inside = tf.Variable(2.0, name="v_inside")
        assert "test_leak_scope" in v_inside.name, \
            f"API Error: Variable name '{v_inside.name}' did not include the scope inside the block."

    # 3. Check global state (name scope) after API usage
    # The PyTorch bug was that the state change persisted. 
    # Here we verify that the scope prefix is NOT present in subsequent operations.
    v_after = tf.Variable(3.0, name="v_after")
    
    assert "test_leak_scope" not in v_after.name, \
        f"Bug Detected: Variable name '{v_after.name}' leaked the scope prefix globally. " \
        "This mimics the PyTorch bug where global state was not restored."

    print("Test passed: tf.keras.name_scope did not leak global state.")

if __name__ == "__main__":
    test_name_scope_global_state_leak()
