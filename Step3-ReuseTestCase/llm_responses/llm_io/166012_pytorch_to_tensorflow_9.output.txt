import tensorflow as tf

def test_name_scope_entries_consistency():
    """
    Test that tf.name_scope generates consistent operation names (entries)
    across different execution modes (Eager vs Graph/Tracing),
    analogous to checking consistency of tlparse entries on cache miss vs hit.
    """
    scope_prefix = "my_test_scope"

    # --- Scenario 1: Eager Execution (Analogous to initial setup) ---
    with tf.name_scope(scope_prefix):
        eager_const = tf.constant(1.0, name="const")
        eager_add = eager_const + 1.0

    # Verify eager entries
    assert eager_const.name.startswith(scope_prefix), \
        f"Eager const name {eager_const.name} missing scope {scope_prefix}"
    assert eager_add.name.startswith(scope_prefix), \
        f"Eager add name {eager_add.name} missing scope {scope_prefix}"

    # --- Scenario 2: Graph Execution (Analogous to Compilation) ---
    @tf.function
    def compiled_func():
        with tf.name_scope(scope_prefix):
            graph_const = tf.constant(1.0, name="const")
            graph_add = graph_const + 1.0
            # Return names to verify consistency
            return graph_const.name, graph_add.name

    # Call 1: Cache Miss (Tracing/Compilation)
    names_miss = compiled_func()

    # Call 2: Cache Hit (Execution from cache)
    names_hit = compiled_func()

    # --- Verification: Consistency between Miss and Hit ---
    # The bug report implies inconsistency between miss and hit artifacts.
    # Here we verify the names (artifacts) are identical.
    assert names_miss == names_hit, \
        f"Inconsistency detected: Miss names {names_miss} != Hit names {names_hit}"

    # Verify scope integrity in the cached graph
    for name in names_hit:
        assert scope_prefix in name, \
            f"Scope {scope_prefix} missing in cached entry {name}"

    print("Test passed: tf.name_scope entries are consistent.")

if __name__ == "__main__":
    test_name_scope_entries_consistency()
