```python
import torch
import torch.nn.functional as F

# Test Case 1: Default dimension (dim=1)
# Using orthogonal vectors [1, 0] and [0, 1] -> similarity 0
x1 = torch.tensor([[1.0, 0.0]]).cuda()
x2 = torch.tensor([[0.0, 1.0]]).cuda()

x1.requires_grad = True
x2.requires_grad = True

# torch.nn.functional.cosine_similarity -> reduce over default dimension (dim=1)
sim_val = F.cosine_similarity(x1, x2)
# Output is a vector of size 1, sum to scalar for backward
sim_val.sum().backward()

print("Case 1 (Default dim):")
print("x1.grad:", x1.grad)
print("x2.grad:", x2.grad)

# Test Case 2: Specific dimension (dim=0)
x1 = torch.tensor([[1.0], [0.0]]).cuda()
x2 = torch.tensor([[0.0], [1.0]]).cuda()

x1.requires_grad = True
x2.requires_grad = True

# torch.nn.functional.cosine_similarity(input, dim=...) -> reduce over specified dimension
sim_val = F.cosine_similarity(x1, x2, dim=0)
sim_val.sum().backward()

print("\nCase 2 (dim=0):")
print("x1.grad:", x1.grad)
print("x2.grad:", x2.grad)
```
