import warnings
import sys

def test_fingerprint_import_no_future_warnings():
    """
    Test case to verify that importing tensorflow.python.saved_model.fingerprinting
    does not introduce FutureWarnings related to functools.partial in Enum classes,
    similar to the issue reported in torch.distributed.algorithms.ddp_comm_hooks.
    """
    
    # Capture warnings
    with warnings.catch_warnings(record=True) as caught_warnings:
        # Ensure all warnings are triggered
        warnings.simplefilter("always")

        # Import the module/class corresponding to the similar API
        # Based on the provided snippet, we target the fingerprinting module
        try:
            from tensorflow.python.saved_model import fingerprinting
        except ImportError:
            print("TensorFlow not installed, skipping test.")
            return

        # Filter for FutureWarnings specifically about functools.partial and enum.member()
        # This matches the warning message in the original bug report.
        partial_enum_warnings = [
            w for w in caught_warnings 
            if issubclass(w.category, FutureWarning) 
            and "functools.partial" in str(w.message) 
            and "enum.member()" in str(w.message)
        ]

        # Assert that no such warnings were raised
        assert len(partial_enum_warnings) == 0, (
            f"Importing tensorflow.python.saved_model.fingerprinting introduced "
            f"{len(partial_enum_warnings)} FutureWarning(s) related to functools.partial in Enum:\n"
            f"{[str(w.message) for w in partial_enum_warnings]}"
        )

if __name__ == "__main__":
    test_fingerprint_import_no_future_warnings()
    print("Test passed: No FutureWarnings detected.")
