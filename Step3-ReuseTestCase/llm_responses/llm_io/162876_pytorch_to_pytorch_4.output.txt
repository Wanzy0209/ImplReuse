```python
import torch
from torch import tensor

# Adapted test case for torch.cumprod based on the torch.aminmax issue
# Original input tensor
input_tensor = torch.tensor([1, -3, 5])

# Call the similar API torch.cumprod
# Note: torch.cumprod requires a 'dim' argument
result = torch.cumprod(input_tensor, dim=0)

# Verify the result
# Expected: [1, 1*-3, 1*-3*5] = [1, -3, -15]
expected = torch.tensor([1, -3, -15])
assert torch.equal(result, expected), f"Expected {expected}, but got {result}"

print("Test passed.")
```
