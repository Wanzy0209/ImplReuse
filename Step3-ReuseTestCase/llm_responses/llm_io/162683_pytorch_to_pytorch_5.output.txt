```python
import torch
import torch.nn.functional as F
import time

torch.manual_seed(0)

# Define configurations suitable for pixel_unshuffle
# Shape format: (Batch, Channels, Height, Width)
# Height and Width must be divisible by downscale_factor
configs = [
    ((1, 12, 10, 64), 2),
    ((1, 12, 20, 20), 2),
]

def benchmark_pixel_unshuffle(input_shape, downscale_factor, dtype=torch.float16, device="cpu", repeat=500):
    # Create input tensor
    A = torch.empty(input_shape, dtype=dtype, device=device).uniform_(0, 1) * 2 - 1
    
    # Warm up
    for _ in range(5000):
        _ = F.pixel_unshuffle(A, downscale_factor)
    
    # Run benchmark
    times = []
    for i in range(repeat):
        start = time.time()
        _ = F.pixel_unshuffle(A, downscale_factor)
        end = time.time()
        if i > 100:
            times.append(round((end - start) * 1000 * 1000))
            
    times.sort()
    print(times)
    avg_time_us = sum(times) / len(times)
    return avg_time_us

if __name__ == "__main__":
    for input_shape, factor in configs:
        # Verify correctness (shape assertion)
        input_tensor = torch.empty(input_shape, dtype=torch.float16)
        output = F.pixel_unshuffle(input_tensor, factor)
        expected_c = input_shape[1] * (factor ** 2)
        expected_h = input_shape[2] // factor
        expected_w = input_shape[3] // factor
        assert output.shape == (input_shape[0], expected_c, expected_h, expected_w), \
            f"Shape mismatch: expected {(input_shape[0], expected_c, expected_h, expected_w)}, got {output.shape}"

        t = benchmark_pixel_unshuffle(input_shape, factor)
        print(f"{input_shape} (factor={factor}) -> {t:.3f} us")
```
