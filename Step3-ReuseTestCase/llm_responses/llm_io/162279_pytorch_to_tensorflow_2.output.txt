```python
import tensorflow as tf
import numpy as np

class AnyDimsModelList(tf.Module):
    """
    Mimics AnyDimsModelEmpty from the PyTorch bug report.
    Uses a list of dimensions for reduction.
    """
    def __init__(self):
        super(AnyDimsModelList, self).__init__()

    def __call__(self, x):
        print('input shape:', x.shape)
        # Mimic torch.ops.aten.any.dims(x, [], False)
        # In PyTorch, [] implies no reduction (identity on shape), but to make 
        # this runnable and distinct in TF, we use a specific axis list [0].
        y = tf.reduce_any(x, axis=[0])
        print('output shape:', y.shape)
        return y

class AnyDimsModelNull(tf.Module):
    """
    Mimics AnyDimsModelNull from the PyTorch bug report.
    Uses None for reduction (reduce all dimensions).
    """
    def __init__(self):
        super(AnyDimsModelNull, self).__init__()

    def __call__(self, x):
        print('input shape:', x.shape)
        # Mimic torch.ops.aten.any.dims(x, None, False)
        y = tf.reduce_any(x, axis=None)
        print('output shape:', y.shape)
        return y

def process(model, x):
    print('model:', model)
    print('running eager mode...')
    model(x)
    print('enabling check numerics (exporting)...')
    # Replacing torch.export.export with tf.debugging.enable_check_numerics
    # This API affects global state and is idempotent.
    tf.debugging.enable_check_numerics()
    # Run again to verify behavior under the API's influence
    model(x)
    print()

if __name__ == "__main__":
    # Setup input
    x = tf.constant([[True, False], [False, False]], dtype=tf.bool)
    
    model_list = AnyDimsModelList()
    model_null = AnyDimsModelNull()

    # Process both models sequentially to check for state pollution
    # The original bug involved the second model having an incorrect output shape.
    process(model_list, x)
    process(model_null, x)
    
    # Assertions to verify correct shapes (mimicking the bug report's concern)
    # ModelList (axis=[0]) should reduce rank by 1
    result_list = model_list(x)
    assert result_list.shape == (2,), f"Expected shape (2,), got {result_list.shape}"
    
    # ModelNull (axis=None) should reduce to scalar
    result_null = model_null(x)
    assert result_null.shape == (), f"Expected shape (), got {result_null.shape}"
    
    print("Test passed. Shapes are correct after sequential processing.")
```
