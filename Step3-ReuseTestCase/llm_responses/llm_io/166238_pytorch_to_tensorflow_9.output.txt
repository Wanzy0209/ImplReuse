import tensorflow as tf
from collections import defaultdict
import unittest

class TestNameScopeDefaultDict(unittest.TestCase):
    def test_defaultdict_creation_in_name_scope(self):
        """
        Adapted from PyTorch Dynamo test for collections.defaultdict creation.
        Verifies that creating a defaultdict inside tf.name_scope does not raise errors
        and behaves as expected, mirroring the logic that caused a regression in PyTorch.
        """
        # The original bug occurred when torch.compile tried to trace the creation
        # of a defaultdict. We test the equivalent operation inside tf.name_scope.
        with tf.name_scope("defaultdict_scope"):
            # Reproduce the core logic: creating a defaultdict with a factory
            # This is the specific pattern that failed in the PyTorch Dynamo tracer.
            d = defaultdict(list)
            d['key1'].append('value1')
            d['key2'].append('value2')

        # Assertions to verify correctness and that the object is functional
        self.assertEqual(d['key1'], ['value1'])
        self.assertEqual(d['key2'], ['value2'])
        
        # Verify default factory works (returns empty list for missing keys)
        self.assertEqual(d['missing_key'], [])

if __name__ == '__main__':
    unittest.main()
