import torch
import os

# Helper function inspired by the pattern of tf.test.benchmark_config.
# It encapsulates the specific configuration required for the test,
# similar to how benchmark_config returns a specific ConfigProto.
def get_aot_compile_config():
    """
    Returns a context manager that patches the dynamo config
    to enable AOT compilation.
    """
    return torch._dynamo.config.patch("enable_aot_compile", True)

class M(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear1 = torch.nn.Linear(2, 4)
        self.relu = torch.nn.ReLU()
        self.linear2 = torch.nn.Linear(4, 8)
    def forward(self, x):
        return self.linear2(self.relu(self.linear1(x)))

def test_aot_precompile_multiple_runs():
    # Determine device, fallback to cpu if cuda unavailable for test stability
    device = "cuda" if torch.cuda.is_available() else "cpu"
    m = M().to(device)
    sample_inputs = (torch.randn(2, 2, device=device),)
    eager_out = m(*sample_inputs)
    
    compiled_fn_path = "./m.pt"

    # The bug title mentions "running multiple times", so we loop to reproduce the issue.
    # This mimics the repetitive nature often associated with benchmarking (tf.test.benchmark_config).
    for _ in range(2):
        # Leverage the helper function to apply the configuration
        with get_aot_compile_config():
            compiled_fn = torch.compile(
                m,
                fullgraph=True
            ).forward.aot_compile((sample_inputs, {}))

            compiled_fn.save_compiled_function(compiled_fn_path)
            torch._dynamo.reset()
            
            with torch.compiler.set_stance("fail_on_recompile"):
                with open(compiled_fn_path, "rb") as f:
                    loaded_fn = torch.compiler.load_compiled_function(f)

        assert loaded_fn is not None
        
        # Fix syntax error from original report (trailing dot)
        compiled_out = loaded_fn(m, *sample_inputs)
        
        assert torch.allclose(eager_out, compiled_out)
        
        # Clean up file to ensure clean state for next iteration
        if os.path.exists(compiled_fn_path):
            os.remove(compiled_fn_path)

if __name__ == "__main__":
    test_aot_precompile_multiple_runs()
