```python
import tensorflow as tf
import tf.experimental.numpy as tnp

# Reproduce the logic of checking stride preservation for the similar API.
# The original issue creates a tensor with specific strides and checks if the operation preserves them.
# Since tf.experimental.numpy does not have a direct equivalent to torch.empty_strided,
# we create a non-contiguous tensor using slicing to introduce custom strides.

# Create a base tensor with dimensions large enough to allow slicing
base_size = [1792, 1899 * 2, 160]
base_tensor = tnp.zeros(base_size)

# Slice the tensor to create a non-contiguous stride (similar to the bug report's intent)
# Taking every other element in the second dimension changes the stride
input_tensor = base_tensor[:, ::2, :]

# Apply the similar API: tf.experimental.numpy.add
# We add 0 to the input to check if the layout is preserved during the operation
output_tensor = tnp.add(input_tensor, 0)

# Assert that the input stride is respected by the output
# This mirrors the assertion in the original bug report
assert tnp.strides(input_tensor) == tnp.strides(output_tensor), \
    f"tf.experimental.numpy.add does not respect input stride. Input: {tnp.strides(input_tensor)}, Output: {tnp.strides(output_tensor)}"
```
