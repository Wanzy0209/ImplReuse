```python
import torch

# Test case for torch.slogdet adapted from the torch.ones OOM issue
# Original failing code: torch.ones(1, device="cuda").item()

if torch.cuda.is_available():
    # Create a minimal square matrix on CUDA
    # torch.ones is used here to replicate the allocation pattern from the bug report
    input_tensor = torch.ones(1, 1, device="cuda")
    
    # Call the similar API
    sign, logabsdet = torch.slogdet(input_tensor)
    
    # Print results to verify execution
    print(f"Sign: {sign.item()}, LogAbsDet: {logabsdet.item()}")
else:
    print("CUDA is not available.")
```
