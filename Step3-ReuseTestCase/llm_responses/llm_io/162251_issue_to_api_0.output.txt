import tensorflow as tf

# Reproduce the logic: Passing a large integer parameter to a spatial reshaping operation
# Original Bug: torch.nn.PixelShuffle(upscale_factor=545460846592) caused an FPE due to integer overflow in shape checks.
# Similar API: tf.nn.with_space_to_batch(dilation_rate=...)

# Create a tensor with shape (5, 5, 9, 3) and complex dtype to match the original context
# Note: tf.complex64 is used as TensorFlow does not support complex32
input_tensor = tf.zeros((5, 5, 9, 3), dtype=tf.complex64)

# The large parameter from the bug report that likely triggers integer overflow in shape calculations
large_dilation_rate = 545460846592

# Define a dummy operation required by with_space_to_batch
def dummy_op(input_tensor, num_spatial_dims, padding):
    return input_tensor

# Padding configuration (0 padding to stress the shape validation logic)
padding = [[0, 0], [0, 0]]

try:
    # Call the similar API with the problematic large parameter
    # PixelShuffle expects a single upscale factor, with_space_to_batch expects a dilation rate per spatial dimension
    result = tf.nn.with_space_to_batch(
        input=input_tensor,
        dilation_rate=[large_dilation_rate, large_dilation_rate],
        padding=padding,
        op=dummy_op
    )
    print("Test passed (no crash detected)")
except Exception as e:
    print(f"Exception caught: {type(e).__name__}: {e}")
