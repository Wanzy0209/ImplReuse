```python
import tensorflow as tf
from collections import namedtuple

def test_namedtuple():
    # Define a NamedTuple similar to the PyTorch example
    Point = namedtuple('Point', 'x y')
    
    # Define the computation function (analogous to torch.nn.Module)
    def computation(x, y):
        return x + y 
    
    # Create input using the NamedTuple
    inp = Point(tf.ones(3), tf.ones(3))
    
    # Verify native execution works
    print("Native execution result:", computation(*inp))
    
    # Test Case 1: Passing namedtuple directly as inputs
    # This mimics the PyTorch bug scenario where strict=False fails with a NamedTuple input.
    # Note: tf.compat.v1.tpu.rewrite expects a list of tensors for 'inputs'.
    # Passing a namedtuple directly tests the API's handling of non-list iterables.
    print("\nAttempting rewrite with namedtuple input...")
    try:
        # In PyTorch: export(M(), inp, strict=False)
        # In TF: rewrite(computation, inputs=inp)
        result = tf.compat.v1.tpu.rewrite(computation, inputs=inp)
        print("Rewrite with namedtuple input succeeded:", result)
    except Exception as e:
        print(f"Rewrite with namedtuple input failed: {type(e).__name__}: {e}")

    # Test Case 2: Workaround - converting to a standard list of tensors
    # This mimics the PyTorch workaround of unpacking the namedtuple.
    print("\nAttempting rewrite with list input (workaround)...")
    try:
        # In PyTorch: export(M(), (), inp_kwargs) or unpacking args
        # In TF: rewrite(computation, inputs=[inp.x, inp.y])
        inp_list = [inp.x, inp.y]
        result = tf.compat.v1.tpu.rewrite(computation, inputs=inp_list)
        print("Rewrite with list input succeeded:", result)
    except Exception as e:
        print(f"Rewrite with list input failed: {type(e).__name__}: {e}")

if __name__ == "__main__":
    test_namedtuple()
```
