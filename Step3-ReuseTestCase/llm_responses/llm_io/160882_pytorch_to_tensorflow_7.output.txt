import tensorflow as tf
import numpy as np

def test_tf_tril_shape_mismatch():
    """
    Adapted test case based on PyTorch Issue 160882.
    Verifies if tf.experimental.numpy.tril handles shape changes 
    within a compiled graph (tf.function) similarly to the reported 
    torch.compile behavior with torch.complex.
    """
    
    # Define the function using the target API: tf.experimental.numpy.tril
    # Note: torch.complex takes two inputs (real, imag), while tril takes one (m).
    # We adapt the function signature accordingly.
    @tf.function
    def f(m: tf.Tensor) -> tf.Tensor:
        return tf.experimental.numpy.tril(m)

    B, F, T = 1, 641, 39

    # Create source input
    # PyTorch equivalent: torch.randn(B, F, T)
    src = tf.random.normal((B, F, T))

    # Create mismatched input (permuted)
    # PyTorch equivalent: r_src.permute(0, 2, 1)
    mismatch = tf.transpose(src, perm=[0, 2, 1])

    # 1. Trace/Compile with source inputs
    # This establishes the initial graph shape
    try:
        out_src = f(src)
        print(f"First call (src) successful. Shape: {out_src.shape}")
    except Exception as e:
        print(f"First call failed: {e}")
        raise

    # 2. Call with mismatched inputs
    # This attempts to reproduce the crash described in the bug report
    # where the compiled graph received different input shapes.
    try:
        out_mismatch = f(mismatch)
        print(f"Second call (mismatch) successful. Shape: {out_mismatch.shape}")
    except Exception as e:
        print(f"Second call failed with error: {e}")
        # In the original bug, this resulted in an AssertionError
        raise

if __name__ == "__main__":
    test_tf_tril_shape_mismatch()
