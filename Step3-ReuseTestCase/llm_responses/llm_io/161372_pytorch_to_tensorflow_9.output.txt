import tensorflow as tf

def test_tf_name_scope_dynamic_shapes():
    """
    Adapted from PyTorch Issue 161372 (torch.compile regression).
    
    The original bug involved torch.compile failing due to a tensor size mismatch
    (expected 77, actual 78) within a cache, triggering a recompile limit error.
    
    This test verifies that tf.name_scope (a context manager similar in usage pattern
    to compilation modes) correctly handles operations on tensors with dynamic 
    shapes without raising errors or failing to scope the operations.
    """
    
    # Simulate the specific size mismatch reported in the bug (77 vs 78)
    input_sizes = [77, 78]
    
    for size in input_sizes:
        # Enter the name scope context
        with tf.name_scope("dynamic_layer"):
            # Create tensors with the current dynamic size
            x = tf.ones([size])
            mask = tf.cast(x > 0.5, tf.bool) # Simulate mask logic
            value = tf.ones_like(x)
            a = tf.zeros_like(x)
            
            # Replicate the torch.where logic found in the original traceback
            # r = torch.where(mask, value, a)
            r = tf.where(mask, value, a)
            
            # Assertions to verify correct behavior
            # 1. Check that the operation was created inside the scope
            assert r.op.name.startswith("dynamic_layer"), \
                f"Operation name '{r.op.name}' does not start with scope 'dynamic_layer'"
            
            # 2. Check that the tensor shape matches the input (no size mismatch errors)
            assert r.shape[0] == size, \
                f"Output size mismatch: expected {size}, got {r.shape[0]}"
            
            # 3. Verify the computation result
            assert tf.reduce_all(r == 1.0), "Computation result is incorrect"

if __name__ == "__main__":
    test_tf_name_scope_dynamic_shapes()
    print("Test passed: tf.name_scope handles dynamic shapes correctly.")
