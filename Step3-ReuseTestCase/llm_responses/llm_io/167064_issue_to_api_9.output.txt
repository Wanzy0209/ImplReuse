import tensorflow as tf
from tensorflow.python.framework.experimental import context_stack

def test_sparse_softmax_cross_entropy_context_isolation():
    """
    Test that tf.nn.sparse_softmax_cross_entropy_with_logits does not
    modify the global execution context.
    
    This test is derived from a PyTorch issue (ID: 167064) where calling
    torch.compile inadvertently modified global distribution validation
    settings (torch.distributions.Distribution.set_default_validate_args).
    
    The similar TensorFlow API (tf.nn.sparse_softmax_cross_entropy_with_logits)
    interacts with the global context stack (context.get_default()). This test
    ensures that calling this API does not produce unintended side effects on
    the global context, preserving the isolation expected from a mathematical
    operation.
    """
    # Setup inputs
    logits = tf.constant([[1.0, 2.0], [3.0, 4.0]])
    labels = tf.constant([0, 1])

    # Capture the initial global context
    initial_context = context_stack.get_default()

    # Call the API
    loss = tf.nn.sparse_softmax_cross_entropy_with_logits(logits, labels)

    # Capture the context after the call
    final_context = context_stack.get_default()

    # Assert that the context has not been modified
    # This mirrors the concern in the PyTorch bug where global state was changed.
    assert initial_context is final_context, \
        "Global context was modified by sparse_softmax_cross_entropy_with_logits"

    # Verify the output is valid (sanity check)
    assert loss.shape == (2,)
    print("Test passed: Context isolation preserved.")

if __name__ == "__main__":
    test_sparse_softmax_cross_entropy_context_isolation()
