import torch
from collections import namedtuple

# Leveraging the similar API pattern (tf.compat.v1.train.SessionRunValues).
# SessionRunValues is a namedtuple used to structure results. 
# We adapt this pattern to structure the output of our compiled function.
DiagonalRunValues = namedtuple("DiagonalRunValues", ["results", "options"])

def test_fill_diagonal_dynamic_shapes():
    """
    Test case for Issue 162271: Tensor.fill_diagonal_ doesn't support dynamic shapes.
    
    This test verifies that fill_diagonal_ works correctly under torch.compile(dynamic=True).
    It adapts the namedtuple structure from the similar API (SessionRunValues) to 
    organize the function's return values.
    """
    
    # The bug occurs specifically with dynamic=True
    @torch.compile(dynamic=True)
    def fill_diagonal_op(x, fill_value):
        # Original bug reproduction logic
        x.fill_diagonal_(fill_value)
        
        # Reusing the pattern from the similar API to return results
        return DiagonalRunValues(results=x, options={"dynamic": True})

    # Test Case 1: Basic square matrix (from the original bug report)
    x = torch.zeros(4, 4)
    run_vals = fill_diagonal_op(x, True)
    
    # Verify the operation succeeded and the structure matches the similar API pattern
    expected = torch.eye(4, dtype=torch.bool)
    assert torch.equal(run_vals.results, expected), "Failed to fill diagonal with boolean True"
    assert isinstance(run_vals, DiagonalRunValues), "Result structure does not match similar API pattern"
    assert run_vals.options["dynamic"] is True

    # Test Case 2: Non-square matrix to ensure robustness with dynamic shapes
    x2 = torch.zeros(3, 5)
    run_vals2 = fill_diagonal_op(x2, 1.0)
    
    expected2 = torch.zeros(3, 5)
    # Manually fill diagonal for verification
    for i in range(min(3, 5)):
        expected2[i, i] = 1.0
        
    assert torch.equal(run_vals2.results, expected2), "Failed to fill diagonal on non-square matrix"

if __name__ == "__main__":
    test_fill_diagonal_dynamic_shapes()
    print("Test passed successfully.")
