import torch
from torch.testing._internal.common_utils import DebugMode

# Test case for torch.any under torch.compile with DebugMode
# Based on Issue 164143: DebugMode silently disables torch.compile

def test_any_compile_debug_mode():
    # Define a function using the similar API: torch.any
    def fn(x):
        return torch.any(x > 0)

    # Compile the function
    # The bug report suggests that DebugMode (a non-infra torch dispatch mode)
    # causes torch.compile to skip compilation silently.
    compiled_fn = torch.compile(fn)

    input_tensor = torch.tensor([0.0, -1.0, 2.0])

    # Execute inside DebugMode to verify the interaction.
    # We expect the function to run correctly (and ideally compile).
    with DebugMode():
        result = compiled_fn(input_tensor)
        
        # Verify the result is correct
        expected = torch.tensor(True)
        assert torch.equal(result, expected), f"Expected {expected}, but got {result}"

if __name__ == "__main__":
    test_any_compile_debug_mode()
