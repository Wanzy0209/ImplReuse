import torch
import torch.nn.functional as F

# Adapted from the context of nn.Linear(5, 7) in the original issue
# Input: Batch size 1, 5 input features
input_tensor = torch.randn(1, 5)
# Weight: 7 output features, 5 input features
weight_tensor = torch.randn(7, 5)
# Bias: 7 output features
bias_tensor = torch.randn(7)

# Call the similar API: torch.nn.functional.linear
output = F.linear(input_tensor, weight_tensor, bias_tensor)

# Verify the output shape matches the expected output features (7)
assert output.shape == (1, 7), f"Expected shape (1, 7), got {output.shape}"
