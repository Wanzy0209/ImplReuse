import tensorflow as tf

# Adapted test case for tf.math.multiply_no_nan
# Mimicking the GPU allocation context of the original torch.ones bug
try:
    # Explicitly target the GPU to mirror the 'device="cuda"' argument
    with tf.device('/GPU:0'):
        # Create input tensors on the GPU
        x = tf.ones([1])
        y = tf.ones([1])
        
        # Execute the similar API
        result = tf.math.multiply_no_nan(x, y)
        
        # Verify the result (equivalent to .item() in PyTorch)
        print(result.numpy())
except Exception as e:
    # Catch potential CUDA/GPU errors similar to the original report
    print(f"Error: {e}")
