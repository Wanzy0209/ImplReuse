import torch

def test_norm_with_sliced_tensor_mps():
    """
    Test case to verify if torch.norm handles sliced tensors on MPS correctly,
    similar to the crash scenario reported for repeat_interleave.
    """
    if not torch.backends.mps.is_available():
        print("MPS device is not available. Skipping test.")
        return

    # Create a tensor on MPS device
    # Mimicking the data setup from the original bug report
    data = torch.arange(3, device="mps")

    # Slice the tensor (non-prefix slice)
    # In the original bug: counts[1:3] caused a segfault in repeat_interleave
    sliced_data = data[1:3]

    # Call the similar API (torch.norm) with the sliced tensor
    # This verifies if torch.norm handles sliced MPS tensors without crashing
    try:
        result = torch.norm(sliced_data)
        
        # Basic assertions to ensure valid output
        assert isinstance(result, torch.Tensor)
        assert result.device.type == "mps"
        assert result.item() >= 0  # Norm should be non-negative
        
        print(f"Test passed. Result: {result}")
    except Exception as e:
        print(f"Test failed with exception: {e}")
        raise

if __name__ == "__main__":
    test_norm_with_sliced_tensor_mps()
