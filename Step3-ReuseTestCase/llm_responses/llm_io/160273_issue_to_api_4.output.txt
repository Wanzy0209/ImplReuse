import tensorflow as tf
import numpy as np

def test_concatenate_gradient_consistency():
    """
    Test to verify gradient behavior of tf.experimental.numpy.concatenate
    with different axis arguments, mirroring the structure of the torch.min
    issue to check for consistency.
    """
    
    # Case 1: tf.experimental.numpy.concatenate -> default axis (-1)
    # Mirroring: torch.min(input) -> reduce over all dimensions (default behavior)
    print("Testing gradient with default axis...")
    a = tf.ones([2, 3])
    b = tf.ones([2, 3])
    
    with tf.GradientTape() as tape:
        tape.watch([a, b])
        # Default axis is -1
        concat_val = tf.experimental.numpy.concatenate((a, b))
        loss = tf.reduce_sum(concat_val)
        
    grads = tape.gradient(loss, [a, b])
    
    # Unlike torch.min (global), concatenate should pass gradients 1-to-1.
    # We assert that gradients are 1.0 for all elements in both tensors.
    assert np.allclose(grads[0].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[0].numpy()}"
    assert np.allclose(grads[1].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[1].numpy()}"
    print("  Gradients for default axis: PASSED")

    # Case 2: tf.experimental.numpy.concatenate(..., axis=0) -> specified axis
    # Mirroring: torch.min(input, dim=...) -> reduce over specified dimension
    print("Testing gradient with specified axis (0)...")
    a = tf.ones([2, 3])
    b = tf.ones([2, 3])
    
    with tf.GradientTape() as tape:
        tape.watch([a, b])
        concat_val = tf.experimental.numpy.concatenate((a, b), axis=0)
        loss = tf.reduce_sum(concat_val)
        
    grads = tape.gradient(loss, [a, b])
    
    # Gradients should remain 1-to-1 regardless of axis specification.
    assert np.allclose(grads[0].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[0].numpy()}"
    assert np.allclose(grads[1].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[1].numpy()}"
    print("  Gradients for specified axis: PASSED")

    # Case 3: tf.experimental.numpy.concatenate(..., axis=-2) -> negative axis handling
    # Mirroring the logic in the similar API snippet: if axis < 0: axis %= rank
    print("Testing gradient with negative axis (-2)...")
    a = tf.ones([2, 3])
    b = tf.ones([2, 3])
    
    with tf.GradientTape() as tape:
        tape.watch([a, b])
        # axis=-2 corresponds to axis=0 for a rank-2 tensor
        concat_val = tf.experimental.numpy.concatenate((a, b), axis=-2)
        loss = tf.reduce_sum(concat_val)
        
    grads = tape.gradient(loss, [a, b])
    
    assert np.allclose(grads[0].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[0].numpy()}"
    assert np.allclose(grads[1].numpy(), np.ones((2, 3))), f"Expected ones, got {grads[1].numpy()}"
    print("  Gradients for negative axis: PASSED")

if __name__ == "__main__":
    test_concatenate_gradient_consistency()
    print("\nAll tests passed.")
