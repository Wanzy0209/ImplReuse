import tensorflow as tf
from packaging import version
import pytest

def test_keras_backend_floatx_version_parsing():
    """
    Test case adapted from the torch.version.hip parsing issue (Issue ID: 166068).
    
    The original issue demonstrates that packaging.version.parse() fails on 
    torch.version.hip because the string format is invalid (e.g., '6.4.43482-0f2d60242').
    
    This test applies the same logic to the similar API tf.keras.backend.floatx.
    Since tf.keras.backend.floatx returns a data type string (e.g., 'float32') 
    rather than a version number, we expect packaging.version.parse to raise 
    an InvalidVersion error, confirming the string is not a version.
    """
    # Get the string from the similar API
    backend_floatx = tf.keras.backend.floatx()
    
    # Preserve the original bug reproduction logic: attempting to parse the string
    # as a version using packaging.version.parse.
    # We assert that this raises InvalidVersion because 'float32' is not a version.
    with pytest.raises(version.InvalidVersion):
        version.parse(backend_floatx)
