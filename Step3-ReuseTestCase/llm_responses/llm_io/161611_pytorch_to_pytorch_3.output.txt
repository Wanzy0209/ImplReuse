import torch

def test_linspace_redundant_dtype_conversion():
    """
    Test case adapted from Issue 161611 for torch.linspace.
    Verifies that specifying dtype in torch.linspace makes subsequent .to() calls redundant,
    similar to the behavior observed with torch.zeros in the original bug report.
    """
    # Define a specific dtype to test
    target_dtype = torch.float32

    # 1. Create a tensor using torch.linspace with the target dtype
    # This mirrors the original code: attn_bias = torch.zeros(..., dtype=query.dtype)
    tensor = torch.linspace(0, 10, steps=5, dtype=target_dtype)

    # 2. Verify the dtype is correct immediately after creation
    assert tensor.dtype == target_dtype, \
        f"Expected dtype {target_dtype} after creation, got {tensor.dtype}"

    # 3. Perform an in-place operation that does not change the dtype
    # This mirrors: attn_bias.masked_fill_(temp_mask.logical_not(), float("-inf"))
    mask = torch.tensor([True, False, True, False, True])
    tensor.masked_fill_(mask, float("-inf"))

    # 4. Verify the dtype is still correct after the in-place operation
    assert tensor.dtype == target_dtype, \
        f"Expected dtype {target_dtype} after in-place op, got {tensor.dtype}"

    # 5. Simulate the redundant conversion found in the bug report
    # Original bug: attn_bias.to(query.dtype) # Redundant! Already query.dtype, and result not assigned
    # We execute it here to verify it doesn't change the tensor state or dtype.
    tensor.to(target_dtype)

    # 6. Final assertion to confirm the tensor remains unchanged and dtype is correct
    assert tensor.dtype == target_dtype, \
        "Dtype check failed after redundant .to() call"
    
    # Verify values to ensure the in-place operation worked as expected
    expected = torch.tensor([float("-inf"), 2.5, float("-inf"), 7.5, float("-inf")], dtype=target_dtype)
    assert torch.equal(tensor, expected), "Tensor values do not match expected output"

if __name__ == "__main__":
    test_linspace_redundant_dtype_conversion()
    print("Test passed: torch.linspace respects initial dtype, making .to() redundant.")
