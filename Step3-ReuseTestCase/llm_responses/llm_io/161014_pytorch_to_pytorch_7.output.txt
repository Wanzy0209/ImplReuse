import torch
import numpy as np

def test_gammainc_negative_inputs():
    """
    Adapted test case based on Issue 161014 (Inconsistent constant_pad_nd behavior with negative padding).
    
    This test verifies that torch.special.gammainc handles negative inputs consistently.
    According to the documentation, if either input is negative, the output should be NaN.
    We check that this behavior is consistent and does not raise unexpected RuntimeErrors.
    """
    
    # Case 1: Negative 'input' tensor
    # Analogous to negative padding values reducing size
    input_neg = torch.tensor([-1.0, -2.0, -5.0])
    other_pos = torch.tensor([1.0, 1.0, 1.0])
    
    # Should return NaN, not crash
    result = torch.special.gammainc(input_neg, other_pos)
    assert torch.isnan(result).all(), "Expected NaN for negative 'input' values"
    
    # Case 2: Negative 'other' tensor
    input_pos = torch.tensor([1.0, 2.0, 3.0])
    other_neg = torch.tensor([-1.0, -2.0, -3.0])
    
    result = torch.special.gammainc(input_pos, other_neg)
    assert torch.isnan(result).all(), "Expected NaN for negative 'other' values"
    
    # Case 3: Mixed negative and positive inputs (checking consistency)
    # Ensures the function handles element-wise operations correctly without crashing
    input_mixed = torch.tensor([-1.0, 0.5, 2.0])
    other_mixed = torch.tensor([1.0, -1.0, 2.0])
    
    result = torch.special.gammainc(input_mixed, other_mixed)
    # input=-1 -> NaN
    assert torch.isnan(result[0]), "Expected NaN for input=-1"
    # other=-1 -> NaN
    assert torch.isnan(result[1]), "Expected NaN for other=-1"
    # both positive -> valid result
    assert not torch.isnan(result[2]), "Expected valid result for positive inputs"

    print("torch.special.gammainc negative input tests passed.")

if __name__ == "__main__":
    test_gammainc_negative_inputs()
