```python
import torch
import pytest

def test_lu_factor_singular_matrix():
    """
    Test that torch.linalg.lu_factor raises a RuntimeError for singular matrices.
    This verifies the fix for the issue where MPS backend did not raise an error
    unlike the CPU backend.
    """
    # Define a singular matrix (determinant is 0)
    # [[1, 2], [2, 4]] -> 1*4 - 2*2 = 0
    t = torch.tensor([[1.0, 2.0], [2.0, 4.0]])

    # Check CPU behavior (Expected behavior)
    with pytest.raises(RuntimeError, match="U\\[2,2\\] is zero"):
        torch.linalg.lu_factor(t)

    # Check MPS behavior if available (Bug fix verification)
    if torch.backends.mps.is_available():
        t_mps = t.to("mps")
        with pytest.raises(RuntimeError, match="U\\[2,2\\] is zero"):
            torch.linalg.lu_factor(t_mps)
```
