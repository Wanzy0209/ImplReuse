import tensorflow as tf

def test_enable_debug_mode_basic_pattern():
    """
    Test case for tf.data.experimental.enable_debug_mode.
    
    This test mirrors the logic of the PyTorch MemPool bug (Issue 167745),
    specifically focusing on the basic pattern of configuring an execution 
    environment and performing operations within that scope.
    
    While PyTorch uses a context manager for scoping, enable_debug_mode 
    applies globally. This test verifies that the mode can be enabled 
    and subsequent dataset operations execute correctly.
    """
    # Setup: Enable debug mode
    # This is analogous to entering the torch.cuda.use_mem_pool context
    tf.data.experimental.enable_debug_mode()

    # Action: Create a dataset and perform operations
    # This mirrors the tensor allocation (torch.randn) inside the PyTorch context
    dataset = tf.data.Dataset.from_tensor_slices([1, 2, 3])

    def map_func(x):
        # In debug mode, transformations are forced to execute synchronously/deterministically
        return x + 10

    dataset = dataset.map(map_func)

    # Verification: Iterate and check results
    # This ensures the operation completed successfully within the configured mode
    results = []
    for item in dataset:
        results.append(item.numpy())

    # Assertions
    assert len(results) == 3, "Dataset iteration did not produce expected number of items"
    assert results == [11, 12, 13], f"Expected [11, 12, 13], got {results}"
    
    print("Test passed: Debug mode enabled and dataset processed successfully.")

if __name__ == "__main__":
    test_enable_debug_mode_basic_pattern()
