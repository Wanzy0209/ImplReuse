```python
import tensorflow as tf
import numpy as np

def addcmul_func(x, y, z):
    return x + (y * z)

# Create tensors
# Note: The original bug report targeted Intel XPU. This test uses the default device
# to ensure the code is runnable in a standard TensorFlow environment, 
# while preserving the logic of the original test case.
x = tf.random.normal((128,))
y = tf.random.normal((128,))
z = tf.random.normal((128,))

# 1. Run in eager mode (baseline)
out = addcmul_func(x, y, z)
print("eager mode passed")

# 2. Run inside the target API: tf.keras.backend.name_scope
# This API acts as a context manager to group operations, similar to how 
# torch.compile wraps the function execution in the original report.
with tf.keras.backend.name_scope("test_scope"):
    out_scoped = addcmul_func(x, y, z)
    print("name_scope passed")

# Verify that the results are consistent
assert np.allclose(out.numpy(), out_scoped.numpy()), "Outputs differ between eager and scoped execution"
```
