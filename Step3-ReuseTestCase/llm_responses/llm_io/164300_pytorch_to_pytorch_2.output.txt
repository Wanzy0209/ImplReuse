```python
import torch
import torch.nn as nn
import torch.nn.utils.prune as prune
import functools

# Define a simple model for testing
class TestModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = nn.Linear(10, 10)

model = TestModel()

# Define a custom pruning function that mimics the structure of the original bug's context_fn
# The original bug used a function that took a policy object.
def custom_pruning_method(module, name, amount):
    prune.l1_unstructured(module, name=name, amount=amount)

# Create a functools.partial, similar to the bug report
# Original: context_fn1 = functools.partial(create_selective_checkpoint_contexts, CustomPolicy())
pruner = functools.partial(custom_pruning_method, amount=0.5)

# Parameters to prune
parameters = ((model.linear, 'weight'),)

# Call the similar API: torch.nn.utils.prune.global_unstructured
# Original API: torch.utils.checkpoint.checkpoint(..., context_fn=context_fn1)
prune.global_unstructured(
    parameters,
    pruning_method=pruner
)

# Assertions to verify the behavior
assert hasattr(model.linear, 'weight_mask'), "Pruning mask was not created"
assert torch.sum(model.linear.weight_mask == 0) > 0, "No weights were pruned"
assert torch.sum(model.linear.weight_mask == 1) > 0, "All weights were pruned"

print("Test passed.")
```
