```python
import tensorflow as tf
import numpy as np

# Set seeds for reproducibility
np.random.seed(0)
tf.random.set_seed(0)

def foo(input):
    # Adapted from torch.nn.functional.interpolate to tf.quantization.quantize_and_dequantize
    # We use a fixed range based on the input generation (0 to 10)
    qdq = tf.quantization.quantize_and_dequantize(
        input,
        input_min=0.0,
        input_max=10.0,
        signed_input=True,
        num_bits=8,
        range_given=True,
        round_mode="HALF_TO_EVEN",
        narrow_range=False,
        axis=None
    )
    # Preserve the post-processing logic from the original test
    squeeze = tf.squeeze(qdq, axis=0)
    argmin = tf.argmin(squeeze, axis=1)
    return argmin

# Generate input data (matching original shape and range)
# Note: tf.quantization.quantize_and_dequantize requires float32 input
x = np.random.uniform(0, 10, size=(1, 40, 1, 1)).astype(np.float32)

# Run in eager mode
eager_res = foo(tf.constant(x))

# Run in graph/compiled mode (equivalent to torch.compile)
cfoo = tf.function(foo)
compile_res = cfoo(tf.constant(x))

# Assert consistency
np.testing.assert_array_equal(eager_res.numpy(), compile_res.numpy())
```
