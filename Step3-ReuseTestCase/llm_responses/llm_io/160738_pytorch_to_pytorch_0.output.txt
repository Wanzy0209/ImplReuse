import torch

def test_var_zero_dim():
    """
    Test case for Issue 160738:
    Verifies that torch.var can handle dim=0 on a zero-dimensional input tensor.
    """
    # Test on CPU (Expected behavior)
    x_cpu = torch.tensor(3.0)
    try:
        out_cpu = torch.var(x_cpu, dim=0)
        # Variance of a single value is undefined/NaN
        assert torch.isnan(out_cpu), "CPU: Expected NaN for variance of a single value"
        print(f"CPU Test Passed: output={out_cpu}")
    except Exception as e:
        print(f"CPU Test Failed: {e}")
        raise

    # Test on MPS (Bug location)
    if torch.backends.mps.is_available():
        x_mps = torch.tensor(3.0, device="mps")
        try:
            out_mps = torch.var(x_mps, dim=0)
            # If the bug is fixed, this should execute and return NaN
            assert torch.isnan(out_mps), "MPS: Expected NaN for variance of a single value"
            print(f"MPS Test Passed: output={out_mps}")
        except RuntimeError as e:
            # This is the error reported in the bug: "var_mps: reduction dim must be in the range of input shape"
            print(f"MPS Test Failed (Bug Reproduced): {e}")
            raise
    else:
        print("MPS device not available. Skipping MPS test.")

if __name__ == "__main__":
    test_var_zero_dim()
