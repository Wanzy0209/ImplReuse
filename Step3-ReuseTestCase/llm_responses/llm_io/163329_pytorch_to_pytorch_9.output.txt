```python
import torch
import torch.distributed as dist
import torch.multiprocessing as mp
import os

def setup(rank, world_size):
    os.environ['MASTER_ADDR'] = 'localhost'
    os.environ['MASTER_PORT'] = '12355'
    # Initialize the process group
    dist.init_process_group("gloo", rank=rank, world_size=world_size)

def cleanup():
    dist.destroy_process_group()

def test_broadcast_object_list(rank, world_size):
    setup(rank, world_size)

    # Adapted from original bug report: setting logs
    torch._logging.set_logs(recompiles=True)

    # Prepare data: Rank 0 holds the configuration, others hold None
    if rank == 0:
        # Simulating pipeline configuration or state
        object_list = [{"prompt": "A cat holding a sign", "height": 1024, "width": 1024}]
    else:
        object_list = [None]

    # Call the similar API: torch.distributed.broadcast_object_list
    dist.broadcast_object_list(object_list, src=0)

    # Verification
    expected_data = {"prompt": "A cat holding a sign", "height": 1024, "width": 1024}
    assert object_list[0] == expected_data, f"Rank {rank} received incorrect data: {object_list[0]}"
    
    print(f"Rank {rank}: Test passed. Received: {object_list[0]}")

    cleanup()

if __name__ == "__main__":
    world_size = 2
    mp.spawn(test_broadcast_object_list, args=(world_size,), nprocs=world_size, join=True)
```
