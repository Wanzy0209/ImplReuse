import torch
import sys

# Reproduce the configuration settings from the original bug report
# which seemed to trigger the eager/compile divergence.
torch._dynamo.config.capture_scalar_outputs = True
torch._dynamo.config.capture_dynamic_output_shape_ops = True
torch._inductor.config.emulate_precision_casts = True

def foo(arg0, arg1):
    # arg0: size=(2, 128, 17, 358), dtype=bfloat16, device=cuda
    # Adjusted size to align with the output of repeat_interleave
    t0 = arg0 
    t1 = t0.max(dim=0).values # size=(128, 17, 358)
    t2 = t1.transpose(1, 0)   # size=(17, 128, 358), non-contiguous stride
    
    # arg1: size=(17, 64, 358), dtype=float32, device=cuda
    t3 = arg1
    t4 = torch.exp(t3)        # size=(17, 64, 358)
    
    # Replace torch.nn.functional.conv1d with torch.repeat_interleave
    # We repeat along dim=1 to match the shape of t2 (128) for the multiplication later.
    # Original conv1d output was (17, 261, 358). 
    # Here we map 64 -> 128 via repeats=2 to maintain structural similarity.
    t7 = torch.repeat_interleave(t4, repeats=2, dim=1) # size=(17, 128, 358)
    
    t8 = t7.clone()
    t8.zero_() # size=(17, 128, 358)
    
    t9 = t2 * t7 * t8 # size=(17, 128, 358)
    output = t9
    return output

if __name__ == '__main__':
    # Check for CUDA availability as the original issue was device-specific
    device = 'cuda' if torch.cuda.is_available() else 'cpu'
    if device == 'cpu':
        print("Warning: CUDA not available, falling back to CPU. The original issue occurred on CUDA.")

    # Adjusted input shapes to accommodate the repeat_interleave logic
    # while preserving the mixed precision (bfloat16/float32) characteristics.
    arg0 = torch.rand([2, 128, 17, 358], dtype=torch.bfloat16, device=device, requires_grad=True)
    arg1 = torch.rand([17, 64, 358], dtype=torch.float32, device=device, requires_grad=True)

    print("Testing Eager Execution...")
    try:
        out_eager = foo(arg0, arg1)
        out_eager.sum().backward()
        print('Eager Success! ✅')
    except Exception as e:
        print(f'Eager Failed! ❌: {e}')
        sys.exit(1)

    print("\nTesting Compiled Execution...")
    try:
        compiled_foo = torch.compile(foo, fullgraph=True, dynamic=True)
        out_compiled = compiled_foo(arg0, arg1)
        out_compiled.sum().backward()
        print('Compile Success! ✅')
    except Exception as e:
        print(f'Compile Failed! ❌: {e}')
        sys.exit(1)
