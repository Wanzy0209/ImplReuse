import torch

def test_torch_full_compile_float64_caching():
    """
    Test case for Issue 166253:
    torch.full gives wrong result with torch.compile and dtype=torch.float64.
    
    The bug causes the value argument of torch.full to be fixed to the value 
    from the first call after torch.compile when using float64 on CPU.
    This test verifies that subsequent calls with different values produce 
    the correct results.
    """
    # Define the function to be tested
    def func(x):
        return torch.full((2, ), x, dtype=torch.float64)

    # Compile the function using torch.compile
    compiled_func = torch.compile(func)

    # First call with value 5.0
    x1 = torch.tensor(5.0, dtype=torch.float64)
    result1 = compiled_func(x1)
    expected1 = torch.tensor([5.0, 5.0], dtype=torch.float64)
    
    # Verify the first call works as expected
    assert torch.equal(result1, expected1), \
        f"First call failed: expected {expected1}, got {result1}"

    # Second call with value 10.0
    # In the bug report, this incorrectly returned [5.0, 5.0]
    x2 = torch.tensor(10.0, dtype=torch.float64)
    result2 = compiled_func(x2)
    expected2 = torch.tensor([10.0, 10.0], dtype=torch.float64)
    
    # Verify the second call updates the value correctly
    assert torch.equal(result2, expected2), \
        f"Second call failed (Bug likely present): expected {expected2}, got {result2}"

    print("Test passed: torch.full correctly handles dynamic values with torch.compile and float64.")

if __name__ == "__main__":
    test_torch_full_compile_float64_caching()
