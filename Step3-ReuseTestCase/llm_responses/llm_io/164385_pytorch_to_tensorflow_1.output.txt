import tensorflow as tf
import numpy as np

def test_he_normal_initializer():
    """
    Adapted test case for tf.keras.initializers.HeNormal.
    
    The original issue (164385) concerned symbolic FloorDiv operations 
    generating incorrect Rational types in PyTorch/SymPy. 
    Since tf.keras.initializers.HeNormal is a weight initializer and not 
    a symbolic arithmetic operator, we adapt the test to verify the 
    initializer's behavior using the constants and structure from the 
    original report where applicable.
    """
    
    # Constants derived from the original FloorDiv expression:
    # FloorDiv((24*s37 + 672)*(((s14*s46)//2016)) + 21, 22)
    # We use these to define the shape and seed for the initializer.
    s14_val = 2016
    s37_val = 24
    s46_val = 22
    
    # Define a shape based on the constants in the original expression
    # (24, 672) derived from (24*s37 + 672)
    shape = (s37_val, 672) 
    
    print(f"Testing HeNormal initializer with shape: {shape}")

    # Initialize the HeNormal initializer
    # Using the Issue ID as a seed to ensure reproducibility
    initializer = tf.keras.initializers.HeNormal(seed=164385)
    
    # Call the initializer to generate weights
    weights = initializer(shape=shape)
    
    print(f"Generated weights shape: {weights.shape}")
    print(f"Generated weights dtype: {weights.dtype}")
    
    # Assertions
    assert isinstance(weights, tf.Tensor), "Output should be a TensorFlow Tensor"
    assert weights.shape == shape, f"Expected shape {shape}, but got {weights.shape}"
    assert weights.dtype == tf.float32, f"Expected dtype float32, but got {weights.dtype}"
    
    # HeNormal draws from a truncated normal distribution. 
    # We check that the values are not all zero (basic sanity check).
    assert not tf.reduce_all(weights == 0.0).numpy(), "Weights should not be all zeros"
    
    # Check that the standard deviation is approximately what we expect 
    # (stddev = sqrt(2 / fan_in)). fan_in = 672.
    # stddev = sqrt(2 / 672) ≈ 0.054
    expected_stddev = np.sqrt(2.0 / 672)
    actual_stddev = tf.math.reduce_std(weights).numpy()
    
    # Allow some tolerance for the random sample
    assert np.isclose(actual_stddev, expected_stddev, atol=0.01), \
        f"Expected stddev ~{expected_stddev}, but got {actual_stddev}"

    print("HeNormal test passed successfully.")

if __name__ == "__main__":
    test_he_normal_initializer()
