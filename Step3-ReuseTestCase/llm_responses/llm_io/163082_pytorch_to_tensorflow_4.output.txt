import tensorflow as tf
import numpy as np

def test_tf_keras_ops_real():
    """
    Adapted test case for tf.keras.ops.real based on the structure of the 
    PyTorch normalize bug report.
    
    Original Bug: torch.nn.functional.normalize outputs vectors with norm>1 
    with torch.compile + cuda.
    
    Target API: tf.keras.ops.real (Returns the real part of a complex (or real) tensor).
    
    This test verifies the behavior of tf.keras.ops.real using the same input 
    values and execution patterns (compiled vs uncompiled, device usage) as the 
    original report.
    """
    
    # Check for GPU availability
    gpus = tf.config.list_physical_devices('GPU')
    device = '/GPU:0' if gpus else '/CPU:0'
    print(f"Using device: {device}")

    # Original input values from the PyTorch bug report
    input_data = [[3.799999, 0.0, 0.0]]

    with tf.device(device):
        # Create tensor
        c = tf.constant(input_data, dtype=tf.float32)
        print("Input vector:", c.numpy()[0])

        # Mimic torch.compile using tf.function
        @tf.function
        def get_real_part_compiled(input_tensor):
            return tf.keras.ops.real(input_tensor)

        def get_real_part_uncompiled(input_tensor):
            return tf.keras.ops.real(input_tensor)

        # Run compiled version
        xyz_compiled = get_real_part_compiled(c)
        print("Real part (tf.function):", xyz_compiled.numpy()[0])

        # Run uncompiled version
        xyz_uncompiled = get_real_part_uncompiled(c)
        print("Real part (no tf.function):", xyz_uncompiled.numpy()[0])

        # Verification
        # Since the input is real, tf.keras.ops.real should return the tensor unchanged.
        # We check if the compiled and uncompiled results match the input.
        assert np.allclose(xyz_compiled.numpy(), c.numpy()), \
            f"Compiled output {xyz_compiled.numpy()} does not match input {c.numpy()}"
        
        assert np.allclose(xyz_uncompiled.numpy(), c.numpy()), \
            f"Uncompiled output {xyz_uncompiled.numpy()} does not match input {c.numpy()}"

        # Additionally test with a complex input to ensure the API works as intended
        # (Adapting the logic to cover the primary use case of the API)
        c_complex = tf.constant([[3.799999 + 1.0j, 0.0, 0.0]], dtype=tf.complex64)
        real_part_complex = get_real_part_compiled(c_complex)
        expected_real = np.array([[3.799999, 0.0, 0.0]], dtype=np.float32)
        
        assert np.allclose(real_part_complex.numpy(), expected_real), \
            f"Complex real part extraction failed: {real_part_complex.numpy()} vs {expected_real}"
        
        print("Test passed.")

if __name__ == "__main__":
    test_tf_keras_ops_real()
