```python
import torch

def test_quantized_index_select():
    # Create a quantized tensor
    quant_input = torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 3.0, 4.0]), scale=1.0, zero_point=0, dtype=torch.quint8)
    
    # Define indices to select
    index = torch.tensor([0, 2])
    
    # Attempt to select indices using the similar API
    # Replaces the original torch.zeros_like(quant_input) call
    out_tensor = torch.index_select(quant_input, dim=0, index=index)
    
    print("Quantized input:", quant_input)
    print("Output tensor:", out_tensor)
    
    # Verify the output is quantized and contains the expected values
    assert out_tensor.is_quantized, "Output tensor should be quantized"
    expected_dequantized = torch.tensor([1.0, 3.0])
    assert torch.allclose(out_tensor.dequantize(), expected_dequantized), "Values do not match"

if __name__ == "__main__":
    test_quantized_index_select()
```
