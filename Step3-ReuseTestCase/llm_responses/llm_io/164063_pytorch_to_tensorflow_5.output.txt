import numpy as np

def foo(arg0, arg1):
    # Adapted from the original PyTorch test case logic
    # Original: t0 = arg0 (bfloat16)
    t0 = arg0
    
    # Original: t1 = t0.reshape((28, 24, 3, 127))
    # Reshaping to match the specific dimensions that triggered the original bug
    t1 = tf.reshape(t0, (28, 24, 3, 127))

    # Adapted: t2 = tf.experimental.numpy.gcd(t1, arg1)
    # Testing the similar API (gcd) with the same tensor shapes and dtype (bfloat16)
    # to see if it exhibits similar type handling issues in eager vs compiled modes.
    t2 = tf.experimental.numpy.gcd(t1, arg1)
    
    return t2
