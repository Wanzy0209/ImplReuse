Issue ID: 160630
Title: torch.zeros_like not implemented for QuantizedCPU tensors
Bug Description:
### 🐛 Describe the bug

torch.zeros_like() fails when used with a QuantizedCPU tensor. The operator aten::zero_ is not implemented for the QuantizedCPU backend, causing any operation that tries to create a zeroed tensor for quantized inputs to fail.
import torch

def test_quantized_zeros_like():
    # Create a quantized tensor
    quant_input = torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 3.0]), scale=1.0, zero_point=0, dtype=torch.quint8)
    
    # Attempt to create a zeroed tensor with the same shape
    out_tensor = torch.zeros_like(quant_input)  # <-- This triggers the error
    
    print("Quantized input:", quant_input)
    print("Output tensor:", out_tensor)

if __name__ == "__main__":
    test_quantized_zeros_like()
