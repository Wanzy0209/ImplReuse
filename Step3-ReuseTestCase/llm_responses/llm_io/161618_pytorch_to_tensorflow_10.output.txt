import tensorflow as tf
import numpy as np

# Adapted from the PyTorch test case dimensions
m = 20120
k = 1536
n = 512

def test_range_input_producer():
    """
    Test case adapted from PyTorch Inductor/Triton bug report (Issue 161618).
    
    Original Logic:
    - Setup large tensors (m=20120, k=1536, n=512).
    - Run a compiled operation (addmm) using these tensors.
    
    Adapted Logic for tf.compat.v1.train.range_input_producer:
    - Use the dimension 'm' as the limit for the range producer.
    - Verify the producer can generate the input range correctly.
    - Note: The specific Inductor/Triton failure is not applicable to TensorFlow,
      but the data scale and execution flow are preserved.
    """
    
    # range_input_producer is a TF v1 API and requires graph mode (disabled eager execution)
    tf.compat.v1.disable_eager_execution()

    with tf.compat.v1.Session() as sess:
        # Initialize local variables required by range_input_producer if num_epochs is used
        sess.run(tf.compat.v1.local_variables_initializer())
        sess.run(tf.compat.v1.global_variables_initializer())

        # Create the range input producer
        # We map the first dimension 'm' from the original matrix multiplication 
        # to the 'limit' of the range producer.
        # We disable shuffle to ensure deterministic behavior similar to the original test.
        queue = tf.compat.v1.train.range_input_producer(
            limit=m,
            shuffle=False,
            capacity=32
        )

        # Dequeue a batch to verify functionality
        # This corresponds to the execution step 'compiled(a, mat1, mat2)' in the original test.
        dequeue_op = queue.dequeue()
        
        # Run the operation
        result = sess.run(dequeue_op)

        # Assertions to verify correct behavior
        assert isinstance(result, (np.int32, int)), f"Expected int, got {type(result)}"
        assert 0 <= result < m, f"Value {result} out of bounds [0, {m})"
        
        print(f"Test passed. Dequeued value: {result} (Limit: {m})")

if __name__ == "__main__":
    test_range_input_producer()
