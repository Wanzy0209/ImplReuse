import tensorflow as tf

# Check for GPU availability to match the original test's context (device="cuda")
gpus = tf.config.list_physical_devices('GPU')
device_name = '/GPU:0' if gpus else '/CPU:0'

print(f"Running test on device: {device_name}")

with tf.device(device_name):
    # Create inputs. We use tf.ones to maintain a link to the original torch.ones call.
    # The original bug involved a tensor of size 1.
    x = tf.ones(1, dtype=tf.float32)
    y = tf.constant(0.0, dtype=tf.float32)

    # Call the similar API: tf.compat.v1.math.multiply_no_nan
    # This computes x * y, but returns 0 if y is 0 (even if x is NaN).
    result = tf.compat.v1.math.multiply_no_nan(x, y)

    # Print the result, mimicking the .item() call in the original test case
    print("Result:", result.numpy()[0])

    # Verify the specific behavior of the API
    # Expected: 1.0 * 0.0 = 0.0
    assert result.numpy()[0] == 0.0, "Expected 0.0 when multiplying by 0"

    # Additional verification for non-zero case
    y_non_zero = tf.constant(5.0, dtype=tf.float32)
    result_non_zero = tf.compat.v1.math.multiply_no_nan(x, y_non_zero)
    assert result_non_zero.numpy()[0] == 5.0, "Expected 5.0 for 1.0 * 5.0"

    print("Test passed.")
