import tensorflow as tf
import tf.experimental.numpy as tnp

# Enable numpy behavior for tf.experimental.numpy
tnp.experimental_enable_numpy_behavior()

def test_standard_normal_shape_preservation():
    """
    Adapted from PyTorch all_gather memory ordering bug report.
    
    Original Bug Logic:
    1. Create a tensor with a specific memory format (channels_last).
    2. Perform operation (all_gather).
    3. Verify output memory format matches input.
    
    Adapted Logic for tf.experimental.numpy.random.standard_normal:
    1. Define a specific shape (analogous to the tensor structure in the bug).
    2. Perform operation (standard_normal generation).
    3. Verify output shape matches the requested shape (data integrity).
    """
    
    # Define the shape similar to the original bug report (2, 2, 2, 2)
    # In the original bug, this was a 4D tensor where memory format (channels_last) was significant.
    size = (2, 2, 2, 2)
    
    # Call the target API: tf.experimental.numpy.random.standard_normal
    # This generates a tensor with the given shape.
    x = tnp.random.standard_normal(size=size)
    
    # Verify the output properties.
    # The original bug checked if the gathered tensor matched the original tensor's storage.
    # Here, we check if the generated tensor matches the requested shape.
    print(f"Input size: {size}")
    print(f"Output shape: {x.shape}")
    print(f"Shape matches: {x.shape == size}")
    
    # Assertion to ensure the API behaves correctly regarding shape preservation
    assert x.shape == size, f"Shape mismatch: expected {size}, got {x.shape}"
    
    # Verify dtype (standard_normal defaults to float)
    expected_dtype = tf.float32
    print(f"Output dtype: {x.dtype}")
    assert x.dtype == expected_dtype, f"Dtype mismatch: expected {expected_dtype}, got {x.dtype}"

    # Print a sample of the data to verify generation
    print(f"Generated tensor sample:\n{x}")

if __name__ == "__main__":
    test_standard_normal_shape_preservation()
