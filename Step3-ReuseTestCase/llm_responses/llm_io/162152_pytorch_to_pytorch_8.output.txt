```python
import torch
import torch.nn as nn
import torch.nn.utils.prune as prune

class SimpleModel(nn.Module):
    def __init__(self, input_size=10, hidden_size=20, output_size=5):
        super(SimpleModel, self).__init__()
        self.linear1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.linear2 = nn.Linear(hidden_size, output_size)

    def forward(self, x):
        x = self.linear1(x)
        x = self.relu(x)
        x = self.linear2(x)
        return x

# Instantiate the model
model = SimpleModel()

# Define parameters to prune globally
parameters_to_prune = [
    (model.linear1, 'weight'),
    (model.linear2, 'weight'),
]

# Apply global unstructured pruning
# This replaces the DataParallel wrapping logic from the original test case
prune.global_unstructured(
    parameters_to_prune,
    pruning_method=prune.L1Unstructured,
    amount=0.2
)

# Verify that the pruning masks have been created
assert hasattr(model.linear1, 'weight_mask'), "Mask not found for linear1.weight"
assert hasattr(model.linear2, 'weight_mask'), "Mask not found for linear2.weight"

# Verify that pruning actually occurred (some weights should be zeroed out)
# Global pruning calculates the threshold based on all parameters together,
# so we check if any mask contains zeros.
linear1_zeros = torch.sum(model.linear1.weight_mask == 0).item()
linear2_zeros = torch.sum(model.linear2.weight_mask == 0).item()

assert (linear1_zeros > 0) or (linear2_zeros > 0), "Pruning did not zero out any weights"

print("success")
```
