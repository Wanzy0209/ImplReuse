import tensorflow as tf
import warnings

def test_rocm_build_check_no_warnings():
    """
    Test case adapted from the PyTorch deprecation warning spam issue.
    
    This test verifies that calling the backend-related API 
    tf.test.is_built_with_rocm does not trigger unwanted deprecation 
    warnings or 'spam', similar to the behavior reported for 
    torch.set_float32_matmul_precision.
    """
    # Capture warnings to detect any 'spam' similar to the original bug report
    with warnings.catch_warnings(record=True) as w:
        warnings.simplefilter("always")

        # Call the similar API (tf.test.is_built_with_rocm)
        # This mirrors the original logic: import -> call backend API
        is_rocm_built = tf.test.is_built_with_rocm()

        # 1. Verify the API returns the expected type
        assert isinstance(is_rocm_built, bool), \
            f"Expected bool, got {type(is_rocm_built)}"

        # 2. Verify no warnings were triggered (addressing the "spam" issue)
        # If warnings are present, we raise an error detailing them, 
        # mimicking the user's complaint about useless warnings.
        if len(w) > 0:
            warning_messages = [str(warning.message) for warning in w]
            raise AssertionError(
                f"API call triggered {len(w)} unwanted warning(s) (spam): {warning_messages}"
            )

if __name__ == '__main__':
    test_rocm_build_check_no_warnings()
