import tensorflow as tf
import numpy as np

def test_stop_gradient_large_tensors():
    """
    Adapted test case for tf.keras.ops.stop_gradient based on the 
    torch.addmm bug report (Issue 167727).
    
    The original bug involved incorrect results for large complex tensors 
    on the MPS device. This test verifies that tf.keras.ops.stop_gradient 
    maintains numerical consistency (identity behavior) between GPU and CPU 
    for both small and large complex tensors.
    """
    
    # Check for GPU availability
    gpus = tf.config.list_physical_devices('GPU')
    device_name = "/GPU:0" if gpus else "/CPU:0"
    print(f"Running test on: {device_name}")

    # Test Case 1: Small tensor (Corresponds to the 'success' case in the bug report)
    # Original dimensions: (64, 300)
    print("Testing small tensor (64, 300)...")
    a_small = tf.complex(
        tf.random.uniform((64, 300)), 
        tf.random.uniform((64, 300))
    )
    
    with tf.device(device_name):
        out_gpu_small = tf.keras.ops.stop_gradient(a_small)
        
    with tf.device("/CPU:0"):
        out_cpu_small = tf.keras.ops.stop_gradient(a_small)

    # stop_gradient should return the tensor values unchanged.
    # We compare GPU result vs CPU result to ensure consistency.
    np.testing.assert_allclose(
        out_gpu_small.numpy(), 
        out_cpu_small.numpy(), 
        rtol=1e-5, 
        atol=1e-5,
        err_msg="Small tensor test failed: GPU and CPU results mismatch."
    )
    print("Small tensor test passed.")

    # Test Case 2: Large tensor (Corresponds to the 'fails' case in the bug report)
    # Original dimensions involved (64, 10000). We use a similar large dimension.
    print("Testing large tensor (64, 10000)...")
    a_large = tf.complex(
        tf.random.uniform((64, 10000)), 
        tf.random.uniform((64, 10000))
    )

    with tf.device(device_name):
        out_gpu_large = tf.keras.ops.stop_gradient(a_large)
        
    with tf.device("/CPU:0"):
        out_cpu_large = tf.keras.ops.stop_gradient(a_large)

    # Verify that the large tensor handling is consistent between GPU and CPU
    np.testing.assert_allclose(
        out_gpu_large.numpy(), 
        out_cpu_large.numpy(), 
        rtol=1e-5, 
        atol=1e-5,
        err_msg="Large tensor test failed: GPU and CPU results mismatch."
    )
    print("Large tensor test passed.")

if __name__ == "__main__":
    test_stop_gradient_large_tensors()
