import tensorflow as tf
import tensorflow.experimental.numpy as tnp

def test_vdot_with_sliced_tensor():
    """
    Adapts the setup from the PyTorch bug report (Issue 167924) 
    to test the similar API tf.experimental.numpy.vdot.
    
    The original bug involved a crash on MPS when using repeat_interleave 
    with a sliced tensor. This test verifies the behavior of vdot 
    with the same tensor shapes and slicing pattern.
    """
    # Setup tensors similar to the original bug report
    # Original: counts = torch.tensor([0, 1, 0], device="mps")
    counts = tnp.array([0, 1, 0])
    
    # Original: data = torch.arange(2, device="mps")
    data = tnp.arange(2)

    # The original bug logic involved slicing the counts tensor: counts[1:3]
    # This results in a tensor [1, 0]
    sliced_counts = counts[1:3]

    # Apply the similar API: tf.experimental.numpy.vdot
    # vdot computes the dot product of the flattened inputs.
    # data = [0, 1], sliced_counts = [1, 0]
    # Expected dot product: (0 * 1) + (1 * 0) = 0
    result = tnp.vdot(data, sliced_counts)

    # Verify the result
    assert result == 0, f"Expected 0, but got {result}"
    print("Test passed: vdot handled the sliced tensor correctly.")

if __name__ == "__main__":
    test_vdot_with_sliced_tensor()
