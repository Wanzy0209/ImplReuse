```python
import sys
import platform
import tensorflow as tf

print("Python:", sys.version)
print("Executable:", sys.executable)
print("TensorFlow:", tf.__version__)
print("macOS:", platform.mac_ver()[0])
print("platform:", platform.platform())
print("arch:", platform.machine())

# Check for GPU support (analogous to is_built)
# In TensorFlow, we check if physical devices are detected or if built with CUDA support.
# On macOS with Metal support, this checks if the plugin/devices are visible.
gpus = tf.config.list_physical_devices('GPU')
print("GPUs found (Built/Supported):", len(gpus) > 0)

# Check MirroredStrategy availability (analogous to is_available)
# This verifies if the tf.types.experimental.distributed.Mirrored 
# functionality (via MirroredStrategy) is operational.
try:
    strategy = tf.distribute.MirroredStrategy()
    print("MirroredStrategy available?: True")
    print("Num replicas:", strategy.num_replicas_in_sync)
except Exception as e:
    print("MirroredStrategy available?: False")
    print("Error:", e)
```
