import tensorflow as tf
import numpy as np
import math

class FastLearnedCellX3(tf.keras.Model):
    def __init__(self, D_in, H, D_out,
                 L_w1=12, L_w2=12, L_b2=12,
                 k1=3, k2=3, k3=3, tau1=1.0, tau2=1.0, tau3=1.0,
                 d_addr=64,           # address bottleneck
                 learn_addr=False,
                 learn_tape_w1=True, learn_tape_w2=True, learn_tape_b2=True):
        super().__init__()
        self.D_in, self.H, self.D_out = D_in, H, D_out

        # keep sizes as plain Python ints
        self.L_w1, self.L_w2, self.L_b2 = int(L_w1), int(L_w2), int(L_b2)
        self.k1, self.k2, self.k3 = int(k1), int(k2), int(k3)
        self.t1, self.t2, self.t3 = float(tau1), float(tau2), float(tau3)

        # address bottleneck
        # Equivalent to nn.Linear(D_in, d_addr, bias=False)
        self.P = tf.keras.layers.Dense(d_addr, use_bias=False, 
                                       kernel_initializer=tf.random_normal_initializer(stddev=1.0/math.sqrt(D_in)))
        if not learn_addr:
            self.P.trainable = False

        def init_U(L, d, learn):
            U = tf.random.normal((L, d))
            U = U - tf.reduce_mean(U, axis=1, keepdims=True)
            U = U / (tf.norm(U, axis=1, keepdims=True) + 1e-8)
            return tf.Variable(U, trainable=learn)

        # three unembeddings in addr-space
        self.U1 = init_U(self.L_w1, d_addr, learn_addr)
        self.U2 = init_U(self.L_w2, d_addr, learn_addr)
        self.U3 = init_U(self.L_b2, d_addr, learn_addr)

        # value tapes
        # F.normalize(torch.randn(...), dim=...)
        def init_tape(shape, learn):
            val = tf.random.normal(shape)
            # Normalize along specified dimensions
            # PyTorch dim=(1,2) for W1 [L, H, D_in] -> normalize H and D_in
            # PyTorch dim=1 for b2 [L, D_out] -> normalize D_out
            if len(shape) == 3:
                val = tf.linalg.l2_normalize(val, axis=(1,2))
            elif len(shape) == 2:
                val = tf.linalg.l2_normalize(val, axis=1)
            return tf.Variable(val, trainable=learn)

        self.W1 = init_tape((self.L_w1, H, D_in), learn_tape_w1)
        self.W2 = init_tape((self.L_w2, D_out, H), learn_tape_w2)
        self.b2 = init_tape((self.L_b2, D_out), learn_tape_b2)

    @staticmethod
    def _tk(z, k, tau):
        # tf.math.top_k returns values and indices
        topv, topi = tf.math.top_k(z, k=k, sorted=True)
        w = tf.nn.softmax(topv / (tau + 1e-8), axis=1)
        return topi, w

    def _address(self, x_addr):
        # fused logits for all three heads
        # U_pack = torch.cat([self.U1, self.U2, self.U3], dim=0)
        U_pack = tf.concat([self.U1, self.U2, self.U3], axis=0)          # [Ltot, d]
        
        # Z = x_addr @ U_pack.t()
        Z = tf.matmul(x_addr, U_pack, transpose_b=True)                 # [N, Ltot]
        
        s1, s2, s3 = self.L_w1, self.L_w2, self.L_b2
        # z1, z2, z3 = torch.split(Z, (s1, s2, s3), dim=1)
        z1, z2, z3 = tf.split(Z, [s1, s2, s3], axis=1)
        
        i1, w1 = FastLearnedCellX3._tk(z1, self.k1, self.t1)
        i2, w2 = FastLearnedCellX3._tk(z2, self.k2, self.t2)
        i3, w3 = FastLearnedCellX3._tk(z3, self.k3, self.t3)
        return (i1, w1), (i2, w2), (i3, w3)

    @staticmethod
    def _apply_mixture(x_flat, topi, weights, W):
        """
        Mix-then-apply (avoids replicating x):
        x_flat : [N, in]
        topi   : [N, k]
        weights: [N, k]
        W      : [L, out, in]
        return : [N, out]
        """
        N = tf.shape(topi)[0]
        k = tf.shape(topi)[1]
        
        # Equivalent to torch.index_select or gather
        # W_selected shape: [N, k, out, in]
        W_selected = tf.gather(W, topi, axis=0)
        
        # Expand weights to [N, k, 1, 1] to broadcast over W_selected
        weights_exp = tf.expand_dims(tf.expand_dims(weights, -1), -1)
        
        # Weighted sum: [N, out, in]
        W_mix = tf.reduce_sum(W_selected * weights_exp, axis=1)
        
        # Apply: x_flat [N, in] @ W_mix.T [in, out] -> [N, out]
        # Using einsum for clarity: 'ni,noi->no'
        return tf.einsum('ni,noi->no', x_flat, W_mix)

    def call(self, x):
        # x: [N, D_in]
        x_addr = self.P(x) # [N, d_addr]
        
        (i1, w1), (i2, w2), (i3, w3) = self._address(x_addr)
        
        # Layer 1
        # x is [N, D_in], W1 is [L, H, D_in]
        h = self._apply_mixture(x, i1, w1, self.W1) # [N, H]
        h = tf.keras.activations.gelu(h)
        
        # Layer 2
        # h is [N, H], W2 is [L, D_out, H]
        y = self._apply_mixture(h, i2, w2, self.W2) # [N, D_out]
        
        # Bias
        # b2 is [L, D_out]
        b = self._apply_mixture(tf.ones_like(y), i3, w3, self.b2) # [N, D_out]
        
        return y + b

def test_fast_learned_cell_rmsprop():
    # Parameters
    D_in, H, D_out = 128, 64, 10
    batch_size = 32
    
    # Instantiate Model
    model = FastLearnedCellX3(D_in, H, D_out)
    
    # Instantiate Optimizer (Similar API: tf.keras.optimizers.RMSprop)
    optimizer = tf.keras.optimizers.RMSprop(learning_rate=0.001)
    
    # Loss function
    loss_fn = tf.keras.losses.MeanSquaredError()
    
    # Dummy data
    x = tf.random.normal((batch_size, D_in))
    y_true = tf.random.normal((batch_size, D_out))
    
    # Training step
    with tf.GradientTape() as tape:
        y_pred = model(x, training=True)
        loss = loss_fn(y_true, y_pred)
        
    gradients = tape.gradient(loss, model.trainable_variables)
    optimizer.apply_gradients(zip(gradients, model.trainable_variables))
    
    # Assertions to verify behavior
    # 1. Check output shape
    assert y_pred.shape == (batch_size, D_out), f"Output shape mismatch: {y_pred.shape}"
    
    # 2. Check that loss is computed
    assert loss.numpy() > 0, "Loss should be positive"
    
    # 3. Check that gradients exist for trainable variables
    # (Note: P is frozen if learn_addr=False, so it shouldn't be in trainable_variables)
    assert len(model.trainable_variables) > 0, "Model should have trainable variables"
    
    # 4. Verify optimizer updated variables
    # We can't easily check the exact values, but we can ensure the step ran without error.
    print("Test passed: RMSprop successfully updated the FastLearnedCellX3 model.")

if __name__ == "__main__":
    test_fast_learned_cell_rmsprop()
