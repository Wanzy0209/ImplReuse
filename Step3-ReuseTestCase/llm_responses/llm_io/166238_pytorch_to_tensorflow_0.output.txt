import tensorflow as tf
from collections import defaultdict
import unittest

class TestTPURewriteDefaultDict(unittest.TestCase):
    """
    Test case to verify the behavior of tf.compat.v1.tpu.rewrite when 
    the computation function creates a collections.defaultdict.
    Adapted from PyTorch Dynamo issue regarding defaultdict tracing.
    """

    @classmethod
    def setUpClass(cls):
        # Initialize TPU system if available to make the test runnable
        cls.has_tpu = False
        try:
            resolver = tf.distribute.cluster_resolver.TPUClusterResolver()
            tf.config.experimental_connect_to_cluster(resolver)
            tf.tpu.experimental.initialize_tpu_system(resolver)
            cls.strategy = tf.distribute.TPUStrategy(resolver)
            cls.has_tpu = True
        except (ValueError, tf.errors.NotFoundError, tf.errors.InternalError):
            # No TPU found or connection failed
            pass

    def test_defaultdict_in_computation(self):
        if not self.has_tpu:
            self.skipTest("TPU hardware not available, skipping tf.compat.v1.tpu.rewrite test")

        # Define the computation that includes the problematic logic
        # from the PyTorch bug: creating a defaultdict inside the compiled function.
        def computation_fn(x):
            # This line caused "Unsupported function call" in PyTorch Dynamo
            d = defaultdict(list)
            d['key'].append(x)
            return x + 1

        # Prepare inputs for the rewrite
        # inputs is a list of lists of tensors
        inputs = [[tf.constant(1.0)]]

        with self.strategy.scope():
            # Execute the API under test
            # We expect this to either succeed (handle the python object) 
            # or fail with a specific error if it cannot trace defaultdict.
            try:
                result = tf.compat.v1.tpu.rewrite(computation_fn, inputs)
                # If successful, verify we got a result
                self.assertIsNotNone(result)
            except Exception as e:
                # Catching the error to mimic the PyTorch failure report
                # PyTorch raised: torch._dynamo.exc.Unsupported
                self.fail(
                    f"tf.compat.v1.tpu.rewrite failed to trace function with defaultdict. "
                    f"Error: {e}"
                )

if __name__ == '__main__':
    unittest.main()
