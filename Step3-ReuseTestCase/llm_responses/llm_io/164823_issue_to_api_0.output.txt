```python
import tensorflow as tf

# This test case mirrors the structure of the PyTorch bug report (Eager vs Compiled)
# but applies it to the tf.control_dependencies API to verify its behavior
# in both execution modes.

class TestModel(tf.Module):
    def __init__(self):
        super().__init__()
        self.v = tf.Variable(0.0)

    def forward(self, x):
        # Use control_dependencies to ensure the variable update happens before the multiplication.
        # This is the "similar API" usage pattern.
        with tf.control_dependencies([self.v.assign_add(1.0)]):
            result = x * 2
        return result

x = tf.random.normal((10, 10))

model = TestModel()

# 1. Eager execution
print("Eager output:", model.forward(x))
print("Eager variable state:", model.v.numpy())

# 2. Compiled execution (tf.function is analogous to torch.compile)
compiled_forward = tf.function(model.forward)
print("Compiled output:", compiled_forward(x))
print("Compiled variable state:", model.v.numpy())

# Verify that the side effect (dependency) was respected in the compiled graph
assert model.v.numpy() == 2.0
```
