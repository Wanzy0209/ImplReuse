import tensorflow as tf
from packaging import version
import pytest

def test_tpu_device_order_mode_version_parsing():
    """
    Test case adapted from Issue 166068 (ROCm version parsing).
    
    Original issue: packaging.version.parse(torch.version.hip) failed due to 
    an invalid version string '6.4.43482-0f2d60242'.
    
    This test applies the same logic (parsing a string representation of an API attribute)
    to the similar API: tf.tpu.experimental.DeviceOrderMode.
    
    Since DeviceOrderMode is an IntEnum, its string representation (e.g., "DeviceOrderMode.AUTO")
    is not a valid PEP 440 version. We expect InvalidVersion to be raised, 
    confirming the parser behaves correctly for this non-version string.
    """
    # Access the similar API
    device_order_mode = tf.tpu.experimental.DeviceOrderMode

    # Reproduce the logic: attempt to parse the string representation of the API members
    for mode in device_order_mode:
        mode_str = str(mode)
        # We expect this to fail because Enum names are not versions
        with pytest.raises(version.InvalidVersion):
            version.parse(mode_str)

if __name__ == "__main__":
    test_tpu_device_order_mode_version_parsing()
