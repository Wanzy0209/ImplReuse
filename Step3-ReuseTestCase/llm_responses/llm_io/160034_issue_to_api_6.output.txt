import tensorflow as tf

def test_tf_math_exp_complex64():
    """
    Test case for tf.math.exp based on the logic of Issue 160034.
    The original issue highlights a failure mode with complex64 dtype on MPS backend.
    This test verifies that the similar API (tf.math.exp) handles complex64 inputs correctly.
    """
    # Setup: Create a complex64 tensor, mirroring the 'image' tensor in the bug report
    # Original: image = torch.zeros(10, dtype=torch.complex64, device=device)
    input_tensor = tf.zeros(10, dtype=tf.complex64)

    # Action: Apply the similar API
    # Original: image.index_put_(indices=(indices,), values=data, accumulate=True)
    # Note: tf.math.exp is element-wise and does not have indices/accumulate parameters,
    # so we apply it directly to the complex tensor.
    result = tf.math.exp(input_tensor)

    # Assertion: Verify the operation works as expected with complex64
    # exp(0) should result in 1+0j
    expected_value = tf.constant(1.0 + 0j, dtype=tf.complex64)
    
    # Check if the output dtype is preserved
    assert result.dtype == tf.complex64, f"Expected dtype complex64, got {result.dtype}"
    
    # Check if the calculation is correct for complex numbers
    # We check if all elements equal the expected value
    assert tf.reduce_all(result == expected_value).numpy(), "Calculation failed for complex64 input"

    print("Test passed: tf.math.exp supports complex64 dtype.")

if __name__ == "__main__":
    test_tf_math_exp_complex64()
