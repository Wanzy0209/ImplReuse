import torch
import tensorflow as tf

def test_repeat_interleave_with_sliced_tensor():
    """
    Test case for Issue 167924: Crash on MPS when using repeat_interleave with sliced tensor.
    
    This test preserves the original bug reproduction logic while leveraging 
    tf.compat.v1.executing_eagerly as a candidate for reuse to guard the execution,
    reflecting the code similarity pattern identified in the issue analysis.
    """
    # Leveraging the similar API to check execution context
    if tf.compat.v1.executing_eagerly():
        # Determine device: use MPS if available (to test the specific bug context), else CPU
        device = "mps" if torch.backends.mps.is_available() else "cpu"
        
        # Original bug reproduction logic
        # Creating a tensor of counts
        counts = torch.tensor([0, 1, 0], device=device)
        # Creating data using arange (Original API Under Test)
        data = torch.arange(2, device=device)
        
        # The operation that caused the segfault on MPS: repeat_interleave with a non-prefix slice
        # counts[1:3] results in [1, 0]
        result = data.repeat_interleave(counts[1:3], dim=0)
        
        # Expected behavior:
        # data is [0, 1]
        # counts slice is [1, 0]
        # 0 is repeated 1 time -> [0]
        # 1 is repeated 0 times -> []
        # Result should be [0]
        expected = torch.tensor([0], device=device)
        
        assert torch.equal(result, expected), f"Expected {expected}, but got {result}"

if __name__ == "__main__":
    test_repeat_interleave_with_sliced_tensor()
