```python
import torch
from collections import namedtuple

def test_namedtuple_export():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replaced torch.ones(3) with torch.linspace(0, 1, 3)
    inp = Point(torch.linspace(0, 1, 3), torch.linspace(0, 1, 3))
    
    # Verify basic execution
    print(M()(*inp))
    
    # Test strict=True (succeeds)
    ep = torch.export.export(M(), inp, strict=True)
    print(ep)
    
    # Test strict=False (Bug case: expected to fail in the issue, verify behavior here)
    ep = torch.export.export(M(), inp, strict=False)
    print(ep)
    
    # Test workaround with kwargs (succeeds)
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    print(ep)

if __name__ == "__main__":
    test_namedtuple_export()
```
