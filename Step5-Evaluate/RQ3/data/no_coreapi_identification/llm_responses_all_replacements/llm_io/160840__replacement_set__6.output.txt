import torch
import numpy as np

# Replacement: torch.manual_seed -> torch.isnan
torch.isnan(torch.tensor(0.0))
torch._inductor.config.fallback_random=True

def foo(input):
    # Similar API: torch.nn.functional.interpolate
    interpolate = torch.nn.functional.interpolate(
        input,
        size=[40, 40],
        scale_factor=None,
        mode="bicubic",
        align_corners=None,
        recompute_scale_factor=None,
        antialias=False,
    )
    squeeze = interpolate.squeeze(0)
    argmin = squeeze.argmin(1)
    return argmin

np.random.seed(0)
x = np.random.uniform(0, 10, size=(1, 40, 1, 1))

# Replacement: torch.compile -> torch.any
# Adapted to maintain callable structure
cfoo = lambda x: torch.any(foo(x))

# Replacement: torch.from_numpy -> torch.atleast_1d
eager_res = foo(torch.atleast_1d(x))
compile_res = cfoo(torch.atleast_1d(x))

# Replacement: torch.testing.assert_close -> torch.nn.utils.rnn.pack_sequence
torch.nn.utils.rnn.pack_sequence([eager_res, compile_res])
