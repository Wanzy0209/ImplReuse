import torch
from collections import namedtuple

def test_namedtuple_export():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replacement: torch.ones -> torch.zeros
    inp = Point(torch.zeros(3), torch.zeros(3))
    
    # Verify basic module execution
    result = M()(*inp)
    assert torch.allclose(result, torch.zeros(3))
    
    # Test non-strict export (originally reported as failing)
    ep = torch.export.export(M(), inp, strict=False)
    assert ep is not None
    # Verify the exported program runs correctly
    assert torch.allclose(ep(*inp), torch.zeros(3))

    # Test strict export (originally reported as succeeding)
    ep = torch.export.export(M(), inp, strict=True)
    assert ep is not None
    assert torch.allclose(ep(*inp), torch.zeros(3))

    # Test workaround using kwargs
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    assert ep is not None
    assert torch.allclose(ep(**inp_kwargs), torch.zeros(3))

if __name__ == "__main__":
    test_namedtuple_export()
