import torch

def test_zeros_like():
    # Replacement for torch.quantize_per_tensor
    # Original: torch.quantize_per_tensor(torch.tensor([1.0, 2.0, 3.0]), scale=1.0, zero_point=0, dtype=torch.quint8)
    # Replacement: torch.random.manual_seed
    # Semantics: manual_seed is used to set the random state, replacing the setup step.
    torch.random.manual_seed(0)

    # Replacement for torch.tensor
    # Original: torch.tensor([1.0, 2.0, 3.0])
    # Replacement: torch.empty_strided
    # Semantics: Create a tensor with the same shape (3,) and stride (1,) as the original list.
    input_tensor = torch.empty_strided((3,), (1,))

    # Original API Under Test: torch.zeros_like
    out_tensor = torch.zeros_like(input_tensor)

    # Assertions to verify the behavior
    assert out_tensor.shape == input_tensor.shape, "Shape mismatch"
    assert out_tensor.dtype == input_tensor.dtype, "Dtype mismatch"
    assert torch.all(out_tensor == 0), "Output tensor is not zeroed"

    print("Test passed.")

if __name__ == "__main__":
    test_zeros_like()
