```python
import torch

def test_var(device):
    # Replaced torch.tensor with torch.empty_strided as requested.
    # A zero-dimensional tensor has size () and stride ().
    x = torch.empty_strided((), (), device=device)
    # Initialize the tensor to match the original test case value of 3.0
    x.fill_(3.0)
    
    try:
        output = torch.var(x, dim=0)
        print(f"var test succeeds for device: {device}. output: {output}")
    except Exception as e:
        print(f"var test fails for device: {device}: {e}")

if __name__ == "__main__":
    test_var(device="cpu")
    # Only run MPS test if the device is available
    if torch.backends.mps.is_available():
        test_var(device="mps")
```
