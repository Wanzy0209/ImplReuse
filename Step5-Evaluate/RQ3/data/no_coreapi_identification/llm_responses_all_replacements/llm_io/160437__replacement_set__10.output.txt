import torch
import torch.nn.functional as F

@torch.compile(backend="eager")
def fn(x, i):
    if i == 1:
        torch._dynamo.graph_break()
    return x + 1

# Replacement: torch.randn -> torch.nn.functional.cosine_embedding_loss
# We need to generate inputs for cosine_embedding_loss to create the 'inp' tensor.
input1 = torch.randn(3)
input2 = torch.randn(3)
target = torch.ones(3)
inp = F.cosine_embedding_loss(input1, input2, target)

# Execute the function to trigger the graph break and subsequent recompilation
fn(inp, 0)
fn(inp, 1)
fn(inp, 2)

# Verify that the function produces the correct output, ensuring the graph is not empty
expected = inp + 1
assert torch.allclose(fn(inp, 0), expected)
assert torch.allclose(fn(inp, 1), expected)
assert torch.allclose(fn(inp, 2), expected)
