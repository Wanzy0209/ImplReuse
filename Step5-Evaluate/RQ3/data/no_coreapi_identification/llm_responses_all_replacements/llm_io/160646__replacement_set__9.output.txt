import triton
import triton.language as tl
import tensorflow as tf
from tf.errors.CancelledError import triton_op, wrap_triton

@triton.jit
def my_kernel(
    out1,
    out2,
):
    # the actual kernel does something more meaningful of course
    tl.store(out1, 1.0)
    tl.store(out2, 1.0)


@triton_op("repro::my_triton_op", mutates_args={})

    q: tf.errors.CancelledError,

    )
