```python
import torch
import torch.nn.functional as F

@torch.compile(backend="eager")
def fn(x, i):
    if i == 1:
        torch._dynamo.graph_break()
    return x + 1

# Original: inp = torch.randn(3)
# Replacement: torch.nn.functional.max_pool2d
# We need a tensor to feed into max_pool2d. Using torch.ones to avoid using torch.randn again.
dummy_input = torch.ones(1, 1, 3, 3)
inp = F.max_pool2d(dummy_input, kernel_size=2)

fn(inp, 0)
fn(inp, 1)
fn(inp, 2)
```
