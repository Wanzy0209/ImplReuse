import torch
from collections import namedtuple

def test_namedtuple():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replacing torch.ones(3) with torch.slogdet
    # torch.slogdet requires a square matrix input.
    # We create a random 3x3 matrix to feed into slogdet.
    # torch.slogdet returns a namedtuple (sign, logabsdet).
    # We take the 'sign' component (index 0) to use as the tensor input.
    mat = torch.randn(3, 3)
    inp = Point(torch.slogdet(mat)[0], torch.slogdet(mat)[0])
    
    print(M()(*inp))
    
    # Bug scenario: Non-strict export
    # According to the bug report, this might fail.
    try:
        ep = torch.export.export(M(), inp, strict=False)
        print("Non-strict export succeeded (Bug might be fixed):")
        print(ep)
    except Exception as e:
        print(f"Non-strict export failed as expected: {e}")

    # Success scenario: Strict export
    ep = torch.export.export(M(), inp, strict=True)
    print("Strict export succeeded:")
    print(ep)

    # Workaround scenario: Convert namedtuple to kwargs
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    print("Kwargs export succeeded:")
    print(ep)

if __name__ == "__main__":
    test_namedtuple()
