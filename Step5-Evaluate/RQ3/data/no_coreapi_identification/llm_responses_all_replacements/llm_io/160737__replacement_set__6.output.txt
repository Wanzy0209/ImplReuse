import torch
import tensorflow as tf

def test_index_select_replaced(device):
    # Replacement 1: torch.ones -> torch.dist
    # Original: x = torch.ones([2, 3], device=device)
    # We use torch.dist to calculate the distance between 0 and 1 (which is 1) to populate the tensor.
    dist_val = torch.dist(torch.tensor(0.0, device=device), torch.tensor(1.0, device=device))
    x = torch.zeros([2, 3], device=device) + dist_val

    # Replacement 2: torch.tensor -> tf.linalg.diag
    # Original: index = torch.tensor(1, device=device)
    # We use tf.linalg.diag to create a diagonal matrix [[1]] and convert it to a scalar PyTorch tensor.
    index_tf = tf.linalg.diag([1])
    index = torch.from_numpy(index_tf.numpy()).to(device).reshape(())

    try:
        output = torch.index_select(x, dim=0, index=index)
        
        # Replacement 3: torch.Size -> tf.keras.backend.exp
        # Original: print(f"... output shape: {output.shape}")
        # We use tf.keras.backend.exp on the shape tensor.
        shape_tf = tf.convert_to_tensor(output.shape, dtype=tf.float32)
        exp_shape = tf.keras.backend.exp(shape_tf)
        
        print(f"index_select test succeeds for device: {device}. output shape (exp): {exp_shape.numpy()}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

# Run the test
test_index_select_replaced(device="cpu")
