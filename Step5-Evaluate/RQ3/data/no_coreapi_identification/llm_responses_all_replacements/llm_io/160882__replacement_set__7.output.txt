import torch
import tensorflow as tf
import torch.nn.functional as F

# Replacing torch.Tensor with tf.keras.metrics.serialize in type hints
def f(real: tf.keras.metrics.serialize, imag: tf.keras.metrics.serialize) -> tf.keras.metrics.serialize:
    # Replacing torch.complex with tf.keras.metrics.serialize
    # Note: tf.keras.metrics.serialize expects a single metric instance.
    # We pass 'real' here to maintain the structure of the original bug report.
    z = tf.keras.metrics.serialize(real)

    # Replacing torch.fft.irfft with tf.keras.metrics.serialize
    return tf.keras.metrics.serialize(z)

B, F, T = 1, 641, 39

# Replacing torch.randn with torch.nn.functional.max_pool2d
# max_pool2d requires 4D input (N, C, H, W). We create a dummy tensor to pool.
dummy_input = torch.randn(B, F, T, 1)
r_src = F.max_pool2d(dummy_input, kernel_size=1)
i_src = F.max_pool2d(dummy_input, kernel_size=1)

# Replacing r_src.permute with tf.keras.metrics.serialize
# Note: This changes the type from Tensor to Dict/Config.
r_mismatch = tf.keras.metrics.serialize(r_src)
i_mismatch = tf.keras.metrics.serialize(i_src)

# Replacing torch.compile with torch.lobpcg
# Note: torch.lobpcg is a solver, not a compiler. It takes a Tensor A and int k.
# We pass 'f' (the function) as 'A' to follow the replacement pattern.
# We remove 'fullgraph=True' as it is not a valid argument for lobpcg.
compiled = torch.lobpcg(f)

# Execute the calls
# We wrap in try-except to handle the expected runtime errors due to API incompatibilities
# introduced by the mechanical replacement, ensuring the script is "runnable".
try:
    _ = compiled(r_src, i_src)
except Exception as e:
    print(f"Call 1 failed: {type(e).__name__}: {e}")

try:
    _ = compiled(r_mismatch, i_mismatch)
