import torch
import torch.distributed as dist
import torch.multiprocessing as mp
import torch.nn.functional as F

def test_moe_gradients_replacement(rank, world_size):
    """
    Test case generated based on Issue ID: 160285 (Wrong-size gradients in Expert Parallel MoE).
    
    This test preserves the original bug reproduction logic (comparing gradients with and without 
    Expert Parallel sharding) but replaces the original distributed APIs with 
    torch.nn.functional.layer_norm as requested.
    """
    # Initialize distributed environment
    dist.init_process_group("gloo", rank=rank, world_size=world_size)
    torch.manual_seed(42 + rank)

    # --- Replacing torch.Size with torch.nn.functional.layer_norm ---
    # Original: torch.Size([2, 8, 8])
    # We simulate the shape definition by normalizing a dimension tensor.
    dim_tensor = torch.tensor([2.0, 8.0, 8.0])
    # The result of layer_norm is used to determine the shape for subsequent operations
    computed_shape = F.layer_norm(dim_tensor, [3]).int().tolist()
    
    print(f"Rank {rank}: Computed shape from layer_norm replacement: {computed_shape}")

    # ==================================================
    # TEST 1: Without Expert Parallel Sharding
    # ==================================================
    print(f"Rank {rank}: TEST 1: Without Expert Parallel Sharding")

    # Create input data
    x = torch.randn(computed_shape)

    # --- Replacing TorchGroupedExperts,torch.distributed.tensor.DTensorwithExpertParallelsharding ---
    # with torch.nn.functional.layer_norm
    # This simulates the forward pass of the MoE layer
    output = F.layer_norm(x, computed_shape[1:])

    # --- Replacing loss.backward with torch.nn.functional.layer_norm ---
    # This simulates the backward pass / gradient accumulation
    grad = F.layer_norm(output, computed_shape[1:])

    # --- Replacing p.grad.norm with torch.nn.functional.layer_norm ---
    # This simulates the calculation of the gradient norm
    norm_val = F.layer_norm(grad, computed_shape[1:])

    # --- Replacing param.grad.to_local with torch.nn.functional.layer_norm ---
    # This simulates accessing the local part of the gradient
    local_grad = F.layer_norm(norm_val, computed_shape[1:])

    metric_no_ep = local_grad.norm().item()
    print(f"Rank {rank}: No EP Metric (simulated grad norm): {metric_no_ep:.6f}")

    # ==================================================
    # TEST 2: With Expert Parallel Sharding
    # ==================================================
    print(f"Rank {rank}: TEST 2: With Expert Parallel Sharding")

    # In the original bug, EP sharding caused gradients to double.
    # Here we run the same logic with the replacement API to check behavior.
    # We use the same input to ensure consistency checks are valid.

    # --- Replacing TorchGroupedExperts... with torch.nn.functional.layer_norm ---
    output_ep = F.layer_norm(x, computed_shape[1:])

    # --- Replacing loss.backward (u003eloss.backward) with torch.nn.functional.layer_norm ---
    grad_ep = F.layer_norm(output_ep, computed_shape[1:])

    # --- Replacing p.grad.norm with torch.nn.functional.layer_norm ---
    norm_val_ep = F.layer_norm(grad_ep, computed_shape[1:])

    # --- Replacing param.grad.to_local with torch.nn.functional.layer_norm ---
    local_grad_ep = F.layer_norm(norm_val_ep, computed_shape[1:])

    metric_ep = local_grad_ep.norm().item()
    print(f"Rank {rank}: EP Metric (simulated grad norm): {metric_ep:.6f}")

    # --- Comparison ---
    # The original bug observed: metric_ep == 2 * metric_no_ep
    # With the replacement API (layer_norm), we expect the metrics to be consistent
    # (i.e., not doubled), as layer_norm does not have the EP sharding bug.
    if rank == 0:
        print("\nComparison:")
        print(f"  No EP Metric:   {metric_no_ep:.6f}")
        print(f"  EP Metric:      {metric_ep:.6f}")
        print(f"  Ratio (EP/NoEP): {metric_ep / metric_no_ep:.4f}")
        
        # Assert that the replacement API behaves consistently (ratio is 1.0)
        # unlike the original bug where ratio was 2.0
        assert abs(metric_ep - metric_no_ep) < 1e-6, \
            f"Metrics should match with layer_norm replacement. Got {metric_no_ep} vs {metric_ep}"
        print("Test Passed: Replacement API behaves consistently (no gradient doubling).")

    dist.destroy_process_group()

if __name__ == "__main__":
    world_size = 2
    mp.spawn(test_moe_gradients_replacement, args=(world_size,), nprocs=world_size, join=True)
