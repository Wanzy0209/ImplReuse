import torch
from collections import namedtuple

def test_namedtuple():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replacing torch.ones(3) with torch.qr(torch.randn(3, 3))[0]
    # torch.qr requires a matrix input, so we generate a random 3x3 matrix.
    # We take the first element of the result (Q) to use as the tensor input.
    inp = Point(torch.qr(torch.randn(3, 3))[0], torch.qr(torch.randn(3, 3))[0])
    print(M()(*inp))
    
    # errors (according to the bug report)
    ep = torch.export.export(M(), inp, strict=False)
    print(ep)

    # succeeds (according to the bug report)
    ep = torch.export.export(M(), inp, strict=True)
    print(ep)

    # workaround could be to convert namedtuple to a kwarg
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    print(ep)

if __name__ == "__main__":
    test_namedtuple()
