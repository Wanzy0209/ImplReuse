import tensorflow as tf
import numpy as np

def test_sparse_categorical_crossentropy_sm75_equivalent():
    """
    Test case generated based on Issue ID: 160980.
    
    Original Issue: PyTorch CUDA13 Binary Cannot Be Built with SM_75 with NVSHMEM.
    The original error involved undefined references during the linking phase 
    (nvlink) for specific architectures (sm_75) when using NVSHMEM extensions.
    
    This test replaces the source API (torch_nvshmem/nvshmem_extension.cu) with
    the similar API: tf.keras.metrics.SparseCategoricalCrossentropy.
    
    The logic is translated from a C++ build/linker verification to a Python
    runtime verification. We ensure the metric can be instantiated and 
    correctly processes sparse integer labels (analogous to the specific 
    type handling in the original C++ template errors).
    """
    
    # Instantiate the metric replacing the NVSHMEM extension logic
    metric = tf.keras.metrics.SparseCategoricalCrossentropy()

    # Define sample data
    # y_true represents sparse labels (integers), similar to the specific 
    # integer types (long) seen in the original nvlink error symbols.
    y_true = [1, 2]
    
    # y_pred represents the probability distribution
    y_pred = [[0.05, 0.95, 0], [0.1, 0.8, 0.1]]

    # Update the metric state
    metric.update_state(y_true, y_pred)

    # Calculate the result
    result = metric.result().numpy()

    # Expected calculation:
    # Loss 1: -log(0.95) ≈ 0.0513
    # Loss 2: -log(0.80) ≈ 0.2231
    # Average: (0.0513 + 0.2231) / 2 ≈ 0.1372
    expected_loss = 0.1372

    # Assert that the metric calculates the loss correctly
    assert np.isclose(result, expected_loss, atol=1e-4), \
        f"Expected loss {expected_loss}, but got {result}"

    print("Test passed: tf.keras.metrics.SparseCategoricalCrossentropy executed successfully.")

if __name__ == "__main__":
    test_sparse_categorical_crossentropy_sm75_equivalent()
