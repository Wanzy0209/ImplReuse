import torch
import torch.nn.functional as F

def loss_func(input1, input2, target):
    # Replacing the arithmetic logic with the replacement API: cosine_embedding_loss
    return F.cosine_embedding_loss(input1, input2, target)

# Setup inputs for cosine_embedding_loss
# Since torch.randn is being replaced, we use torch.ones/zeros to create the required tensors.
# cosine_embedding_loss requires input1, input2 (embeddings) and target (labels 1 or -1).
# We keep the device "xpu" as per the original bug report context.
input1 = torch.ones(128, requires_grad=True).to("xpu")
input2 = torch.ones(128, requires_grad=True).to("xpu")
target = torch.ones(128).to("xpu") # Target of 1 means inputs should be similar

# Eager mode execution
out = loss_func(input1, input2, target)
print("eager mode passed")

# Compile the function
loss_func_compiled = torch.compile(loss_func)

# Compiled execution
out = loss_func_compiled(input1, input2, target)
print("torch.compile passed")

# Basic assertion to check output validity
assert out.item() >= 0.0, "Loss should be non-negative"
