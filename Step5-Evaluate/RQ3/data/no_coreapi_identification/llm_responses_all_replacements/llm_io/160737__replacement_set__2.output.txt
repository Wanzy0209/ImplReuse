import torch
import tensorflow as tf

def test_index_select_replaced(device):
    # Replacement: torch.ones -> torch.eye
    x = torch.eye(2, 3, device=device)

    # Replacement: torch.tensor -> torch.isinf
    # Note: torch.isinf returns a boolean tensor.
    index = torch.isinf(torch.tensor(1.0, device=device))

    try:
        output = torch.index_select(x, dim=0, index=index)
        # Replacement: torch.Size -> tf.compat.v1.global_variables_initializer
        # Replacing the usage of the shape object in the print statement
        print(f"index_select test succeeds for device: {device}. output shape: {tf.compat.v1.global_variables_initializer()}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select_replaced(device="cpu")
test_index_select_replaced(device="mps")
