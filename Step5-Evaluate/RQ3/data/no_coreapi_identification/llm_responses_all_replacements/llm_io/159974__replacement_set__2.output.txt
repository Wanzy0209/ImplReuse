```python
import torch
import tensorflow as tf

def addcmul_func(x, y, z):
    return x + (y * z)

# Helper to generate data using the replacement API (TensorFlow) 
# and converting it to PyTorch XPU tensors
def get_xpu_tensor(shape):
    # tf.random.stateless_truncated_normal requires a seed tensor of shape [2]
    seed = tf.constant([1, 2], dtype=tf.int32)
    # Generate tensor using the replacement API
    tf_tensor = tf.random.stateless_truncated_normal(shape, seed=seed)
    # Convert TensorFlow tensor to PyTorch tensor via NumPy and move to XPU
    return torch.from_numpy(tf_tensor.numpy()).to("xpu")

# Generate inputs
x = get_xpu_tensor([128])
y = get_xpu_tensor([128])
z = get_xpu_tensor([128])

# Test eager mode
out = addcmul_func(x, y, z)
assert out is not None
print("eager mode passed")

# Test torch.compile
addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
assert out is not None
print("torch.compile passed")
```
