import torch
import tensorflow as tf

def test_index_select(device):
    # Replacement: torch.ones -> torch.det
    # Original: x = torch.ones([2, 3], device=device)
    # We create a square matrix [2, 3, 3] to compute determinant, resulting in tensor x of shape [2].
    # This preserves the logic of having a tensor x to select from.
    mat = torch.eye(3, device=device).unsqueeze(0).repeat(2, 1, 1)
    x = torch.det(mat)

    # Replacement: torch.tensor -> torch.cummin
    # Original: index = torch.tensor(1, device=device)
    # We use torch.cummin to generate indices. 
    # Input [2.0, 1.0] -> cummin indices [0, 1]. We select the last one to get value 1.
    dummy_input = torch.tensor([2.0, 1.0], device=device)
    _, indices = torch.cummin(dummy_input, dim=0)
    index = indices[1]

    try:
        output = torch.index_select(x, dim=0, index=index)
        
        # Replacement: torch.Size -> tf.math.exp
        # Original: output.shape (which is a torch.Size)
        # We replace the usage of the shape object with tf.math.exp.
        # We convert the shape dimensions to a TF tensor and apply exponential.
        shape_list = list(output.shape)
        tf_shape_tensor = tf.constant(shape_list, dtype=tf.float32)
        exp_shape = tf.math.exp(tf_shape_tensor)
        
        print(f"index_select test succeeds for device: {device}. output shape exp: {exp_shape}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select(device="cpu")
# Check for MPS availability to ensure the test is runnable on non-Mac devices
if torch.backends.mps.is_available():
    test_index_select(device="mps")
