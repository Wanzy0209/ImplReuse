import torch
import tensorflow as tf

def test_index_select(device):
    # Replacement: torch.ones -> torch.zeros
    x = torch.zeros([2, 3], device=device)
    
    # Replacement: torch.tensor -> torch.isnan
    # We use torch.isnan on a scalar element from x to generate the index tensor.
    # We cast to long to ensure compatibility with index_select (runnable requirement).
    index = torch.isnan(x[0, 0]).long()
    
    try:
        output = torch.index_select(x, dim=0, index=index)
        
        # Replacement: torch.Size -> tf.linalg.LinearOperatorScaledIdentity
        # We replace the usage of the shape object (torch.Size) with the similar API.
        # We construct the TF operator using the dimensions from the output.
        tf_op = tf.linalg.LinearOperatorScaledIdentity(num_rows=output.shape[0], multiplier=1.0)
        
        print(f"index_select test succeeds for device: {device}. output shape: {tf_op.shape}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select(device = "cpu")
test_index_select(device = "mps")
