```python
import torch
from collections import namedtuple

def test_namedtuple():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replaced torch.ones(3) with torch.eye(3) as requested
    inp = Point(torch.eye(3), torch.eye(3))
    
    # Verify basic execution
    output = M()(*inp)
    assert torch.allclose(output, 2 * torch.eye(3))
    
    # Test non-strict export (Bug report indicates this might error)
    try:
        ep = torch.export.export(M(), inp, strict=False)
        print("Non-strict export succeeded")
    except Exception as e:
        print(f"Non-strict export failed: {e}")
        # Re-raise to ensure the test fails if the bug is present
        raise
    
    # Test strict export (Bug report indicates this succeeds)
    ep = torch.export.export(M(), inp, strict=True)
    assert ep is not None
    print("Strict export succeeded")
    
    # Test workaround with kwargs
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    assert ep is not None
    print("Kwargs export succeeded")

if __name__ == "__main__":
    test_namedtuple()
```
