```python
import torch
import torch.distributed as dist
import os

def main():
    # Setup
    gpu_id = int(os.environ["LOCAL_RANK"])
    device = f"cuda:{gpu_id}"
    torch.cuda.set_device(device)

    # Initialize the default distributed process group.
    # Note: This is a prerequisite for torch.distributed.broadcast_object_list.
    dist.init_process_group(backend='nccl', device_id=gpu_id)

    # Test torch.distributed.broadcast_object_list
    # This replaces the original dist.barrier() logic to verify the similar API.
    obj_list = [gpu_id]
    dist.broadcast_object_list(obj_list, src=0)

    # Assertion to verify the broadcast worked correctly
    assert obj_list[0] == 0, f"Expected 0, got {obj_list[0]}"

    dist.destroy_process_group()

if __name__ == "__main__":
    main()
```
