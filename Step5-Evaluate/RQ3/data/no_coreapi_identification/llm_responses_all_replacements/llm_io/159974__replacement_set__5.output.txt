```python
import torch
import torch.nn.functional as F

def addcmul_func(x, y, z):
    return x + (y * z)

# Create a dummy input for the pooling operation since adaptive_max_pool3d requires an input tensor.
# Using torch.ones as a base generator since torch.randn is being replaced.
dummy_input = torch.ones(1, 1, 4, 4, 4).to("xpu")

# Replace torch.randn(128).to("xpu") with torch.nn.functional.adaptive_max_pool3d
# The output size is set to (2, 2, 2) to generate valid 5D tensors for the operation.
x = F.adaptive_max_pool3d(dummy_input, output_size=(2, 2, 2))
y = F.adaptive_max_pool3d(dummy_input, output_size=(2, 2, 2))
z = F.adaptive_max_pool3d(dummy_input, output_size=(2, 2, 2))

# Eager mode execution
out = addcmul_func(x, y, z)
assert out is not None
assert out.shape == x.shape
print("eager mode passed")

# Compile mode execution
addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
assert out is not None
assert out.shape == x.shape
print("torch.compile passed")
```
