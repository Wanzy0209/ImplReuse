```python
import torch
from collections import namedtuple

def test_namedtuple_export():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Replacement: torch.ones -> torch.normal
    # Original: torch.ones(3)
    # New: torch.normal(0, 1, (3,))
    inp = Point(torch.normal(0, 1, (3,)), torch.normal(0, 1, (3,)))
    print(M()(*inp))
    
    # errors
    try:
        ep = torch.export.export(M(), inp, strict=False)
        print(ep)
    except Exception as e:
        print(f"strict=False failed: {e}")

    # succeeds
    try:
        ep = torch.export.export(M(), inp, strict=True)
        print(ep)
    except Exception as e:
        print(f"strict=True failed: {e}")

    # workaround could be to convert namedtuple to a kwarg
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    try:
        ep = torch.export.export(M(), (), inp_kwargs)
        print(ep)
    except Exception as e:
        print(f"kwargs export failed: {e}")

if __name__ == "__main__":
    test_namedtuple_export()
```
