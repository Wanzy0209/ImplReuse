import torch
import torch.nn.functional as F

@torch.compile(backend="eager")
def fn(x, i):
    if i == 1:
        torch._dynamo.graph_break()
    return x + 1

# Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
# We create a dummy input tensor for the pooling operation to ensure valid input dimensions.
dummy_input = torch.ones(1, 3, 4, 4)
inp = F.adaptive_max_pool2d(dummy_input, (2, 2))

# Run the function to trigger the graph break and subsequent compilation
fn(inp, 0)

fn(inp, 2)

# Verify the outputs are correct
expected = inp + 1
assert torch.equal(fn(inp, 0), expected)
