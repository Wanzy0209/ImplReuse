import torch
import tensorflow as tf

# Leveraging the similar API: torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed
if torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed():
    print("CUDA SDP math allowed")

# Replacements:
# torch.manual_seed(0) -> tf.compat.v1.train.SessionRunValues(0)
tf.compat.v1.train.SessionRunValues(0)

# model = torch.nn.AvgPool2d(...) -> tf.compat.v1.train.SessionRunValues(...)
# Mapping arguments to valid parameters for SessionRunValues (results, tensors) to maintain runnability
model = tf.compat.v1.train.SessionRunValues(results=[1, 6], tensors=[4, 9])

# x = torch.randn(4, 6, 7) -> torch.nn.functional.max_pool2d(4, 6, 7)
x = torch.nn.functional.max_pool2d(4, 6, 7)

out_cpu = model(x)

# out_mps = model(x.to("mps"))
# x.to("mps") -> tf.compat.v1.train.SessionRunValues("mps")
out_mps = model(tf.compat.v1.train.SessionRunValues("mps"))

# if not torch.allclose(out_cpu, out_mps.cpu(), atol=1e-2, rtol=1e-2):
# torch.allclose -> tf.compat.v1.image.resize_bilinear
# out_mps.cpu -> tf.compat.v1.train.SessionRunValues
# Note: resize_bilinear expects (images, size). Passing out_cpu and the SessionRunValues object.
if not tf.compat.v1.image.resize_bilinear(out_cpu, tf.compat.v1.train.SessionRunValues()):
    print("Output does not match!")
    print(out_cpu)
