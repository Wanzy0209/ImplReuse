import torch
import torch.nn.functional as F

@torch.compile(backend="eager")
def fn(x, i):
    if i == 1:
        torch._dynamo.graph_break()
    return x + 1

# Replacing torch.randn(3) with torch.nn.functional.adaptive_max_pool1d
# We create a dummy input tensor to feed into the pooling layer
dummy_input = torch.ones(1, 1, 5)
# Apply adaptive_max_pool1d to get a tensor of size 3, then squeeze to match original shape (3,)
inp = F.adaptive_max_pool1d(dummy_input, output_size=3).squeeze()

# Execute the function to trigger graph compilation and graph breaks
fn(inp, 0)
fn(inp, 1)
fn(inp, 2)
