```python
import torch
import torch.nn.functional as F

def addcmul_func(x, y, z):
    return x + (y * z)

# Replacing torch.randn(128).to("xpu") with torch.nn.functional.adaptive_max_pool2d
# We create a base tensor to serve as input for the pooling operation.
# We aim to generate 1D tensors of size 128.
# Shape (1, 1, 16, 8) has 128 elements. We pool to (16, 8) and flatten.
base_tensor = torch.ones(1, 1, 16, 8).to("xpu")

x = F.adaptive_max_pool2d(base_tensor, (16, 8)).flatten()
y = F.adaptive_max_pool2d(base_tensor, (16, 8)).flatten()
z = F.adaptive_max_pool2d(base_tensor, (16, 8)).flatten()

out = addcmul_func(x, y, z)
print("eager mode passed")
assert out is not None
assert out.device.type == "xpu"

addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
print("torch.compile passed")
assert out is not None
assert out.device.type == "xpu"
```
