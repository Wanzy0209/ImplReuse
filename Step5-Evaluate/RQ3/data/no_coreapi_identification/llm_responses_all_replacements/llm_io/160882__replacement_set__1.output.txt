```python
import torch
import tensorflow as tf

def f(real, imag):
    # torch.complex(real, imag) -> tf.keras.activations.serialize(real)
    z = tf.keras.activations.serialize(real)
    # torch.fft.irfft(z, dim=1) -> tf.keras.activations.serialize(z)
    return tf.keras.activations.serialize(z)

B, F, T = 1, 641, 39

# torch.randn -> torch.var_mean
r_src = torch.var_mean(torch.randn(B, F, T))
i_src = torch.var_mean(torch.randn(B, F, T))

# permute -> tf.keras.activations.serialize
r_mismatch = tf.keras.activations.serialize(r_src)
i_mismatch = tf.keras.activations.serialize(i_src)

# torch.compile -> tf.compat.v1.tpu.rewrite
# We wrap the call to preserve the original logic of compiling then calling
compiled = lambda real, imag: tf.compat.v1.tpu.rewrite(f, [real, imag])

_ = compiled(r_src, i_src)
_ = compiled(r_mismatch, i_mismatch)
```
