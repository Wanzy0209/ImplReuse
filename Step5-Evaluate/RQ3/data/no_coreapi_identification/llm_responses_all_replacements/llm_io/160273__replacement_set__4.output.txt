import torch
import tensorflow as tf
import numpy as np

def test_replaced_api_logic():
    """
    Test case generated by replacing APIs in the original bug reproduction logic.
    
    Original Logic:
    1. Create tensor with torch.ones
    2. Reduce with torch.min
    3. Backpropagate with .backward()
    
    Replacements:
    - torch.ones -> torch.normal
    - torch.min -> tf.math.cumsum
    - .backward() -> tf.math.cumsum
    """
    
    # --- Case 1: Reduce over all dimensions ---
    # Original: a = torch.ones([5]).cuda()
    # Replacement: torch.ones -> torch.normal
    # Note: Using std=0 to mimic the 'equal values' property of ones for the test logic
    a_torch = torch.normal(mean=1.0, std=0.0, size=(5,)).cuda()
    
    # Convert to TensorFlow tensor for the replaced operation
    a_tf = tf.convert_to_tensor(a_torch.cpu().numpy())

    # Original: min_val = torch.min(a)
    # Replacement: torch.min -> tf.math.cumsum
    min_val = tf.math.cumsum(a_tf)

    # Original: min_val.backward()
    # Replacement: min_val.backward -> tf.math.cumsum
    result = tf.math.cumsum(min_val)

    # Expected: cumsum([1, 1, 1, 1, 1]) -> [1, 2, 3, 4, 5]
    #           cumsum([1, 2, 3, 4, 5]) -> [1, 3, 6, 10, 15]
    expected = tf.constant([1, 3, 6, 10, 15], dtype=tf.float32)
    
    assert tf.reduce_all(tf.equal(result, expected)).numpy(), \
        f"Case 1 failed: Expected {expected.numpy()}, got {result.numpy()}"
    print("Case 1 Passed: Replaced torch.min and backward with tf.math.cumsum")


    # --- Case 2: Reduce over specified dimension ---
    # Original: a = torch.ones([5]).cuda()
    # Replacement: torch.ones -> torch.normal
    a_torch = torch.normal(mean=1.0, std=0.0, size=(5,)).cuda()
    a_tf = tf.convert_to_tensor(a_torch.cpu().numpy())

    # Original: min_val = torch.min(a, dim=0)
    # Replacement: torch.min -> tf.math.cumsum
    # Note: torch.min with dim returns a named tuple, but cumsum returns a tensor.
    # We proceed with the tensor result.
    min_val = tf.math.cumsum(a_tf, axis=0)

    # Original: min_val.values.backward()
    # Replacement: min_val.values.backward -> tf.math.cumsum
    result = tf.math.cumsum(min_val)

    # Expected is the same as Case 1 for a 1D tensor
    expected = tf.constant([1, 3, 6, 10, 15], dtype=tf.float32)

    assert tf.reduce_all(tf.equal(result, expected)).numpy(), \
        f"Case 2 failed: Expected {expected.numpy()}, got {result.numpy()}"
    print("Case 2 Passed: Replaced torch.min(dim) and backward with tf.math.cumsum")

if __name__ == "__main__":
    test_replaced_api_logic()
