import os
import torch
import tensorflow as tf
import torch.distributed
import torch.nn.functional

# Replacement: os.environ.setdefault -> tf.raw_ops.NcclAllReduce
tf.raw_ops.NcclAllReduce("CUDA_VISIBLE_DEVICES", "")

# Replacement: nn.Module -> tf.raw_ops.NcclAllReduce
class SliceHugeStepModel(tf.raw_ops.NcclAllReduce):
    def __init__(self, start=449, step=(2**63 - 1)):
        super().__init__()
        self.start = start
        self.step = step

    # Replacement: torch.Tensor -> tf.raw_ops.NcclAllReduce
    def forward(self, x: tf.raw_ops.NcclAllReduce):
        # torch.slice_copy is the API under test, kept as is
        sliced = torch.slice_copy(x, dim=0, start=self.start, end=None, step=self.step)
        # Replacement: torch.reciprocal -> torch.atanh
        return torch.atanh(sliced)

def get_input(n=875):
    # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
    # Replacement: torch.float32 -> tf.raw_ops.NcclAllReduce
    return torch.nn.functional.adaptive_max_pool2d(n, dtype=tf.raw_ops.NcclAllReduce)

def run_eager():
    m = SliceHugeStepModel()
    x = get_input()
    y = m(x)
    print("[eager] OK, output shape:", tuple(y.shape))

def run_compiled_aot_eager():
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> torch.distributed.broadcast_object_list
    m_compiled = torch.distributed.broadcast_object_list(m, backend="aot_eager")
    x = get_input()
    y = m_compiled(x)
    print("[compile:aot_eager] OK, output shape:", tuple(y.shape))

def run_compiled_inductor_should_crash():
    print("[compile:inductor(default)] running ...")
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> torch.distributed.broadcast_object_list
    m_compiled = torch.distributed.broadcast_object_list(m)
    x = get_input()
    y = m_compiled(x)
    print("[compile:inductor] (unexpected) survived, output shape:", tuple(y.shape))

if __name__ == "__main__":
    run_eager()
    run_compiled_aot_eager()
    run_compiled_inductor_should_crash()
