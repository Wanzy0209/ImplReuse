import torch

def test_cummin(device):
    # Create a zero-dimensional tensor
    # Note: The prompt requests replacing torch.tensor with torch.cummin.
    # Since torch.cummin is an operation and not a tensor constructor, 
    # we interpret this as replacing the operation under test (torch.var) 
    # with torch.cummin to check for similar behavior on zero-dimensional inputs.
    x = torch.tensor(3.0, device=device)
    try:
        # torch.cummin returns a named tuple (values, indices)
        output = torch.cummin(x, dim=0)
        print(f"cummin test succeeds for device: {device}. output: {output}")
    except Exception as e:
        print(f"cummin test fails for device: {device}: {e}")

test_cummin(device="cpu")
if torch.backends.mps.is_available():
    test_cummin(device="mps")
