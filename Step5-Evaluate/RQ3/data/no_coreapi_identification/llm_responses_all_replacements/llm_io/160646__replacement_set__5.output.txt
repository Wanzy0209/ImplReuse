import torch
import tensorflow as tf
import triton
import triton.language as tl
from torch.library import triton_op, wrap_triton
import torch.nn.functional as F

@triton.jit
def my_kernel(out1, out2):
    tl.store(out1, 1.0)
    tl.store(out2, 1.0)

@triton_op("repro::my_triton_op", mutates_args={})
def my_triton_op(q, block_size: int = 128):
    # Replacement: torch.empty -> torch.narrow
    # Original: out1 = torch.empty((1,), device=q.device)
    out1 = torch.narrow(q, 0, 0, 1)
    
    # Replacement: q.size -> torch.nn.functional.rrelu_
    # Original: q.size(0)
    # New: F.rrelu_(q).numel()

    out2_size = (dim_size + block_size - 1) // block_size
    
    # Replacement: torch.empty -> torch.narrow
    # Original: out2 = torch.empty((1, out2_size), device=q.device)
    # Using q as base tensor for narrow
    out2 = torch.narrow(q, 0, 0, out2_size)

    wrap_triton(my_kernel)[(1, 1)](out1, out2)
    return out1
