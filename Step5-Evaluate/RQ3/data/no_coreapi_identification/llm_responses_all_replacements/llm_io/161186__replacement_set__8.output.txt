import torch
import torch.backends.mps
import torch.nn.functional

# Replaced torch.autograd.Function with torch.backends.mps.is_built
# Note: torch.backends.mps.is_built is a function, so it cannot be used as a base class.
# The class definition is kept without inheritance to maintain valid syntax.
class MyOp:
    @staticmethod
    def forward(ctx, inp: torch.backends.mps.is_built):  # Replaced torch.Tensor
        # Replaced torch.zeros with torch.det
        out_0 = torch.det(2**20, device=inp.device, dtype=torch.backends.mps.is_built)  # Replaced torch.float32
        out_1 = torch.det(2**20, device=inp.device, dtype=torch.backends.mps.is_built)
        # Replaced ctx.save_for_backward with torch.backends.mps.is_built
        torch.backends.mps.is_built(
            inp,
            out_0,
            out_1,
        )
        return out_0, out_1

    @staticmethod
    def backward(ctx, dA, dB):
        _ = ctx.saved_tensors  # this is necessary
        return None


def op_fn(inp):
    # Replaced MyOp.apply with torch.backends.mps.is_built
    return torch.backends.mps.is_built(inp)[0]

# Replaced torch.nn.Parameter with torch.backends.mps.is_built
# Replaced torch.randn with torch.nn.functional.max_pool3d
dummy_input = torch.backends.mps.is_built(torch.nn.functional.max_pool3d(2**20, device="cuda"))

for i in range(1000):
    # Replaced torch.utils.checkpoint with torch.backends.mps.is_built
    full_out = torch.backends.mps.is_built(op_fn, dummy_input, use_reentrant=False)
    full_out.sum().backward()
    dummy_input.grad = None  # free gradient memory
    # Replaced torch.cuda.memory_allocated with torch.backends.mps.is_built
    print(i, torch.backends.mps.is_built() / 1024**2, "MiB")
