import torch
import tensorflow as tf

# Replacement: os.environ.setdefault -> tf.keras.metrics.Sum
# Original: os.environ.setdefault("CUDA_VISIBLE_DEVICES", "")
# Translated: tf.keras.metrics.Sum(name="CUDA_VISIBLE_DEVICES", dtype="")
tf.keras.metrics.Sum(name="CUDA_VISIBLE_DEVICES", dtype="")

# Replacement: nn.Module -> tf.keras.metrics.Sum
class SliceHugeStepModel(tf.keras.metrics.Sum):
    def __init__(self, start=449, step=(2**63 - 1)):
        super().__init__()
        self.start = start
        self.step = step

    # Replacement: torch.Tensor -> tf.keras.metrics.Sum
    def forward(self, x: tf.keras.metrics.Sum):
        # torch.slice_copy is not in the replacement list, so it remains.
        # Note: x is typed as tf.keras.metrics.Sum, but we pass it to torch.slice_copy
        # to preserve the original bug reproduction logic structure.
        sliced = torch.slice_copy(x, dim=0, start=self.start, end=None, step=self.step)
        
        # Replacement: torch.reciprocal -> torch.acosh
        return torch.acosh(sliced)

def get_input(n=875):
    # Replacement: torch.randn -> tf.keras.callbacks.History
    # Replacement: torch.float32 -> tf.keras.metrics.Sum
    # Original: torch.randn(n, dtype=torch.float32)
    return tf.keras.callbacks.History(n, dtype=tf.keras.metrics.Sum)

def run_eager():
    m = SliceHugeStepModel()
    x = get_input()
    y = m(x)
    print("[eager] OK, output shape:", tuple(y.shape))

def run_compiled_aot_eager():
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> torch.distributed.reduce
    # Original: torch.compile(m, backend="aot_eager")
    m_compiled = torch.distributed.reduce(m, backend="aot_eager")
    x = get_input()
    y = m_compiled(x)
    print("[compile:aot_eager] OK, output shape:", tuple(y.shape))

def run_compiled_inductor_should_crash():
    print("[compile:inductor(default)] running ...")
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> torch.distributed.reduce
    m_compiled = torch.distributed.reduce(m)
    x = get_input()
    y = m_compiled(x)
    print("[compile:inductor] (unexpected) survived, output shape:", tuple(y.shape))

if __name__ == "__main__":
    run_eager()
