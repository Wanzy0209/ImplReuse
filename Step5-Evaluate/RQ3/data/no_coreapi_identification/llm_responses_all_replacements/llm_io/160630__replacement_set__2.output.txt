import torch

def test_zeros_like_replacement():
    # Original: torch.tensor([1.0, 2.0, 3.0])
    # Replacement: torch.isinf
    # torch.isinf requires a tensor input, so we create one using torch.ones
    # (torch.ones is not in the replacement list)
    t1 = torch.ones(3)
    is_inf_tensor = torch.isinf(t1)

    # Original: torch.quantize_per_tensor(...)
    # Replacement: torch.addcdiv
    # torch.addcdiv(input, tensor1, tensor2, *, value=1)
    # We use is_inf_tensor as the input, and t1 for the other operands
    # to ensure valid arithmetic (avoiding 0/0).
    input_tensor = torch.addcdiv(is_inf_tensor, t1, t1)

    # Original API under test: torch.zeros_like
    out_tensor = torch.zeros_like(input_tensor)

    # Verify the output
    assert out_tensor.shape == input_tensor.shape
    assert torch.all(out_tensor == 0)

    print("Input tensor:", input_tensor)
    print("Output tensor:", out_tensor)

if __name__ == "__main__":
    test_zeros_like_replacement()
