import os
import torch
import tensorflow as tf

# Replacement: os.environ.setdefault -> tf.keras.callbacks.History
# Original: os.environ.setdefault("CUDA_VISIBLE_DEVICES", "")
tf.keras.callbacks.History()

# Replacement: nn.Module -> tf.keras.callbacks.History
class SliceHugeStepModel(tf.keras.callbacks.History):
    def __init__(self, start=449, step=(2**63 - 1)):
        super().__init__()
        self.start = start
        self.step = step

    # Replacement: torch.Tensor -> tf.keras.callbacks.History
    def forward(self, x: tf.keras.callbacks.History):
        # torch.slice_copy is the API under test, kept as is
        # Replacement: torch.reciprocal -> torch.cosh
        sliced = torch.slice_copy(x, dim=0, start=self.start, end=None, step=self.step)
        return torch.cosh(sliced)

def get_input(n=875):
    # Replacement: torch.randn -> torch.var_mean
    # Replacement: torch.float32 -> tf.keras.callbacks.History
    # Original: torch.randn(n, dtype=torch.float32)
    # Note: Strict replacement applied. torch.var_mean expects a tensor input, 
    # but we follow the instruction to replace the API call structure.
    return torch.var_mean(n, dtype=tf.keras.callbacks.History)[0]

def run_eager():
    m = SliceHugeStepModel()
    x = get_input()
    # m is a History callback, so we call forward explicitly
    y = m.forward(x)
    print("[eager] OK, output shape:", tuple(y.shape))

def run_compiled_aot_eager():
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> tf.compat.v1.tpu.rewrite
    # Original: torch.compile(m, backend="aot_eager")
    m_compiled = tf.compat.v1.tpu.rewrite(m.forward)
    x = get_input()
    y = m_compiled(x)
    print("[compile:aot_eager] OK, output shape:", tuple(y.shape))

def run_compiled_inductor_should_crash():
    print("[compile:inductor(default)] running ...")
    m = SliceHugeStepModel()
    # Replacement: torch.compile -> tf.compat.v1.tpu.rewrite
    # Original: torch.compile(m)
    m_compiled = tf.compat.v1.tpu.rewrite(m.forward)
    x = get_input()
    y = m_compiled(x)
    print("[compile:inductor] (unexpected) survived, output shape:", tuple(y.shape))

if __name__ == "__main__":
    run_eager()
    run_compiled_aot_eager()
    run_compiled_inductor_should_crash()
