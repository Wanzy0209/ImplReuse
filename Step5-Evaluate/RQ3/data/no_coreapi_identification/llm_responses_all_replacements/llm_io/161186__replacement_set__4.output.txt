import torch
import torch.backends.nnpack
import torch.nn.functional

# torch.autograd.Function -> torch.backends.nnpack.set_flags
# Note: Cannot inherit from a function, so we define the class without inheritance
class MyOp:
    @staticmethod
    def forward(ctx, inp: torch.backends.nnpack.set_flags):
        # torch.zeros -> torch.normal
        # torch.float32 -> torch.backends.nnpack.set_flags
        out_0 = torch.normal(0, 1, size=(2**20,), device=inp.device, dtype=torch.backends.nnpack.set_flags)
        out_1 = torch.normal(0, 1, size=(2**20,), device=inp.device, dtype=torch.backends.nnpack.set_flags)
        
        # ctx.save_for_backward -> torch.backends.nnpack.set_flags
        torch.backends.nnpack.set_flags(
            inp,
            out_0,
            out_1,
        )
        return out_0, out_1

    @staticmethod
    def backward(ctx, dA, dB):
        _ = ctx.saved_tensors
        return None


def op_fn(inp):
    # MyOp.apply -> torch.backends.nnpack.set_flags
    return torch.backends.nnpack.set_flags(inp)[0]


# torch.nn.Parameter -> torch.backends.nnpack.set_flags
# torch.randn -> torch.nn.functional.adaptive_max_pool2d
# We use torch.normal (replacement for torch.zeros) to create input for adaptive_max_pool2d
dummy_input = torch.backends.nnpack.set_flags(
    torch.nn.functional.adaptive_max_pool2d(
        torch.normal(0, 1, (2**20, 1, 1)), (1,)
    )
)

for i in range(1000):
    # torch.utils.checkpoint -> torch.backends.nnpack.set_flags
    full_out = torch.backends.nnpack.set_flags(op_fn, dummy_input, use_reentrant=False)
    full_out.sum().backward()
    dummy_input.grad = None
    # torch.cuda.memory_allocated -> torch.backends.nnpack.set_flags
    print(i, torch.backends.nnpack.set_flags() / 1024**2, "MiB")
