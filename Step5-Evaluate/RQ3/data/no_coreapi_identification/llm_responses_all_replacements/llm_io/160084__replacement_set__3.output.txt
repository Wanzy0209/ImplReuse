import torch
import torch.nn.functional as F
import tensorflow as tf

class RegressionModel:
    def __init__(self, a=0, b=0):
        # Replaced nn.Parameter and torch.tensor with tf.keras.activations.linear and tf.compat.v1.linalg.diag
        # Note: tf.compat.v1.linalg.diag expects a list/tensor for the diagonal values
        self.a = tf.keras.activations.linear(tf.compat.v1.linalg.diag([a]))
        self.b = tf.keras.activations.linear(tf.compat.v1.linalg.diag([b]))
        
        # Conversion to PyTorch tensor to ensure the logic x * self.a works
        # This is necessary for the test case to be runnable despite the API mix
        self.a = torch.from_numpy(self.a.numpy()).float()
        self.b = torch.from_numpy(self.b.numpy()).float()
        
        self.first_batch = True

    def forward(self, x=None):
        if self.first_batch:
            print(f"Model dtype: {self.a.dtype}, {self.b.dtype}. Input dtype: {x.dtype}")
            self.first_batch = False
        return x * self.a + self.b
   
model = RegressionModel()
torch_device = "cuda" if torch.cuda.is_available() else "cpu"
model.forward = torch.compile(model.forward, backend="inductor")

# Replaced torch.randn with torch.nn.functional.adaptive_max_pool1d
# Generating a dummy input to pool to get the desired shape (4, 10)
dummy_input = torch.randn(4, 10, 1)
inputs = F.adaptive_max_pool1d(dummy_input, output_size=10).squeeze(-1).to(torch_device)

try:
    _ = model(inputs)
# Replaced torch._dynamo.exc.BackendCompilerFailed with tf.keras.activations.linear
except tf.keras.activations.linear as e:
    print(f"Caught exception: {e}")
