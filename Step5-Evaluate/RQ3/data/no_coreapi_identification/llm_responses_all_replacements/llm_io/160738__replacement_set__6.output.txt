import tensorflow as tf

def test_var(device):
    try:
        with tf.device(device):
            # Replacement: torch.tensor -> tf.linalg.diag
            # Original: x = torch.tensor(3.0, device=device)
            # Note: tf.linalg.diag creates a diagonal matrix. 
            # We use [3.0] to create a 1x1 matrix [[3.0]] to mimic the single value input.
            x = tf.linalg.diag([3.0])
            
            # Original: torch.var(x, dim=0)
            # Semantics: Calculate variance along axis 0.
            output = tf.math.reduce_variance(x, axis=0)
            
        print(f"var test succeeds for device: {device}. output: {output}")
    except tf.errors.OperatorNotAllowedInGraphError as e:
        # Leveraging the similar API: tf.errors.OperatorNotAllowedInGraphError
        print(f"var test fails for device: {device} (OperatorNotAllowedInGraphError): {e}")
    except Exception as e:
        print(f"var test fails for device: {device}: {e}")

# Running the test with the original device arguments
# Note: TensorFlow device strings might differ (e.g., "/CPU:0"), 
# but we keep the original logic strings as requested.
test_var(device="cpu")
test_var(device="mps")
