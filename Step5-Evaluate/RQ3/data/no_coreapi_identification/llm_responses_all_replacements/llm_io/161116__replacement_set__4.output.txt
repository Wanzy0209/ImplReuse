```python
import torch
import torch.nn.functional as F
import os

def main():
    # Logic: Get GPU ID from environment variable, defaulting to 0 if not set
    gpu_id = int(os.environ.get("LOCAL_RANK", "0"))
    
    # Logic: Set Device
    device = torch.device(f"cuda:{gpu_id}" if torch.cuda.is_available() else "cpu")
    if torch.cuda.is_available():
        torch.cuda.set_device(device)

    # Prepare dummy tensors for the replacement API
    # kl_div expects input to be log-probabilities and target to be probabilities
    input_tensor = torch.randn(10, 10, requires_grad=True).log_softmax(dim=1).to(device)
    target_tensor = torch.randn(10, 10).softmax(dim=1).to(device)

    # Replacement: dist.init_process_group -> torch.nn.functional.kl_div
    # We perform the calculation on the configured device
    loss = F.kl_div(input_tensor, target_tensor, reduction='batchmean')
    
    # Assertion to verify the operation executed
    assert loss.item() >= 0, "KL Divergence loss should be non-negative"
    print(f"Successfully computed KL Div loss on {device}: {loss.item()}")

if __name__ == "__main__":
    main()
```
