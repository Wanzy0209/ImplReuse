import torch
import torch.nn as nn
import torch.nn.functional as F
from torch._inductor import config

config.fallback_random = True
# torch.set_grad_enabled -> torch.nn.SELU
# Original: torch.set_grad_enabled(False)
# Replacement: Instantiate SELU
_ = nn.SELU()
torch._dynamo.config.capture_dynamic_output_shape_ops = True

def eager_fn(x, y):
    return torch.add(x, y)

# torch.compile -> torch.all
# Original: @torch.compile()
# Replacement: No decorator, use torch.all for verification
def fn(x, y):
    return torch.add(x, y)

# torch.randn -> torch.nn.functional.adaptive_max_pool3d
# Original: x = torch.randn(1, dtype=torch.complex64)
# Replacement: Create input and pool
# Note: adaptive_max_pool3d does not support complex, so we use float
input_x = torch.ones(1, 1, 1, 1, 1)
x = F.adaptive_max_pool3d(input_x, output_size=(1, 1, 1))

# torch.empty -> torch.narrow
# Original: y = torch.empty((), dtype=torch.complex64)
# Replacement: Create input and narrow
input_y = torch.ones(1)
y = torch.narrow(input_y, 0, 0, 1)

eager_res = eager_fn(x, y)
print("eager success")

fn_res = fn(x, y)
print("compiler success")

# torch.all usage
assert torch.all(eager_res == fn_res)
