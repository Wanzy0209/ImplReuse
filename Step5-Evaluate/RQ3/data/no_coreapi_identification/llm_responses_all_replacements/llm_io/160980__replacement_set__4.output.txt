import tensorflow as tf
import numpy as np

def test_sparse_categorical_crossentropy_with_gpu_support():
    """
    Test case generated based on Issue 160980 (PyTorch CUDA13 NVSHMEM build failure).
    
    Original Issue Logic:
    The issue describes a build failure (undefined reference) when attempting to use
    NVSHMEM extensions (torch_nvshmem/nvshmem_extension.cu) with specific CUDA architectures.
    
    Adaptation:
    - The 'torch_nvshmem/nvshmem_extension.cu' dependency is replaced by 
      'tf.test.is_built_with_gpu_support' to check for the necessary build environment.
    - The similar API 'tf.keras.losses.SparseCategoricalCrossentropy' is used as the 
      functional component being tested, analogous to the distributed operations in the issue.
    """
    
    # Replacement for torch_nvshmem/nvshmem_extension.cu
    # Check if the build environment supports the necessary GPU capabilities.
    # This mirrors the prerequisite check that failed in the original bug report.
    if not tf.test.is_built_with_gpu_support():
        print("Test skipped: TensorFlow not built with GPU support.")
        return

    # Similar API: tf.keras.losses.SparseCategoricalCrossentropy
    # We instantiate the loss function to verify it operates correctly given the build support.
    scce = tf.keras.losses.SparseCategoricalCrossentropy()

    # Define sample inputs
    # y_true: Integer class labels
    # y_pred: Probability distributions
    y_true = [1, 2]
    y_pred = [[0.05, 0.95, 0], [0.1, 0.8, 0.1]]

    # Execute the loss calculation
    # In the original bug, this stage would fail with undefined references.
    # Here we assert successful execution.
    loss = scce(y_true, y_pred)

    # Assertions
    # Verify the loss is computed and matches the expected value from the API documentation.
    assert loss is not None, "Loss calculation returned None"
    expected_loss = 1.177
    assert np.isclose(loss.numpy(), expected_loss, atol=0.01), \
        f"Expected loss ~{expected_loss}, got {loss.numpy()}"

if __name__ == "__main__":
    test_sparse_categorical_crossentropy_with_gpu_support()
