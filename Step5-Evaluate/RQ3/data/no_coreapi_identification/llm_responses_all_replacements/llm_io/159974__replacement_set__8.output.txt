```python
import torch
import torch.nn.functional as F

def addcmul_func(x, y, z):
    return x + (y * z)

# Replacing torch.randn with torch.nn.functional.max_pool3d
# max_pool3d requires a 5D input tensor (N, C, D, H, W).
# We use torch.ones to create the necessary input for the replacement API.
dummy_input = torch.ones(1, 1, 4, 4, 4)

x = F.max_pool3d(dummy_input, kernel_size=2).to("xpu")
y = F.max_pool3d(dummy_input, kernel_size=2).to("xpu")
z = F.max_pool3d(dummy_input, kernel_size=2).to("xpu")

out = addcmul_func(x, y, z)
print("eager mode passed")

addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
print("torch.compile passed")
```
