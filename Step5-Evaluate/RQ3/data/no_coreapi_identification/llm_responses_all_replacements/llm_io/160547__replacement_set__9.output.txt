import torch
from collections import namedtuple

def test_namedtuple():
    Point = namedtuple('Point', 'x y')

    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y

    # Replacing torch.ones(3) with torch.matrix_power
    # torch.matrix_power requires a square matrix, so we use torch.eye(3) as the base.
    base_matrix = torch.eye(3)
    inp = Point(torch.matrix_power(base_matrix, 1), torch.matrix_power(base_matrix, 1))
    
    print(M()(*inp))
    
    # errors in the original bug report
    ep = torch.export.export(M(), inp, strict=False)
    print(ep)

    # succeeds in the original bug report
    ep = torch.export.export(M(), inp, strict=True)
    print(ep)

    # workaround could be to convert namedtuple to a kwarg
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    print(ep)
