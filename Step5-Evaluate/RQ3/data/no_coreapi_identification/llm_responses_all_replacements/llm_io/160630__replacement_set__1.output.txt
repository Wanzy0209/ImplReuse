import torch

def test_quantized_zeros_like():
    # Replacement: torch.tensor -> torch.isnan
    # We pass a list to isnan to avoid using torch.tensor constructor explicitly.
    # We cast to float because isnan returns bool and quantize_per_channel expects float.
    input_data = torch.isnan([1.0, 2.0, 3.0]).float()

    # Replacement: torch.quantize_per_tensor -> torch.quantize_per_channel
    # Adapt arguments for per-channel quantization (axis 0, 3 channels)
    scales = torch.tensor([1.0, 1.0, 1.0])
    zero_points = torch.tensor([0, 0, 0])
    axis = 0

    quant_input = torch.quantize_per_channel(
        input_data, scales, zero_points, axis, dtype=torch.quint8
    )

    # Original API Under Test: torch.zeros_like
    # This triggers the error described in the bug report
    out_tensor = torch.zeros_like(quant_input)

    print("Quantized input:", quant_input)
    print("Output tensor:", out_tensor)

if __name__ == "__main__":
    test_quantized_zeros_like()
