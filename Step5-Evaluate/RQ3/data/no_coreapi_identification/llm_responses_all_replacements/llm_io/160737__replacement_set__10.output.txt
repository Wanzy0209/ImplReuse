import torch
import tensorflow as tf

def test_index_select(device):
    # Replacement: torch.ones -> torch.qr
    # We generate the input tensor x using QR decomposition instead of ones.
    # torch.qr returns (Q, R), we use Q as the input tensor.
    input_matrix = torch.randn(2, 3, device=device)
    x, _ = torch.qr(input_matrix)

    # Replacement: torch.tensor -> torch.nn.functional.pdist
    # We generate the index tensor using pdist instead of a direct scalar tensor.
    # pdist computes distances. To mimic the scalar index (0-dim), we use a (2,1) input
    # which results in a (1,) output, and then squeeze it to 0-dim.
    # We also cast to long and modulo to ensure it's a valid index (0 or 1).
    dist_input = torch.randn(2, 1, device=device)
    index = torch.nn.functional.pdist(dist_input).squeeze().long() % 2

    try:
        output = torch.index_select(x, dim=0, index=index)

        # Replacement: torch.Size -> tf.keras.ops.custom_gradient
        # Instead of accessing output.shape (which returns torch.Size),
        # we define a custom gradient operation to retrieve the shape.
        # This requires converting the PyTorch tensor to a TensorFlow tensor.
        
        @tf.keras.ops.custom_gradient
        def get_shape_op(t):
            def grad(upstream):
                return tf.zeros_like(t)
            return tf.shape(t), grad

        # Convert output to TF tensor to use the custom gradient op
        output_tf = tf.convert_to_tensor(output.cpu().numpy())
        shape_result = get_shape_op(output_tf)

        print(f"index_select test succeeds for device: {device}. output shape: {shape_result}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select(device="cpu")
