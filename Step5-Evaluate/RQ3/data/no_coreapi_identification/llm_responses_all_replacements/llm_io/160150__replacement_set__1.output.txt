import tensorflow as tf
import numpy as np

def test_fx_node_output_includes_none():
    """
    Test case adapted from Issue 160150: [ONNX] Exporter crashes when fx node output includes None.
    
    Original Issue: torch.onnx.export crashes when a model returns a tuple containing None.
    Adaptation: This test verifies the behavior of tf.compat.v1.Summary.Value when processing
    similar outputs that include None, replacing the torch.onnx.export logic.
    """
    # Setup data mimicking the bug report scenario
    # 'output' is a valid tensor
    output = tf.constant([[1.0, 2.0], [3.0, 4.0]], dtype=tf.float32)
    # 'image_tokens_masks' could be None
    image_tokens_masks = None
    return_dict = False

    # Original logic from the bug report
    if not return_dict:
        node_outputs = (output, image_tokens_masks)

    # Replacement: Use tf.compat.v1.Summary.Value instead of torch.onnx.export
    # We iterate through the outputs to create summary values, simulating the export process.
    summary_values = []
    
    for i, val in enumerate(node_outputs):
        if val is not None:
            # Create a Summary.Value for the valid tensor
            # This replaces the ONNX node generation logic
            value = tf.compat.v1.Summary.Value(
                tag=f"output_{i}",
                tensor=tf.make_tensor_proto(val)
            )
            summary_values.append(value)
        else:
            # The PyTorch exporter crashed here because it did not handle None.
            # In this adaptation, we check if tf.compat.v1.Summary.Value can handle None.
            # Since Summary.Value fields are strongly typed (e.g., simple_value is float),
            # None is not a valid input for the value fields.
            # We explicitly handle the None to avoid a crash, reflecting the fix logic.
            pass

    # Assertions to verify the test logic
    # We expect one valid Summary.Value to be created for the tensor output
    assert len(summary_values) == 1
    assert summary_values[0].tag == "output_0"
    
    # Verify the tensor content is preserved
    expected_tensor = tf.make_tensor_proto(output)
    assert summary_values[0].tensor.tensor_content == expected_tensor.tensor_content

if __name__ == "__main__":
    test_fx_node_output_includes_none()
    print("Test passed.")
