```python
import torch
import torch.nn.functional as F

def addcmul_func(x, y, z):
    return x + (y * z)

# Replacing torch.randn(128).to("xpu") with torch.nn.functional.adaptive_max_pool1d
# We use torch.arange to generate input data for the pooling layer to avoid using torch.randn.
# The goal is to produce a 1D tensor of size 128 on the "xpu" device.

def generate_tensor(seed_offset):
    # Create a 3D input tensor (Batch=1, Channels=128, Length=10) on XPU
    input_tensor = torch.arange(1 + seed_offset, 1 + seed_offset + 1280, dtype=torch.float32).view(1, 128, 10).to("xpu")
    # Apply adaptive_max_pool1d with output_size=1 to reduce the last dimension
    # Result shape is (1, 128, 1)
    pooled = F.adaptive_max_pool1d(input_tensor, output_size=1)
    # Squeeze to get the final shape (128,)
    return pooled.squeeze()

x = generate_tensor(0)
y = generate_tensor(1280)
z = generate_tensor(2560)

# Verify shapes
assert x.shape == (128,)
assert y.shape == (128,)
assert z.shape == (128,)

# Eager execution
out = addcmul_func(x, y, z)
print("eager mode passed")

# Compile execution
addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
print("torch.compile passed")
```
