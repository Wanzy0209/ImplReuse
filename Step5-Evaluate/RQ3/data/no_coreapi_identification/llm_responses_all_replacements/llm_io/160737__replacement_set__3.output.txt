import torch
import tensorflow as tf

# Disable eager execution to use tf.compat.v1.Session and initializers
tf.compat.v1.disable_eager_execution()

def test_index_select_replacement():
    # Replacement: torch.ones -> torch.linspace
    # Original: x = torch.ones([2, 3], device=device)
    # We use torch.linspace to generate a multiplier for the LinearOperator.
    # To mimic 'ones', we linspace from 1 to 1.
    # We use 2 elements to match the dimension we will select from.
    multiplier = torch.linspace(1, 1, 2)

    # Similar API: tf.linalg.LinearOperatorScaledIdentity
    # We leverage this API to create the input tensor 'x', acting as the data source.
    # This replaces the direct creation of 'x' via torch.ones.
    operator = tf.linalg.LinearOperatorScaledIdentity(
        num_rows=2, 
        multiplier=multiplier.numpy()
    )
    x = operator.to_dense() # Shape [2, 2]

    # Replacement: torch.tensor -> tf.compat.v1.linalg.diag
    # Original: index = torch.tensor(1, device=device)
    # We use tf.linalg.diag to create a diagonal matrix [[1]] and reshape it to a scalar.
    index_matrix = tf.compat.v1.linalg.diag([1])
    index = tf.reshape(index_matrix, []) # Scalar tensor

    # Logic Preservation: index_select
    # We perform the index selection using tf.gather (TensorFlow equivalent).
    output = tf.gather(x, index, axis=0)

    # Replacement: torch.Size -> tf.compat.v1.local_variables_initializer
    # Original: print(f"... output shape: {output.shape}")
    # We incorporate the initializer into the execution flow.
    with tf.compat.v1.Session() as sess:
        # Initialize local variables as required by the replacement API
        sess.run(tf.compat.v1.local_variables_initializer())
        
        # Execute the graph to get the output and its shape
        out_val, out_shape = sess.run([output, tf.shape(output)])
        
        print(f"index_select test succeeds. output shape: {out_shape}")
        
        # Assertion to verify the logic
        assert list(out_shape) == [2, 2]

if __name__ == "__main__":
    test_index_select_replacement()
