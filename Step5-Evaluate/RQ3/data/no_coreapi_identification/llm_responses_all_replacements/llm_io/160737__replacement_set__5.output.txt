import torch
import tensorflow as tf

def test_index_select(device):
    # Original: x = torch.ones([2, 3], device=device)
    # Replacement: torch.ones -> torch.norm
    # Similar API: tf.linalg.LinearOperatorScaledIdentity
    # We use torch.norm to generate a multiplier for the LinearOperator to create the input tensor x.
    # This preserves the logic of creating a tensor of shape [2, 3] for the operation.
    multiplier = torch.norm(torch.randn(2, device=device))
    multiplier_tf = tf.constant(multiplier.cpu().numpy())
    
    # Leverage the similar API to construct the input tensor
    op = tf.linalg.LinearOperatorScaledIdentity(num_rows=3, multiplier=multiplier_tf)
    x_tf = op.to_dense()
    x = torch.from_numpy(x_tf.numpy()).to(device)

    # Original: index = torch.tensor(1, device=device)
    # Replacement: torch.tensor -> tf.keras.ops.diag
    # We use tf.keras.ops.diag to create the index. 
    # Passing [[1]] extracts the diagonal as a 1D tensor [1], mimicking the scalar index logic.
    index_tf = tf.keras.ops.diag([[1]])
    index = torch.from_numpy(index_tf.numpy()).to(device)

    try:
        output = torch.index_select(x, dim=0, index=index)
        
        # Original: print(... output.shape)
        # Replacement: torch.Size -> tf.experimental.numpy.exp
        # We apply tf.exp to the shape tuple as requested.
        shape_exp = tf.experimental.numpy.exp(output.shape)
        
        print(f"index_select test succeeds for device: {device}. output shape exp: {shape_exp}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select(device="cpu")
