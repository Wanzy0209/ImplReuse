import torch
import tensorflow as tf

# Define the function to be compiled
@torch.compile(backend="eager")
def fn(x, i):
    if i == 1:
        torch._dynamo.graph_break()
    return x + 1

# Replacement: torch.randn -> tf.random.stateless_truncated_normal
# We generate the input using TensorFlow and convert it to a PyTorch tensor
# to maintain compatibility with the torch.compile logic.
seed = [1, 2]
shape = [3]
tf_tensor = tf.random.stateless_truncated_normal(shape=shape, seed=seed)
inp = torch.from_numpy(tf_tensor.numpy())

# Execute the function with different arguments to trigger graph breaks
# and verify the behavior.
out1 = fn(inp, 0)
out2 = fn(inp, 1)
out3 = fn(inp, 2)

# Assertions to ensure the function logic remains correct
assert torch.allclose(out1, inp + 1)
assert torch.allclose(out2, inp + 1)
assert torch.allclose(out3, inp + 1)
