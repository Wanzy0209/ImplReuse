import torch
import torch.nn.functional as F

# Replacement for torch.set_grad_enabled(False)
# torch.nn.Softshrink is a module, so we instantiate it to replace the context manager call
shrink = torch.nn.Softshrink()

# Replacement for torch.randn(1, dtype=torch.complex64)
# adaptive_max_pool1d requires real input, so we create a real input, pool it, and cast to complex
# to maintain the original test's dtype context as much as possible.
dummy_input = torch.randn(10).unsqueeze(0)
x = F.adaptive_max_pool1d(dummy_input, 1).to(torch.complex64)

# Replacement for torch.empty((), dtype=torch.complex64)
# torch.set_rng_state returns None
y = torch.set_rng_state(torch.get_rng_state())

def eager_fn(x, y):
    # Original call site x + y replaced with torch.add
    return torch.add(x, y)

# Replacement for @torch.compile()
# torch.compile is removed as it is a decorator and torch.prod is a function.
# We will apply torch.prod to the result later.
def fn(x, y):
    return torch.add(x, y)

try:
    eager_fn(x, y)
    print("eager success")
except Exception as e:
    print(f"eager failed: {e}")

try:
    # Applying torch.prod replacement logic
    res = fn(x, y)
    torch.prod(res)
    print("compiler success")
except Exception as e:
    print(f"compiler failed: {e}")
