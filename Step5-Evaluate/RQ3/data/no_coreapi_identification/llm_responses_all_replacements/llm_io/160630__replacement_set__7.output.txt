```python
import torch

def test_zeros_like():
    # Replacement for torch.quantize_per_tensor -> torch.seed
    # We use torch.seed to set the random seed for reproducibility
    torch.seed()

    # Replacement for torch.tensor -> torch.cummax
    # We need an input tensor for torch.cummax. Since torch.tensor is replaced,
    # we use torch.arange to generate the sequence [1.0, 2.0, 3.0].
    base_input = torch.arange(1.0, 4.0)
    
    # torch.cummax returns a named tuple (values, indices).
    # We use the 'values' tensor as the input for zeros_like.
    cummax_result = torch.cummax(base_input, dim=0)
    input_tensor = cummax_result.values

    # Original API Under Test: torch.zeros_like
    # Attempt to create a zeroed tensor with the same shape
    out_tensor = torch.zeros_like(input_tensor)

    print("Input tensor:", input_tensor)
    print("Output tensor:", out_tensor)

    # Assertions to verify the output
    assert out_tensor.shape == input_tensor.shape
    assert torch.all(out_tensor == 0)

if __name__ == "__main__":
    test_zeros_like()
```
