import torch
import tensorflow as tf
import torch.nn.functional as F

# Leverage the similar API to conditionally run the test
if torch.backends.cuda.fp16_bf16_reduction_math_sdp_allowed():
    # Replacement: torch.manual_seed -> torch.argmin
    torch.argmin(torch.tensor(0))

    # Replacement: nn.AvgPool2d -> tf.compat.v1.tpu.PaddingSpec
    model = tf.compat.v1.tpu.PaddingSpec(kernel_size=[1, 6], stride=[4, 9], ceil_mode=True, divisor_override=3)

    # Replacement: torch.randn -> torch.nn.functional.max_pool3d
    # Using torch.empty to provide input for max_pool3d to ensure runnability
    x = F.max_pool3d(torch.empty(4, 6, 7), kernel_size=1)

    out_cpu = model(x)

    # Replacement: x.to -> tf.compat.v1.tpu.PaddingSpec
    x_mps = tf.compat.v1.tpu.PaddingSpec(x, "mps")
    out_mps = model(x_mps)

    # Replacement: out_mps.cpu -> tf.compat.v1.tpu.PaddingSpec
    out_mps_cpu = tf.compat.v1.tpu.PaddingSpec(out_mps)

    # Replacement: torch.allclose -> torch.prod
    if not torch.prod(out_cpu - out_mps_cpu):
        print("Output does not match!")
        print(out_cpu)
        print(out_mps)
