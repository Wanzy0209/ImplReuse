```python
import torch
from collections import namedtuple

def test_namedtuple():
    Point = namedtuple('Point', 'x y')
    
    class M(torch.nn.Module):
        def forward(self, x, y):
            return x + y 
    
    # Create square matrices for torch.det
    mat1 = torch.eye(3)
    mat2 = torch.eye(3)

    # Replacement: torch.ones -> torch.det
    # Original: Point(torch.ones(3), torch.ones(3))
    inp = Point(torch.det(mat1), torch.det(mat2))
    
    # Verify basic execution
    output = M()(*inp)
    assert torch.allclose(output, torch.tensor(2.0))
    
    # Test strict=False (Bug case: errors in original report)
    # This assertion checks if the bug is fixed
    ep = torch.export.export(M(), inp, strict=False)
    assert ep is not None

    # Test strict=True (Succeeds in original report)
    ep = torch.export.export(M(), inp, strict=True)
    assert ep is not None

    # Test workaround (Succeeds in original report)
    inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
    ep = torch.export.export(M(), (), inp_kwargs)
    assert ep is not None
```
