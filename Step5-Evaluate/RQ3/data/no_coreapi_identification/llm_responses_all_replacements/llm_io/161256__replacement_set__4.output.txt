import torch
import tensorflow as tf
from tensorflow.python.framework.errors_impl import OperatorNotAllowedInGraphError

# Original: torch.set_default_dtype(torch.float64)
# Replacements: torch.set_default_dtype -> tf.keras.backend.std, torch.float64 -> tf.keras.activations.linear
# We create a dummy tensor to satisfy the argument requirements of the replacement APIs.
dummy_input = tf.constant(1.0)
# Apply the replacements
tf.keras.backend.std(tf.keras.activations.linear(dummy_input))

# Original: x = torch.randn(1000, 1000, device="cuda")
# Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
# We use torch.empty to provide the necessary input tensor for the pooling operation.
input_x = torch.empty(1, 1, 1000, 1000, device="cuda")
x = torch.nn.functional.adaptive_max_pool2d(input_x, (1, 1))

# Original: y = torch.randn(1000, 1000, device="cuda")
input_y = torch.empty(1, 1, 1000, 1000, device="cuda")
y = torch.nn.functional.adaptive_max_pool2d(input_y, (1, 1))

# Original: z = x @ y
# Replacement: @ -> tf.keras.activations.linear
# We apply the linear activation to x. Note: x is a PyTorch tensor.
# This tests the interop boundary.
z = tf.keras.activations.linear(x)

# Original: print(z[0, 0])
print(z[0, 0])

# Leveraging the similar API: tf.errors.OperatorNotAllowedInGraphError
# We check if the operation context is valid (e.g., not raising this error in a graph context if we were in one).
# Since we are in eager mode, we just assert the API is available.
assert OperatorNotAllowedInGraphError is not None
