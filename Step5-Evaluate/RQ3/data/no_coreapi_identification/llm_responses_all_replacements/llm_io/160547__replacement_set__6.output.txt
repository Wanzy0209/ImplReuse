import torch
from collections import namedtuple
import unittest

class TestExportNamedTuple(unittest.TestCase):
    def test_namedtuple(self):
        Point = namedtuple('Point', 'x y')
        
        class M(torch.nn.Module):
            def forward(self, x, y):
                return x + y 
        
        # Replacing torch.ones(3) with torch.dist
        # torch.dist computes the p-norm distance between two tensors.
        # We create dummy tensors to serve as inputs for torch.dist.
        t1 = torch.tensor([1.0, 2.0, 3.0])
        t2 = torch.tensor([4.0, 5.0, 6.0])
        
        # torch.dist returns a scalar tensor (0-d)
        inp = Point(torch.dist(t1, t2), torch.dist(t1, t2))
        
        # Verify basic execution
        result = M()(*inp)
        self.assertIsNotNone(result)
        
        # Test strict=False (Bug case: originally failed with NamedTuple)
        # This assertion will fail if the bug is not fixed.
        ep = torch.export.export(M(), inp, strict=False)
        self.assertIsNotNone(ep)
        
        # Test strict=True (Success case)
        ep_strict = torch.export.export(M(), inp, strict=True)
        self.assertIsNotNone(ep_strict)

        # Test kwargs workaround
        inp_kwargs = {field: getattr(inp, field) for field in inp._fields}
        ep_kwargs = torch.export.export(M(), (), inp_kwargs)
        self.assertIsNotNone(ep_kwargs)

if __name__ == '__main__':
    unittest.main()
