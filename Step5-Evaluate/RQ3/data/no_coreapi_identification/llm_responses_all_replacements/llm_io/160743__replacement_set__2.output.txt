import torch
import tensorflow as tf

# torch.manual_seed(0) -> tf.compat.v1.assign
# We create a variable to hold the seed state and assign the initial value.
seed_var = tf.Variable([0, 0], dtype=tf.int32)
tf.compat.v1.assign(seed_var, [0, 0])

# model = torch.nn.AvgPool2d(kernel_size=[1, 6], stride=[4, 9], ceil_mode=True, divisor_override=3)
# Replacement: nn.AvgPool2d -> tf.errors.OperatorNotAllowedInGraphError
# Note: Replacing the layer definition with the specified Error class as per instructions.
model = tf.errors.OperatorNotAllowedInGraphError

# x = torch.randn(4, 6, 7)
# Replacement: torch.randn -> tf.random.stateless_truncated_normal
x = tf.random.stateless_truncated_normal([4, 6, 7], seed_var)

# out_cpu = model(x)
out_cpu = model(x)

# out_mps = model(x.to("mps"))
# Replacement: x.to -> tf.errors.OperatorNotAllowedInGraphError
out_mps = model(tf.errors.OperatorNotAllowedInGraphError(x))

# if not torch.allclose(out_cpu, out_mps.cpu(), atol=1e-2, rtol=1e-2):
# Replacement: out_mps.cpu -> tf.errors.OperatorNotAllowedInGraphError
# Replacement: torch.allclose -> torch.sum
# We adapt the comparison logic to use torch.sum on the difference.
# Note: torch.sum does not accept atol/rtol, so we check if the sum of differences is significant.
diff = torch.sum(out_cpu - tf.errors.OperatorNotAllowedInGraphError(out_mps))
if diff > 1e-2:
    print("Output does not match!")
    print(out_cpu)
    print(out_mps)
