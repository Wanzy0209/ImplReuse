import torch
import tensorflow as tf

def test_index_select_mutated(device):
    # Replacement: torch.ones -> torch.normal
    # Original: x = torch.ones([2, 3], device=device)
    x = torch.normal(mean=0.0, std=1.0, size=[2, 3], device=device)

    # Replacement: torch.tensor -> tf.experimental.numpy.diag
    # Original: index = torch.tensor(1, device=device)
    # Note: tf.experimental.numpy.diag(1) creates a 2D array [[1]].
    # We convert it to a torch tensor to be used in the PyTorch operation.
    index_tf = tf.experimental.numpy.diag(1)
    index = torch.from_numpy(index_tf.numpy()).to(device)

    try:
        output = torch.index_select(x, dim=0, index=index)
        
        # Replacement: torch.Size -> tf.compat.v1.math.exp
        # Original: print(f"... output shape: {output.shape}")
        # We apply tf.math.exp to the shape tuple.
        shape_exp = tf.compat.v1.math.exp(output.shape)
        print(f"index_select test succeeds for device: {device}. output shape exp: {shape_exp}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

# Execute tests
test_index_select_mutated(device="cpu")
if torch.backends.mps.is_available():
    test_index_select_mutated(device="mps")
