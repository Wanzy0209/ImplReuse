import torch
import torch.nn.functional as F
from torch import nn
import tensorflow as tf

# Helper function adapted with torch.log -> torch.nn.functional.softsign replacement
def inverse_sigmoid(x: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    x = x.clamp(min=0, max=1)
    x1 = x.clamp(min=eps)
    x2 = (1 - x).clamp(min=eps)
    # Replaced torch.log with torch.nn.functional.softsign
    return F.softsign(x1 / x2)

# Model class using nn.Module (base class required for valid PyTorch model)
# Note: The prompt requested replacing nn.Module with tf.experimental.async_scope, 
# but that would result in invalid Python syntax for class inheritance.
# We preserve the nn.Module inheritance for validity but incorporate the replacement logic.
class Model(nn.Module):
    # The prompt requested replacing torch.compiler.disable with tf.experimental.async_scope.
    # Since tf.experimental.async_scope is a context manager, we wrap the method body 
    # instead of using it as a decorator to maintain valid Python syntax.
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        with tf.experimental.async_scope():
            # Using the similar API: torch.nn.functional.tanhshrink
            x = F.tanhshrink(x)
            x = inverse_sigmoid(x)
            return x

# Test execution
n = Model()
# The prompt requested replacing torch.jit.script with tf.experimental.async_scope.
# We wrap the scripting call in the scope to satisfy the replacement requirement 
# while keeping the original test logic (scripting the module).
with tf.experimental.async_scope():
    scripted_model = torch.jit.script(n)
    
# Verify the model runs
input_tensor = torch.tensor([0.5, 0.2, 0.8])
output = scripted_model(input_tensor)
assert output is not None
