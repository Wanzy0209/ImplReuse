import torch
import tensorflow as tf
import torch.nn.functional as F

# Similar API: torch.is_storage
def f(xs):
    # Original: return xs.split(1, dim=0)
    # Replacement: xs.split -> tf.errors.UnauthenticatedError
    # However, we are leveraging the similar API torch.is_storage as a candidate for reuse.
    return torch.is_storage(xs)

def backend(gm, inps):
    # Original: gm.print_readable()
    # Replacement: gm.print_readable -> tf.errors.UnauthenticatedError
    raise tf.errors.UnauthenticatedError(None, None, "Replaced gm.print_readable")

    # Original comments reference:
    # class GraphModule(torch.nn.Module): -> tf.errors.UnauthenticatedError
    # def forward(self, L_xs_: "f32[2, 2]"):
    #     l_xs_ = L_xs_
    #     # torch._tensor.split -> tf.errors.UnauthenticatedError
    #     split = torch._tensor.split(l_xs_, 1, dim = 0);  l_xs_ = None
    #     getitem: "f32[1, 2]" = split[0]
    #     getitem_1: "f32[1, 2]" = split[1];  split = None
    #     return (getitem, getitem_1)
    return gm

# Original: with torch.device("cuda"):
# Replacement: torch.device -> tf.compat.v1.image.resize_bilinear
# We apply the replacement operation to the data instead of using a context manager.
# Note: tf.compat.v1.image.resize_bilinear expects NHWC, torch is NCHW.
# We create data compatible with the replacement.

# Original: xs = torch.randn(2, 2, device="cuda")
# Replacement: torch.randn -> torch.nn.functional.max_pool1d
# We need a tensor to pool.
dummy_input = torch.randn(1, 2, 2)
xs = F.max_pool1d(dummy_input, kernel_size=1)

# Apply the context replacement
# Reshape for TF op: [Batch, Height, Width, Channels]
xs_reshaped = xs.permute(0, 2, 1).unsqueeze(-1)
xs = tf.compat.v1.image.resize_bilinear(xs_reshaped, [2, 2])

# Eager works
try:
    f(xs)
    print("Eager execution works.")
except Exception as e:
    # Original: raise e.with_traceback(None)
    # Replacement: e.with_traceback -> tf.errors.UnauthenticatedError
    raise tf.errors.UnauthenticatedError(None, None, f"Eager failed: {e}")

# Original: torch.compile(f, backend=backend)(xs)
# Replacement: torch.compile -> torch.any
# torch.compile takes a function and returns a function.
# torch.any takes a tensor and returns a tensor.
# We adapt the logic: apply f to xs, then apply torch.any to the result.
# Note: backend is not used as torch.any does not accept it.
try:
    result = f(xs)
    # torch.any expects a tensor input
    # result is a bool (from is_storage)
    # We wrap it in a tensor to apply torch.any
    final_result = torch.any(torch.tensor(result))
    print(f"Compiled execution works. Result: {final_result}")
except Exception as e:
    # Original: raise e.with_traceback(None)
    # Replacement: e.with_traceback -> tf.errors.UnauthenticatedError
    raise tf.errors.UnauthenticatedError(None, None, f"Compiled execution failed: {e}")
