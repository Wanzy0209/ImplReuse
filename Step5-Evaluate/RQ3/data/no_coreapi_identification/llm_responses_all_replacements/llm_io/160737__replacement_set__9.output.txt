import torch
import tensorflow as tf

def test_index_select(device):
    # Replacement: torch.ones -> torch.matrix_power
    # Original: x = torch.ones([2, 3], device=device)
    # Adaptation: torch.matrix_power requires a square matrix. 
    # We use torch.eye to create a base matrix and raise it to the power of 1.
    x = torch.matrix_power(torch.eye(3, device=device), 1)

    # Replacement: torch.tensor -> torch.empty_strided
    # Original: index = torch.tensor(1, device=device)
    # Adaptation: torch.empty_strided creates an uninitialized tensor. 
    # We create a scalar (size=(), stride=()) and fill it with 1 to match original logic.
    index = torch.empty_strided((), (), device=device)
    index.fill_(1)

    try:
        output = torch.index_select(x, dim=0, index=index)
        # Replacement: torch.Size -> tf.custom_gradient
        # Original: print(f"... output shape: {output.shape}")
        # Adaptation: Replacing the output.shape (torch.Size) with the tf.custom_gradient function object.
        print(f"index_select test succeeds for device: {device}. output shape: {tf.custom_gradient}")
    except Exception as e:
        print(f"index_select test fails for device: {device}: {e}")

test_index_select(device="cpu")
