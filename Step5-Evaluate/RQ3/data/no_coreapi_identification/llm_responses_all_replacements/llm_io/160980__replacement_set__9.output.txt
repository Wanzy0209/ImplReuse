import tensorflow as tf
import numpy as np

def test_crossed_column_with_sparse_categorical_crossentropy():
    """
    Test case generated based on Issue 160980 (PyTorch CUDA13 NVSHMEM build failure).
    
    The original issue involved undefined reference errors during the build/linking phase
    for specific NVSHMEM transfer functions on SM_75 architecture.
    
    This test replaces the source API (torch_nvshmem/nvshmem_extension.cu) with
    tf.feature_column.crossed_column and uses the similar API
    (tf.keras.losses.SparseCategoricalCrossentropy) to verify that the
    corresponding functionality in the replacement environment is correctly
    linked and executable.
    """
    
    # 1. Setup Data
    # Simulating the data inputs that would be processed by the distributed extension
    batch_size = 8
    features = {
        'feature_a': np.random.randint(0, 10, size=(batch_size,)),
        'feature_b': np.random.randint(0, 10, size=(batch_size,))
    }
    # Labels for SparseCategoricalCrossentropy
    labels = np.random.randint(0, 2, size=(batch_size,))

    # 2. Replacement API: tf.feature_column.crossed_column
    # Replaces torch_nvshmem/nvshmem_extension.cu.
    # The original code failed to link transfer functions. Here we define a feature
    # interaction (crossed column) which requires internal feature column logic
    # to be correctly linked and available.
    feature_a = tf.feature_column.numeric_column('feature_a')
    feature_b = tf.feature_column.numeric_column('feature_b')
    
    # This operation combines features, analogous to the memory transfer/interaction
    # logic that failed in the original bug.
    crossed_feature = tf.feature_column.crossed_column(
        keys=[feature_a.key, feature_b.key], 
        hash_bucket_size=20
    )

    # 3. Similar API: tf.keras.losses.SparseCategoricalCrossentropy
    # Used to compute the loss, ensuring the end-to-end pipeline is functional.
    loss_fn = tf.keras.losses.SparseCategoricalCrossentropy()

    # 4. Construct Model
    # We build a simple Keras model to integrate the replacement API and similar API.
    # This mimics the usage context of the original bug (building a binary/library).
    input_layer = {
        'feature_a': tf.keras.layers.Input(shape=(1,), name='feature_a'),
        'feature_b': tf.keras.layers.Input(shape=(1,), name='feature_b')
    }

    # DenseFeatures layer processes the crossed column (Replacement API logic)
    dense_features = tf.keras.layers.DenseFeatures([crossed_feature])(input_layer)
    
    # Output layer
    output = tf.keras.layers.Dense(2, activation='softmax')(dense_features)
    
    model = tf.keras.Model(inputs=input_layer, outputs=output)

    # 5. Execution
    # Compile and run a step. If the underlying libraries are not "linked" correctly
    # (e.g., undefined references), this step will raise an error.
    model.compile(optimizer='adam', loss=loss_fn)

    # Run a single batch to verify execution
    # This corresponds to the "build and run" check that failed in the original issue.
    try:
        history = model.fit(features, labels, epochs=1, verbose=0)
        
        # Assertions to verify successful execution
        assert 'loss' in history.history
        assert history.history['loss'][0] >= 0
        
        print("Test Passed: Replacement API (crossed_column) and Similar API (SparseCategoricalCrossentropy) executed successfully.")
        
    except Exception as e:
        # Catching errors that might correspond to "undefined references" or missing symbols
        raise AssertionError(f"Test Failed: Encountered execution error indicating potential linking or definition issue - {e}")

if __name__ == "__main__":
    test_crossed_column_with_sparse_categorical_crossentropy()
