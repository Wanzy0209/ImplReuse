```python
import torch
import torch.nn.functional as F

def addcmul_func(x, y, z):
    return x + (y * z)

# Replacing torch.randn with torch.nn.functional.max_pool1d
# We create a base tensor to apply the pooling operation.
# Using torch.ones to ensure valid input data for the operation.
base_input = torch.ones(1, 1, 256).to("xpu")

# Apply max_pool1d to generate the tensors.
# Kernel size 2 reduces the length from 256 to 128, matching the original shape.
x = F.max_pool1d(base_input, kernel_size=2).squeeze()
y = F.max_pool1d(base_input, kernel_size=2).squeeze()
z = F.max_pool1d(base_input, kernel_size=2).squeeze()

out = addcmul_func(x, y, z)
print("eager mode passed")

addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
print("torch.compile passed")
```
