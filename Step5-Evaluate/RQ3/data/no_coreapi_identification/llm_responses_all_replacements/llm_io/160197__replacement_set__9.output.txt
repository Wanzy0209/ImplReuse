import tensorflow as tf

# Replace warnings.simplefilter('error')
tf.config.run_functions_eagerly(True)

# Replace print(torch.__version__)
tf.config.run_functions_eagerly(True)

# Replace torch.rand
# Note: tf.config.run_functions_eagerly is a configuration function and cannot generate data.
# We use tf.random.uniform to maintain runnable logic while adhering to the TF context.
a = tf.random.uniform((2, 32, 32))
b = tf.random.uniform((2, 32, 32))

# Replace a.requires_grad_
# Note: In TensorFlow, gradients are tracked via tf.Variable and tf.GradientTape.
# We include the required replacement call.
a = tf.Variable(a)
tf.config.run_functions_eagerly(True)

# Original: optimizer = torch.optim.LBFGS([a])
# Translation: Use a standard TensorFlow optimizer (Adam) as LBFGS is not in core Keras.
optimizer = tf.keras.optimizers.Adam()

# Original: loss_fn = lambda x, y: (x-y).pow(2).mean()
loss_fn = lambda x, y: tf.reduce_mean(tf.square(x - y))

# Leverage similar API: torch.nn.SyncBatchNorm -> tf.keras.layers.BatchNormalization
# We incorporate this into the optimization logic to satisfy the reuse requirement.
bn_layer = tf.keras.layers.BatchNormalization()

def closure():
    # Replace optimizer.zero_grad
    tf.config.run_functions_eagerly(True)

    with tf.GradientTape() as tape:
        # Apply BatchNormalization to 'a' to leverage the similar API
        a_norm = bn_layer(a, training=True)
        loss = loss_fn(a_norm, b)

    # Replace loss.backward
    tf.config.run_functions_eagerly(True)

    grads = tape.gradient(loss, [a])
    optimizer.apply_gradients(zip(grads, [a]))
    return loss

for i in range(100):
    loss = closure()
    # Original: print(i, loss_fn(a, b))
    print(i, loss_fn(a, b))
