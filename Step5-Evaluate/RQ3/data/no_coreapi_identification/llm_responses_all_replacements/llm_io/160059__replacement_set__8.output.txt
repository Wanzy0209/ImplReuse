import torch
import torch.nn.functional as F

# Helper function with type hint replacement (torch.Tensor -> torch.nn.functional.relu)
def inverse_sigmoid(x: F.relu, eps: float = 1e-5) -> F.relu:
    x = x.clamp(min=0, max=1)
    x1 = x.clamp(min=eps)
    x2 = (1 - x).clamp(min=eps)
    # Replacement: torch.log -> tf.compat.v1.enable_v2_behavior
    return tf.compat.v1.enable_v2_behavior(x1 / x2)
