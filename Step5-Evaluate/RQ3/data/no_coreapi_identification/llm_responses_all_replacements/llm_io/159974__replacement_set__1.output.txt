```python
import torch

def addcmul_func(x, y, z):
    return x + (y * z)

# Replacing torch.randn with torch.var_mean
# To maintain the tensor shape (128) and ensure non-trivial computation,
# we generate a base tensor using torch.arange and compute its variance along dim 0.
# This prevents potential constant folding that might occur with constant inputs.
base_tensor = torch.arange(128, dtype=torch.float32).unsqueeze(0).repeat(128, 1)

x = torch.var_mean(base_tensor, dim=0)[0].to("xpu")
y = torch.var_mean(base_tensor, dim=0)[0].to("xpu")
z = torch.var_mean(base_tensor, dim=0)[0].to("xpu")

# Test eager mode
out = addcmul_func(x, y, z)
assert torch.isfinite(out).all(), "Eager mode produced NaN/Inf"
print("eager mode passed")

# Test torch.compile
addcmul_func_compiled = torch.compile(addcmul_func)
out = addcmul_func_compiled(x, y, z)
assert torch.isfinite(out).all(), "Compiled mode produced NaN/Inf"
print("torch.compile passed")
```
