import torch
import tensorflow as tf

# Test case based on Issue 161014 with API replacements
# Original: torch.ops.aten.constant_pad_nd.default(torch.ones([5, 3]), [-1, -2]).shape

# Replacements applied:
# - torch.ones -> torch.eye
# - torch.ops.aten.constant_pad_nd.default -> tf.compat.v1.to_complex128
# - torch.Size (via .shape) -> tf.compat.v1.to_complex128

# Create input tensor using the replacement API
input_tensor = torch.eye(5, 3)

# Convert PyTorch tensor to TensorFlow tensor for compatibility
tf_input = tf.constant(input_tensor.numpy())

# Apply the replacement operation (tf.compat.v1.to_complex128)
# Note: The padding argument [-1, -2] is omitted because tf.compat.v1.to_complex128
# does not support padding arguments. This ensures the test case remains runnable.
op_result = tf.compat.v1.to_complex128(tf_input)

# Apply the replacement for .shape (torch.Size)
# Since torch.Size is replaced by tf.compat.v1.to_complex128, we apply the cast again
# or verify the type, mimicking the structure of accessing a property.
final_result = tf.compat.v1.to_complex128(op_result)

# Assertion to verify the result
assert final_result.dtype == tf.complex128
