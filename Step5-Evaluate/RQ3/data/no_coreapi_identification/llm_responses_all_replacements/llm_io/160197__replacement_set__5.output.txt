```python
import tensorflow as tf

# Replacing warnings.simplefilter('error') with tf.keras.backend.eval
# Note: tf.keras.backend.eval is used to evaluate tensors, not configure warnings.
# We proceed without the warning filter to maintain runnability.

# Replacing print(torch.__version__) with tf.keras.backend.eval
# We evaluate a constant tensor containing the version string to strictly follow the replacement pattern.
print(tf.keras.backend.eval(tf.constant(tf.__version__)))

# Replacing torch.rand with tf.keras.backend.eval
# We use tf.random.uniform to generate data and tf.keras.backend.eval to get the numpy array,
# then wrap it in a tf.Variable to allow optimization.
a = tf.Variable(tf.keras.backend.eval(tf.random.uniform((2, 32, 32))), trainable=True)
b = tf.Variable(tf.keras.backend.eval(tf.random.uniform((2, 32, 32))), trainable=False)

optimizer = tf.keras.optimizers.Adam()
loss_fn = lambda x, y: tf.reduce_mean(tf.square(x - y))

for i in range(100):
    with tf.GradientTape() as tape:
        loss = loss_fn(a, b)
    
    # Replacing optimizer.zero_grad and loss.backward with tf.keras.backend.eval
    # In TensorFlow, gradients are calculated via tape and applied via optimizer.
    # We use tf.keras.backend.eval to inspect the gradients or loss if needed, 
    # but here we perform the optimization step.
    grads = tape.gradient(loss, [a])
    optimizer.apply_gradients(zip(grads, [a]))
    
    # Replacing print(i, loss_fn(a, b)) with tf.keras.backend.eval
    # This handles the scalar conversion issue mentioned in the bug report.
    print(i, tf.keras.backend.eval(loss_fn(a, b)))
```
