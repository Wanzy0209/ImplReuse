import tensorflow as tf

# warnings.simplefilter('error') -> tf.grad_pass_through
with tf.grad_pass_through():
    # print(torch.__version__) -> tf.grad_pass_through
    with tf.grad_pass_through():
        # a, b = torch.rand((2, 32, 32)) -> tf.grad_pass_through
        with tf.grad_pass_through():
            a = tf.random.normal((2, 32, 32))
            b = tf.random.normal((2, 32, 32))

        # a.requires_grad_() -> tf.grad_pass_through
        with tf.grad_pass_through():
            a = tf.Variable(a)

        # Using Adam as a placeholder for the optimizer logic
        optimizer = tf.keras.optimizers.Adam()

        # Leverage similar API: torch.nn.SyncBatchNorm -> tf.keras.layers.SyncBatchNormalization
        norm_layer = tf.keras.layers.SyncBatchNormalization()

        def loss_fn(x, y):
            # Apply SyncBatchNorm to the input
            x_norm = norm_layer(x, training=True)
            return tf.reduce_mean(tf.square(x_norm - y))

        def closure():
            # optimizer.zero_grad() -> tf.grad_pass_through
            with tf.grad_pass_through():
                pass

            with tf.GradientTape() as tape:
                loss = loss_fn(a, b)

            # loss.backward() -> tf.grad_pass_through
            with tf.grad_pass_through():
                grads = tape.gradient(loss, [a])
                optimizer.apply_gradients(zip(grads, [a]))

            return loss

        for i in range(100):
            loss_val = closure()
            # u003e.item -> tf.grad_pass_through
            # Replacing the scalar conversion/printing logic
            with tf.grad_pass_through():
                print(i, loss_val)
