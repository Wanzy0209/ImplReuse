### 🐛 Describe the bug

The following script will fail on the latest pytorch nightly and Triton main branch ( 2ac060b8b955cf4c6412458bea9df554342c1b1a):

```python

import triton
import torch

import torch._inductor.config as inductor_config

m = 20120
k = 1536
n = 512

a = torch.randn((m, n)).requires_grad_(False).cuda()
mat1 = torch.randn((m, k)).requires_grad_(False).cuda()
mat2 = torch.randn((k, n)).requires_grad_(False).cuda()
f = lambda a, mat1, mat2: torch.addmm(a, mat1, mat2)

with inductor_config.patch(
    max_autotune=True,
    max_autotune_gemm_backends="TRITON",
    autotune_fallback_to_aten=False,):
    compiled = torch.compile(f, dynamic=False)
    compiled(a, mat1, mat2)
