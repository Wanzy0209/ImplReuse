class Isin(torch.nn.Module):

        self.device = device
        self.x = torch.randint(-50, 50, (1,), dtype=torch.int64, device=device)

    def forward(self):

        print(self.y)
        out = torch.isin(self.x, self.y, assume_unique=False, invert=False)
        return {'out': out}
