```python
import torch
import torch.nn as nn


class Config:
    def __repr__(self):
        return "Config()"


def forward(x, config):
    # Calling repr() on non-constant user object
    # This triggers the bug without the fix

compiled = torch.compile(forward, fullgraph=True)
```
