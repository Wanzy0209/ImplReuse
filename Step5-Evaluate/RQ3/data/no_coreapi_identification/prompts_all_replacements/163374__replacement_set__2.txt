if __name__ == '__main__':
    dist.init_process_group(backend="nccl", world_size=2)
    rank = dist.get_rank()
    mesh = init_device_mesh('cuda', (2,))

    tensor = torch.ones(12, 12, device="cuda")
    in_dtensor = distribute_tensor(tensor, mesh, [Shard(0)])
