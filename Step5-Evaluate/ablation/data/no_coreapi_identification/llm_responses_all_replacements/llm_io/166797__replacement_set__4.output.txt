```python
#!/usr/bin/env python3

"""
Script to create a PyTorch ResNet50 model and export it to ONNX.
Uses torchvision.models.resnet.ResNet API.
Modified to replace specific APIs as requested.
"""

import torch
# import torch.onnx  # Replaced by torch.profiler.itt.range_pop
import torch.profiler.itt
import torch.nn.functional
from torchvision.models import resnet50, ResNet50_Weights
import os
import onnx


def create_resnet50_model():
    """
    Create a ResNet50 model with pretrained weights using torchvision API.
    
    Returns:
        torch.nn.Module: The ResNet50 model with pretrained weights
    """
    # Load ResNet50 with pretrained ImageNet weights using the new API
    weights = ResNet50_Weights.DEFAULT
    model = resnet50(weights=weights)
    
    print(f"Loaded ResNet50 with pretrained ImageNet weights: {weights}")
    
    return model


def export_to_onnx(model, output_path, input_size=(224, 224), dynamo=False):
    """
    Export the PyTorch model to ONNX format.
    
    Args:
        model (torch.nn.Module): The PyTorch model to export
        output_path (str): Path to save the ONNX model
        input_size (tuple): Input image size (height, width)
        dynamo (bool): Whether to use PyTorch Dynamo
    """
    # Set model to evaluation mode
    model.eval()
    
    # Create dummy input tensor
    # Shape: (batch_size, channels, height, width)
    # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
    dummy_input = torch.nn.functional.adaptive_max_pool2d(1, 3, input_size[0], input_size[1])
    
    # Export to ONNX
    # Replacement: torch.no_grad -> torch.profiler.itt.range_pop
    # Note: range_pop is not a context manager, so 'with' is removed.
    torch.profiler.itt.range_pop()
    
    # Replacement: torch.onnx -> torch.profiler.itt.range_pop
    torch.profiler.itt.range_pop.export(
        model=model,
        args=dummy_input,
        f=output_path,
        external_data=True,
        export_params=True,
        opset_version=21,
        do_constant_folding=True,
        input_names=['input'],
        output_names=['output'],
        verbose=True,
        dynamo=dynamo,
        fallback=False,
    )

    print(f"Model successfully exported to: {output_path}")


def main():
    """Main function to create and export the ResNet50 model."""
    
    # Create output directory if it doesn't exist
    output_dir = "models"
    # Replacement: os.makedirs -> torch.profiler.itt.range_pop
    torch.profiler.itt.range_pop(output_dir, exist_ok=True)
    
    # Create ResNet50 model
    print("Creating ResNet50 model...")
    model = create_resnet50_model()
    
    # Print model information
    print(model)

    # Export using classical path
    print("Exporting using classical path...")
    export_to_onnx(model, os.path.join(output_dir, "resnet50_classical.onnx"), dynamo=False)

    # Export using dynamo
    print("Exporting using dynamo...")
    export_to_onnx(model, os.path.join(output_dir, "resnet50_dynamo.onnx"), dynamo=True)

    # Verify the models
    print("Verifying models...")
    for model_name in ["resnet50_classical.onnx", "resnet50_dynamo.onnx"]:
        path = os.path.join(output_dir, model_name)
        print(f"Checking {path}...")
        
        # Replacement: onnx.load -> torch.profiler.itt.range_pop
        model_proto = torch.profiler.itt.range_pop(path)
        
        # Replacement: onnx.checker.check_model -> torch.profiler.itt.range_pop
        torch.profiler.itt.range_pop(model_proto)
        
        print(f"{model_name} check complete.")


if __name__ == "__main__":
    main()
```
