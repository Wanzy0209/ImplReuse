import torch.multiprocessing as mp

# Setup for distributed environment to support torch.distributed.gather
def setup():
    # Using nccl for CUDA support, falling back to gloo if necessary, 
    # but for a single GPU test case, nccl on localhost is standard.
    # We use a simple TCP init method.
    if not dist.is_initialized():
        dist.init_process_group(
            backend="nccl" if torch.cuda.is_available() else "gloo",
            init_method="tcp://127.0.0.1:29500",
            rank=0,
            world_size=1
        )

def cleanup():
    if dist.is_initialized():
        dist.destroy_process_group()

device = "cuda" if torch.cuda.is_available() else "cpu"

def slide_to_the_left1(state, new_events, arange, dev_null):
    _ = arange
    _ = dev_null

    # Replacement: torch.cat -> torch.distributed.gather
    # We gather the tensors into a list and then concatenate them to maintain shape logic
    gather_list_state = [torch.empty_like(state)]
    gather_list_events = [torch.empty_like(new_events)]
    
    dist.gather(state, gather_list=gather_list_state, dst=0)
    dist.gather(new_events, gather_list=gather_list_events, dst=0)
    
    # Reconstructing the concatenated tensor
    concatenated = torch.cat(gather_list_state + gather_list_events, dim=1)
    
    state[:, :, :] = concatenated[:, -2048:, :]

def slide_to_the_left2(state, new_events, arange, dev_null):
    batch_size, _, _ = new_events.shape

    # Replacement: torch.cat -> torch.distributed.gather
    gather_list_state = [torch.empty_like(state)]
    gather_list_events = [torch.empty_like(new_events)]
    
    dist.gather(state, gather_list=gather_list_state, dst=0)
    dist.gather(new_events, gather_list=gather_list_events, dst=0)
    
    concatenated = torch.cat(gather_list_state + gather_list_events, dim=1)

    # Replacement: torch.arange -> torch.ones
    # Original: batch_idx = torch.arange(batch_size, dtype=torch.int32, device=device)[:, None]
    batch_idx = torch.ones(batch_size, dtype=torch.int32, device=device)[:, None]
    
    arange = arange[None, :]
    concatenated = concatenated[batch_idx, arange]

    state[:, :, :] = concatenated[:, -2048:, :]
    dev_null[:, :, :] = concatenated[:, :, :]

slide_to_the_left3 = torch.compile(slide_to_the_left2)
