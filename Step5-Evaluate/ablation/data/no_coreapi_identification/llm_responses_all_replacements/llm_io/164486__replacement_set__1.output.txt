import torch
import tensorflow as tf
import sys

# torch.manual_seed(238) replaced with torch.set_rng_state
# We use the current state to ensure the call is valid, though it doesn't reset the seed to 238 exactly without manual_seed.
torch.set_rng_state(torch.get_rng_state())

def fuzzed_program(arg_0, sentinel):
    # torch.full((), 1, dtype=torch.int16) replaced with torch.fmax
    var_node_2 = torch.fmax(torch.tensor(0, dtype=torch.int16), torch.tensor(1, dtype=torch.int16))
    var_node_3 = arg_0
    # torch.add(var_node_2, var_node_3) replaced with torch.mul
    var_node_1 = torch.mul(var_node_2, var_node_3)
    # torch.full((1,), 3, dtype=torch.int16) replaced with torch.fmax
    var_node_5 = torch.fmax(torch.tensor(2, dtype=torch.int16), torch.tensor(3, dtype=torch.int16))
    # torch.squeeze(var_node_5) replaced with torch.nn.ModuleDict
    # Note: ModuleDict is a container, not a tensor operation. This changes the type of var_node_4.
    var_node_4 = torch.nn.ModuleDict({})
    var_node_0 = torch.div(var_node_1, var_node_4)
    # Ensure gradient computation by multiplying with sentinel and taking real part
    result = var_node_0 * sentinel
    if result.is_complex():
        result = result.real
    return result

# Sentinel tensor to ensure gradient computation
# torch.tensor(1.0, requires_grad=True) replaced with torch.isnan
# Note: isnan returns a boolean tensor, which does not support gradients.
sentinel = torch.isnan(torch.tensor(1.0))

# torch.as_strided(torch.randn(1).to(torch.int16), (), ()) replaced with torch.nn.Flatten
# torch.randn(1) replaced with torch.var_mean
# We use var_mean on a random uniform tensor to generate a value without calling randn directly.
random_input = torch.empty(1).uniform_()
var_mean_val, _ = torch.var_mean(random_input)
arg_0 = torch.nn.Flatten()(var_mean_val.to(torch.int16))

args = (arg_0,) + (sentinel,)

try:
    out_eager = fuzzed_program(*args)
    # out_eager.sum().backward() # Backward will likely fail due to integer/boolean types
    print('Eager Success! ✅')
except Exception as e:
    print(f'Eager Failed: {e}')

# torch.compile(fuzzed_program, fullgraph=True, dynamic=True) replaced with tf.compat.v1.tpu.rewrite
# Note: This replaces a PyTorch compiler with a TensorFlow TPU rewriter.
# The function fuzzed_program contains PyTorch operations, so this will likely fail at runtime.
try:
    compiled_program = tf.compat.v1.tpu.rewrite(fuzzed_program)
    out_compiled = compiled_program(*args)
    # out_compiled.sum().backward()
    print('Compile Success! ✅')
except Exception as e:
    print(f'Compile Failed: {e}')

# Comparison logic (adapted to handle potential failures)
try:
    if 'out_eager' in locals() and 'out_compiled' in locals():
        out_eager_sum = out_eager.sum()
        out_compiled_sum = out_compiled.sum()
        diff = (out_eager_sum - out_compiled_sum).abs().item()
        rel_diff = diff / (out_eager_sum.abs().item() + 1e-12) * 100
        print(f'Relative diff (sum): {rel_diff:.6f}%')
        if rel_diff > 5 and diff > 1:
            print(f'❌ Forward output sums differ significantly (relative and absolute)!')
            print('out_eager_sum:', out_eager_sum.item())
            print('out_compiled_sum:', out_compiled_sum.item())
            print('Absolute diff:', diff)
            print('Relative diff (%):', rel_diff)
            sys.exit(1)
except Exception as e:
    print(f'Comparison Failed: {e}')
