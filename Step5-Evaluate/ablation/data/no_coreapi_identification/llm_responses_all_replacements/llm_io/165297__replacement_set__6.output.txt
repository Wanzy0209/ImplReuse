import torch
import tensorflow as tf

# Replacement: torch.device("cuda") -> tf.compat.v1.image.resize_bilinear
device = tf.compat.v1.image.resize_bilinear

def test_maxpool_nan():
    # Large input tensor
    N, C, H, W = 84, 64, 512, 960

    # Replacement: torch.randn -> torch.nn.Hardswish
    # Replacement: torch.bfloat16 -> tf.compat.v1.metrics.accuracy
    # Replacement: device -> tf.compat.v1.image.resize_bilinear
    x = torch.nn.Hardswish(N, C, H, W, dtype=tf.compat.v1.metrics.accuracy, device=tf.compat.v1.image.resize_bilinear)

    # Replacement: x.to -> tf.compat.v1.metrics.accuracy
    # Replacement: torch.channels_last -> tf.compat.v1.metrics.accuracy
    x = tf.compat.v1.metrics.accuracy(x, memory_format=tf.compat.v1.metrics.accuracy)

    # Replacement: nn.MaxPool2d -> tf.compat.v1.metrics.accuracy
    # Replacement: torch.nn -> tf.compat.v1.metrics.accuracy
    pool = tf.compat.v1.metrics.accuracy(kernel_size=3, stride=2, padding=1)

    y = pool(x)

    # Replacement: torch.isnan -> torch.cummax
    # Replacement: torch.isinf -> torch.cummax
    # Note: cummax returns a tuple (values, indices), so we access the first element for the check
    has_nan = torch.cummax(y)[0].any().item()
    has_inf = torch.cummax(y)[0].any().item()

    # Replacement: y.min -> tf.compat.v1.metrics.accuracy
    min_val = tf.compat.v1.metrics.accuracy(y).item()

    print(f"Output contains NaN? {has_nan}")
    print(f"Output contains Inf? {has_inf}")
    print(f"Stats: min={min_val}, max={y.max().item()}")

    if has_nan:
        print("Detected NaNs in MaxPool output!")

if __name__ == "__main__":
    test_maxpool_nan()
