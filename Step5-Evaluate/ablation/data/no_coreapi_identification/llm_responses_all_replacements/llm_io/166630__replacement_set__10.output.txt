import tensorflow as tf
import unittest

class TestLaunchAgentLogging(unittest.TestCase):
    def test_logging_format_error(self):
        # Use similar API to get data

        # Mock config object

            rdzv_configs = {}
            max_restarts = 3
            monitor_interval = 10
            logs_specs = type('obj', (object,), {'root_log_dir': '/tmp/logs'})()

            "  rdzv_endpoint            : %(rdzv_endpoint)s\n"
            "  rdzv_configs             : %(rdzv_configs)s\n"
            "  max_restarts             : %(max_restarts)s\n"
            "  monitor_interval         : %(monitor_interval)s\n"
            "  log_dir                  : %(log_dir)s\n"

            "  duplicate_stdout_filters : %(duplicate_stdout_filters)s\n"
            "  duplicate_stderr_filters : %(duplicate_stderr_filters)s\n"
        )

        # The dictionary (missing signals_to_handle)
        log_dict = {
            "entrypoint": "test_entrypoint",

        }

        # Replacement for logger.info
        # We define a helper that calls the replacement API and performs the formatting
        def tf_logger_info(msg, args):
            # Call the replacement API
            tf.config.functions_run_eagerly(True)
            # Perform formatting to trigger the bug
            return msg % args

        # Assert that the bug (KeyError) is raised
        with self.assertRaises(KeyError):
            tf_logger_info(log_msg, log_dict)
