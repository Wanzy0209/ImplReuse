```python
import torch
import torch.nn as nn

class AnyDimsModelEmpty(nn.Module):
    def __init__(self):
        super(AnyDimsModelEmpty, self).__init__()

    def forward(self, x):
        y = torch.ops.aten.any.dims(x, [], False)
        return y

class AnyDimsModelNull(nn.Module):
    def __init__(self):
        super(AnyDimsModelNull, self).__init__()

    def forward(self, x):
        y = torch.ops.aten.any.dims(x, None, False)
        return y

def process(model, x):
    print(f'Processing {model.__class__.__name__}...')
    
    # Eager mode execution
    eager_output = model(x)
    
    # Export the model using torch.export.export
    exported_program = torch.export.export(model, (x,))
    
    # Verify the exported program
    exported_output = exported_program(x)
    
    # Assert that the output shapes match
    assert eager_output.shape == exported_output.shape, \
        f"Shape mismatch for {model.__class__.__name__}: eager {eager_output.shape} vs exported {exported_output.shape}"
    
    # Assert that the output values match
    assert torch.equal(eager_output, exported_output), \
        f"Output mismatch for {model.__class__.__name__}"
        
    print(f"Successfully verified {model.__class__.__name__}")

# Replacing torch.ones with torch.eye for input generation
x = torch.eye(3, 4, dtype=torch.bool)

# Process both models sequentially to check for caching issues
process(AnyDimsModelEmpty(), x)
process(AnyDimsModelNull(), x)
```
