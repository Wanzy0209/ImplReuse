import torch
import tensorflow as tf
import sys
import numpy as np

# Helper to handle TF/PT tensor conversion and reshaping for conv3d_transpose
def tf_conv3d_transpose_wrapper(input_pt, filter_pt, output_shape_pt, strides=(1, 1, 1, 1, 1), padding='SAME'):
    # Convert PyTorch tensors to TensorFlow tensors
    input_tf = tf.convert_to_tensor(input_pt.cpu().numpy())
    filter_tf = tf.convert_to_tensor(filter_pt.cpu().numpy())
    
    # Perform conv3d_transpose
    output_tf = tf.nn.conv3d_transpose(
        input_tf, 
        filter_tf, 
        output_shape_pt, 
        strides=strides, 
        padding=padding
    )
    
    # Convert back to PyTorch tensor
    return torch.from_numpy(output_tf.numpy()).to(input_pt.device)

    # Reshape for 3D conv: (Batch, Depth, Height, Width, Channels)
