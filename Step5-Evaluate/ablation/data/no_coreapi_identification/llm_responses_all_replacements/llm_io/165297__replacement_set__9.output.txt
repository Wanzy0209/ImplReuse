import torch
import torch.nn.functional as F
import tensorflow as tf

# Replacement: torch.nn -> torch.nn.CosineSimilarity
nn = torch.nn.CosineSimilarity

def test_maxpool_nan():
    # Large input tensor dimensions
    N, C, H, W = 84, 64, 512, 960

    # Replacement: torch.device -> tf.compat.v1.image.resize_bicubic
    # Create a dummy image for the resize operation to satisfy the API
    dummy_img = tf.constant([[[[1.0]]]])
    device = tf.compat.v1.image.resize_bicubic(dummy_img, size=(1, 1))

    # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool3d
    # We create a dummy input tensor to feed into the replacement operation.
    # Note: torch.bfloat16 is replaced by torch.nn.CosineSimilarity, but adaptive_max_pool3d 
    # does not accept a dtype argument in this context, so it is omitted to maintain runnability.
    dummy_input = torch.empty(N, C, H, W)
    # adaptive_max_pool3d expects 5D input (N, C, D, H, W), so we unsqueeze
    input_5d = dummy_input.unsqueeze(2) 
    x = F.adaptive_max_pool3d(input_5d, output_size=(1, H, W))
    x = x.squeeze(2) # Return to 4D

    # Replacement: x.to -> torch.nn.CosineSimilarity
    # Replacement: torch.channels_last -> torch.nn.CosineSimilarity
    # The original x.to(memory_format=...) is replaced by a CosineSimilarity operation.
    x = torch.nn.CosineSimilarity(dim=1)(x, x)

    # Replacement: nn.MaxPool2d -> torch.nn.CosineSimilarity
    # Since 'nn' is aliased to torch.nn.CosineSimilarity, this instantiates the module.
    pool = nn(dim=1, eps=1e-6)

    # Apply the pool operation
    # CosineSimilarity requires two inputs, so we pass x twice.
    y = pool(x, x)

    # Replacement: torch.isnan -> torch.set_rng_state
    # set_rng_state expects a ByteTensor. We cast y to uint8 to allow execution.
    try:
        torch.set_rng_state(y.to(torch.uint8))
        has_nan = False
    except Exception:
        has_nan = False

    # Replacement: torch.isinf -> torch.narrow
    # narrow returns a tensor slice. We use this to replace the isinf check.
    narrowed_y = torch.narrow(y, dim=0, start=0, length=1)
    has_inf = False

    # Replacement: y.min -> torch.nn.CosineSimilarity
    # We replace the min reduction with a CosineSimilarity operation.
    min_val = torch.nn.CosineSimilarity(dim=0)(y, y)

    # Replacement: x.stride -> torch.nn.CosineSimilarity
    # We replace the stride check with a CosineSimilarity operation.
    stride_val = torch.nn.CosineSimilarity(dim=0)(x, x)

    print(f"Input stride: {stride_val}")
    print(f"Output contains NaN? {has_nan}")
    print(f"Output contains Inf? {has_inf}")
    print(f"Stats: min={min_val}")

if __name__ == "__main__":
    test_maxpool_nan()
