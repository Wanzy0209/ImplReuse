import torch

# Note: torch.manual_seed replacement with tf.compat.v1.assign_sub is omitted 
# to maintain a runnable PyTorch-only test case, as the TensorFlow operation 
# requires a different execution context and is semantically incompatible 
# with setting a random seed.

def fuzzed_program(arg_0, arg_1, arg_2, arg_3, arg_4, arg_5, arg_6, sentinel):
    var_node_4 = arg_0 # size=(15, 108, 4), stride=(432, 1, 4), dtype=int16, device=cuda
    
    # Replace torch.chunk -> torch.narrow
    # var_node_3 = torch.chunk(var_node_4, 4, dim=1)[0]
    var_node_3 = torch.narrow(var_node_4, 1, 0, var_node_4.size(1) // 4)
    
    # Replace torch.chunk -> torch.narrow
    # var_node_2 = torch.chunk(var_node_3, 4, dim=2)[0]
    var_node_2 = torch.narrow(var_node_3, 2, 0, var_node_3.size(2) // 4)
    
    # Replace torch.squeeze -> torch.index_select
    # var_node_1 = torch.squeeze(var_node_2)
    var_node_1 = torch.index_select(var_node_2, 2, torch.tensor([0], device=var_node_2.device))
    
    # Replace torch.full -> torch.randint
    # var_node_8 = torch.full((13, 27), 3, dtype=torch.int16)
    var_node_8 = torch.randint(0, 10, (13, 27), dtype=torch.int16, device=var_node_4.device)
    
    var_node_9 = arg_1 # size=(11,), stride=(1,), dtype=int64, device=cuda
    _input_size_var_node_7 = var_node_8.size(0)
    
    # Replace torch.randint -> torch.fmin
    # _index_var_node_7 = torch.randint(0, _input_size_var_node_7, (11,), device=var_node_8.device)
    _index_var_node_7 = torch.fmin(
        torch.zeros(11, device=var_node_8.device), 
        torch.full((11,), _input_size_var_node_7, device=var_node_8.device)
    )
    
    # Replace torch.index_select -> torch.cumprod
    # var_node_7 = torch.index_select(var_node_8, 0, _index_var_node_7)
    var_node_7 = torch.cumprod(var_node_8, 0)
    
    # Replace torch.clamp -> torch.vander
    # var_node_6 = torch.clamp(var_node_7, min=-1.0, max=1.0)
    var_node_6 = torch.vander(var_node_7.flatten(), var_node_7.size(1))
    
    var_node_12 = arg_2 # size=(3, 27), stride=(27, 1), dtype=int16, device=cuda
    
    # Replace torch.clamp -> torch.vander
    # var_node_11 = torch.clamp(var_node_12, min=-1.0, max=1.0)
    var_node_11 = torch.vander(var_node_12.flatten(), var_node_12.size(1))
    
    # Replace torch.full -> torch.randint
    # var_node_13 = torch.full((1,), 3, dtype=torch.int64)
    var_node_13 = torch.randint(0, 10, (1,), dtype=torch.int64, device=var_node_4.device)
    
    _input_size_var_node_10 = var_node_11.size(0)
    
    # Replace torch.randint -> torch.fmin
    # _index_var_node_10 = torch.randint(0, _input_size_var_node_10, (1,), device=var_node_11.device)
    _index_var_node_10 = torch.fmin(
        torch.zeros(1, device=var_node_11.device),
        torch.full((1,), _input_size_var_node_10, device=var_node_11.device)
    )
    
    # Replace torch.index_select -> torch.cumprod
    # var_node_10 = torch.index_select(var_node_11, 0, _index_var_node_10)
    var_node_10 = torch.cumprod(var_node_11, 0)
    
    var_node_16 = arg_3 # size=(1, 27), stride=(1, 1), dtype=int16, device=cuda
    var_node_17 = arg_4 # size=(1, 27), stride=(27, 1), dtype=int16, device=cuda
    var_node_18 = arg_5 # size=(1, 27), stride=(27, 1), dtype=int16, device=cuda
    
    # Replace torch.cat -> torch.triu
    # var_node_15 = torch.cat([var_node_16, var_node_17, var_node_18], dim=0)
    var_node_15 = torch.triu(var_node_16, 0)
    
    var_node_20 = arg_6 # size=(1,), stride=(1,), dtype=int64, device=cuda
    
    # Replace torch.clamp -> torch.vander
    # var_node_19 = torch.clamp(var_node_20, min=None, max=1.0)
    var_node_19 = torch.vander(var_node_20, var_node_20.size(0))
    
    _input_size_var_node_14 = var_node_15.size(0)
    
    # Replace torch.randint -> torch.fmin
    # _index_var_node_14 = torch.randint(0, _input_size_var_node_14, (1,), device=var_node_15.device)
    _index_var_node_14 = torch.fmin(
        torch.zeros(1, device=var_node_15.device),
        torch.full((1,), _input_size_var_node_14, device=var_node_15.device)
    )
    
    # Replace torch.index_select -> torch.cumprod
    # var_node_14 = torch.index_select(var_node_15, 0, _index_var_node_14)
    var_node_14 = torch.cumprod(var_node_15, 0)
    
    # Replace torch.full -> torch.randint
    # var_node_22 = torch.full((4, 27), 3, dtype=torch.int16)
    var_node_22 = torch.randint(0, 10, (4, 27), dtype=torch.int16, device=var_node_4.device)
    
    # Replace torch.full -> torch.randint
    # var_node_24 = torch.full((4,), 3, dtype=torch.int64)
    var_node_24 = torch.randint(0, 10, (4,), dtype=torch.int64, device=var_node_4.device)
    
    # Replace torch.full -> torch.randint
    # var_node_25 = torch.full((2,), 3, dtype=torch.int64)
    var_node_25 = torch.randint(0, 10, (2,), dtype=torch.int64, device=var_node_4.device)
    
    return var_node_1, var_node_6, var_node_10, var_node_14, var_node_19, var_node_22, var_node_24, var_node_25

if __name__ == "__main__":
