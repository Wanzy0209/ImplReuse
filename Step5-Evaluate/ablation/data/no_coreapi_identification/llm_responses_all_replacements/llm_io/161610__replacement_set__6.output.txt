import tensorflow as tf
from typing import NamedTuple

# Define the NamedTuple structure
class MyNamedTuple(NamedTuple):
    # Replacing torch.Tensor with tf.Tensor (type hint)
    # Note: tf.keras.backend.batch_set_value is a function, not a type, so it is used in the logic below.
    first: tf.Tensor
    second: tf.Tensor

# Subclass to allow dynamic attributes (standard Python NamedTuple is immutable)
class MyNamedTupleSubclass(MyNamedTuple):
    __slots__ = ()

def fn(tup: MyNamedTupleSubclass) -> MyNamedTupleSubclass:
    # Replacement: torch.tensor(4.0) -> tf.linalg.diag([4.0])
    extra_info = tf.linalg.diag([4.0])

    # Replacement: torch.Tensor -> tf.keras.backend.batch_set_value
    # Using batch_set_value to modify the tensor in-place to satisfy the API replacement requirement
    tf.keras.backend.batch_set_value([(tup.first, [5.0])])

    # Add dynamic attribute
    tup.extra_info = extra_info
    return tup

print("\nTesting NamedTuple with __setattr__ (Eager):")
# Replacement: torch.tensor([2.0]) -> tf.linalg.diag([2.0])
extended_tup = MyNamedTupleSubclass(first=tf.linalg.diag([2.0]), second=tf.linalg.diag([1.0]))
setattr_result = fn(extended_tup)
print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")

print("\nTesting NamedTuple with __setattr__ (Compiled):")
extended_tup = MyNamedTupleSubclass(first=tf.linalg.diag([2.0]), second=tf.linalg.diag([1.0]))
# Replacement: torch.compile -> tf.function
compiled_fn = tf.function(fn)
setattr_result = compiled_fn(extended_tup)
print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
