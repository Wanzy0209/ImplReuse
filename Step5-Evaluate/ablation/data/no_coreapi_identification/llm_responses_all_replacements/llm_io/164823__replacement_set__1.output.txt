# Replacing nn.Module with a function that utilizes tf.control_dependencies
# to structure the execution flow, mirroring the original model's logic.
def test_model(x):

    # We use tf.control_dependencies to ensure the input is ready before conversion.
    with tf.control_dependencies([x]):
        # Create a sparse tensor from the dense input
        indices = tf.where(tf.not_equal(x, 0))
        values = tf.gather_nd(x, indices)
        x_sparse = tf.SparseTensor(indices, values, tf.cast(tf.shape(x), tf.int64))

    # Replacing the multiplication logic (x_sparse * 2).
    # We use tf.control_dependencies to sequence the operation after the sparse conversion.
    with tf.control_dependencies([x_sparse]):
        result = x_sparse * 2

    # Replacing result.to_dense() logic.
    # We use tf.control_dependencies to sequence the dense conversion after the math.
    with tf.control_dependencies([result]):
        dense_result = tf.sparse.to_dense(result)

    return dense_result
