import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset
import torch.optim.swa_utils

# 1. Data generation: torch.arange -> torch.linspace
# Original: [torch.arange(0, n) for n in (10, 20, 30)]
# We create compatible tensors for the subsequent operations.
# To ensure the test is runnable with torch.cdist, we use fixed dimensions.
x1 = torch.linspace(0, 10, steps=10).unsqueeze(0)
x2 = torch.linspace(0, 10, steps=10).unsqueeze(0)

# 2. Replacement for torch.nested.nested_tensor -> torch.optim.swa_utils.update_bn
# This function requires a model and a data loader.
# We set up a dummy model and loader to satisfy the API call.
model = nn.Linear(10, 10)
dummy_data = torch.randn(10, 10)
loader = DataLoader(TensorDataset(dummy_data), batch_size=2)

# Call the replacement function
torch.optim.swa_utils.update_bn(loader, model)

# 3. Replacement for torch.nested.nested_tensor(...,layout=torch.jagged).max(dim=...) -> torch.cdist
# Original: x.max(dim=1).values
# New: torch.cdist(x1, x2)
result = torch.cdist(x1, x2)

# 4. Expected Result and Assertion
# Original: tensor([9, 19, 29])
# New: Distance between identical vectors is 0.
expected = torch.zeros(1, 1)

assert torch.allclose(result, expected), f"Expected {expected}, got {result}"
