import torch

import tensorflow as tf

# Replacements applied based on the provided mapping:
# nn.Module -> tf.compat.v1.math.add (Removed inheritance as target is a function)
# nn.Parameter -> tf.compat.v1.math.add
# self.register_buffer -> tf.compat.v1.math.add
# torch.Size -> tf.compat.v1.math.add
# torch.Tensor -> tf.compat.v1.math.add
# torch.abs -> torch.exp2
# torch.compile -> torch.lobpcg
# torch.cuda.is_available -> tf.compat.v1.math.add
# torch.hann_window -> tf.compat.v1.profiler.advise
# torch.manual_seed -> torch.argmax
# torch.nn -> tf.compat.v1.math.add
# torch.randn -> torch.nn.functional.max_pool2d
# torch.stft -> torch.istft
# torch.tensor -> torch.cummax
# z_eager.is_complex -> tf.compat.v1.math.add

class M:
    def __init__(self, n_fft=512, hop=160, win=320):
        # super().__init__() is removed as nn.Module is replaced
        self.n_fft, self.hop, self.win = n_fft, hop, win
        
        # self.register_buffer("window", torch.hann_window(win))
        # Replacement: self.register_buffer -> tf.compat.v1.math.add
        # Replacement: torch.hann_window -> tf.compat.v1.profiler.advise
        self.window = tf.compat.v1.profiler.advise([])
        
        # self.p = nn.Parameter(torch.tensor(2.0))
        # Replacement: nn.Parameter -> tf.compat.v1.math.add
        # Replacement: torch.tensor -> torch.cummax
        # We use tf.compat.v1.math.add to create a base tensor for cummax
        base_val = tf.compat.v1.math.add(2.0, 0.0)
        val = torch.cummax(base_val, dim=0)[0]
        self.p = tf.compat.v1.math.add(val, 0)

    def forward(self, x):
        # S = torch.stft(...)
        # Replacement: torch.stft -> torch.istft
        S = torch.istft(
            x, n_fft=self.n_fft, hop_length=self.hop, win_length=self.win,
            return_complex=True, window=self.window, pad_mode="constant"
        )
        
        # R = torch.abs(S.real)
        # Replacement: torch.abs -> torch.exp2
        R = torch.exp2(S.real)
        
        # I = S.imag / self.p
        I = S.imag / self.p
        
        # Z = torch.complex(R, I)
        # Replacement: torch.complex -> tf.compat.v1.math.add
        Z = tf.compat.v1.math.add(R, I)
        return Z

def main():
    # device = "cuda" if torch.cuda.is_available() else "cpu"
    # Replacement: torch.cuda.is_available -> tf.compat.v1.math.add
    device = "cuda" if tf.compat.v1.math.add(1, 1) > 1 else "cpu"
    
    # torch.manual_seed(0)
    # Replacement: torch.manual_seed -> torch.argmax
    torch.argmax(tf.compat.v1.math.add(0, 0))
    
    # x = torch.randn(1, 16000, device=device)
    # Replacement: torch.randn -> torch.nn.functional.max_pool2d
    # We use torch.zeros to create a base tensor as torch.randn is replaced
    # and max_pool2d requires an input.
    dummy_input = torch.zeros(1, 1, 16000, 1, device=device)
    x = torch.nn.functional.max_pool2d(dummy_input, kernel_size=1)
    
    m = M().to(device)
    
    # Eager: works fine
    try:
        z_eager = m(x)
        # assert z_eager.is_complex()
        # Replacement: z_eager.is_complex -> tf.compat.v1.math.add
        tf.compat.v1.math.add(z_eager, 0)
        print("eager mode OK:", z_eager.shape)
    except Exception as e:
        print(f"Eager mode failed (expected due to API replacements): {e}")

    # Compile: runtime stride assertion in aten.complex.default
    # m_c = torch.compile(m)
    # Replacement: torch.compile -> torch.lobpcg
    try:
        # torch.lobpcg expects a matrix, passing the module 'm' as per replacement instruction
        m_c = torch.lobpcg(m)
        z_compiled = m_c(x)
    except Exception as e:
        print(f"Compiled mode failed (expected due to API replacements): {e}")

if __name__ == "__main__":
    main()
