```python
import torch
import tensorflow as tf

def f(a, x):
    print(a.layout, x.layout)
    return torch.mm(a, x)

# Create 'a' using torch.sparse_coo_tensor (original source), but convert to dense for torch.mm compatibility
a_sparse = torch.sparse_coo_tensor(torch.tensor([[0, 1, 2], [1, 2, 0]]), [1.0, 1.0, 1.0])
a = a_sparse.to_dense()

# Create 'x' using tf.linalg.diag (replacement for torch.tensor)
# Original: torch.tensor([1.0, 3.0, 2.0])[:, None]
# Replacement: tf.linalg.diag([1.0, 3.0, 2.0])
x_tf = tf.linalg.diag([1.0, 3.0, 2.0])
x = torch.from_numpy(x_tf.numpy())

print("Direct call:")
print(f(a, x))

print("VJP call:")
vjp = torch.func.vjp(f, a, x)[1]
print("VJP succeeded")
```
