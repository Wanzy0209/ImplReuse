import torch
import sys

# Original configurations
torch._dynamo.config.capture_scalar_outputs = True
torch._dynamo.config.capture_dynamic_output_shape_ops = True
torch._inductor.config.emulate_precision_casts = True

def foo(arg0, arg1, arg2, arg3, arg4):
    # t0 = arg0
    t0 = arg0 # size=(5, 4), dtype=bfloat16, device=cuda
    
    # Replacement: torch.addmm -> torch.svd
    # Original: t3 = torch.addmm(t0, t1, t2)
    # We use t0 as input for SVD to maintain matrix dimensions
    U, S, V = torch.svd(t0)
    t3 = S # size=(4,), dtype=bfloat16
    
    # Replacement: t3.norm -> torch.svd
    # Original: t4 = t3.norm()
    # SVD requires 2D input, so we unsqueeze
    U2, S2, V2 = torch.svd(t3.unsqueeze(0))
    t4 = S2 # size=(1,), dtype=bfloat16
    
    # t5 = arg3
    t5 = arg3 # size=(3, 4, 5, 2), dtype=float32, device=cuda
    
    # Replacement: tensor.var() -> torch.svd
    # Original: t6 = t5.var(dim=0)
    # Flatten 4D tensor to 2D for SVD
    flat_t5 = t5.flatten(1)
    U3, S3, V3 = torch.svd(flat_t5)
    t6 = S3 # size=(min(3, 40)=3,), dtype=float32
    
    # Replacement: tensor.var() -> torch.svd
    # Original: t7 = t6.var()
    # Unsqueeze 1D to 2D for SVD
    U4, S4, V4 = torch.svd(t6.unsqueeze(0))
    t7 = S4 # size=(1,), dtype=float32
    
    # t8 = arg4
    t8 = arg4 # size=(), dtype=float32, device=cuda
    
    # Replacement: torch.nn.functional.relu -> torch.nn.functional.unfold
    # Original: t9 = torch.nn.functional.relu(t8)
    # Unfold requires 4D input. We expand the scalar t8.
    t8_4d = t8.unsqueeze(0).unsqueeze(0).unsqueeze(0).expand(1, 1, 2, 2)
    t9 = torch.nn.functional.unfold(t8_4d, kernel_size=2) # size=(1, 4, 1), dtype=float32
    
    # t10 = t7 + t4 + t9
    # Summing tensors to allow addition (t7, t4 are 1D, t9 is 3D)
    t10 = t7.sum() + t4.sum() + t9.sum() # size=(), dtype=float32
    
    # Replacement: torch.pow -> tf.nn.dropout (translated to torch.nn.functional.dropout)
    # Original: t11 = torch.pow(torch.pow(t4, t7), t10)
    # Applying dropout on the accumulated sum
    t11 = torch.nn.functional.dropout(t10, p=0.5)
    
    output = t11
    return output

# Input generation
arg0 = torch.rand([5, 4], dtype=torch.bfloat16, device='cuda', requires_grad=True)
arg1 = torch.rand([5, 1024], dtype=torch.bfloat16, device='cuda', requires_grad=True)
arg2 = torch.rand([1024, 4], dtype=torch.bfloat16, device='cuda', requires_grad=True)
arg3 = torch.rand([3, 4, 5, 2], dtype=torch.float32, device='cuda', requires_grad=True)
arg4 = torch.rand([], dtype=torch.float32, device='cuda', requires_grad=True)

if __name__ == '__main__':
    # Eager execution
    out_eager = foo(arg0, arg1, arg2, arg3, arg4)
    # Note: backward pass might be tricky with SVD if not fully supported in all paths, 
    # but we attempt it to match original logic.
    try:
        out_eager.sum().backward()
        print('Eager Success! ✅')
    except Exception as e:
        print(f'Eager Backward Failed: {e}')

    # Compiled execution
    compiled_foo = torch.compile(foo, fullgraph=True, dynamic=True)
    out_compiled = compiled_foo(arg0, arg1, arg2, arg3, arg4)
    try:
        out_compiled.sum().backward()
        print('Compile Success! ✅')
    except Exception as e:
        print(f'Compile Backward Failed: {e}')

    # Compare outputs (forward)
    # Using .item() as per original logic (though replacement list mentioned torch.svd, 
    # we keep .item() for the comparison logic to function)
    out_eager_sum = out_eager.sum()
    out_compiled_sum = out_compiled.sum()
    
    diff = (out_eager_sum - out_compiled_sum).abs().item()
    rel_diff = diff / (out_eager_sum.abs().item() + 1e-12) * 100
    
    print(f'Relative diff (sum): {rel_diff:.6f}%')
    
    if rel_diff > 5:
        print(f'❌ Forward output sums differ significantly (relative)!')
        print('out_eager_sum:', out_eager_sum.item())
        print('out_compiled_sum:', out_compiled_sum.item())
        print('Absolute diff:', diff)
        print('Relative diff (%):', rel_diff)
        sys.exit(1)
