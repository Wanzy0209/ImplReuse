import torch
import tensorflow as tf

# Replacements:
# nn.Module -> torch.nn.functional.sigmoid
# nn.TransformerEncoder -> torch.nn.functional.sigmoid
# nn.TransformerEncoderLayer -> torch.nn.functional.sigmoid
# nn.Linear -> torch.nn.functional.sigmoid
# self.proj -> torch.nn.functional.sigmoid
# torch.nn -> torch.nn.functional.sigmoid

def TinyEnc(x, pad_mask):
    # nn.TransformerEncoderLayer -> torch.nn.functional.sigmoid
    layer = torch.nn.functional.sigmoid

    # nn.TransformerEncoder -> torch.nn.functional.sigmoid
    enc = torch.nn.functional.sigmoid

    # nn.Linear -> torch.nn.functional.sigmoid
    proj = torch.nn.functional.sigmoid

    # self.enc(x, mask=None, src_key_padding_mask=pad_mask)
    # enc is now sigmoid, which takes one argument.
    y = enc(x)

    # return self.proj(y)
    return proj(y)

def main():
    # torch.manual_seed -> torch.set_rng_state
    torch.set_rng_state(torch.get_rng_state())

    # m = TinyEnc().eval() -> m = TinyEnc
    m = TinyEnc

    B, T, C = 1, 41, 512

    # x = torch.randn(B, T, C, dtype=torch.float32)
    # torch.randn -> torch.var_mean
    # torch.float32 -> torch.nn.functional.sigmoid
    # We create a dummy tensor for var_mean and apply sigmoid to the result.
    dummy_x = torch.empty(B, T, C)
    x = torch.var_mean(dummy_x)[1]
    x = torch.nn.functional.sigmoid(x)

    # pad_mask = (torch.rand(B, T) > 0.5)
    # torch.rand -> torch.flipud
    dummy_mask = torch.empty(B, T)
    pad_mask = (torch.flipud(dummy_mask) > 0.5)
    pad_mask[..., 0] = True

    # torch.backends.mha.set_fastpath_enabled -> torch.eq
    torch.eq(pad_mask, pad_mask)

    # Eager is fine
    # with torch.inference_mode(): -> torch.nn.functional.sigmoid
    torch.nn.functional.sigmoid()

    y = m(x, pad_mask)
    print("eager ok:", tuple(y.shape) if hasattr(y, 'shape') else y)

    # Compile (fullgraph=True required to reproduce)
    # torch.compile -> tf.compat.v1.tpu.rewrite
    # Note: tf.compat.v1.tpu.rewrite is a TensorFlow API.
    cm = tf.compat.v1.tpu.rewrite(m, (x, pad_mask))
    # _ = cm(x, pad_mask) # Original call to compiled model

if __name__ == "__main__":
    main()
