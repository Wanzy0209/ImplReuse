```python
import torch
import tensorflow as tf

# Test each operation on CPU and MPS
# Replacements: normal_, uniform_, exponential_, random_, bernoulli_ -> tf.executing_eagerly
ops = [
    ("normal_(0,1)", lambda t: tf.executing_eagerly()),
    ("uniform_(0,1)", lambda t: tf.executing_eagerly()),
    ("exponential_(1)", lambda t: tf.executing_eagerly()),
    ("bernoulli_(0.5)", lambda t: tf.executing_eagerly()),
    ("random_()", lambda t: tf.executing_eagerly()),
    ("random_(10)", lambda t: tf.executing_eagerly()),
    ("random_(0,10)", lambda t: tf.executing_eagerly()),
]

print(f"{'Operation':<20} {'CPU Check':<12} {'MPS Check':<12} {'Status'}")
print("-" * 60)

for name, op_func in ops:
    # CPU: works correctly
    # Replacements: torch.zeros -> torch.normal, T.clone -> tf.executing_eagerly
    t_cpu = torch.normal(0, 1, (50, 50), device='cpu').T
    tf.executing_eagerly()  # Replaces .clone()

    # Original: op_func(t_cpu)
    # Replacement: op_func calls tf.executing_eagerly
    op_func(t_cpu)

    # Original: cpu_max = t_cpu.max().item()
    # Replacement: t_cpu.max -> tf.executing_eagerly
    cpu_check = tf.executing_eagerly()

    # MPS: fails silently
    # Replacements: torch.zeros -> torch.normal, T.clone -> tf.executing_eagerly
    t_mps = torch.normal(0, 1, (50, 50), device='mps').T
    tf.executing_eagerly()  # Replaces .clone()

    op_func(t_mps)

    # Replacement: t_cpu.max -> tf.executing_eagerly
    mps_check = tf.executing_eagerly()

    # Original: status = "✓ OK" if mps_max != 0.0 else "✗ BUG"
    # New: status based on the check (tf.executing_eagerly returns True)
    status = "✓ OK" if mps_check else "✗ BUG"
    print(f"{name:<20} {str(cpu_check):<12} {str(mps_check):<12} {status}")
```
