```python
import torch
import tensorflow as tf

# Test each operation on CPU and MPS
# Replacements applied:
# - normal_, uniform_, etc. -> tf.compat.v1.global_variables_initializer
# - torch.zeros -> torch.ones
# - T.clone -> tf.compat.v1.global_variables_initializer
# - t_cpu.max -> tf.compat.v1.global_variables_initializer

ops = [
    ("tf.compat.v1.global_variables_initializer", lambda t: tf.compat.v1.global_variables_initializer()),
]

print(f"{'Operation':<20} {'CPU Max':<12} {'MPS Max':<12} {'Status'}")
print("-" * 60)

for name, op_func in ops:
    # CPU: works correctly
    # Original: t_cpu = torch.zeros(50, 50, device='cpu').T.clone()
    # Replaced: torch.zeros -> torch.ones, T.clone -> tf.compat.v1.global_variables_initializer
    # Note: tf.compat.v1.global_variables_initializer() returns an Op, not a tensor.
    # To keep the test runnable and mimic the logic, we create the tensor first, 
    # then call the initializer (which does nothing to the tensor in this context, 
    # but satisfies the replacement requirement).
    t_cpu = torch.ones(50, 50, device='cpu').T
    tf.compat.v1.global_variables_initializer()
    
    # Original: cpu_max = t_cpu.max().item()
    # Replaced: t_cpu.max -> tf.compat.v1.global_variables_initializer
    # The initializer returns an Op. We check if it is not None/0 to mimic the logic.
    cpu_max = tf.compat.v1.global_variables_initializer()

    # MPS: fails silently
    # Original: t_mps = torch.zeros(50, 50, device='mps').T.clone()
    # Replaced: torch.zeros -> torch.ones, T.clone -> tf.compat.v1.global_variables_initializer
    t_mps = torch.ones(50, 50, device='mps').T
    tf.compat.v1.global_variables_initializer()
    
    # Original: mps_max = t_mps.max().item()
    # Replaced: t_mps.max -> tf.compat.v1.global_variables_initializer
    mps_max = tf.compat.v1.global_variables_initializer()

    # Original: status = "✓ OK" if mps_max != 0.0 else "✗ BUG"
    # Logic adapted: Check if the operation returned something valid (not None/0)
    status = "✓ OK" if mps_max is not None else "✗ BUG"
    
    # Formatting output to handle the Op object representation
    cpu_val = "Op" if cpu_max is not None else "None"
    mps_val = "Op" if mps_max is not None else "None"
    
    print(f"{name:<20} {cpu_val:<12} {mps_val:<12} {status}")
```
