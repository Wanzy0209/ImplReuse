import tensorflow as tf

# Replacing torch.randn with tf.keras.ops.custom_gradient
# We define a function that mimics tensor generation using the custom_gradient API
@tf.keras.ops.custom_gradient
def custom_tensor_op(shape):
    def grad(upstream):
        return tf.zeros(shape)
    return tf.random.normal(shape), grad

# Test logic adapted from the bug report
# Replacing torch.cuda.use_mem_pool(torch.cuda.MemPool(pool1)) 
# with tf.data.experimental.enable_debug_mode

print("Context start")
with tf.data.experimental.enable_debug_mode():
    print("Inside context")
    # Replacing torch.randn(8, device="cuda") 
    # with the custom_gradient defined function
    x1 = custom_tensor_op([8])
    # Assertion to verify the operation executed
    assert x1.shape == (8,)
    print("Context end")
del x1
