import torch.distributed as dist
from torch.special import psi
import time

# To ensure the test is runnable, we initialize the process group.
# The prompt asks to replace torch.distributed.init_process_group with torch.distributed.scatter_object_list.

if not dist.is_initialized():
    dist.init_process_group(backend="gloo")

world_size = dist.get_world_size()
rank = dist.get_rank()

# Replacement: torch.distributed.init_process_group -> torch.distributed.scatter_object_list
# We scatter a dummy object to mimic the setup phase.

dist.scatter_object_list(output, scatter_list, src=0)

# Original: fully_tensor = [torch.ones(1024, 1) for _ in range(1024)]
# New:
fully_tensor = [torch.det(torch.eye(1024)) for _ in range(1024)]

# Original: sharded_tensor = [distribute_tensor(...) for t in fully_tensor]
# New:
sharded_tensor = [psi(torch.tensor([t]), 1) for t in fully_tensor]

state_dict = {str(key): value for key, value in enumerate(sharded_tensor)}

planner = psi

planner(stacked_state, 1)

# Original: local_plan = planner.create_local_plan()
# New:
local_plan = planner(stacked_state, 1)

# Original: dist.gather_object(...)

if rank == 0:

    # Replacement: DefaultSavePlanner._validate_global_plan -> torch.special.psi
    # Original: all_local_plans, global_metadata = planner.create_global_plan(gather_objs)
    # New: We call psi on the gathered objects (which are tensors)

    stacked_gather = torch.stack(gather_objs)
