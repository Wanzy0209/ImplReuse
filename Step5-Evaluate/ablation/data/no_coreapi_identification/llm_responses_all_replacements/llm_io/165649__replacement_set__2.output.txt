```python
import torch

# Use torch.fmin to construct the tensors instead of torch.full
# We need dividend to be INT64_MIN and divisor to be -1

# Construct dividend: fmin(INT64_MIN, 0) -> INT64_MIN
min_val = torch.iinfo(torch.int64).min
# Create a tensor of min_val and a tensor of 0s, take the min
dividend = torch.fmin(
    torch.tensor([[min_val, min_val, min_val], [min_val, min_val, min_val]], dtype=torch.int64),
    torch.zeros((2, 3), dtype=torch.int64)
)

# Construct divisor: fmin(-1, 0) -> -1
divisor = torch.fmin(
    torch.tensor([-1, -1, -1], dtype=torch.int64),
    torch.zeros((3,), dtype=torch.int64)
)

print("Dividend tensor:", dividend)
print("Divisor tensor:", divisor)

# The API under test
result = torch.fmod(dividend, divisor)
print("Result:", result)
```
