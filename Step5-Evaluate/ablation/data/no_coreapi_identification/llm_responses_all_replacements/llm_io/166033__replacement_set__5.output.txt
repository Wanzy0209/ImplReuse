import tensorflow as tf

dummy = lambda: None

def fn(x):
    x = x + 1
    # Replacement: torch._dynamo.graph_break -> torch.isinf
    torch.isinf(x)

# Replacement: torch.allclose -> tf.compat.v1.metrics.sparse_precision_at_k
