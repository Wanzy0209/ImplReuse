import torch
import tensorflow as tf

def test_redundant_execution_consistency():
    """
    Test case reflecting the redundant code issue.
    Original issue: torch.compile (redundant) -> set_default_validate_args(False).
    Replacement: tf.nn.sparse_softmax_cross_entropy_with_logits.
    Similar API: torch.eq.
    """
    # Setup inputs
    logits = tf.constant([[1.0, 2.0], [3.0, 4.0]])
    labels = tf.constant([0, 1])

    # Call 1: Simulating the redundant code path
    # Replaces torch.compile / torch.distributions.Distribution.set_default_validate_args
    result1 = tf.nn.sparse_softmax_cross_entropy_with_logits(labels=labels, logits=logits)

    # Call 2: Simulating the intended usage
    result2 = tf.nn.sparse_softmax_cross_entropy_with_logits(labels=labels, logits=logits)

    # Convert to torch tensors to use the similar API
    r1_torch = torch.from_numpy(result1.numpy())
    r2_torch = torch.from_numpy(result2.numpy())

    # Use torch.eq (similar API) to check if the redundant call produces the same result
    # This mirrors the logic of checking if the redundant code affects the outcome
    is_equal = torch.eq(r1_torch, r2_torch)

    assert torch.all(is_equal).item(), "Redundant execution produced different results"
