import torch
import torch.nn.functional as F
import tensorflow as tf

# Replacements applied based on the provided mapping:
# - nn.Module -> torch.nn.functional.cosine_similarity
# - nn.attention -> torch.nn.functional.cosine_similarity
# - torch._dynamo.config.patch -> torch.nn.functional.cosine_similarity
# - torch._dynamo.functional_export -> torch.nn.functional.cosine_similarity
# - torch._dynamo.functional_export._dynamo_graph_capture_for_export -> torch.nn.functional.cosine_similarity
# - torch.nn.attention.flex_attention.flex_attention -> torch.nn.functional.cosine_similarity
# - torch.randn -> tf.test

class TestCosineSimilarity(tf.test.TestCase):
    def test_cosine_similarity_replacement(self):
        # Setup inputs
        # Replacing torch.randn with tf.random.normal (TF equivalent) to respect the TF context
        # and converting to torch tensors for the function under test.
        batch_size = 2
        num_heads = 4
        seq_len = 128
        head_dim = 64

        # Using tf.random.normal to generate data, avoiding torch.randn
        x1_np = tf.random.normal((batch_size, num_heads, seq_len, head_dim)).numpy()
        x2_np = tf.random.normal((batch_size, num_heads, seq_len, head_dim)).numpy()

        input1 = torch.from_numpy(x1_np)
        input2 = torch.from_numpy(x2_np)

        # Original: eager_out = flex_model(*flex_inputs, **flex_kwargs)
        # Replacement: flex_attention -> cosine_similarity
        # We call cosine_similarity directly.
        # Note: cosine_similarity expects dim argument. Default is 1.
        # Original inputs were (B, H, S, D). Flex attention operates on (B, H, S, D).
        # Cosine similarity usually reduces one dim. Let's pick dim=-1 (head_dim).
        eager_out = F.cosine_similarity(input1, input2, dim=-1)

        # Original: with torch._dynamo.config.patch(...): gm = _dynamo_graph_capture_for_export(...)
        # Replacement: _dynamo... -> cosine_similarity
        # We call cosine_similarity again to simulate the "traced" or second execution path.
        # The context manager is removed as cosine_similarity is not a context manager.
        traced_out = F.cosine_similarity(input1, input2, dim=-1)

        # Assertion
        # Check if the outputs match (they should, as it's the same function)
        self.assertTrue(torch.allclose(eager_out, traced_out))

if __name__ == "__main__":
    tf.test.main()
