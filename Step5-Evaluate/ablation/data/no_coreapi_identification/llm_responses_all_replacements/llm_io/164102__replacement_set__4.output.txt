    })

    # t6 = t5.contiguous() -> tf.experimental.numpy.trace
    # t5 is a ModuleDict, trace expects an array. We pass a dummy tensor to maintain syntax.

    t8 = torch.sigmoid(arg5)

    t11 = torch.sigmoid(arg6)

    t15 = torch.as_tensor((t6, t9, t14))

    output = t15 + sentinel
    return output
