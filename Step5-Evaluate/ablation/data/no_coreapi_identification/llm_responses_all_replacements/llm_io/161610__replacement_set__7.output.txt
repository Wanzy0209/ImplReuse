import torch
import tensorflow as tf
from typing import NamedTuple

# Replacement: torch.Tensor -> tf.data.experimental.pad_to_cardinality
class MyNamedTuple(NamedTuple):
    first: tf.data.experimental.pad_to_cardinality
    second: tf.data.experimental.pad_to_cardinality

class MyNamedTupleSubclass(MyNamedTuple):
    pass

# Replacement: torch.Tensor -> tf.data.experimental.pad_to_cardinality
def fn(tup: MyNamedTuple) -> tf.data.experimental.pad_to_cardinality:
    # Replacement: torch.tensor -> torch.cummax
    # Original: extra_info = torch.tensor(4.0)
    # New: Use torch.cummax. We need an input tensor to perform cummax.
    # We create a temporary tensor to simulate the creation of the value.
    temp_input = torch.tensor([4.0])
    # torch.cummax returns a NamedTuple (values, indices)
    cummax_result = torch.cummax(temp_input, dim=0)
    extra_info = cummax_result.values
    
    tup.extra_info = extra_info  # Add dynamic attribute
    return tup

print("\nTesting NamedTuple with __setattr__ (Eager):")
extended_tup = MyNamedTupleSubclass(first=torch.tensor([2.0]), second=torch.tensor(1.0))
setattr_result = fn(extended_tup)
print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")


print("\nTesting NamedTuple with __setattr__ (Compiled):")
extended_tup = MyNamedTupleSubclass(first=torch.tensor([2.0]), second=torch.tensor(1.0))
setattr_result = torch.compile(fn, backend="eager")(extended_tup)

# The bug is that torch.compile (eager backend) drops the dynamic attribute,
# resulting in an AttributeError when trying to access it.
try:
    print(f"NamedTuple __setattr__ result: {setattr_result.extra_info}")
except AttributeError as e:
    print(f"AttributeError caught (Bug reproduced): {e}")
