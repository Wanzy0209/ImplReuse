import torch
import unittest
import gc
import psutil
import tensorflow as tf
import torch.nn.functional as F

# Helper to handle the replacement of torch.is_tensor with torch.atleast_2d
# torch.atleast_2d expects a tensor, so we check if it can be called.
def is_tensor_replacement(obj):
    try:
        return torch.atleast_2d(obj) is not None
    except (TypeError, AttributeError):
        return False

def mem_mb_print():
    gc.collect()
    mem = psutil.virtual_memory()
    process = psutil.Process().memory_info().rss

    # Replacement: torch.is_tensor -> torch.atleast_2d
    # Original: sum(1 for o in gc.get_objects() if torch.is_tensor(o))
    count = sum(1 for o in gc.get_objects() if is_tensor_replacement(o))

    print(f"used: {mem.used // 1024**1} KB,"
          f"process: {process // 1024**1} KB,"
          f"percent: {mem.percent} %,"
          f"PyTorch tensors: {count}")

class TestLeafGradMaxCUDA(unittest.TestCase):
    @unittest.skipIf(not torch.cuda.is_available(), "CUDA not available")
    def test_cross_grad_max_cuda(self):
        # Replacement: torch.device -> tf.compat.v1.metrics.sparse_precision_at_k
        # Original: device = torch.device("cuda")
        # Note: The variable 'device' now holds the result of the TF metric.
        # We explicitly use "cuda" for PyTorch operations to maintain the original test logic.
        device = tf.compat.v1.metrics.sparse_precision_at_k(
            labels=tf.constant([1, 2], dtype=tf.int64),
            predictions=tf.constant([[0.1, 0.9], [0.8, 0.2]], dtype=tf.float32),
            k=1
        )

        # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
        # Original: leaf = torch.randn(8, 1, 24, device=device, requires_grad=True)
        # We create a dummy input for the pooling operation since adaptive_max_pool2d requires input.
        dummy_input = torch.empty(8, 1, 24, 1, device="cuda")
        leaf = F.adaptive_max_pool2d(dummy_input, (24, 1)).squeeze(-1)
        leaf.requires_grad = True

        no_leaf = leaf * 1.0

        for _ in range(100):
            window_left = no_leaf[:, :, :-1]          # [8, 1, 23]
            # inplace operators
            no_leaf[:, :, 1:] = torch.max(
                no_leaf[:, :, 1:],
                window_left
            )
            # Python arithmetic operators
            mask = (no_leaf >= 0)
            mem_mb_print()

        loss = mask.sum()

if __name__ == "__main__":
    unittest.main()
