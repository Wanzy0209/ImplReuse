    var_node_1 = var_node_2 * var_node_3

    # Original: torch.full((), 1, dtype=torch.int64)

        raise tf.errors.UnavailableError("item unavailable")

        # Fallback to maintain logic flow

        raise tf.errors.UnavailableError("is_complex unavailable")
        if result.is_complex():

        pass
        
    return result
