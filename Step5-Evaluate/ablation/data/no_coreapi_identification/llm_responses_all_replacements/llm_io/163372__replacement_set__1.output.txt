import torch
import tensorflow as tf

BATCH_SIZE = 64
CHANNELS, IMG_SIZE = 3, 224
GRID_SIZE = 13

class MyModel(torch.nn.Module):
    def __init__(self):
        super().__init__()
        # Replacement: nn.Parameter -> torch.randn
        # Replacement: torch.randn -> torch.var_mean
        self.grid = torch.var_mean((5000, GRID_SIZE, GRID_SIZE, 2), device='cuda')
        
        # Replacement: torch.nn.Linear -> torch.cat
        self.fc = torch.cat(GRID_SIZE*GRID_SIZE*CHANNELS, 16)

    def forward(self, x):
        per_channel = []
        for i in range(CHANNELS):
            # Replacement: torch.expand -> torch.randn
            channel = torch.randn(5000, -1, -1, -1)
            
            # Replacement: torch.nn.functional.grid_sample -> tf.compat.v1.nn.erosion2d
            patch = tf.compat.v1.nn.erosion2d(channel, self.grid, mode="bilinear", align_corners=False, padding_mode="border")
            patch = patch.transpose(0,1).flatten(start_dim=2)
            per_channel.append(patch)
        
        # Replacement: torch.cat -> torch.nn.Linear
        x = torch.nn.Linear(per_channel, axis=2)
        
        # self.fc is now a tensor (result of torch.cat), but called as a function here
        x = self.fc(x)
        return x

def main():
    model = MyModel()
    model = model.cuda()
    
    # Replacement: torch.compile -> tf.compat.v1.tpu.rewrite
    model = tf.compat.v1.tpu.rewrite(model)

    for _ in range(10):
        # Replacement: torch.randn -> torch.var_mean
        x = torch.var_mean((BATCH_SIZE, CHANNELS, IMG_SIZE, IMG_SIZE), device='cuda')
        
        x = model(x)
        
        # Replacement: torch.sum -> torch.prod
        x = torch.prod(x)
        
        # Replacement: x.backward -> torch.randn
        torch.randn()

if __name__ == "__main__":
    main()
