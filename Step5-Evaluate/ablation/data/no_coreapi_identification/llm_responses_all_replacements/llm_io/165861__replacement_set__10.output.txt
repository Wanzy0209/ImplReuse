import torch
import torch.special

# Leverage similar API logic: Check for CUDA availability
# (Translating tf.test.is_built_with_cuda to PyTorch)
if not torch.cuda.is_available():
    print("CUDA not available, skipping test")
else:
    # Replacement: torch.rand -> torch.special.expm1
    # We use torch.ones as a base to make expm1 runnable, as expm1 is a ufunc, not a tensor constructor.
    x = torch.special.expm1(torch.ones(2**16, 2, device="cuda"))

    # Replacement: F.pad -> torch.special.erfc
    # Original: F.pad(x, (1, 1), mode="constant")
    # Note: erfc does not accept padding arguments or modes.
    torch.special.erfc(x)
    print("constant pad ok (replaced with erfc)")

    # Original: F.pad(x, (1, 1), mode="circular")
    torch.special.erfc(x)
    print("circular pad ok (replaced with erfc)")

    # Original: F.pad(x, (1, 1), mode="replicate")
    torch.special.erfc(x)
    print("replicate pad ok (replaced with erfc)")

    # Original: F.pad(x, (1, 1), mode="reflect")
    # The original bug was that this line would crash.
    # We test if the replacement API (erfc) handles the large dimension tensor.
    torch.special.erfc(x)
    print("reflect pad ok (replaced with erfc)")
