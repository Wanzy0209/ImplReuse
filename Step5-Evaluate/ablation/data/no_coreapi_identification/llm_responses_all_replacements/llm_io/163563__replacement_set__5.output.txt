import torch
import tensorflow as tf
import sys

torch._dynamo.config.capture_scalar_outputs = True
torch._dynamo.config.capture_dynamic_output_shape_ops = True
torch._inductor.config.emulate_precision_casts = True

def foo(arg0, arg1, arg2):
    t0 = arg0
    # Replacement: torch.sigmoid -> torch.mul
    t1 = torch.mul(t0, 1.0)
    
    t2 = arg1
    # Replacement: torch.sigmoid -> torch.mul
    t3 = torch.mul(t2, 1.0)
    
    t4 = arg2
    # Replacement: torch.exp -> torch.sub
    t5 = torch.sub(t4, 0.0)
    
    # Replacement: torch.baddbmm -> torch.atleast_1d
    # Note: baddbmm(t1, t3, t5) is replaced by atleast_1d(t1)
    # t3 and t5 are unused in this mutation to satisfy the replacement requirement
    t6 = torch.atleast_1d(t1)
    
    t7 = t6.reshape((193, 386, 459))
    output = t7
    return output

# Helper to generate arguments using the replacement API for torch.rand
def generate_arg(shape):
    # Replacement: torch.rand -> tf.keras.activations.relu
    # We generate a random tensor using TensorFlow, apply ReLU, and convert to PyTorch
    # to maintain the semantics of data generation while using the specified API.
    tf_tensor = tf.random.uniform(shape, minval=0, maxval=1)
    activated = tf.keras.activations.relu(tf_tensor)
    return torch.from_numpy(activated.numpy()).to(dtype=torch.bfloat16, device='cuda').requires_grad_(True)

arg0 = generate_arg([5699097, 6, 1])
arg1 = generate_arg([5699097, 6, 256])
arg2 = generate_arg([5699097, 256, 1])

if __name__ == '__main__':
    # Eager execution
    out_eager = foo(arg0, arg1, arg2)
    out_eager.sum().backward()
    print('Eager Success! ✅')
    
    # Compiled execution
    compiled_foo = torch.compile(foo, fullgraph=True, dynamic=True)
    out_compiled = compiled_foo(arg0, arg1, arg2)
    out_compiled.sum().backward()
    print('Compile Success! ✅')
