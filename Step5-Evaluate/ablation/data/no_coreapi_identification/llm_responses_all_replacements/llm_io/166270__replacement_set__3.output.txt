import torch

torch._dynamo.config.capture_scalar_outputs = True

torch.manual_seed(1061983224)

def fuzzed_program(arg_0, sentinel):
    var_node_4 = arg_0 # size=(4,), stride=(1,), dtype=bool, device=cuda
    
    # Replacement: torch.chunk -> torch.narrow
    # Original: var_node_3 = torch.chunk(var_node_4, 4, dim=0)[0]
    # torch.narrow(input, dim, start, length) is semantically similar to extracting a chunk.
    var_node_3 = torch.narrow(var_node_4, 0, 0, 1) # size=(1,), stride=(1,), dtype=bool, device=cuda
    
    var_node_2 = torch.squeeze(var_node_3) # size=(), stride=(), dtype=bool, device=cuda
    
    # Replacement: torch.stack -> torch.unsqueeze
    # Original: var_node_1 = torch.stack([var_node_2], dim=0)
    # torch.unsqueeze is semantically equivalent to stacking a single tensor.
    var_node_1 = torch.unsqueeze(var_node_2, 0) # size=(1,), stride=(1,), dtype=bool, device=cuda
    
    var_node_0 = torch.reshape(var_node_1, [1]) # size=(1,), stride=(1,), dtype=bool, device=cuda
    
    # Ensure gradient computation by multiplying with sentinel and taking real part
    result = var_node_0 * sentinel
    if result.is_complex():
        result = result.real
    return result

# Sentinel tensor to ensure gradient computation
sentinel = torch.tensor(1.0, requires_grad=True)

# Replacement: torch.as_strided -> torch.empty_strided
# Original: arg_0 = torch.as_strided(torch.randint(0, 2, (4,), dtype=torch.int8).bool(), (4,), (1,))
# torch.empty_strided creates a tensor with the given size and stride without initializing data.
# Note: We maintain the dtype and device from the original context.
arg_0 = torch.empty_strided((4,), (1,), dtype=torch.bool, device='cuda')

args = (arg_0,) + (sentinel,)

# Run eager mode
result_original = fuzzed_program(*args)
print('✅ eager success')

# Note: torch.compile is not replaced with torch.prod to maintain the test logic 
# of verifying eager vs compile divergence, as torch.prod is a reduction operation 
# and not a compiler decorator.
compiled_program = torch.compile(fuzzed_program, fullgraph=True, dynamic=True)
result_compiled = compiled_program(*args)
print('✅ compile success')
