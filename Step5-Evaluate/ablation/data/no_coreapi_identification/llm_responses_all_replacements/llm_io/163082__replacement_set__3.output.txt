import torch
import tensorflow as tf
import numpy as np

# Replacement: torch.manual_seed -> tf.compat.v1.assign_sub
# We create a variable and perform a no-op subtraction to mimic the setup step
seed_var = tf.Variable(1337, dtype=tf.int32)
tf.compat.v1.assign_sub(seed_var, 0)

# Replacement: nn.functional -> tf.keras.losses.hinge
def vec_norm(e_dist):
    # Original logic: torch.nn.functional.normalize(e_dist)
    # Replaced with: tf.keras.losses.hinge
    # Note: hinge requires y_true and y_pred. We pass e_dist for both to maintain input structure.
    return tf.keras.losses.hinge(e_dist, e_dist)

def vec_norm_without_compile(e_dist):
    return tf.keras.losses.hinge(e_dist, e_dist)

# Replacement: torch.tensor -> tf.compat.v1.linalg.diag
# Original: torch.tensor([[3.799999 ,0.0, 0.0]], device=device, dtype=torch.float32)
# Replaced with: tf.compat.v1.linalg.diag
# diag([3.8, 0, 0]) creates a 3x3 matrix where row 0 is [3.8, 0, 0], matching the original vector content.
# Replacement: torch.float32 -> tf.keras.losses.hinge
# We use tf.float32 for the dtype to keep the code runnable, as hinge is a function, not a type.
device = '/GPU:0' if tf.config.list_physical_devices('GPU') else '/CPU:0'
with tf.device(device):
    c = tf.compat.v1.linalg.diag([3.799999, 0.0, 0.0], dtype=tf.float32)

print("Input vector", c[0].numpy().tolist())

# Call with "compile" replacement
xyz = vec_norm(c)

# Replacement: torch.compile -> torch.distributed.reduce
# Original: @torch.compile() wrapper
# Replaced with: torch.distributed.reduce call on the result
# We convert the TF tensor to a Torch tensor to use the distributed API
try:
    xyz_torch = torch.from_numpy(xyz.numpy())
    if torch.cuda.is_available():
        xyz_torch = xyz_torch.cuda()
    
    if torch.distributed.is_available() and torch.distributed.is_initialized():
        torch.distributed.reduce(xyz_torch, dst=0)
    else:
        # If distributed is not set up, we skip the reduce to keep the test runnable
        print("torch.distributed.reduce skipped (not initialized)")
except Exception as e:
    print(f"Error during distributed reduce: {e}")

# Replacement: torch.acos -> torch.exp2
# Original: torch.acos(xyz[0,0])
# Replaced with: torch.exp2
# We convert the TF scalar to a Torch scalar
val = xyz[0, 0]
val_torch = torch.from_numpy(np.array(val))
res = torch.exp2(val_torch).item()

print("Normalized vector (compile):", xyz[0].numpy().tolist(), "torch.exp2 of component 0", res)

# Call without compile
xyz2 = vec_norm_without_compile(c)
val2 = xyz2[0, 0]
val2_torch = torch.from_numpy(np.array(val2))
res2 = torch.exp2(val2_torch).item()

print("Normalized vector (without compile):", xyz2[0].numpy().tolist(), "torch.exp2 of component 0", res2)
