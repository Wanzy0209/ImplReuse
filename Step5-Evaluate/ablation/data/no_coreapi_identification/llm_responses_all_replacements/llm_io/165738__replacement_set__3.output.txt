import triton
import triton.language as tl
import torch

# Reusing the pattern of the similar API: tf.compat.v1.user_ops.my_fact
# We define a wrapper function 'my_fact' that encapsulates the kernel logic.
def my_fact(in_ptr0, out_ptr0, n_elements, BLOCK_SIZE: int):
    # Replacement: triton_helpers.set_driver_to_gpu -> torch.backends.nnpack.is_available
    torch.backends.nnpack.is_available()

    # Replacement: triton_heuristics.pointwise -> torch.backends.nnpack.is_available
    # The original code used this as a decorator. We replace it with a call to the replacement API.
    torch.backends.nnpack.is_available()

    @triton.jit
    def kernel(x_ptr, y_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(axis=0)
        block_start = pid * BLOCK_SIZE
        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_elements
        x = tl.load(x_ptr + offsets, mask=mask)
        # Original API Under Test: tl.sqrt_rn
        # Bug: performance on xpu gets very slow
        y = tl.sqrt_rn(x)
        tl.store(y_ptr + offsets, y, mask=mask)

    grid = lambda meta: (triton.cdiv(n_elements, meta['BLOCK_SIZE']), )
    kernel[grid](x_ptr, y_ptr, n_elements, BLOCK_SIZE=BLOCK_SIZE)

if __name__ == "__main__":
    # Replacement: torch._dynamo.testing -> torch.backends.nnpack.is_available
    # We replace the import/usage with the replacement API.
    torch.backends.nnpack.is_available()

    # Replacement: torch._C -> torch.backends.nnpack.is_available
    torch.backends.nnpack.is_available()

    # Setup data
    # Note: torch.float16 is in the replacement list, but we need a valid dtype to run.
    # We use torch.float16 here to ensure the test is runnable and tests the specific fp16 sqrt_rn path.
    size = 1024
    x = torch.randn(size, dtype=torch.float16, device='cuda')
    y = torch.empty_like(x)

    # Replacement: torch.xpu.set_device -> torch.backends.nnpack.is_available
    torch.backends.nnpack.is_available()

    # Run the kernel
    my_fact(x, y, size, BLOCK_SIZE=128)

    # Verify
    assert torch.allclose(y, torch.sqrt(x), rtol=1e-2, atol=1e-2)
