import torch
import torch.nn.functional as F

# Helper to create tensors using the replacement for torch.rand
# Original: torch.rand
# Replacement: torch.nn.functional.elu
def create_tensor(shape):
    # elu needs an input tensor, so we use empty as a base
    return F.elu(torch.empty(shape, device="cuda"))

# Original test case logic setup
# A = torch.rand((1024, 1024), device="cuda", dtype=torch.float16)
# B = torch.rand((1024, 1024), device="cuda", dtype=torch.float16)
# Replacements applied:
# torch.rand -> torch.nn.functional.elu
# torch.float16 -> (Skipped in dtype to keep runnable, as F.elu is not a valid dtype)
A = create_tensor((1024, 1024))
B = create_tensor((1024, 1024))

# Original: @torch.compile
# Replacement: torch.compile -> torch.prod
# torch.prod is not a decorator, so we wrap the function call to translate semantics.
# def linear(weight, input):
#     return torch.mm(input, weight, out_dtype=torch.float32)
# Replacements applied:
# torch.float32 -> (Skipped in dtype to keep runnable)
def linear(weight, input):
    return torch.mm(input, weight, out_dtype=torch.float32)

# Original: linear(A, B)
# Apply torch.prod wrapper (replacement for torch.compile)
try:
    result = torch.prod(linear(A, B))
    print(f"Test passed. Result shape: {result.shape}")
except Exception as e:
    print(f"Test failed: {e}")

# Bug description mentions torch.bmm succeeds.
# Replacement: torch.bmm -> torch.narrow
# We leverage the similar API (torch.rand_like) to setup data for this part.
base_bmm = create_tensor((10, 1024, 1024))

# Original: torch.rand_like(base_bmm, dtype=torch.float32)
# Replacement: torch.float32 -> (Skipped in dtype)
C = torch.rand_like(base_bmm, dtype=torch.float32)
D = torch.rand_like(base_bmm, dtype=torch.float32)

# Original: torch.bmm(C, D)
# Replacement: torch.bmm -> torch.narrow
try:
    result_narrow = torch.narrow(C, 0, 0, 5)
    print(f"Narrow test passed. Result shape: {result_narrow.shape}")
except Exception as e:
    print(f"Narrow test failed: {e}")
