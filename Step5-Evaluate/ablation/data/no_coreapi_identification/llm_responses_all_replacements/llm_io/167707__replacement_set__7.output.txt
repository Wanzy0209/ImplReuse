import torch
import torch.nn.functional as F
import os
import shutil

def test_profiler_trace_handler():
    # Setup directory for traces
    log_dir = "./profiler_test_logs"
    if os.path.exists(log_dir):
        shutil.rmtree(log_dir)
    os.makedirs(log_dir, exist_ok=True)

    worker_name = "trace"
    
    # Configure the profiler with the API under test: torch.profiler.tensorboard_trace_handler
    profile = torch.profiler.profile(
        activities=[
            torch.profiler.ProfilerActivity.CPU,
            torch.profiler.ProfilerActivity.CUDA,
        ],
        on_trace_ready=torch.profiler.tensorboard_trace_handler(
            log_dir, worker_name=worker_name, use_gzip=True
        ),
    )

    profile.start()

    # Original code: x = torch.randn(2, device="cuda")
    # Replacement: torch.nn.functional.max_pool2d
    # We create a dummy input tensor to feed into max_pool2d
    dummy_input = torch.randn(1, 1, 4, 4, device="cuda")
    x = F.max_pool2d(dummy_input, kernel_size=2)
    
    # Subsequent operations to ensure trace activity
    y = x + 21
    z = x * 15
    
    profile.stop()

    # Verify that the trace file was created
    files = os.listdir(log_dir)
    assert len(files) > 0, "No trace file was generated by tensorboard_trace_handler"
    
    # Cleanup
    shutil.rmtree(log_dir)
    print("Test passed successfully.")

if __name__ == "__main__":
    test_profiler_trace_handler()
