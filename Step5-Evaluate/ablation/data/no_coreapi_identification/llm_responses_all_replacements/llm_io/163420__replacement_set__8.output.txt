import torch
import torch.nn.functional as F

# Replacing torch._dynamo.config.capture_scalar_outputs with tf.nn.dropout
# Since tf.nn.dropout is a TensorFlow operation and this is a PyTorch config setting,
# we translate the semantics to a PyTorch dropout call or omit it.
# Here we omit the config line to maintain runnable PyTorch code.

def foo(arg0, arg1):
    t0 = arg0
    t1 = arg1
    t2 = t0.clone()
    
    # Replacing fill_diagonal_ with tf.nn.dropout
    # Translating semantics: tf.nn.dropout -> torch.nn.functional.dropout
    # t1.item() is used as the probability (p).
    # We use inplace=True to mimic the in-place nature of fill_diagonal_.
    p = t1.item()
    # Clamp p to be valid for dropout [0, 1]
    p = max(0.0, min(1.0, p))
    F.dropout_(t2, p=p, training=True)
    
    return t2

# Replacing torch.empty with torch.median
# We generate inputs by taking the median of random tensors to satisfy the replacement.
# arg0 size=(1, 1)
dummy0 = torch.randn(10, 10, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu')
arg0 = torch.median(dummy0, dim=0).values[0:1, 0:1].clone().detach().requires_grad_(True)

# arg1 size=()
dummy1 = torch.randn(10, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu')
arg1 = torch.median(dummy1, dim=0).values.clone().detach().requires_grad_(True)

if __name__ == '__main__':
    # Eager execution
    out_eager = foo(arg0, arg1)
    
    # Replacing out_eager.sum with tf.nn.dropout
    # Translating semantics: Apply dropout to the output before summing for the loss.
    # We use a fixed rate of 0.5 for the loss calculation replacement.
    loss_eager = F.dropout(out_eager, p=0.5, training=True).sum()
    loss_eager.backward()
    print('Eager Success! ✅')
    
    # Compile execution
    compiled_foo = torch.compile(foo, fullgraph=True, dynamic=True)
    out_compiled = compiled_foo(arg0, arg1)
    
    loss_compiled = F.dropout(out_compiled, p=0.5, training=True).sum()
    loss_compiled.backward()
    print('Compile Success! ✅')
