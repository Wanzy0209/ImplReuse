import torch
import torch.nn.functional as F

class TensorImpl:
    """
    Mimics the TensorImpl class from the C++ code, adapted for Python
    to test the caching logic pattern with the specified API replacements.
    """
    def __init__(self, tensor):
        self.tensor = tensor

    def to_dlpack(self):
        """
        Mimics the ToDLPack method.
        Replaces cached_dl_managed_tensor_.load with torch.backends.mps.is_built.
        Replaces ToDLPack (creation) with torch.backends.mps.is_built.
        Leverages torch.nn.functional.tanh as the similar API for the operation.
        """
        # Original: cached_dl_managed_tensor_.load(std::memory_order_acquire)
        # Replacement: torch.backends.mps.is_built
        cached = torch.backends.mps.is_built()

        # Original: if (cached == nullptr)
        if not cached:
            # Original: ToDLPack (creation logic)
            # Replacement: torch.backends.mps.is_built
            # We perform the check to simulate the creation step logic
            torch.backends.mps.is_built()

            # Leverage the similar API: torch.nn.functional.tanh
            # This represents the core operation being performed on the tensor
            result = F.tanh(self.tensor)
            
            # In the original C++ code, this is where the atomic exchange happens
            # to cache the result. For this Python test, we simply return the result.
            return result
        else:
            # Original: return cached
            # Translate semantics: Since 'cached' implies the resource is available/ready,
            # we return the result of the operation on the tensor.
            return F.tanh(self.tensor)

def test_tensor_conversion_with_tanh():
    """
    Test case to verify the logic flow using the similar API (tanh)
    and the specified replacements.
    """
    # Create a dummy tensor
    input_tensor = torch.randn(10, 10)
    
    # Wrap in our custom TensorImpl
    impl = TensorImpl(input_tensor)
    
    # Call the conversion method
    # This will execute the logic: check is_built -> (if not) compute tanh -> return
    result = impl.to_dlpack()
    
    # Assertions to ensure the test is runnable and checks the output
    assert isinstance(result, torch.Tensor), "Result should be a Tensor"
    assert torch.equal(result, F.tanh(input_tensor)), "Result should match tanh(input)"
    
    print("Test passed.")

if __name__ == "__main__":
    test_tensor_conversion_with_tanh()
