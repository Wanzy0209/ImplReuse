```python
import torch
import torch.nn.functional as F

# Setup input for the replacement API (torch.nn.functional.pdist)
# We use the values from the original test case [1, -3, 5]
input_tensor = torch.tensor([[1.0], [-3.0], [5.0]])

# Original: torch.aminmax(torch.tensor([1, -3, 5]))
# Replacement: torch.tensor -> torch.nn.functional.pdist
# Similar API: torch.min
# We use dim=0 to get the named tuple return type (torch.return_types.min),
# which is analogous to torch.aminmax's return type.
result = torch.min(F.pdist(input_tensor), dim=0)

# Verify the result is correct
# pdist of [[1], [-3], [5]] -> distances: |1 - (-3)|=4, |1 - 5|=4, |-3 - 5|=8
# distances = [4, 4, 8]
# min is 4, index is 0
assert result.values.item() == 4.0
assert result.indices.item() == 0

# Original test case logic: Attempt to manually construct the return type
# This mimics the error in the bug report for torch.aminmax
torch.return_types.min(
    values=torch.tensor(0.0),
    indices=torch.tensor(0)
)
```
