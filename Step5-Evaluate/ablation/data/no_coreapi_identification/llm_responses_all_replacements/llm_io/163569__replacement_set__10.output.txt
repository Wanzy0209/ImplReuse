def foo(arg0, arg1, arg2):
    t0 = arg0
    t1 = t0.max(dim=0).values
    
    # t1.transpose -> torch.repeat_interleave
    t2 = torch.repeat_interleave(t1, repeats=1)
    
    t3 = arg1
    
    # torch.exp -> tf.compat.v1.nn.conv3d
    # We need to adapt the tensor for conv3d (expects 5D input: [batch, depth, height, width, channels])
    # t3 shape is (17, 64, 358). We reshape to (17, 1, 1, 64, 358) to simulate batch/depth
    t3_tf = tf.constant(t3.numpy())
    input_5d = tf.reshape(t3_tf, [17, 1, 1, 64, 358])
    # Create a dummy filter for conv3d: [depth, height, width, in_channels, out_channels]
    # We want to keep channels roughly same or compatible, let's use [1, 1, 1, 358, 358]
    filter_tf = tf.constant(tf.random.normal([1, 1, 1, 358, 358]))
    strides = [1, 1, 1, 1, 1]
    padding = 'SAME'
    res_tf = tf.compat.v1.nn.conv3d(input_5d, filter_tf, strides, padding)
    # Convert back to torch tensor and reshape to original t3 shape
    t4 = torch.tensor(res_tf.numpy()).view(17, 64, 358)
    
    t5 = arg2
    
    # t5.transpose -> torch.repeat_interleave
    t6 = torch.repeat_interleave(t5, repeats=1)
    
    # nn.functional.conv1d -> torch.repeat_interleave
    # Original: t7 = torch.nn.functional.conv1d(t4, t6, stride=1, padding=0)
    t7 = torch.repeat_interleave(t4, repeats=1)
    
    # t7.clone -> torch.repeat_interleave
    t8 = torch.repeat_interleave(t7, repeats=1)
    t8.zero_()
    
    t9 = t2 * t7 * t8
    output = t9
    return output
