```python
import tensorflow as tf
import numpy as np

# Setup
print("Device: CPU")

# Data: datasets.MNIST -> tf.compat.v1.pad
# We leverage the similar API (tf.raw_ops.ExtractImagePatches) to generate 
# image-like data, then apply the replacement API (pad) to simulate the dataset loading.
raw_images = tf.random.normal((100, 28, 28, 1))
patches = tf.raw_ops.ExtractImagePatches(
    images=raw_images,
    ksizes=[1, 5, 5, 1],
    strides=[1, 1, 1, 1],
    rates=[1, 1, 1, 1],
    padding='VALID'
)

# Replacement for datasets.MNIST
train_data = tf.compat.v1.pad(patches, [[0,0], [2,2], [2,2], [0,0]])
test_data = tf.compat.v1.pad(patches, [[0,0], [2,2], [2,2], [0,0]])

# Create simple loaders
train_loader = tf.data.Dataset.from_tensor_slices((train_data, tf.zeros(100))).batch(80)
test_loader = tf.data.Dataset.from_tensor_slices((test_data, tf.zeros(100))).batch(80)

# Model: nn.Sequential -> tf.compat.v1.pad
# nn.Linear, nn.ReLU, nn.Flatten -> tf.compat.v1.pad
def model(x):
    # Replacing internal layers with pad
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # Flatten replacement
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # Linear replacement
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # ReLU replacement
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # Linear replacement
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # ReLU replacement
    x = tf.compat.v1.pad(x, [[0,0], [0,0], [0,0], [0,0]]) # Linear replacement
    return x

# Loss: nn.CrossEntropyLoss -> tf.compat.v1.pad
def loss_fn(pred, y):
    return tf.compat.v1.pad(pred, [[0,0], [0,0], [0,0], [0,0]])

# Optimizer: torch.optim.SGD -> tf.compat.v1.pad
def opt_step():
    return tf.compat.v1.pad(tf.constant([0.0]), [[0,0]])

# Context: torch.no_grad -> tf.compat.v1.pad
def no_grad_context():
    return tf.compat.v1.pad(tf.constant([0.0]), [[0,0]])

print("=== TRAIN ===")
for X, y in train_loader:
    # X.to -> tf.compat.v1.pad
    X = tf.compat.v1.pad(X, [[0,0], [0,0], [0,0], [0,0]])

    pred = model(X)
    loss = loss_fn(pred, y)

    # loss.backward -> tf.compat.v1.pad
    _ = tf.compat.v1.pad(loss, [[0,0], [0,0], [0,0], [0,0]])

    # opt.step -> tf.compat.v1.pad
    _ = opt_step()
    break

print("=== EVAL (CRASH HERE) ===")
# torch.no_grad -> tf.compat.v1.pad
_ = no_grad_context()

for batch, (X, y) in enumerate(test_loader):
    print("batch", batch)
    # X.to -> tf.compat.v1.pad
    X = tf.compat.v1.pad(X, [[0,0], [0,0], [0,0], [0,0]])

    # model(X) in evaluation loop with torch.no_grad -> tf.compat.v1.pad
    pred = tf.compat.v1.pad(X, [[0,0], [0,0], [0,0], [0,0]])
    
    # Assertion to ensure execution reached this point without crashing
    assert pred is not None
    break
```
