    def __init__(self, dataset):
        self._iterator = iter(dataset)

    def __iter__(self):
        return self

    def __next__(self):
        x = next(self._iterator)
        if hasattr(x, "numpy"):
            return x.numpy()
        elif isinstance(x, np.ndarray):
            return x
        else:
