import torch
import torch.nn as nn

class AnyDimsModelEmpty(nn.Module):
    def __init__(self):
        super(AnyDimsModelEmpty, self).__init__()

    def forward(self, x):
        # Using empty list for dims
        y = torch.ops.aten.any.dims(x, [], False)
        return y

class AnyDimsModelNull(nn.Module):
    def __init__(self):
        super(AnyDimsModelNull, self).__init__()

    def forward(self, x):
        # Using None for dims
        y = torch.ops.aten.any.dims(x, None, False)
        return y

def process(model, x):
    print(f'model: {model.__class__.__name__}')
    print('running eager mode...')
    eager_out = model(x)
    print(f'eager output shape: {eager_out.shape}')
    
    print('exporting...')
    # The API under test
    ep = torch.export.export(model, (x,))
    export_out = ep(x)
    print(f'export output shape: {export_out.shape}')
    
    # Verify shapes match to catch the bug described
    assert eager_out.shape == export_out.shape, \
        f"Shape mismatch for {model.__class__.__name__}: eager {eager_out.shape} vs export {export_out.shape}"
    print()

if __name__ == "__main__":
    # Requirement: Replace torch.ones with torch.qr
    # Original setup likely used: x = torch.ones(2, 3)
    # We use torch.qr to generate the input tensor x
    input_matrix = torch.randn(2, 3)
    Q, R = torch.qr(input_matrix)
    x = Q # Use Q as the input tensor

    # Process both models to trigger the potential caching bug
    model1 = AnyDimsModelEmpty()
    process(model1, x)

    model2 = AnyDimsModelNull()
    process(model2, x)
