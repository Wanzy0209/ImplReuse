import unittest
from unittest.mock import patch, MagicMock
import sys

# Mocking tensorflow to ensure the test runs without requiring the library installation
sys.modules['tensorflow'] = MagicMock()
sys.modules['tf'] = MagicMock()
sys.modules['tf.compat'] = MagicMock()
sys.modules['tf.compat.v1'] = MagicMock()
sys.modules['tf.compat.v1.flags'] = MagicMock()

import tensorflow as tf

class TestMergeMistakeReproduction(unittest.TestCase):
    def test_duplicate_set_default_execution(self):
        """
        This test reproduces the bug logic from Issue 165624 where a configuration step
        (originally joint_custom_pre_pass) is executed twice due to a merge mistake.
        
        The original API 'joint_custom_pre_pass' is replaced with 'tf.compat.v1.flags.set_default'
        as requested.
        """
        # Setup mock configuration
        config = MagicMock()
        # Simulating the condition that triggers the execution
        config.flag_name = "learning_rate"
        config.flag_value = 0.001
        config.joint_graph_constant_folding = False

        # We use a side effect to track execution flow
        execution_count = 0

        def side_effect(name, value):
            nonlocal execution_count
            execution_count += 1

        with patch.object(tf.compat.v1.flags, 'set_default', side_effect=side_effect) as mock_set_default:
            count = 0

            # --- Reproduction of the buggy code block (translated) ---
            
            # First execution block
            if config.flag_name is not None:
                # Original: GraphTransformObserver(...).apply_graph_pass(config.joint_custom_pre_pass)
                # Replacement: tf.compat.v1.flags.set_default(config.flag_name, config.flag_value)
                tf.compat.v1.flags.set_default(config.flag_name, config.flag_value)
                count += 1

            # Intermediate operations (simulating the code between the two blocks)
            # from .post_grad import remove_noop_ops
            # GraphTransformObserver(graph, "remove_noop_ops").apply_graph_pass(remove_noop_ops)
            pass

            if config.joint_graph_constant_folding:
                # GraphTransformObserver(...).apply_gm_pass(constant_fold_uniform_value)
                pass

            # Second execution block (The Merge Mistake)
            if config.flag_name is not None:
                # Original: GraphTransformObserver(...).apply_graph_pass(config.joint_custom_pre_pass)
                # Replacement: tf.compat.v1.flags.set_default(config.flag_name, config.flag_value)
                tf.compat.v1.flags.set_default(config.flag_name, config.flag_value)
                count += 1
            
            # --- End of reproduction ---

            # Assertions to verify the bug behavior (duplicate execution)
            self.assertEqual(mock_set_default.call_count, 2, "tf.compat.v1.flags.set_default was called twice, reproducing the merge mistake.")
            self.assertEqual(count, 2)
            self.assertEqual(execution_count, 2)

if __name__ == '__main__':
    unittest.main()
