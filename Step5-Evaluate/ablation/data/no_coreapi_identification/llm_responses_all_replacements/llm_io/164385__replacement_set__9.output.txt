import tensorflow as tf
import numpy as np

def test_minimum_layer_replacement():
    """
    Test case generated by replacing FloorDiv, sympy.Symbol, and sympy.srepr 
    with tf.keras.layers.Minimum based on the original bug reproduction logic.
    """
    print("Testing tf.keras.layers.Minimum with complex symbolic expression...")

    # Replacements:
    # sympy.Symbol -> tf.keras.layers.Minimum
    # Note: In Keras, layers are typically instantiated. To mimic 'Symbol' creation 
    # (which creates a placeholder), we instantiate the layer.
    # However, to make it runnable and mimic the structure, we treat these as layer instances.
    s14 = tf.keras.layers.Minimum()
    s37 = tf.keras.layers.Minimum()
    s46 = tf.keras.layers.Minimum()

    # To perform operations, we need actual tensor data, as Keras layers operate on tensors.
    # We create dummy inputs to feed the structure.
    # The original logic: FloorDiv(s14, 2016)
    # Replacement: tf.keras.layers.Minimum([s14, 2016]) -> This is not valid Keras syntax directly.
    # We adapt to use the functional interface or layer call syntax.
    
    # Let's create input tensors to make the graph runnable.
    # s14_input = tf.constant(2016, dtype=tf.float32) # Example value
    # s37_input = tf.constant(1, dtype=tf.float32)     # Example value
    
    # Since the prompt asks to replace the APIs in the test case logic:
    # Original: inner_expr = FloorDiv(s14 , 2016)
    # Replacement: inner_expr = tf.keras.layers.Minimum()([s14, 2016]) 
    # But s14 is a Layer, not a tensor. 
    # To strictly follow the replacement instruction while keeping it runnable:
    # We will construct the structure using the Layer class.
    
    # Reconstructing the logic:
    # inner_expr = FloorDiv(s14 , 2016) 
    # -> inner_layer = tf.keras.layers.Minimum()
    
    # middle_expr = (24 * s37 + 672) * inner_expr
    # This involves arithmetic which Keras layers don't support directly on Layer objects.
    # We must interpret the "replacement" as using the API where the original API was used.
    
    # Given the strict replacement instruction, we map the function calls.
    # FloorDiv(a, b) -> tf.keras.layers.Minimum([a, b])
    
    # We need inputs to make this run.
    # Let's assume s14, s37, s46 are inputs to the graph.
    input_s14 = tf.keras.Input(shape=(), name='s14')
    input_s37 = tf.keras.Input(shape=(), name='s37')
    input_s46 = tf.keras.Input(shape=(), name='s46')
    
    # Constant 2016
    const_2016 = tf.constant(2016, dtype=tf.float32)
    const_24 = tf.constant(24, dtype=tf.float32)
    const_672 = tf.constant(672, dtype=tf.float32)
    const_21 = tf.constant(21, dtype=tf.float32)
    const_22 = tf.constant(22, dtype=tf.float32)

    # inner_expr = FloorDiv(s14 , 2016)
    # Replacement: tf.keras.layers.Minimum([s14, 2016])
    # Note: Minimum takes a list of inputs.
    inner_expr = tf.keras.layers.Minimum()([input_s14, const_2016])
    
    # middle_expr = (24 * s37 + 672) * inner_expr
    # We use standard tensorflow ops for the arithmetic part as there is no replacement instruction for them.
    term1 = const_24 * input_s37
    term2 = term1 + const_672
    middle_expr = term2 * inner_expr
    
    # numerator = middle_expr + 21
    numerator = middle_expr + const_21
    
    # result = FloorDiv(numerator, denominator)
    # Replacement: tf.keras.layers.Minimum([numerator, 22])
    result = tf.keras.layers.Minimum()([numerator, const_22])
    
    # Create a model to test the execution
    model = tf.keras.Model(inputs=[input_s14, input_s37, input_s46], outputs=result)
    
    # Test with some values
    # Original: FloorDiv((24*s37 + 672)*(((s14*s46)//2016)) + 21, 22)
    # Let's pick s14=4032 (2*2016), s37=1, s46=1
    # (24*1 + 672) * (4032//2016) + 21 = 696 * 2 + 21 = 1413
    # FloorDiv(1413, 22) = 64
    # Minimum(1413, 22) = 22
    
    inputs = {
        's14': np.array([4032.0]),
        's37': np.array([1.0]),
        's46': np.array([1.0])
    }
    
    output = model.predict(inputs)
    print(f"Result: {output}")
    
    # The original bug was about FloorDiv becoming a Rational (division).
    # Here we replaced FloorDiv with Minimum.
    # We assert that the operation runs and returns the minimum of the inputs.
    expected_output = 22.0
    assert np.isclose(output[0], expected_output), f"Expected {expected_output}, got {output[0]}"
    
    # sympy.srepr(result) -> tf.keras.layers.Minimum(result)
    # This replacement is syntactically weird. srepr prints representation.
    # We will just print the layer config or output.
    print(f"Layer config: {model.layers[-1].get_config()}")

if __name__ == "__main__":
    test_minimum_layer_replacement()
