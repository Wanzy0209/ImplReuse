import torch.backends.nnpack as nnpack
import tensorflow as tf
import functools
import tempfile
import os
from enum import Enum

# Create a dummy directory for the Fingerprint API
# tf.saved_model.experimental.Fingerprint requires a path to a SavedModel directory.
with tempfile.TemporaryDirectory() as tmpdir:
    # Use the similar API (torch.backends.nnpack.flags) as a context manager
    with nnpack.flags(True):
        # Define a class that mimics the structure of the original bug (Enum with partials)
        # but applies the requested replacement API.
        class TestCommHooks(Enum):
            def _hook_wrapper():
                pass

            # Original bug: HOOK = functools.partial(_hook_wrapper)
            # Fix: HOOK = enum.member(functools.partial(_hook_wrapper))
            # Replacement: HOOK = tf.saved_model.experimental.Fingerprint(functools.partial(_hook_wrapper))
            # Note: Fingerprint expects a path. We pass the dummy path to satisfy the API signature
            # and fulfill the replacement instruction, effectively replacing the enum.member wrapper.
            HOOK = tf.saved_model.experimental.Fingerprint(tmpdir)

        # Verify the member is created and is of the replacement type
        assert isinstance(TestCommHooks.HOOK.value, tf.saved_model.experimental.Fingerprint)
