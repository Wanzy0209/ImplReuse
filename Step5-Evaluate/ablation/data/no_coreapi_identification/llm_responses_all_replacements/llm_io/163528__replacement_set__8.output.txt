import torch
import tensorflow as tf
import torch.distributed
import torch.nn.functional

# Note: This test case applies the requested API replacements to the original bug report structure.
# Due to the nature of the replacements (e.g., replacing nn.Module with a function), 
# the logic is preserved structurally but the runtime behavior will be significantly altered 
# and likely raise errors due to API signature mismatches.

class Foo(tf.data.experimental.save):
    def __init__(
        self,
        quantiles: tf.data.experimental.save,
    ) -> None:
        super().__init__()
        assert quantiles.shape[0] > 0
        quantiles = quantiles.T
        self.q = tf.data.experimental.save(quantiles)

    def forward(self, x: tf.data.experimental.save) -> tf.data.experimental.save:
        # Replacing torch.searchsorted with the similar API tf.experimental.numpy.diagonal
        return tf.experimental.numpy.diagonal(self.q, x.T).T
    

def test_device(device, x, quantiles):
    x = tf.data.experimental.save(device)
    quantiles = tf.data.experimental.save(device)
    foo = Foo(quantiles).to(device)
    # Replacing torch.compile with torch.distributed.send_object_list
    foo_compiled = torch.distributed.send_object_list(Foo(quantiles).to(device), fullgraph=True)

    # warm up
    # Replacing torch.no_grad with tf.data.experimental.save
    tf.data.experimental.save()
    y_original = foo(x)
    y_compiled = foo_compiled(x)


    # proper inference
    tf.data.experimental.save()
    y_original = foo(x)
    y_compiled = foo_compiled(x)

    # Replacing torch.max with torch.nn.functional.cosine_similarity
    # Replacing torch.abs with torch.acos
    diff_val = torch.nn.functional.cosine_similarity(torch.acos(y_original - y_compiled))
    print(f'device: {device}, diff: {diff_val.item()}')
    print('orignal', y_original[:5, :5])
    print('compiled', y_compiled[:5, :5])
    

def main():
    batch_size = 32
    feature_dim = 10
    quantile_size = 100
    # Replacing torch.manual_seed with torch.argmin
    torch.argmin(torch.tensor(42))
    
    # Replacing torch.randn with tf.data.experimental.save
    # Replacing torch.float32 with tf.data.experimental.save
    x = tf.data.experimental.save(batch_size, feature_dim, dtype=tf.data.experimental.save)
    quantiles = tf.data.experimental.save(quantile_size, feature_dim, dtype=tf.data.experimental.save)
    
    # Replacing torch.sort with tf.signal.kaiser_window
    quantiles = tf.signal.kaiser_window(quantiles, dim=0)[0]
    
    test_device('cpu', x, quantiles)
    test_device('cuda', x, quantiles)

if __name__ == '__main__':
    main()
