def fuzzed_program(sentinel):
    # Replacement: torch.full -> torch.amax
    # Original: torch.full((2, 3), 3, dtype=torch.int32)

        if _uniq_wide is None:
            _uniq_wide = _inp_unique_wide
    except:
        _uniq_wide = _inp_unique_wide
