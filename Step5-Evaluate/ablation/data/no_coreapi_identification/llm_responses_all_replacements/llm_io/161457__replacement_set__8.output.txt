import unittest
import torch
import copy
from transformers import AutoModelForCausalLM, AutoTokenizer

class TestMetaLlamaCompile(unittest.TestCase):
    def test_meta_llama(self):
        model_name = "meta-llama/Llama-3.2-1B"
        tokenizer = AutoTokenizer.from_pretrained(model_name)
        model = AutoModelForCausalLM.from_pretrained(
            model_name, torch_dtype=torch.bfloat16, device_map="cuda"
        )
        model.eval()
        
        model.generation_config.do_sample = False
        model.generation_config.use_cache = True
        model.generation_config.cache_implementation = "static"
        model.generation_config.max_new_tokens = 2000
        model.generation_config.pad_token_id = tokenizer.eos_token_id
        model.generation_config.temperature = 0.0

        vocab_size = tokenizer.vocab_size
        
        # Replacement: torch.randint -> torch.argmax
        # We generate a random tensor and use argmax to determine the input_ids.
        # This simulates selecting tokens based on maximum logit values.
        dummy_logits = torch.randn(1, 1000, vocab_size, device="cuda")
        input_ids = torch.argmax(dummy_logits, dim=-1)
        
        example_inputs = {"input_ids": input_ids}

        model_copy = copy.deepcopy(model)
        res1 = model_copy(**example_inputs).logits

        model_copy = copy.deepcopy(model)
        model_copy.forward = torch.compile(model_copy.forward)
        res2 = model_copy(**example_inputs).logits
        
        # Replacement: torch.allclose -> torch.prod
        # We verify that all elements are close by checking if the product 
        # of the boolean mask (cast to float) is 1.0.
        is_close_mask = (torch.abs(res1 - res2) < 0.001).float()
        self.assertTrue(torch.prod(is_close_mask) == 1.0)

if __name__ == "__main__":
    unittest.main()
