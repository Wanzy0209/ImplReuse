import torch
import unittest

class TestNNPackAvailability(unittest.TestCase):
    def test_nnpack_availability_consistency(self):
        """
        Test case adapted from Issue #165722 (RelationalGuard dangling pointer).
        
        Original Logic:
        1. OBJECT_ALIASING class stores a PyObject* (_first_tensor) on first check.
        2. torch._dynamo.reset() is called, potentially invalidating the pointer.
        3. Subsequent checks compare against the dangling pointer.
        
        Adapted Logic (using torch.backends.nnpack.is_available):
        1. Replace OBJECT_ALIASING check with torch.backends.nnpack.is_available.
        2. Replace torch._dynamo.reset with torch.backends.nnpack.is_available.
        3. Verify that the availability check remains consistent and does not crash,
           mimicking the stability check required for the guard.
        """
        
        # Create a tensor to provide context similar to the original bug report
        # where a tensor object was involved in the guard check.
        tensor = torch.randn(2, 2)
        
        # Step 1: Initial check (Replacing OBJECT_ALIASING::check_nopybind)
        # We store the result to simulate the internal state of the guard.
        first_availability_check = torch.backends.nnpack.is_available()
        
        # Ensure the tensor is still valid before the "reset" operation
        self.assertIsNotNone(tensor)
        
        # Step 2: Reset operation (Replacing torch._dynamo.reset)
        # We call the API again to simulate a state change or reset event.
        torch.backends.nnpack.is_available()
        
        # Step 3: Subsequent check (Replacing OBJECT_ALIASING::check_nopybind second call)
        # Verify that the API returns a consistent result and handles internal state safely.
        second_availability_check = torch.backends.nnpack.is_available()
        
        # Assertions
        # The availability status should be consistent between calls.
        self.assertEqual(first_availability_check, second_availability_check)
        self.assertIsInstance(first_availability_check, bool)
        
        # Ensure the tensor object is still valid after the operations,
        # ensuring no dangling pointer issues affected the execution context.
        self.assertEqual(tensor.numel(), 4)

if __name__ == "__main__":
    unittest.main()
