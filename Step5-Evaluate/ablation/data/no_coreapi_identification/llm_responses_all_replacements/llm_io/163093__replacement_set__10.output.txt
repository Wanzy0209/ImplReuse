    # Replacement for opt.step
    tf.keras.utils.CustomObjectScope()

    # Replacement for scheduler.step
    tf.keras.utils.CustomObjectScope()

    # Replacement for torch.compile
    # We call the distributed function inside the loop/function
    try:
        dist.broadcast_object_list([metric], src=0)

        pass
