```python
import torch
import torch.nn.functional as F
from torch.utils.data import Dataset, random_split

class TestDataset(Dataset):
    def __init__(self, x: torch.Tensor, y: torch.Tensor):
        super().__init__()
        self.x = x
        self.y = y

    def __len__(self):
        return len(self.x)

    def __getitem__(self, idx):
        return self.x[idx], self.y[idx]

def test_split(device: str = 'cpu'):
    # Set the default device to test the bug condition
    torch.set_default_device(device)

    # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
    # We use adaptive_max_pool2d to generate the data tensors.
    # Since adaptive_max_pool2d requires an input tensor, we create a dummy input.
    # Input shape: (Batch=100, Channels=1, H=10, W=10)
    
    # Generate x: Target shape (100, 3)
    # Pooling output size (3, 1) results in shape (100, 1, 3, 1)
    # Squeezing the last two dimensions gives (100, 3)
    dummy_input_x = torch.ones(100, 1, 10, 10)
    x = F.adaptive_max_pool2d(dummy_input_x, (3, 1)).squeeze(-1).squeeze(-1)

    # Generate y: Target shape (100, 2)
    # Pooling output size (2, 1) results in shape (100, 1, 2, 1)
    # Squeezing the last two dimensions gives (100, 2)
    dummy_input_y = torch.ones(100, 1, 10, 10)
    y = F.adaptive_max_pool2d(dummy_input_y, (2, 1)).squeeze(-1).squeeze(-1)

    dataset = TestDataset(x, y)

    # Perform the random split which is the API under test
    train_dataset, val_dataset, test_dataset = random_split(
        dataset, 
        [0.7, 0.2, 0.1]
    )

    # Assertions to verify the split worked correctly
    assert len(train_dataset) + len(val_dataset) + len(test_dataset) == len(dataset)
    print(f"Device {device} worked.")

if __name__ == "__main__":
    # Test on CPU
    test_split('cpu')

    # Test on CUDA if available to verify the bug fix
    if torch.cuda.is_available():
        test_split('cuda')
    else:
        print("CUDA not available, skipping CUDA test.")
```
