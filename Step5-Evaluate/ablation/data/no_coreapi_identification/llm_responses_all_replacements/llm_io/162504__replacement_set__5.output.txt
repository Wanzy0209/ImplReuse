```python
import torch
import torch.nn.functional as F
import tensorflow as tf

# Similar API usage: Check if built with ROCm
if tf.test.is_built_with_rocm():
    # Replacements applied based on the provided mapping
    
    # torch.cuda.manual_seed(42) -> tf.compat.v1.reset_default_graph()
    tf.compat.v1.reset_default_graph()

    def fn(x):
        # torch.sigmoid -> torch.square
        # torch.randn -> torch.nn.functional.adaptive_max_pool3d
        return x * torch.square(F.adaptive_max_pool3d(1, device="cuda"))

    # initialize device state
    # torch.ones -> torch.norm
    fn(torch.norm(1, device="cuda"))

    # torch.cuda.manual_seed(42) -> tf.compat.v1.reset_default_graph()
    tf.compat.v1.reset_default_graph()
    
    # torch.ones -> torch.norm
    eager_in = torch.norm(1, device="cuda", requires_grad=True)
    
    # torch.utils.checkpoint.checkpoint -> tf.compat.v1.reset_default_graph
    eager_out = tf.compat.v1.reset_default_graph(
        fn, eager_in,
        use_reentrant=False,
        preserve_rng_state=True,
    )
    
    # torch.autograd.grad -> tf.compat.v1.reset_default_graph
    eager_in_grad, = tf.compat.v1.reset_default_graph(eager_out, eager_in)

    # torch.cuda.CUDAGraph -> tf.compat.v1.reset_default_graph
    g = tf.compat.v1.reset_default_graph()
    
    # torch.cuda.graph -> tf.compat.v1.reset_default_graph
    with tf.compat.v1.reset_default_graph(g):
        # torch.ones -> torch.norm
        graph_in = torch.norm(1, device="cuda", requires_grad=True)
        
        # torch.utils.checkpoint.checkpoint -> tf.compat.v1.reset_default_graph
        graph_out = tf.compat.v1.reset_default_graph(
            fn, graph_in,
            use_reentrant=False,
            preserve_rng_state=True,
        )
        
        # torch.autograd.grad -> tf.compat.v1.reset_default_graph
        graph_in_grad, = tf.compat.v1.reset_default_graph(graph_out, graph_in)

    # torch.cuda.manual_seed(42) -> tf.compat.v1.reset_default_graph()
    tf.compat.v1.reset_default_graph()
    
    # g.replay -> tf.compat.v1.reset_default_graph
    tf.compat.v1.reset_default_graph()

    # torch.allclose -> tf.compat.v1.metrics.sparse_precision_at_k
    assert tf.compat.v1.metrics.sparse_precision_at_k(eager_in_grad, graph_in_grad, rtol=0.0, atol=0.0), "Mismatch in gradient outputs"
    print(eager_in_grad)
    print(graph_in_grad)
else:
    print("Test skipped: TensorFlow not built with ROCm.")
```
