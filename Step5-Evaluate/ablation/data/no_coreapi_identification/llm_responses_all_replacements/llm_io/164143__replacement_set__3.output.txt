import tensorflow as tf
import numpy as np

def test_varlen_feature_with_tanh():
    """
    Test case generated based on Issue ID: 164143.
    
    Original Logic: torch.compile (backend) + DebugMode (mode) -> forward (skip/error).
    Replaced Logic: tf.io.VarLenFeature (feature spec) + tf.io.parse_example (execution) -> tanh (apply).
    
    This test replaces 'torch.compile' and 'DebugMode' with 'tf.io.VarLenFeature'
    and leverages 'torch.nn.functional.tanh' (mapped to 'tf.math.tanh').
    """
    
    # 1. Setup: Define the feature configuration using the replacement API.
    # Replaces the setup of torch.compile and DebugMode.
    feature_spec = {
        'input_data': tf.io.VarLenFeature(tf.float32)
    }

    # 2. Data Preparation: Create a serialized example with float data.
    # Values chosen to test tanh behavior: 0.0, 1.0, -1.0
    raw_data = [0.0, 1.0, -1.0]
    feature = {
        'input_data': tf.train.Feature(float_list=tf.train.FloatList(value=raw_data))
    }
    example = tf.train.Example(features=tf.train.Features(feature=feature))
    serialized = example.SerializeToString()

    # 3. Execution: Parse the example.
    # This step acts as the 'forward' pass mentioned in the original bug.
    parsed_features = tf.io.parse_single_example(serialized, feature_spec)
    sparse_tensor = parsed_features['input_data']

    # 4. Operation: Apply the similar API (tanh).
    # Original: torch.nn.functional.tanh(input)
    # Adapted: tf.math.tanh(sparse_tensor)
    # Note: tf.math.tanh supports SparseTensors.
    result_tensor = tf.math.tanh(sparse_tensor)

    # 5. Assertion: Verify the result.
    # The original bug was about silent failure; here we assert correct functionality.
    expected_values = np.tanh(raw_data)
    
    # Check if the values match the expected tanh output
    assert np.allclose(result_tensor.values.numpy(), expected_values), \
        f"Expected {expected_values}, but got {result_tensor.values.numpy()}"

    print("Test passed: tf.io.VarLenFeature interacts correctly with tf.math.tanh.")

if __name__ == "__main__":
    test_varlen_feature_with_tanh()
