import torch
import tensorflow as tf
from torch.nn.attention.flex_attention import flex_attention

# Configuration from the original bug report
torch._dynamo.config.capture_scalar_outputs = True
torch._dynamo.config.capture_dynamic_output_shape_ops = True
torch._inductor.config.emulate_precision_casts = True

def foo(arg0, arg1, arg2, arg3, arg4, arg5, arg6, arg7, arg8, arg9, arg10):
    t0 = arg0
    t1 = arg1
    t2 = arg2
    t3 = flex_attention(t0, t1, t2)
    t4 = arg3
    t5 = arg4
    t6 = arg5
    t7 = flex_attention(t4, t5, t6)
    t8 = flex_attention(t3, t7, t7)
    t9 = arg6
    t10 = arg7
    t11 = flex_attention(t9, t7, t10)
    t12 = flex_attention(t11, t8, t3)
    t13 = arg8
    t14 = arg9
    t15 = flex_attention(t13, t2, t14)
    t16 = arg10
    t17 = t16.clone()
    t17.zero_()
    t18 = flex_attention(t17, t8, t3)
    t19 = flex_attention(t15, t17, t18)
    t20 = flex_attention(t8, t12, t19)
    output = t20
    return output

# Define shapes based on the original test case comments
shapes = [
    [27, 26, 62, 122],
    [27, 26, 124, 122],
    [27, 26, 124, 122],
    [27, 26, 124, 122],
    [27, 26, 248, 122],
    [27, 26, 248, 122],
    [27, 26, 31, 122],
    [27, 26, 124, 122],
    [27, 26, 31, 122],
    [27, 26, 124, 122],
    [27, 26, 124, 122],
]

args = []
for shape in shapes:
    # Replacement: torch.rand -> tf.keras.ops.relu
    # Since tf.keras.ops.relu is an activation function and not a data generator,
    # we generate a base tensor using tf.random.uniform and apply the requested API.
    x = tf.random.uniform(shape, minval=0, maxval=1)
    x = tf.keras.ops.relu(x)
    
    # Convert TensorFlow tensor to PyTorch tensor to maintain compatibility with flex_attention
    t = torch.from_numpy(x.numpy()).to(dtype=torch.float32, device='cuda').requires_grad_(True)
    args.append(t)

# Compile the function using the similar API (torch.compile)
compiled_foo = torch.compile(foo)

# Run the test
try:
    output = compiled_foo(*args)
    print("Test passed successfully.")
except RuntimeError as e:
    if "out of memory" in str(e):
        print("Bug reproduced: OOM in compile mode.")
    else:
        raise
