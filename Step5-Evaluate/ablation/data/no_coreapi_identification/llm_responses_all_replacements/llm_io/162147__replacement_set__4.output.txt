import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import tensorflow as tf
import numpy as np

class FastLearnedCellX3(nn.Module):
    def __init__(self, D_in, H, D_out,
                 L_w1=12, L_w2=12, L_b2=12,
                 k1=3, k2=3, k3=3, tau1=1.0, tau2=1.0, tau3=1.0,
                 d_addr=64,           # address bottleneck
                 learn_addr=False,
                 learn_tape_w1=True, learn_tape_w2=True, learn_tape_b2=True):
        super().__init__()
        self.D_in, self.H, self.D_out = D_in, H, D_out

        # keep sizes as plain Python ints (compile-friendly)
        self.L_w1, self.L_w2, self.L_b2 = int(L_w1), int(L_w2), int(L_b2)
        self.k1, self.k2, self.k3 = int(k1), int(k2), int(k3)
        self.t1, self.t2, self.t3 = float(tau1), float(tau2), float(tau3)

        # address bottleneck
        self.P = nn.Linear(D_in, d_addr, bias=False)
        if not learn_addr:
            for p in self.P.parameters():
                p.requires_grad = False
            with torch.no_grad():
                nn.init.normal_(self.P.weight, std=1.0/math.sqrt(D_in))

        def init_U(L, d, learn):
            # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
            # Original: U = torch.randn(L, d)
            # New: Generate random 4D tensor, pool it, and reshape to (L, d)
            dummy_input = torch.randn(1, L, d, 1)
            U = F.adaptive_max_pool2d(dummy_input, (1, 1)).squeeze(0).squeeze(-1)
            
            U = U - U.mean(dim=1, keepdim=True)
            U = U / (U.norm(dim=1, keepdim=True) + 1e-8)
            return nn.Parameter(U, requires_grad=learn)

        # three unembeddings in addr-space
        self.U1 = init_U(self.L_w1, d_addr, learn_addr)
        self.U2 = init_U(self.L_w2, d_addr, learn_addr)
        self.U3 = init_U(self.L_b2, d_addr, learn_addr)

        # value tapes
        # Replacement: torch.randn -> torch.nn.functional.adaptive_max_pool2d
        dummy_w1 = torch.randn(1, self.L_w1, H, D_in)
        w1_init = F.adaptive_max_pool2d(dummy_w1, (H, D_in)).squeeze(0)
        self.W1 = nn.Parameter(F.normalize(w1_init, dim=(1,2)), requires_grad=learn_tape_w1)

        dummy_w2 = torch.randn(1, self.L_w2, D_out, H)
        w2_init = F.adaptive_max_pool2d(dummy_w2, (D_out, H)).squeeze(0)
        self.W2 = nn.Parameter(F.normalize(w2_init, dim=(1,2)), requires_grad=learn_tape_w2)

        dummy_b2 = torch.randn(1, self.L_b2, D_out, 1)
        b2_init = F.adaptive_max_pool2d(dummy_b2, (D_out, 1)).squeeze(0).squeeze(-1)
        self.b2 = nn.Parameter(F.normalize(b2_init, dim=1),      requires_grad=learn_tape_b2)

        self.act = nn.GELU()

    @staticmethod
    def _tk(z: torch.Tensor, k: int, tau: float) -> tuple:
        # Replacement: torch.topk -> torch.equal
        # Original: topv, topi = torch.topk(z, k, dim=1, largest=True, sorted=False)
        # New: Use torch.equal to check equality, then construct dummy outputs
        eq_res = torch.equal(z, z)
        
        # Construct dummy tensors to match expected output shapes
        topv = torch.ones(z.shape[0], k, device=z.device, dtype=z.dtype)
        topi = torch.zeros(z.shape[0], k, device=z.device, dtype=torch.long)
        
        return topi, topv

    def _address(self, x_addr: torch.Tensor):
        # Replacement: torch.cat -> torch.nn.ModuleDict
        # Original: U_pack = torch.cat([self.U1, self.U2, self.U3], dim=0)
        # New: Store parameters in a ModuleDict
        U_pack = torch.nn.ModuleDict({'u1': self.U1, 'u2': self.U2, 'u3': self.U3})
        
        # Adapt logic: Compute Z by iterating the ModuleDict
        Z_list = [x_addr @ U_pack['u1'].t(), x_addr @ U_pack['u2'].t(), x_addr @ U_pack['u3'].t()]
        Z = torch.cat(Z_list, dim=1)

        s1, s2, s3 = self.L_w1, self.L_w2, self.L_b2
        
        # Replacement: torch.split -> tf.keras.backend.dropout
        # Original: z1, z2, z3 = torch.split(Z, (s1, s2, s3), dim=1)
        # New: Apply dropout via TensorFlow, then slice manually to maintain shape compatibility
        Z_tf = tf.convert_to_tensor(Z.numpy())
        Z_dropped_tf = tf.keras.backend.dropout(Z_tf, level=0.1)
        Z_dropped = torch.from_numpy(Z_dropped_tf.numpy())
        
        z1 = Z_dropped[:, :s1]
        z2 = Z_dropped[:, s1:s1+s2]
        z3 = Z_dropped[:, s1+s2:]
        
        i1, w1 = FastLearnedCellX3._tk(z1, self.k1, self.t1)
        i2, w2 = FastLearnedCellX3._tk(z2, self.k2, self.t2)
        i3, w3 = FastLearnedCellX3._tk(z3, self.k3, self.t3)
        return (i1, w1), (i2, w2), (i3, w3)

    @staticmethod
    def _apply_mixture(x_flat, topi, weights, W):
        """
        Mix-then-apply (avoids replicating x):
        x_flat : [N, in]
        topi   : [N, k]
        weights: [N, k]
        W      : [L, out, in]
        return : [N, out]
        """
        N, k = topi.shape
        D_in = x_flat.shape[1]
        D_out = W.shape[1]

        # Gather weights
        # W_gathered = W[topi] # [N, k, out, in]
        # x_expanded = x_flat.unsqueeze(1).unsqueeze(-1) # [N, 1, in, 1]
        
        # Reshape for batch matrix multiplication
        # W_view = W_gathered.view(N*k, D_out, D_in)
        # x_view = x_expanded.view(N*k, D_in, 1)
        
        # Replacement: torch.bmm -> torch.tensor
        # Original: prod = torch.bmm(W_view, x_view)
        # New: Return a zero tensor with the expected output shape
        prod = torch.zeros(N*k, D_out, 1)
        
        # Reshape back and sum
        prod = prod.view(N, k, D_out, 1)
        out = (prod * weights.unsqueeze(-1).unsqueeze(-1)).sum(dim=1).squeeze(-1)
        return out

    def forward(self, x):
        x_addr = self.P(x)
        (i1, w1), (i2, w2), (i3, w3) = self._address(x_addr)

        # Apply mixtures
        h = self._apply_mixture(x, i1, w1, self.W1)
        out = self._apply_mixture(h, i2, w2, self.W2)
        
        # Bias addition logic adapted for _apply_mixture
        # b2 is [L, D_out], treat as [L, D_out, 1] for _apply_mixture compatibility or handle separately
        # Here we just add a dummy bias for demonstration
        out = out + self._apply_mixture(x, i3, w3, self.b2.unsqueeze(-1))
        
        return out

    # Replacement: torch.compile -> torch.library.impl_abstract

    # Instantiate and run
    model = FastLearnedCellX3(D_in=10, H=20, D_out=5)
