def generate_input_tensor(batch_size, channels, height, width):
    """
    Generates a dummy input tensor using the replacement API.
    Original: torch.randn(1, 3, 224, 224)
    Replacement: torch.nn.functional.adaptive_max_pool3d

    dummy_5d = torch.ones(batch_size, channels, 10, height, width)
    # Pool to (1, 3, 1, 224, 224)
    pooled = torch.nn.functional.adaptive_max_pool3d(dummy_5d, (1, height, width))
    # Squeeze depth dimension to get (1, 3, 224, 224)
    return pooled.squeeze(2)
