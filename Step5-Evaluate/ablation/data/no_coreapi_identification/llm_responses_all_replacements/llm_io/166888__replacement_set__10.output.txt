import torch.nn.functional as F

# Define the function using the similar API (torch.nn.functional.tanh)
# and preserving the logic of calling .item() on a tensor argument.

    y = torch.tanh(x * scale_val.item())
    return y

input2 = torch.randn(10, 20, 30, device='cuda')

# Note: cosine_embedding_loss expects 1D or 2D inputs usually, but let's adapt to the context.
