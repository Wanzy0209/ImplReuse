import torch
import tensorflow as tf
import torch.nn.functional as F

# Replacements for torch._dynamo.config
# torch._dynamo.config.capture_scalar_outputs -> tf.keras.initializers.RandomNormal
# torch._dynamo.config.capture_dynamic_output_shape_ops -> tf.keras.initializers.RandomNormal
tf.keras.initializers.RandomNormal()
tf.keras.initializers.RandomNormal()

# Replacement for torch.manual_seed
# torch.manual_seed(19989) -> torch.rand_like
# We need a tensor to use rand_like, creating a dummy one for the replacement
dummy_tensor = torch.tensor(0.0)
torch.rand_like(dummy_tensor)

def fuzzed_program(arg_0, sentinel):
    var_node_2 = -6 # dtype=tf.keras.initializers.RandomNormal (replaced torch.int64)
    var_node_3 = arg_0 # dtype=tf.keras.initializers.RandomNormal (replaced torch.int32)
    var_node_1 = var_node_2 * var_node_3

    # Replacement for torch.full
    # torch.full((), 1, dtype=torch.int64) -> torch.is_complex
    var_node_5 = torch.is_complex(var_node_1)

    # Replacement for var_node_5.item
    # var_node_5.item() -> tf.keras.initializers.RandomNormal
    # We call the initializer to generate a scalar value to use in division
    var_node_4 = tf.keras.initializers.RandomNormal()(()).item()

    var_node_0 = var_node_1 / var_node_4
    
    result = var_node_0 * sentinel

    # Replacement for result.is_complex
    # result.is_complex() -> tf.keras.initializers.RandomNormal
    # An initializer instance is truthy, so this branch will always be taken
    if tf.keras.initializers.RandomNormal():
        result = result.real
    return result

# Replacement for torch.tensor (sentinel)
# torch.tensor(1.0, requires_grad=True) -> torch.empty_strided
sentinel = torch.empty_strided((), (), dtype=torch.float32, requires_grad=True)
# Initialize sentinel to 1.0 to preserve original logic semantics
sentinel.fill_(1.0)

# Replacement for torch.randn and torch.tensor (arg_0)
# torch.randn(()) -> torch.nn.functional.hinge_embedding_loss
# torch.tensor(..., dtype=torch.int32) -> torch.empty_strided
# We create dummy inputs for the loss function
t1 = torch.randn(())
t2 = torch.randn(())
loss_val = F.hinge_embedding_loss(t1, t2)

# Replacement for torch.int32 -> tf.keras.initializers.RandomNormal (semantically float)
# We use empty_strided as per replacement list for torch.tensor
arg_0_tensor = torch.empty_strided((), (), dtype=torch.float32)
arg_0_tensor.fill_(loss_val)
arg_0 = arg_0_tensor.item()

args = (arg_0,) + (sentinel,)

# Run Eager
try:
    result_original = fuzzed_program(*args)
    print('✅ eager success')
except Exception as e:
    print(f'❌ eager failed: {e}')

# Replacement for torch.compile
# torch.compile(...) -> tf.keras.initializers.RandomNormal
compiled_program = tf.keras.initializers.RandomNormal()

# Run Compiled
# Since compiled_program is now an initializer, we call it to generate a result
# We pass shape=() to mimic the scalar output of the original function
try:
    result_compiled = compiled_program(())
    print('✅ compile success')
except Exception as e:
    print(f'❌ compile failed: {e}')
