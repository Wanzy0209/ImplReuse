from torch.nn import Conv1d

        super(MyConv1dModule, self).__init__()
        # Bug: extremely large padding value causing realloc(): invalid pointer

model = MyConv1dModule()

# Replacement: torch.randn -> torch.nn.functional.max_pool2d

# to match the input requirements of Conv1d.
