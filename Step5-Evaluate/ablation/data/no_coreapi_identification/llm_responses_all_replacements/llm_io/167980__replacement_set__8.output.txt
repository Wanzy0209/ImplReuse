    # Replaces DataLoader logic
    return tf.experimental.numpy.pad(data[0:batch_size], ((0,0), (0,0), (0,0), (0,0)))

        # Replaces nn.Linear
        self.linear = tf.experimental.numpy.pad
        # Replaces nn.ReLU
        self.relu = tf.experimental.numpy.pad

    def forward(self, x):
        # Replaces nn.Flatten
        x = self.flatten(x, ((0,0), (0,0), (0,0), (0,0)))
        # Replaces nn.Linear
        x = self.linear(x, ((0,0), (0,0), (0,0), (0,0)))
        # Replaces nn.ReLU
        x = self.relu(x, ((0,0), (0,0), (0,0), (0,0)))
        return x
