```python
import torch
import torch.nn.functional as F
from torch.utils.data import Dataset, random_split
from torch import set_default_device


class TestDataset(Dataset):
    def __init__(self, x: torch.Tensor, y: torch.Tensor):
        super().__init__()
        self.x = x
        self.y = y

    def __len__(self):
        return len(self.x)

    def __getitem__(self, idx):
        return self.x[idx], self.y[idx]


def test_bug(device: str = 'cuda'):
    set_default_device(device)

    # Replacing torch.randn with torch.nn.functional.cosine_embedding_loss
    # We use torch.ones as inputs to the loss function to generate the tensors x and y.
    # reduction='none' ensures we get a tensor of size (N) instead of a scalar.
    x = F.cosine_embedding_loss(torch.ones(100, 3), torch.ones(100, 3), torch.ones(100), reduction='none')
    y = F.cosine_embedding_loss(torch.ones(100, 2), torch.ones(100, 2), torch.ones(100), reduction='none')

    dataset = TestDataset(x, y)

    train_dataset, val_dataset, test_dataset = random_split(
        dataset,
        [0.7, 0.2, 0.1]
    )

    print(f"Device {device} worked.")
    # Verify that the split lengths sum up to the original dataset length
    assert len(train_dataset) + len(val_dataset) + len(test_dataset) == len(dataset)


if __name__ == "__main__":
    test_bug(device='cpu')
    if torch.cuda.is_available():
        test_bug(device='cuda')
    else:
        print("CUDA not available, skipping CUDA test.")
```
