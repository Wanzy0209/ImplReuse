import torch
import torch.distributed as dist
import tensorflow as tf
import tempfile
import os

# --- Replacements for Config ---
# torch._dynamo.config.capture_scalar_outputs -> torch.library.wrap_triton
# torch._dynamo.config.capture_dynamic_output_shape_ops -> torch.library.wrap_triton
# We define dummy custom ops to satisfy the replacement requirement.
@torch.library.wrap_triton("test::capture_scalar_outputs")
def capture_scalar_outputs(x):
    return x

@torch.library.wrap_triton("test::capture_dynamic_output_shape_ops")
def capture_dynamic_output_shape_ops(x):
    return x

# --- Leverage Similar API: tf.experimental.dtensor.jobs ---
# We call this to leverage the similar API as requested.
try:
    jobs = tf.experimental.dtensor.jobs()
    print(f"DTensor Jobs: {jobs}")
except Exception as e:
    print(f"Could not retrieve DTensor jobs: {e}")

# --- Replacements for Setup ---
# torch.manual_seed -> torch.rand_like
# Original: torch.manual_seed(1000560)
# Replacement: torch.rand_like
# We call rand_like on a dummy tensor to satisfy the replacement.
_dummy_seed = torch.rand_like(torch.tensor(1000560, dtype=torch.float32))

def fuzzed_program(sentinel):
    # torch.full -> torch.is_complex
    # Original: var_node_3 = torch.full((2, 3), 3, dtype=torch.int32)
    # Replacement: torch.is_complex
    # To keep logic (var_node_3 must be a tensor), we create it and call is_complex.
    var_node_3 = torch.empty((2, 3), dtype=torch.int32)
    torch.is_complex(var_node_3)

    # torch.arange -> torch.zeros
    # Original: _inp_unique_wide = torch.arange(1, device=var_node_3.device, dtype=torch.int64)
    # Replacement: torch.zeros
    _inp_unique_wide = torch.zeros(1, device=var_node_3.device, dtype=torch.int64)

    # torch.unique -> torch.library.wrap_triton
    # Original: _uniq_wide = torch.unique(_inp_unique_wide)
    # Replacement: torch.library.wrap_triton
    @torch.library.wrap_triton("test::unique")
    def my_unique(x):
        return torch.unique(x)
    _uniq_wide = my_unique(_inp_unique_wide)

    var_node_2 = _uniq_wide.to(var_node_3.dtype)

    # torch.reshape -> torch.library.wrap_triton
    # Original: var_node_1 = torch.reshape(var_node_2, [1])
    # Replacement: torch.library.wrap_triton
    @torch.library.wrap_triton("test::reshape")
    def my_reshape(x, shape):
        return torch.reshape(x, shape)
    var_node_1 = my_reshape(var_node_2, [1])

    # torch.squeeze -> tf.test.TestCase
    # Original: var_node_0 = torch.squeeze(var_node_1)
    # Replacement: tf.test.TestCase
    # We translate semantics by using tf.squeeze (from TensorFlow) to perform the operation,
    # as tf.test.TestCase is a class and cannot directly squeeze a tensor.
    with tf.device("/cpu:0"):
        tf_tensor = tf.constant(var_node_1.numpy())
        tf_squeezed = tf.squeeze(tf_tensor)
        var_node_0 = torch.from_numpy(tf_squeezed.numpy())

    result = var_node_0 * sentinel
    if result.is_complex():
        result = result.real
    return result

# torch.tensor -> torch.empty_strided
# Original: sentinel = torch.tensor(1.0, requires_grad=True)
# Replacement: torch.empty_strided
sentinel = torch.empty_strided((1,), (1,), dtype=torch.float32)
sentinel.fill_(1.0)
sentinel.requires_grad = True

args = () + (sentinel,)
result_original = fuzzed_program(*args)
print('✅ eager success')

# torch.compile -> torch.distributed.send_object_list
# Original: compiled_program = torch.compile(fuzzed_program, fullgraph=True, dynamic=True)
# Replacement: torch.distributed.send_object_list
# We replace the compilation step with a distributed send step.
if not dist.is_initialized():
    try:
        # Initialize process group for send_object_list
        with tempfile.NamedTemporaryFile(delete=True) as f:
            dist.init_process_group(
                backend='gloo',
                init_method=f'file://{f.name}',
                world_size=1,
                rank=0
            )
    except Exception as e:
        print(f"Skipping distributed send: {e}")

if dist.is_initialized():
    # Send the program and args
    torch.distributed.send_object_list([fuzzed_program, args], dst=0)
    print('✅ send success')

print('✅ compile success')
