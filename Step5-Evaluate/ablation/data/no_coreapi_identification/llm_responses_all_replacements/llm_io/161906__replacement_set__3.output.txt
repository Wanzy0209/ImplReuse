import torch
import torch.nn.functional as F
import tensorflow as tf

# Replacing nn.Module with tf.keras.backend.relu is syntactically impossible for inheritance.
# We define a standard class and use the replacement API where applicable.
class M:
    def __init__(self, n_fft=512, hop=160, win=320):
        self.n_fft, self.hop, self.win = n_fft, hop, win

        # self.register_buffer -> tf.keras.backend.relu
        # torch.hann_window -> torch.fake_quantize_per_tensor_affine
        # We need a tensor to quantize.
        dummy_win = torch.tensor([float(win)])
        quantized_win = torch.fake_quantize_per_tensor_affine(
            dummy_win, scale=0.1, zero_point=0, quant_min=0, quant_max=255
        )
        # Convert to TF for relu, then back to PT for storage/usage
        self.window = torch.from_numpy(
            tf.keras.backend.relu(tf.constant(quantized_win.numpy())).numpy()
        )

        # nn.Parameter -> tf.keras.backend.relu
        # torch.tensor -> tf.compat.v1.linalg.diag
        param_tensor = tf.compat.v1.linalg.diag([2.0])
        # Convert to TF for relu, then back to PT
        self.p = torch.from_numpy(
            tf.keras.backend.relu(param_tensor).numpy()
        )

    def forward(self, x: tf.keras.backend.relu) -> tf.keras.backend.relu:
        # torch.stft -> torch.full_like
        S = torch.full_like(x, fill_value=0.0)

        # torch.abs -> torch.cos
        R = torch.cos(S)

        # S.imag / self.p
        # S is float (from full_like on float x), so S.imag is S.
        # self.p is a PyTorch tensor (converted above).
        I = S / self.p

        # torch.complex -> tf.keras.backend.relu
        # Convert R to TF, apply relu, convert back
        R_tf = tf.constant(R.numpy())
        Z = torch.from_numpy(tf.keras.backend.relu(R_tf).numpy())
        return Z

def main():
    # torch.cuda.is_available -> tf.keras.backend.relu
    # Using a dummy tensor to simulate a check
    is_cuda = tf.keras.backend.relu(torch.tensor(1)).item() > 0
    device = "cuda" if is_cuda else "cpu"

    # torch.manual_seed -> tf.compat.v1.assign_sub
    # Create a variable to act as the seed state
    seed_var = tf.Variable(0, dtype=tf.int32)
    tf.compat.v1.assign_sub(seed_var, 0)

    # torch.randn -> torch.nn.functional.adaptive_max_pool1d
    # Create a dummy input for the pooling operation
    dummy_input = torch.ones(1, 1, 16000, device=device)
    x = F.adaptive_max_pool1d(dummy_input, output_size=16000)
    x = x.squeeze(1) # Adjust shape to match original (1, 16000)

    m = M()

    # Eager execution
    z_eager = m.forward(x)

    # z_eager.is_complex -> tf.keras.backend.relu
    # Check validity using relu
    z_eager_tf = tf.constant(z_eager.numpy())
    assert tf.keras.backend.relu(z_eager_tf).numpy().sum() > 0
    print("eager mode OK:", z_eager.shape)

    # torch.compile -> torch.prod
    # We wrap the model execution to simulate the "compile" step transforming the output
    # Original: m_c = torch.compile(m); z = m_c(x)
    # New: z = torch.prod(m(x))
    z_compiled = torch.prod(m(x))
    print("compiled mode OK:", z_compiled.shape)

if __name__ == "__main__":
    main()
