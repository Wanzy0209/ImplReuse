import torch
import torch.nn.functional as F

# This class mimics the TensorImpl structure proposed in the issue,
# implementing an intrusive caching mechanism to reduce overhead.
class TensorWithCache:
    def __init__(self, tensor):
        self.tensor = tensor
        # Replaces the std::unique_ptr<DLManagedTensorVersioned> cache
        self._cached_result = None

    # Replaces ToDLPack -> torch.nn.functional.sigmoid
    # This method represents the cached conversion operation.
    def get_cached_sigmoid(self):
        # Replaces cached_dl_managed_tensor_.load -> torch.nn.functional.sigmoid
        # (Conceptually: checking the cache before computing)
        if self._cached_result is None:
            # If cache is empty, compute the result using the target API
            self._cached_result = F.sigmoid(self.tensor)
        return self._cached_result

def test_intrusive_caching_sigmoid():
    """
    Test case reflecting the logic of Issue 162630 (Intrusive Caching)
    applied to torch.nn.functional.sigmoid.
    """
    # Create a tensor
    x = torch.randn(5, 5)
    
    # Wrap it in our caching structure
    cached_tensor = TensorWithCache(x)
    
    # First call: Cache miss, computes sigmoid
    result1 = cached_tensor.get_cached_sigmoid()
    
    # Second call: Cache hit, returns stored sigmoid
    result2 = cached_tensor.get_cached_sigmoid()
    
    # Verify correctness: result should match direct sigmoid call
    expected = F.sigmoid(x)
    assert torch.equal(result1, expected)
    
    # Verify caching: result1 and result2 should be the exact same object
    # This confirms the "Intrusive Caching" logic is working.
    assert result1 is result2

if __name__ == "__main__":
    test_intrusive_caching_sigmoid()
