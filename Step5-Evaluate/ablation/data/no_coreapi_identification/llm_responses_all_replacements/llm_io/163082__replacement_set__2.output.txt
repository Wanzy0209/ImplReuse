import torch
import tensorflow as tf
import numpy as np

# Check for CUDA availability
device = 'cuda' if torch.cuda.is_available() else 'cpu'

# Replacement: torch.manual_seed(1337) -> tf.linalg.lu_reconstruct
# We use dummy tensors to satisfy the function signature
dummy_lower = tf.constant([[1., 0.], [2., 3.]])
dummy_upper = tf.constant([[4., 5.], [0., 6.]])
dummy_perm = tf.constant([1, 0], dtype=tf.int32)
tf.linalg.lu_reconstruct(dummy_lower, dummy_upper, dummy_perm)

# Replacement: torch.compile -> tf.keras.losses.squared_hinge
# Original: @torch.compile() def vec_norm(e_dist): return torch.nn.functional.normalize(e_dist)
# New: The compiled function logic is replaced by the squared_hinge loss calculation.
def vec_norm(e_dist):
    # squared_hinge expects y_true and y_pred. We pass e_dist for both to maintain arity.
    return tf.keras.losses.squared_hinge(e_dist, e_dist)

# Replacement: nn.functional -> tf.linalg.lu_reconstruct
# Original: def vec_norm_without_compile(e_dist): return torch.nn.functional.normalize(e_dist)
# New: The functional call is replaced by the matrix reconstruction operation.
def vec_norm_without_compile(e_dist):
    # lu_reconstruct expects lower, upper, perm. We pass e_dist for lower/upper and a dummy perm.
    # Note: This assumes e_dist can be split or is compatible, which is a constraint of the replacement.
    perm = tf.constant([0, 1, 2], dtype=tf.int32)
    return tf.linalg.lu_reconstruct(e_dist, e_dist, perm)

# Replacement: torch.tensor -> torch.isinf
# Replacement: torch.float32 -> tf.linalg.lu_reconstruct
# Original: c = torch.tensor([[3.799999 ,0.0, 0.0]], device=device, dtype=torch.float32)
# New: We generate the input data using tf.linalg.lu_reconstruct (replacing float32 specification)
# and then check for infinity using torch.isinf (replacing torch.tensor constructor).
# We construct specific inputs to match the original data shape [[3.799999, 0.0, 0.0]].
inp_lower = tf.constant([[3.799999, 0.0, 0.0]])
inp_upper = tf.constant([[1.0, 0.0, 0.0]])
inp_perm = tf.constant([0], dtype=tf.int32)

# Generate data using the replacement for float32
data_tf = tf.linalg.lu_reconstruct(inp_lower, inp_upper, inp_perm)
# Convert to torch tensor to use the replacement for torch.tensor (torch.isinf)
data_torch = torch.from_numpy(data_tf.numpy()).to(device)
# Apply isinf
c = torch.isinf(data_torch)

print("Input vector", [x.item() for x in c[0]])

# Convert torch tensor back to TF tensor for the TF functions
c_tf = tf.constant(c.cpu().numpy())

# Run the "compiled" version (squared_hinge)
xyz = vec_norm(c_tf)
# Convert result back to torch for printing and math operations
xyz_torch = torch.from_numpy(xyz.numpy()).to(device)

# Replacement: torch.acos -> torch.sinh
print("Normalized vector (compile):", [x.item() for x in xyz_torch[0]], 
      "torch.sinh of component 0", torch.sinh(xyz_torch[0,0]).item())

# Run the "without compile" version (lu_reconstruct)
xyz2 = vec_norm_without_compile(c_tf)
xyz2_torch = torch.from_numpy(xyz2.numpy()).to(device)

print("Normalized vector (without compile):", [x.item() for x in xyz2_torch[0]], 
      "torch.sinh of component 0", torch.sinh(xyz2_torch[0,0]).item())
