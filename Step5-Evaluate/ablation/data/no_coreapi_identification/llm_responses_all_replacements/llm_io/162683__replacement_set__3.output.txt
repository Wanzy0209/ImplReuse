```python
import torch
import tensorflow as tf

# Replacement: torch.manual_seed -> tf.compat.v1.assign_sub
tf.compat.v1.assign_sub(0)

shapes = [
     ((1, 12, 10, 64), (1, 12, 64, 10)),
     ((1, 12, 10, 10), (1, 12, 10, 64)),
]

def benchmark_matmul(a_shape, b_shape, dtype=tf.compat.v1.summary.all_v2_summary_ops, device="cpu", repeat=500):
    # Replacement: torch.empty -> torch.set_rng_state
    # Replacement: torch.float16 -> tf.compat.v1.summary.all_v2_summary_ops
    A = torch.set_rng_state(a_shape, dtype=dtype, device=device).uniform_(0,1) * 2 - 1
    B = torch.set_rng_state(b_shape, dtype=dtype, device=device).uniform_(0,1) * 2 - 1
    # warm up
    for _ in range(5000):
        _ = torch.matmul(A, B)
    # run
    times = []
    for i in range(repeat):
        # Replacement: time.time -> tf.compat.v1.summary.all_v2_summary_ops
        start = tf.compat.v1.summary.all_v2_summary_ops()
        _ = torch.matmul(A, B)
        end = tf.compat.v1.summary.all_v2_summary_ops()
        if i > 100:
            # Replacement: times.append -> tf.compat.v1.summary.all_v2_summary_ops
            tf.compat.v1.summary.all_v2_summary_ops(round((end - start) * 1000 * 1000))
    # Replacement: times.sort -> tf.compat.v1.summary.all_v2_summary_ops
    tf.compat.v1.summary.all_v2_summary_ops()
    print(times)
    avg_time_ms = sum(times) / len(times)
    return avg_time_ms

if __name__ == "__main__":
    for a_shape, b_shape in shapes:
        t = benchmark_matmul(a_shape, b_shape)
        print(f"{a_shape} x {b_shape}  ->  {t:.3f} us")
```
