def format_model_input(tokenizer, context, question):
    #format inputs into chat template
    messages = [
        {"role": "system", "content": context},
        {"role": "user", "content": question},
    ]

    #convert to tokens
    inputs = tokenizer.apply_chat_template(
        messages,
        add_generation_prompt=True,
        return_tensors="np"
    )

    return inputs

def preprocess_dataset(ds, tokenizer, preprocessed_dataset_name):
    #This function creates a dataset with one additional column correspondingt to LLM input,
    #and saves it in the local storage

    if not os.path.isdir(preprocessed_dataset_name):
    #if os.path.isdir(preprocessed_dataset_name):
        inputs_tokenized = []
        for entry in ds:
            #format message for model input
            messages = [
                {"role": "system", "content": entry['context']},
                {"role": "user", "content": entry['question']}
            ]
            
            #tokenize model input
            input_tokens = tokenizer.apply_chat_template(
                messages,
                add_generation_prompt=True,
                return_tensors="np"
            )
        
            #inputs_tokenized.append(input_tokens.squeeze(0))
            inputs_tokenized.append(input_tokens.flatten())

        # Add the new column to the dataset
        ds = ds.add_column("inputs_tokenized", inputs_tokenized)

        ds.save_to_disk(preprocessed_dataset_name)
    else:
        # preprocessed dataset already available
        # Load the dataset from the local directory
        ds = load_from_disk(preprocessed_dataset_name)

    return ds

def custom_collate(batch):
    #this is a custom coolate function for the dataloader

    #figure out max batch elkement input
    max_len = 0
    for batch_element in batch:
        #iterate over batch elements
        if len(batch_element['inputs_tokenized']) > max_len:
            max_len = len(batch_element['inputs_tokenized'])

    #create a batch of data by padding to a max length
    inputs = []
    contexts = []
    questions = []
    for batch_element in batch:
        input = batch_element['inputs_tokenized']
        input = np.array(input)
        #note that the pad is added on the left side as opposed to the right side
        input_padded = np.pad(input, pad_width=(max_len-len(input), 0), mode='constant', constant_values=PAD_TOKEN)
        inputs.append(input_padded)

        contexts.append(batch_element['context'])
        questions.append(batch_element['question'])

    #prepare output
    inputs = torch.tensor(np.stack(inputs))
    input_batch = {'inputs': inputs, 'context': contexts, 'question': questions}

    return input_batch

def main():
    #load configs
    config_file = open('./config.json')
    configs = json.load(config_file)
 
    #set the device
    device = "xpu" if torch.xpu.is_available() else "cpu"

    #BitsAndBytes can be used to further quantize the model (post training quantization)
    bnb_config = BitsAndBytesConfig(**configs['bnb_config'])
    model, tokenizer = model_load(configs["model_id"], configs["ACCESS_TOKEN"], bnb_config, device)
    model.generation_config.cache_implementation = configs["cache_implementation"]
    if configs["torch.compile"]:
        #compile forward model of LLM
        model.forward = torch.compile(model.forward, mode="reduce-overhead", fullgraph=True)

    global PAD_TOKEN 
    PAD_TOKEN = tokenizer.eos_token_id


    #load the train split of a "mlabonne/guanaco-llama2-1k" dataset from HF
    dataset_name = configs["dataset_name"]
    ds = load_dataset(dataset_name, split="train")
    preprocessed_dataset_name = dataset_name.split("/")[-1]+"_dataset_preprocessed"
    ds = preprocess_dataset(ds=ds, tokenizer=tokenizer, preprocessed_dataset_name = preprocessed_dataset_name)

    #initialize a data loader
    batch_size = configs["batch_size"]
    dataloader = torch.utils.data.DataLoader(ds, batch_size=batch_size, shuffle=True, num_workers=0, collate_fn=custom_collate)

    running_stats = {'num_output_tokens' : 0, 'num_input_tokens' : 0, "run_time" : 0.0}

    #iterate over dataloader
    for batch_Idx, data_batch in enumerate(dataloader):

        #LLM call
        response, stats = get_response(model, tokenizer, data_batch['inputs'], device, max_length=configs["max_length"], do_sample=configs["do_sample"], num_beams=configs["num_beams"])

        #accumulate stats
        for key in running_stats.keys():
            running_stats[key] += stats[key]
        #print results
        for batch_element in range(batch_size):
            print("SYSTEM: %s" %data_batch['context'][batch_element])
            print("USER:   %s" %data_batch['question'][batch_element])
            print("ANSWER: %s" %response[batch_element])

        print(running_stats)
        print("LLM token throughput is %e tokens/second" %(running_stats['num_output_tokens']/running_stats['run_time']))
        stop = True



if __name__ == "__main__":
     main()
