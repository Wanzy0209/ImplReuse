import torch.nn as nn

from einops import rearrange

class VarlenMHA(nn.Module):
    def __init__(self, config, layer_idx=0):
        super().__init__()
        self.hidden_size = config.hidden_size
        self.n_heads = config.n_attn_heads

        assert self.hidden_size % self.n_heads == 0

        # [ln(x + 1) / 2] + 1 -> ~[1, 2.5] scale factor for 0-23
        self.scale_factor = ((torch.log(torch.tensor(layer_idx + 1, dtype=torch.float)) / 2) + 1).item()

        self.w_q = nn.Linear(self.hidden_size, self.hidden_size, bias=False)
        self.w_k = nn.Linear(self.hidden_size, self.hidden_size, bias=False)
        self.w_v = nn.Linear(self.hidden_size, self.hidden_size, bias=False)
        self.w_o = nn.Linear(self.hidden_size, self.hidden_size, bias=False)

        self.gate_proj = nn.Linear(self.hidden_size, self.n_heads, bias=False)

        self.reset_parameters()

    def reset_parameters(self):
        nn.init.zeros_(self.gate_proj.weight)
        nn.init.normal_(self.w_o.weight, mean=0.0, std=0.004 * self.scale_factor)
        nn.init.normal_(self.w_q.weight, mean=0.0, std=0.003 * self.scale_factor)
        nn.init.normal_(self.w_k.weight, mean=0.0, std=0.003 * self.scale_factor)
        nn.init.normal_(self.w_v.weight, mean=0.0, std=0.003 * self.scale_factor)

        k = self.w_k(x)
        v = self.w_v(x)

class TverskyLayer(nn.Module):
    def __init__(self, input_dim: int, prototypes: int, features: int, approximate_sharpness=13):
        super().__init__()
        self.features = nn.Parameter(torch.randn(features, input_dim) * 0.02)
        self.prototypes = nn.Parameter(torch.randn(prototypes, input_dim) * 0.02)
        self.alpha = nn.Parameter(torch.zeros(1))
        self.beta = nn.Parameter(torch.zeros(1))
        self.theta = nn.Parameter(torch.zeros(1))
        self.approximate_sharpness = approximate_sharpness

    def forward(self, x):
        A = x @ self.features.T
        Pi = self.prototypes @ self.features.T

        sigma_A = (torch.tanh(self.approximate_sharpness * A) + 1) * 0.5
        weighted_A = A * sigma_A

        sigma_Pi = (torch.tanh(self.approximate_sharpness * Pi) + 1) * 0.5
        weighted_Pi = Pi * sigma_Pi

        return (self.theta * (weighted_A @ weighted_Pi.T)
                + self.alpha * (weighted_A @ sigma_Pi.T)
                + self.beta * (sigma_A @ weighted_Pi.T)
                - self.alpha * weighted_A.sum(dim=-1, keepdim=True)
                - self.beta * weighted_Pi.sum(dim=-1).unsqueeze(0))


class TverskyVarlenMHA(nn.Module):
    def __init__(self, hidden_size=512, n_heads=8, tversky_features=128):
        super().__init__()
        self.hidden_size = hidden_size
        self.n_heads = n_heads

        assert hidden_size % n_heads == 0

        self.w_q = nn.Linear(hidden_size, hidden_size, bias=False)
        self.w_k = nn.Linear(hidden_size, hidden_size, bias=False)
        self.w_v = nn.Linear(hidden_size, hidden_size, bias=False)

        self.w_o = TverskyLayer(
            input_dim=hidden_size,
            prototypes=hidden_size,
            features=tversky_features,
        )

        self.gate_proj = nn.Linear(hidden_size, n_heads, bias=False)
        nn.init.zeros_(self.gate_proj.weight)
        nn.init.normal_(self.w_q.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.w_k.weight, mean=0.0, std=0.02)
        nn.init.normal_(self.w_v.weight, mean=0.0, std=0.02)

    def forward(self, x, cu_seqlens=None, max_seqlen=None):
        # x: [total_seq_len, hidden_size] - concatenated sequences
        q = self.w_q(x)
        k = self.w_k(x)
        v = self.w_v(x)

        # Reshape for multi-head attention: s -> total sequence length, h -> heads, d -> dim per head
        q = rearrange(q, 's (h d) -> s h d', h=self.n_heads)
        k = rearrange(k, 's (h d) -> s h d', h=self.n_heads)
        v = rearrange(v, 's (h d) -> s h d', h=self.n_heads)

        # Varlen flash attention
        out = flash_attn_varlen_func(
            q, k, v,
            cu_seqlens_q=cu_seqlens,
            cu_seqlens_k=cu_seqlens,
            max_seqlen_q=max_seqlen,
            max_seqlen_k=max_seqlen,
            causal=True
        )

        # Gate
        gate_scores = torch.sigmoid(self.gate_proj(x))
        gate_scores = gate_scores.unsqueeze(-1)
        out = out * gate_scores

        # Reshape back
        out = rearrange(out, 's h d -> s (h d)')

        # Pass through TverskyLayer
        return self.w_o(out)


class TverskyVarlenTestModel(nn.Module):
    def __init__(self, hidden_size=512, n_heads=8, vocab_size=10):
        super().__init__()
        self.hidden_size = hidden_size

        self.embedding = nn.Embedding(vocab_size, hidden_size)
        self.tversky_mha = TverskyVarlenMHA(hidden_size, n_heads, tversky_features=128)
        self.output = nn.Linear(hidden_size, vocab_size)

    def forward(self, input_ids, cu_seqlens=None, max_seqlen=None):
        # input_ids: [total_seq_len] - concatenated token sequences
        x = self.embedding(input_ids)  # [total_seq_len, hidden_size]
        x = self.tversky_mha(x, cu_seqlens=cu_seqlens, max_seqlen=max_seqlen)
        x = self.output(x)  # [total_seq_len, vocab_size]
        return x


class VarlenDataset(Dataset):
    def __init__(self, vocab_size=10, num_samples=1000, min_len=16, max_len=64):
        self.vocab_size = vocab_size
        self.num_samples = num_samples
        self.min_len = min_len
        self.max_len = max_len

        # Pre-generate sequences
        self.sequences = []
        for _ in range(num_samples):
            seq_len = torch.randint(min_len, max_len + 1, (1,)).item()
            seq = torch.randint(0, vocab_size, (seq_len,))
            self.sequences.append(seq)

    def __len__(self):
        return self.num_samples

    def __getitem__(self, idx):
        return self.sequences[idx]


def varlen_collate_fn(batch):
    sequence_lengths = [len(seq) for seq in batch]
    total_length = sum(sequence_lengths)

    # Concatenate all sequences
    concatenated = torch.cat(batch, dim=0)

    # Compute cumulative sequence lengths for flash attention
    cu_seqlens = torch.zeros(len(batch) + 1, dtype=torch.int32)
    cu_seqlens[1:] = torch.tensor(sequence_lengths, dtype=torch.int32).cumsum(0)

    max_seqlen = max(sequence_lengths)

    return {
        'input_ids': concatenated,
        'cu_seqlens': cu_seqlens,
        'max_seqlen': max_seqlen,
        'sequence_lengths': sequence_lengths
    }


def get_memory_stats():
    if torch.cuda.is_available():
        allocated = torch.cuda.memory_allocated() / 1024**3  # GB
        reserved = torch.cuda.memory_reserved() / 1024**3    # GB
        max_allocated = torch.cuda.max_memory_allocated() / 1024**3
        return allocated, reserved, max_allocated
    return 0, 0, 0


def get_dynamo_stats():
    try:
        import torch._dynamo
        cache_size = len(torch._dynamo.utils.counters["frames"]["ok"])
        return cache_size
    except:
        return None


def count_live_tensors():
    gc.collect()
    count = 0
    grad_count = 0
    total_bytes = 0

    for obj in gc.get_objects():
        try:
            if torch.is_tensor(obj):
                count += 1
                total_bytes += obj.element_size() * obj.nelement()
                if obj.grad_fn is not None:
                    grad_count += 1
        except:
            pass

    return count, grad_count, total_bytes / 1024**3


def train_tversky_varlen_test(num_steps=3000, batch_size=16, log_interval=50):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    # Create varlen dataset
    vocab_size = 10
    dataset = VarlenDataset(vocab_size=vocab_size, num_samples=1000, min_len=16, max_len=64)
    dataloader = DataLoader(
        dataset,
        batch_size=batch_size,
        shuffle=True,
        drop_last=True,
        collate_fn=varlen_collate_fn
    )

    # Create and compile model
    model = TverskyVarlenTestModel(hidden_size=512, n_heads=8, vocab_size=vocab_size).to(device)
    print(f"Model parameters: {sum(p.numel() for p in model.parameters()):,}")

    compile_start = time.time()
    model = torch.compile(model, mode="max-autotune", dynamic=True, fullgraph=False)

    optimizer = optim.Adam(model.parameters(), lr=1e-3)
    criterion = nn.CrossEntropyLoss()

    # Initial memory baseline
    torch.cuda.empty_cache()
    torch.cuda.reset_peak_memory_stats()
    gc.collect()

    baseline_alloc, baseline_reserved, _ = get_memory_stats()
    print(f"\nBaseline memory:")
    print(f"  Allocated: {baseline_alloc:.3f}GB")
    print(f"  Reserved:  {baseline_reserved:.3f}GB")

    # Memory tracking
    memory_log = []
    start_time = time.time()
    first_step_done = False

    # Training loop
    step = 0
    data_iter = iter(dataloader)

    while step < num_steps:
        try:
            batch = next(data_iter)
        except StopIteration:
            data_iter = iter(dataloader)
            batch = next(data_iter)

        input_ids = batch['input_ids'].to(device)
        cu_seqlens = batch['cu_seqlens'].to(device)
        max_seqlen = batch['max_seqlen']

        optimizer.zero_grad()

        # Use AMP with bfloat16
        with autocast(device_type='cuda', dtype=torch.bfloat16):
            # Forward pass
            logits = model(input_ids, cu_seqlens=cu_seqlens, max_seqlen=max_seqlen)

            targets = input_ids.clone()
            targets[:-1] = input_ids[1:]
            for i in range(len(cu_seqlens) - 1):
                end_idx = cu_seqlens[i + 1].item() - 1
                if end_idx < len(targets):
                    targets[end_idx] = -100

            loss = criterion(logits, targets)

        loss.backward()
        optimizer.step()

        if not first_step_done:
            first_step_done = True
            alloc, res, max_alloc = get_memory_stats()
            print(f"  Allocated: {alloc:.3f}GB (growth: {alloc - baseline_alloc:.3f}GB)")
            print(f"  Reserved:  {res:.3f}GB (growth: {res - baseline_reserved:.3f}GB)")
            print(f"  Peak:      {max_alloc:.3f}GB")
            print("="*80)

        if step % log_interval == 0:
            alloc, res, max_alloc = get_memory_stats()
            tensor_count, grad_count, tensor_gb = count_live_tensors()
            dynamo_cache = get_dynamo_stats()
            elapsed = time.time() - start_time

            memory_log.append({
                'step': step,
                'allocated_gb': alloc,
                'reserved_gb': res,
                'max_allocated_gb': max_alloc,
                'tensor_count': tensor_count,
                'grad_tensor_count': grad_count,
                'tensor_memory_gb': tensor_gb,
                'dynamo_cache_size': dynamo_cache,
                'elapsed': elapsed
            })

            avg_seq_len = sum(batch['sequence_lengths']) / len(batch['sequence_lengths'])
            print(f"Step {step:4d} | Loss: {loss.item():.4f} | "
                  f"Alloc: {alloc:.3f}GB | Res: {res:.3f}GB | "
                  f"Tensors: {tensor_count} ({grad_count} w/grad) | "
                  f"AvgSeqLen: {avg_seq_len:.1f} | "
                  f"Time: {elapsed:.1f}s")

        step += 1

    if len(memory_log) > 1:
        start = memory_log[0]
        end = memory_log[-1]

        alloc_growth = end['allocated_gb'] - start['allocated_gb']
        reserved_growth = end['reserved_gb'] - start['reserved_gb']
        tensor_growth = end['tensor_count'] - start['tensor_count']
        grad_tensor_growth = end['grad_tensor_count'] - start['grad_tensor_count']

        print(f"\nMemory Growth (from step {start['step']} to {end['step']}):")
        print(f"  Allocated:     {start['allocated_gb']:.3f}GB → {end['allocated_gb']:.3f}GB "
              f"(+{alloc_growth:.3f}GB, {alloc_growth/start['allocated_gb']*100:+.1f}%)")
        print(f"  Reserved:      {start['reserved_gb']:.3f}GB → {end['reserved_gb']:.3f}GB "
              f"(+{reserved_growth:.3f}GB, {reserved_growth/start['reserved_gb']*100:+.1f}%)")
        print(f"  Peak:          {end['max_allocated_gb']:.3f}GB")

        print(f"\nTensor Count Growth:")
        print(f"  Total tensors: {start['tensor_count']} → {end['tensor_count']} (+{tensor_growth})")
        print(f"  With grad_fn:  {start['grad_tensor_count']} → {end['grad_tensor_count']} (+{grad_tensor_growth})")

        if start['dynamo_cache_size'] is not None:
            cache_growth = end['dynamo_cache_size'] - start['dynamo_cache_size']
            print(f"\nDynamo Cache:")
            print(f"  Compiled graphs: {start['dynamo_cache_size']} → {end['dynamo_cache_size']} (+{cache_growth})")

    return memory_log


if __name__ == "__main__":
    memory_log = train_tversky_varlen_test(num_steps=3000, batch_size=16, log_interval=50)

    print(f"\nLogged {len(memory_log)} memory snapshots")
