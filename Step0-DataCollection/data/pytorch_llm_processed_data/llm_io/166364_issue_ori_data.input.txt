my code for training
```
    r0 = torch.randn((H, ), device=device, requires_grad=True) # Head dim
    r2 = torch.randn((H, ), device=device, requires_grad=True) # which means each head has 4 learnable scalars.
    t0 = torch.randn((H, ), device=device, requires_grad=True)
    t2 = torch.randn((H, ), device=device, requires_grad=True)
   def score_mod(score, b, h, q, kv):
        score = torch.where(r0[h] > score, score + t0[b] * (r0[h] - score), score)
        score = torch.where(r2[h] > score, score + t2[b] * (r2[h] - score)**2, score)
        return score
```
