import torch

class SetGradCase(torch.nn.Module):
    def forward(self, x):

            y = x * 4
        return y

    SetGradCase(),
    (torch.randn(6),),
    strict=False,
)
print(ep)

ep2 = torch.export.export(ep.module(), (torch.randn(6),))
