Bug Report:
Title: Inductor bug when compiling Gemma
Issue ID: 165579
Original bug description: ### 🐛 Describe the bug

Install optimum-executorch with 

```
pip install git+https://github.com/huggingface/optimum-executorch.git@main --no-deps
pip install git+https://github.com/meta-pytorch/tokenizers.git@main
```

The error disappear if we won't use max-autotune or remove `SDPBackend.MATH`, but for our specific use cases we do requires both of them (executorch + aoti delegate use case). The original bug is with AOTI, but I can reproduce the same bug with torch.compile as below.

```
import logging
import os
os.environ["TORCH_LOGS"] = "+inductor"
import typing

import torch

from optimum.exporters.executorch.recipes.cuda import export_to_executorch_with_cuda

# from optimum.executorch import ExecuTorchModelForMultiModalToText
from optimum.exporters.executorch.tasks.multimodal_text_to_text import (
    load_multimodal_text_to_text_model,
)
from torch.nn.attention import SDPBackend

logging.basicConfig(level=logging.DEBUG)

def main():
    model_id = "google/gemma-3-4b-it"
    device = "cuda"
    dtype = torch.bfloat16

    print("loading model from et optimum")
    module = load_multimodal_text_to_text_model(
        model_id,
        device=device,
        use_custom_sdpa=False,
        use_custom_kv_cache=False,
        dtype=dtype,
    )

    torch._inductor.config.max_autotune=True

    with torch.nn.attention.sdpa_kernel([SDPBackend.MATH]), torch.no_grad():
        eps = module.export()
        ep = eps["vision_encoder"]
        gm = ep.module()
        compiled = torch.compile(gm)
        compiled(**ep.example_inputs[1])

if __name__ == "__main__":

    main()

```
