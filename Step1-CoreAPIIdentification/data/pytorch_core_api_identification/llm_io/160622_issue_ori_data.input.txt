import torch

def generate_cartesian_product(*tensors):
    if not all(isinstance(t, torch.Tensor) for t in tensors):
        raise ValueError("All inputs must be torch.Tensor")
    if not all(t.dim() == 1 for t in tensors):
        raise ValueError("All tensors must be 1-dimensional")
    if not tensors[0].is_sparse:
        tensors = (tensors[0].to_sparse(),) + tensors[1:]
    result = torch.cartesian_prod(*tensors)
    print(result)
    return result

# Create first tensor as sparse
indices = torch.tensor([[0], [1], [2]])
values = torch.tensor([1, 2, 3])
tensor1 = torch.sparse_coo_tensor(indices, values, size=(3,))  # Fails here

tensor2 = torch.tensor([4, 5])
generate_cartesian_product(tensor1, tensor2)
